   119→/**
   120→ * موقعیت شبکهٔ هر خانه — شماره‌گذاری مارپیچی کلاسیک:
   121→ * ردیف پایین (b = 0) از چپ بالا می‌رود، ردیف‌های فرد از راست (زیگزاگ).
   122→ * چیدمان RTL است، پس ستون شبکه = ۱۰ − فاصله از چپ؛
   123→ * مثلاً خانهٔ ۱ → col 10/row 10 (پایین-چپ) و خانهٔ ۱۰۰ → col 10/row 1 (بالا-چپ).
   124→ */
   125→const CELL_POS: { col: number; row: number }[] = Array.from({ length: 100 }, (_, i) => {
   126→  const n = i + 1;
   127→  const b = Math.floor((n - 1) / 10);
   128→  const fromLeft = b % 2 === 0 ? (n - 1) % 10 : 9 - ((n - 1) % 10);
   129→  return { col: 10 - fromLeft, row: 9 - b + 1 };
   130→});

   137→function wrongIndexOf(q: QuizQuestionSeed): number {
   138→  const wrongs = [0, 1, 2, 3].filter((i) => i !== q.answer);
   139→  return wrongs[Math.floor(Math.random() * wrongs.length)] ?? q.answer;
   140→}

   221→  /** زمان‌بندی داخل زنجیرهٔ ref‌شده — بعد از اجرا خودش را از فهرست حذف می‌کند */
   222→  function later(fn: () => void, ms: number): void {
   223→    const t = window.setTimeout(() => {
   224→      timersRef.current = timersRef.current.filter((x) => x !== t);

   237→  function drawNext(): void {

   252→  function bumpStats(p: number, ok: boolean): void {
   253→    const s = statsRef.current[p];
   254→    s.asked += 1;
   255→    if (ok) s.correct += 1;
   256→    else s.wrong += 1;

   260→  function beginTurn(p: number): void {
   261→    curRef.current = p;
   262→    setCur(p);
   263→    setReveal(null);
   264→    setDiceFace(null);
   265→    turnLockRef.current = false;

   268→      later(() => botAnswer(p), BOT_DELAY);
   269→    }
   270→  }
   271→
   272→  /** تاس: چند چرخش نمایشی → عدد قطعی → شروع حرکت مهره */
   273→  function proceedAfterCorrect(p: number): void {
   274→    const d = 1 + Math.floor(Math.random() * 6);
   275→    let flips = 0;
   276→    function cycle(): void {
   277→      flips += 1;
   278→      if (flips <= DICE_FLIPS) {
   279→        setDiceFace(1 + Math.floor(Math.random() * 6));
   280→        setDicePop((k) => k + 1);
   281→        later(cycle, 90);
   282→      } else {
   283→        setDiceFace(d);
   284→        setDicePop((k) => k + 1);
   285→        later(() => hop(p, d), 420);
   286→      }
   287→    }
   288→    later(cycle, 350);
   289→  }
   290→
   291→  /** حرکت مهره خانه‌به‌خانه با زنجیرهٔ setTimeout پله‌ای (~۱۲۰ms) */
   292→  function hop(p: number, d: number): void {
   293→    const from = posRef.current[p];
   294→    const to = Math.min(100, from + d);
   295→    let at = from;
   296→    function step(): void {
   297→      if (at < to) {
   298→        at += 1;
   299→        posRef.current[p] = at;
   300→        setPositions([...posRef.current]);
   301→        later(step, STEP_MS);
   302→      } else {
   303→        land(p, d);
   304→      }
   305→    }
   306→    step();
   307→  }
   308→
   309→  /** بررسی مار/پله در خانهٔ فرود و اعلام انتقال با toast */
   310→  function land(p: number, d: number): void {
   311→    const pos = posRef.current[p];
   312→    const snakeTo = SNAKES[pos];
   313→    const ladderTo = LADDERS[pos];
   314→    if (snakeTo !== undefined) {
   315→      toast.error(`مار! برگشتی به خانهٔ ${faNum(snakeTo)}`);
   316→      later(() => {
   317→        posRef.current[p] = snakeTo;
   318→        setPositions([...posRef.current]);
   319→        afterJump(p, d);
   320→      }, 550);
   321→    } else if (ladderTo !== undefined) {
   322→      toast.success(`پله! رفتی به خانهٔ ${faNum(ladderTo)}`);
   323→      later(() => {
   324→        posRef.current[p] = ladderTo;
   325→        setPositions([...posRef.current]);
   326→        afterJump(p, d);
   327→      }, 550);
   328→    } else {
   329→      afterJump(p, d);
   330→    }
   331→  }
   332→
   333→  /** بعد از مار/پله: پایان بازی یا نوبت بعدی (تاس ۶ = نوبت اضافه) */
   334→  function afterJump(p: number, d: number): void {
   335→    if (posRef.current[p] >= 100) {
   336→      finish(p);
   337→      return;
   338→    }
   339→    if (d === 6) {
   340→      toast.success(`تاس ${faNum(6)} — نوبت اضافه برای ${playersRef.current[p]?.name ?? ""}!`);
   341→      beginTurn(p);
   342→    } else {
   343→      beginTurn((p + 1) % playersRef.current.length);
   344→    }
   345→  }
   346→
   347→  /** پاسخ ربات — نمایش گزینهٔ انتخابی‌اش و طی همان جریان بازیکن انسانی */
   348→  function botAnswer(p: number): void {
   349→    const cq = qRef.current;
   350→    if (!cq) return;
   351→    const ok = Math.random() < BOT_SKILL;
   352→    const idx = ok ? cq.answer : wrongIndexOf(cq);
   353→    setReveal({ picked: idx, ok, byBot: true });
   354→    bumpStats(p, ok);
   355→    if (ok) {
   356→      later(() => proceedAfterCorrect(p), 600);
   357→    } else {
   358→      toast.error(`${playersRef.current[p]?.name ?? "ربات"} بند شد!`);
   359→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   360→    }
   361→  }
   362→
   363→  function answerHuman(idx: number): void {
   364→    if (turnLockRef.current || !q) return;
   365→    const p = curRef.current;
   366→    if (playersRef.current[p]?.isBot) return;
   367→    turnLockRef.current = true;
   368→    const ok = idx === q.answer;
   369→    setReveal({ picked: idx, ok, byBot: false });
   370→    bumpStats(p, ok);
   371→    if (ok) {
   372→      later(() => proceedAfterCorrect(p), 350);
   373→    } else {
   374→      toast.error("بند شدی!");
   375→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   376→    }
   377→  }

   393→  function start(): void {
   394→    if (pool.length < MIN_POOL) {
   395→      toast.error("سؤال کافی در این محدوده نیست — محدودهٔ دیگری را انتخاب کن");
   396→      return;
   397→    }
   398→    clearTimers();
   399→    const ps = buildPlayers();
   400→    playersRef.current = ps;
   401→    setPlayers(ps);
   402→    posRef.current = ps.map(() => 0);
   403→    setPositions([...posRef.current]);
   404→    statsRef.current = ps.map(() => ({ asked: 0, correct: 0, wrong: 0 }));
   405→    setStats(statsRef.current.map((x) => ({ ...x })));
   406→    // سؤال یکتا روی متن — کل استخر شافل‌شده؛ تمام شد؟ در drawNext از نو شافل می‌شود
   407→    deckRef.current = pickUnique(pool, pool.length, (item) => item.text);
   408→    qiRef.current = -1;
   409→    curRef.current = 0;
   410→    secsRef.current = 0;
   411→    setSecs(0);
   412→    setQNum(0);
   413→    setReveal(null);
   414→    setDiceFace(null);
   415→    setDicePop(0);
   416→    setWinner(null);
   417→    setIsRecord(false);
   418→    setPhase("playing");
   419→    beginTurn(0);
   420→  }

   422→  function finish(winnerIdx: number): void {
   423→    clearTimers();
   424→    turnLockRef.current = true;
   425→    const spent = secsRef.current;
   426→    setSecs(spent);
   427→    touchStreak();
   428→    recordStudySession();
   429→    bumpPlays(activeBookId, GAME_KEY);
   430→    const ps = playersRef.current;
   431→    // آمار Attempt فقط از بازیکن‌های انسانی (total = سؤال‌های پرسیده‌شده از انسان‌ها)
   432→    const humanIdxs = ps.map((_, i) => i).filter((i) => !ps[i]?.isBot);
   433→    const total = humanIdxs.reduce((a, i) => a + (statsRef.current[i]?.asked ?? 0), 0);
   434→    const correct = humanIdxs.reduce((a, i) => a + (statsRef.current[i]?.correct ?? 0), 0);
   435→    if (total > 0) {
   436→      saveAttempt(activeBookId, GAME_KEY, {
   437→        lesson: lesson === ALL_LESSON ? 0 : lesson,
   438→        total,
   439→        correct,
   440→        duration: spent,
   441→        detail: {
   442→          winner: ps[winnerIdx]?.name ?? "",
   443→          players: ps.map((pl, i) => ({
   444→            name: pl.name,
   445→            isBot: pl.isBot,
   446→            asked: statsRef.current[i]?.asked ?? 0,
   447→            correct: statsRef.current[i]?.correct ?? 0,
   448→            wrong: statsRef.current[i]?.wrong ?? 0,
   449→            position: posRef.current[i] ?? 0,
   450→          })),
   451→        },
