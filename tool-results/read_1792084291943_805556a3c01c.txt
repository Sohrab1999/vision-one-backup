   259→function fmtDuration(sec: number): string {
   260→  if (!sec) return "—";
   261→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   262→  const m = Math.floor(sec / 60);
   263→  const s = sec % 60;
   264→  if (m === 0) return `${fa(s)} ثانیه`;
   265→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   266→}
   267→
   268→function fmtDate(iso: string): string {
   269→  try {
   270→    return new Date(iso).toLocaleDateString("fa-IR", { month: "short", day: "numeric" });
   271→  } catch {
   272→    return "";
   273→  }
   274→}
