   103→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
   104→function Confetti() {
   105→  const pieces = useMemo(
   106→    () =>
   107→      Array.from({ length: 32 }, (_, i) => ({
   108→        id: i,
   109→        left: Math.random() * 100,
   110→        delay: Math.random() * 1.2,
   111→        duration: 2.4 + Math.random() * 1.6,
   112→        size: 6 + Math.random() * 6,
   113→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   114→        drift: Math.round((Math.random() - 0.5) * 120),
   115→        round: Math.random() > 0.5,
   116→      })),
   117→    [],
   118→  );
   119→  return (
   120→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   121→      {pieces.map((p) => (
   122→        <span
   123→          key={p.id}
   124→          className="confetti-piece absolute top-[-14px] block"
   125→          style={{
   126→            left: `${p.left}%`,
   127→            width: p.size,
   128→            height: p.size * (p.round ? 1 : 0.45),
   129→            background: p.color,
   130→            borderRadius: p.round ? "50%" : 2,
   131→            animationDelay: `${p.delay}s`,
   132→            animationDuration: `${p.duration}s`,
   133→            "--confetti-drift": `${p.drift}px`,
   134→          } as React.CSSProperties}
   135→        />
   136→      ))}
   137→    </div>
   138→  );
   139→}

   359→  function chooseOption(optIdx: number) {
   360→    if (selected !== null) return;

   362→      answeredLeftRef.current = Math.max(0, Math.ceil((deadlineRef.current - Date.now()) / 1000));

   366→      const next = [...prev];
   367→      next[index] = optIdx;
   368→      return next;
   369→    });

   371→      if (optIdx === current.answer) {
   372→        // در مرور اشتباهات و مرور ترکیبی، پاسخ درست سؤال را از دفتر اشتباهات حذف می‌کند

   374→      } else {
   375→        recordMistake(current.id);
   376→      }
   377→    }
   378→  }

   403→  function nextQuestion() {
   404→    if (advanceRef.current) clearTimeout(advanceRef.current);
   405→    if (index + 1 >= questions.length) {
   406→      finish();
   407→      return;
   408→    }
   409→    setIndex((i) => i + 1);
   410→    setSelected(answers[index + 1]);
   411→    setAiExplanation(null);
   412→    deadlineRef.current = mode === "daily" ? 0 : Date.now() + timed * 1000;
   413→  }

   415→  // میان‌برهای کیبورد حین آزمون: ۱-۴ (فارسی/انگلیسی) انتخاب گزینه، Enter سؤال بعدی
   416→  useEffect(() => {
   417→    if (phase !== "playing") return;
   418→    function onKey(e: KeyboardEvent) {
   419→      const target = e.target as HTMLElement | null;
   420→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA")) return;
   421→      if (e.key === "Enter" && selected !== null) {
   422→        e.preventDefault();
   423→        nextQuestion();
   424→        return;
   425→      }
   426→      if (selected !== null) return;
   427→      const fa = "۱۲۳۴".indexOf(e.key);
   428→      const en = "1234".indexOf(e.key);
   429→      const digit = fa >= 0 ? fa : en;
   430→      if (digit >= 0 && digit < (current?.options.length ?? 0)) {
   431→        e.preventDefault();
   432→        chooseOption(digit);
   433→      }
   434→    }
   435→    window.addEventListener("keydown", onKey);
   436→    return () => window.removeEventListener("keydown", onKey);
   437→  }, [phase, selected, index, questions]);

   439→  function finish() {
   440→    if (timerRef.current) clearInterval(timerRef.current);
   441→    if (advanceRef.current) clearTimeout(advanceRef.current);
   442→    setPhase("result");
   443→    recordStudySession();
   444→    const streak = touchStreak();
   445→    if (mode === "daily") recordDailyResult(todayKey(), scorePct, questions.length);
   446→    const detail = questions.map((q, i) => ({ id: q.id, chosen: answers[i], answer: q.answer }));
   447→    fetch("/api/attempts", {
   448→      method: "POST",
   449→      headers: { "Content-Type": "application/json" },
   450→      body: JSON.stringify({
   451→        category:
   452→          mode === "review"
   453→            ? "review"
   454→            : mode === "daily"
   455→              ? "daily"
   456→              : mode === "starred"
   457→                ? "starred"
   458→                : mode === "combined"
   459→                  ? "combined"
   460→                  : category,
   461→        lesson: lesson === "all" ? 0 : Number(lesson),
   462→        total: questions.length,
   463→        correct: correctCount,
   464→        duration: elapsed,
   465→        name: userName || undefined,
   466→        detail,
   467→      }),
   468→    })
   469→      .then(() =>
   470→        toast.success(
   471→          mode === "daily"
   472→            ? `چالش امروز ثبت شد — ${scorePct.toLocaleString("fa-IR")}٪ ${scorePct >= 80 ? "🏆" : "💪"}`
   473→            : mode === "review"
   474→              ? `دور مرور تمام شد! ${mistakeCount.toLocaleString("fa-IR")} سؤال دیگر در دفتر اشتباهات مانده`
   475→              : mode === "starred"
   476→                ? `تمرین ستاره‌دارها ثبت شد — ${scorePct.toLocaleString("fa-IR")}٪ ⭐`
   477→                : mode === "combined"
   478→                  ? `مرور ترکیبی ثبت شد — ${scorePct.toLocaleString("fa-IR")}٪ 🧩`
   479→                  : `نتیجه آزمون ذخیره شد ${userName ? `— موفق باشی ${userName}!` : "🎉"}${streak.count > 1 ? ` 🔥 زنجیره ${streak.count.toLocaleString("fa-IR")} روزه` : ""}`,
   480→        ),
   481→      )
   482→      .catch(() => undefined);
   483→  }

   485→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   486→  async function exportMistakesCsv() {

   493→      if (!qs.length) {
