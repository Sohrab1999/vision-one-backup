    71→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
    72→function Confetti() {
    73→  const pieces = useMemo(
    74→    () =>
    75→      Array.from({ length: 32 }, (_, i) => ({
    76→        id: i,
    77→        left: Math.random() * 100,
    78→        delay: Math.random() * 1.2,
    79→        duration: 2.4 + Math.random() * 1.6,
    80→        size: 6 + Math.random() * 6,
    81→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
    82→        drift: Math.round((Math.random() - 0.5) * 120),
    83→        round: Math.random() > 0.5,
    84→      })),
    85→    [],
    86→  );
    87→  return (
    88→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
    89→      {pieces.map((p) => (
    90→        <span
    91→          key={p.id}
    92→          className="confetti-piece absolute top-[-14px] block"
    93→          style={{
    94→            left: `${p.left}%`,
    95→            width: p.size,
    96→            height: p.size * (p.round ? 1 : 0.45),
    97→            background: p.color,
    98→            borderRadius: p.round ? "50%" : 2,
    99→            animationDelay: `${p.delay}s`,
   100→            animationDuration: `${p.duration}s`,
   101→            "--confetti-drift": `${p.drift}px`,
   102→          } as React.CSSProperties}
   103→        />
   104→      ))}
   105→    </div>
   106→  );
   107→}

   109→function fmtTime(s: number) {
   110→  const m = Math.floor(s / 60);
   111→  const sec = s % 60;
   112→  return `${m.toLocaleString("fa-IR")}:${sec.toString().padStart(2, "0")}`;
   113→}

   115→export function QuizGame({ lessonFilter }: { lessonFilter?: number }) {
   116→  const [phase, setPhase] = useState<Phase>("setup");
   117→  const [mode, setMode] = useState<Mode>("normal");
   118→  const [category, setCategory] = useState("all");
   119→  const [lesson, setLesson] = useState(lessonFilter !== undefined ? String(lessonFilter) : "all");
   120→  const [count, setCount] = useState(10);
   121→
   122→  const [questions, setQuestions] = useState<Question[]>([]);
   123→  const [loading, setLoading] = useState(false);
   124→
   125→  const [index, setIndex] = useState(0);
   126→  const [selected, setSelected] = useState<number | null>(null);
   127→  const [answers, setAnswers] = useState<(number | null)[]>([]);
   128→  const [aiExplaining, setAiExplaining] = useState(false);
   129→  const [aiExplanation, setAiExplanation] = useState<string | null>(null);
   130→  const [elapsed, setElapsed] = useState(0);
   131→  const userName = useUserName();

   133→  const { ids: practiceIds } = usePracticeIds();
   134→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   135→
   136→  async function startPracticeRound(ids: string[], nextMode: Mode = "normal") {
   137→    setLoading(true);
   138→    setMode(nextMode);
   139→    try {
   140→      const params = new URLSearchParams({ count: String(Math.max(ids.length, 1)) });
   141→      params.set("ids", ids.join(","));
   142→      const res = await fetch(`/api/questions?${params.toString()}`);
   143→      const data = await res.json();
   144→      if (!res.ok || !data.questions?.length) {
   145→        toast.error("سؤالی برای تمرین یافت نشد.");
   146→        return;
   147→      }
   148→      setQuestions(data.questions as Question[]);
   149→      setAnswers(new Array(data.questions.length).fill(null));
   150→      setIndex(0);
   151→      setSelected(null);
   152→      setAiExplanation(null);
   153→      setElapsed(0);
   154→      setPhase("playing");
   155→      clearPracticeIds();
   156→    } catch {
   157→      toast.error("خطا در دریافت سؤالات");
   158→    } finally {
   159→      setLoading(false);
   160→    }
   161→  }
   162→
   163→  useEffect(() => {
   164→    if (lessonFilter !== undefined) setLesson(String(lessonFilter));
   165→  }, [lessonFilter]);

   168→    if (phase === "playing") {
   169→      timerRef.current = setInterval(() => setElapsed((s) => s + 1), 1000);
   170→    }
   171→    return () => {
   172→      if (timerRef.current) clearInterval(timerRef.current);

   174→  }, [phase]);
   175→
   176→  const current = questions[index];
   177→  const correctCount = useMemo(
   178→    () => questions.reduce((acc, q, i) => acc + (answers[i] === q.answer ? 1 : 0), 0),
   179→    [questions, answers],
   180→  );
   181→  const scorePct = questions.length
   182→    ? Math.round((correctCount / questions.length) * 100)
   183→    : 0;

   185→  async function startQuiz(nextMode: Mode = "normal") {
   186→    setLoading(true);
   187→    setMode(nextMode);
   188→    try {
   189→      const params = new URLSearchParams({
   190→        category: category === "all" ? "all" : category,
   191→        lesson: lesson === "all" ? "all" : lesson,
   192→        count: String(count),
   193→      });
   194→      if (nextMode === "review") params.set("ids", mistakeIds.join(","));
   195→      const res = await fetch(`/api/questions?${params.toString()}`);
   196→      const data = await res.json();
   197→      if (!res.ok || !data.questions?.length) {
   198→        toast.error("سؤالی یافت نشد — تنظیمات دیگری را امتحان کنید.");
   199→        return;
   200→      }
   201→      setQuestions(data.questions as Question[]);
   202→      setAnswers(new Array(data.questions.length).fill(null));
   203→      setIndex(0);
   204→      setSelected(null);
   205→      setAiExplanation(null);
   206→      setElapsed(0);
   207→      setPhase("playing");
   208→    } catch {
   209→      toast.error("خطا در دریافت سؤالات");
   210→    } finally {
   211→      setLoading(false);
   212→    }
   213→  }

   215→  function chooseOption(optIdx: number) {
   216→    if (selected !== null) return;
   217→    setSelected(optIdx);
   218→    setAnswers((prev) => {
   219→      const next = [...prev];
   220→      next[index] = optIdx;
   221→      return next;
   222→    });
   223→    if (current) {
   224→      if (optIdx === current.answer) {
   225→        if (mode === "review") resolveMistake(current.id);
   226→      } else {
   227→        recordMistake(current.id);
   228→      }
   229→    }
   230→  }
   231→
   232→  async function askAiExplanation() {
   233→    if (!current) return;
   234→    setAiExplaining(true);
   235→    try {
   236→      const res = await fetch("/api/explain", {
   237→        method: "POST",
   238→        headers: { "Content-Type": "application/json" },
   239→        body: JSON.stringify({
   240→          text: current.text,
   241→          options: current.options,
   242→          answer: current.answer,
   243→        }),
   244→      });
   245→      const data = await res.json();
   246→      if (res.ok) setAiExplanation(data.explanation);
   247→      else toast.error(data.error ?? "خطا در توضیح هوشمند");
   248→    } catch {
   249→      toast.error("خطا در ارتباط با هوش مصنوعی");
   250→    } finally {
   251→      setAiExplaining(false);
   252→    }
   253→  }
   254→
   255→  function nextQuestion() {
   256→    if (index + 1 >= questions.length) {
   257→      finish();
   258→      return;
   259→    }
   260→    setIndex((i) => i + 1);
   261→    setSelected(answers[index + 1]);
   262→    setAiExplanation(null);
   263→  }
   264→
   265→  function finish() {
   266→    if (timerRef.current) clearInterval(timerRef.current);
   267→    setPhase("result");
   268→    const streak = touchStreak();
   269→    const detail = questions.map((q, i) => ({ id: q.id, chosen: answers[i], answer: q.answer }));
   270→    fetch("/api/attempts", {
   271→      method: "POST",
   272→      headers: { "Content-Type": "application/json" },
   273→      body: JSON.stringify({
   274→        category: mode === "review" ? "review" : category,
   275→        lesson: lesson === "all" ? 0 : Number(lesson),
   276→        total: questions.length,
   277→        correct: correctCount,
   278→        duration: elapsed,
   279→        name: userName || undefined,
   280→        detail,
   281→      }),
   282→    })
   283→      .then(() =>
   284→        toast.success(
   285→          mode === "review"
   286→            ? `دور مرور تمام شد! ${mistakeCount.toLocaleString("fa-IR")} سؤال دیگر در دفتر اشتباهات مانده`
   287→            : `نتیجه آزمون ذخیره شد ${userName ? `— موفق باشی ${userName}!` : "🎉"}${streak.count > 1 ? ` 🔥 زنجیره ${streak.count.toLocaleString("fa-IR")} روزه` : ""}`,
   288→        ),
   289→      )
   290→      .catch(() => undefined);
   291→  }

   316→            </div>

   319→          {/* دور تمرین سؤال‌های پراشتباه (ارسالی از آمار) */}

   335→                  </p>
   336→                </div>
   337→                <Button
