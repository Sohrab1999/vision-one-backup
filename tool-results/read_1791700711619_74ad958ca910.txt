    90→    90→      aria-label={`سطح ${faNum(level)} — پیشرفت ${faNum(safe)} درصد`}

   330→   330→// ─── بخش رده‌بندی هفتگی (fetch + skeleton) ───
   331→   331→
   332→   332→function LeaderboardSection() {
   333→   333→  const { bookId } = useBook();
   334→   334→  const [loaded, setLoaded] = useState<{ bookId: string; data: BoardData } | null>(null);
   335→   335→  const [failedFor, setFailedFor] = useState<string | null>(null);
   336→   336→
   337→   337→  useEffect(() => {
   338→   338→    if (!bookId) return;
   339→   339→    let cancelled = false;
   340→   340→    const ctrl = new AbortController();
   341→   341→    fetch(`/api/leaderboard?book=${bookId}`, { signal: ctrl.signal })
   342→   342→      .then((r) => (r.ok ? r.json() : null))
   343→   343→      .then((raw: unknown) => {
   344→   344→        if (cancelled || !raw) return;
   345→   345→        const d = raw as Partial<BoardData>;
   346→   346→        setLoaded({
   347→   347→          bookId,
   348→   348→          data: {
   349→   349→            weekStart: typeof d.weekStart === "string" ? d.weekStart : "",
   350→   350→            rows: Array.isArray(d.rows) ? d.rows : [],
   351→   351→            me: d.me ?? null,
   352→   352→          },
   353→   353→        });
   354→   354→        setFailedFor(null);
   355→   355→      })
   356→   356→      .catch(() => {
   357→   357→        if (!cancelled) setFailedFor(bookId);
   358→   358→      });
   359→   359→    return () => {
   360→   360→      cancelled = true;
   361→   361→      ctrl.abort();
   362→   362→    };
   363→   363→  }, [bookId]);
   364→   364→
   365→   365→  const board = loaded && loaded.bookId === bookId ? loaded.data : null;
   366→   366→  const isFailed = bookId !== null && failedFor === bookId;
   367→   367→
   368→   368→  if (!bookId) {
   369→   369→    return (
   370→   370→      <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center text-sm text-muted-foreground">
   371→   371→        📚 برای دیدن رده‌بندی هفتگی، یک جزوه را از قفسه باز کن.
   372→   372→      </div>
   373→   373→    );
   374→   374→  }
   375→   375→
   376→   376→  if (!board) {
   377→   377→    if (isFailed) {
   378→   378→      return (
   379→   379→        <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center text-sm text-muted-foreground">
   380→   380→          🌱 رده‌بندی همین حالا در دسترس نیست — بعداً دوباره امتحان کن.
   381→   381→        </div>
   382→   382→      );
   383→   383→    }
   384→   384→    return (
   385→   385→      <div className="space-y-3" aria-label="در حال بارگذاری رده‌بندی" aria-busy="true">
   386→   386→        {Array.from({ length: 5 }).map((_, i) => (
   387→   387→          <Skeleton key={i} className="h-14 w-full rounded-2xl" />
   388→   388→        ))}
   389→   389→      </div>
   390→   390→    );
   391→   391→  }
   392→   392→
   393→   393→  if (board.rows.length === 0) {
   394→   394→    return (
   395→   395→      <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center">
   396→   396→        <div className="text-3xl" aria-hidden>
   397→   397→          🌱
   398→   398→        </div>
   399→   399→        <p className="mt-2 text-sm font-bold text-foreground">
   400→   400→          هنوز این هفته کسی امتیاز نگرفته — تو اولین باش!
   401→   401→        </p>
   402→   402→        <p className="mt-1 text-xs text-muted-foreground">
   403→   403→          با آزمون و بازی امتیاز جمع کن و از شنبه بالای جدول بایست.
   404→   404→        </p>
   405→   405→      </div>
   406→   406→    );
   407→   407→  }
   408→   408→
   409→   409→  return (
   410→   410→    <div className="space-y-3">
   411→   411→      {board.me && (
   412→   412→        <div className="flex items-center gap-2 rounded-2xl border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-xs font-bold text-amber-700 dark:text-amber-400">
   413→   413→          <Trophy className="size-3.5 shrink-0" aria-hidden />
   414→   414→          <span>
   415→   415→            جای تو در این هفته: ردهٔ {faNum(board.me.rank)} با {faNum(board.me.xp)} XP
   416→   416→          </span>
   417→   417→        </div>
   418→   418→      )}
   419→   419→      <ul className="space-y-2.5" aria-label="جدول رده‌بندی هفتگی">
   420→   420→        {board.rows.map((row, i) => {
   421→   421→          const rank = i + 1;
   422→   422→          return (
   423→   423→            <li
   424→   424→              key={`${row.name}-${rank}`}
   425→   425→              className={cn(
   426→   426→                "flex flex-wrap items-center gap-3 rounded-2xl border bg-background/50 px-3 py-2.5",
   427→   427→                rank <= 3 && "border-amber-500/30 bg-amber-500/5",
   428→   428→              )}
   429→   429→            >
   430→   430→              <span
   431→   431→                className="w-8 shrink-0 text-center text-sm font-extrabold tabular-nums"
   432→   432→                aria-label={`ردهٔ ${faNum(rank)}`}
   433→   433→              >
   434→   434→                {rank <= 3 ? (RANK_MEDALS[rank - 1] ?? "") : faNum(rank)}
   435→   435→              </span>
   436→   436→              {row.avatar ? (
   437→   437→                <span className="text-2xl leading-none" aria-hidden>
   438→   438→                  {row.avatar}
   439→   439→                </span>
