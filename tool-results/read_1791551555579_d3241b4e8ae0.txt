   220→function fmtDuration(sec: number): string {
   221→  if (!sec) return "—";
   222→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   223→  const m = Math.floor(sec / 60);
   224→  const s = sec % 60;
   225→  if (m === 0) return `${fa(s)} ثانیه`;
   226→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   227→}
   228→
   229→function fmtDate(iso: string): string {
   230→  try {
   231→    return new Date(iso).toLocaleDateString("fa-IR", { month: "short", day: "numeric" });
   232→  } catch {
   233→    return "";
   234→  }
   235→}
