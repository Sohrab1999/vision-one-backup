   106→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
   107→function Confetti() {
   108→  const pieces = useMemo(
   109→    () =>
   110→      Array.from({ length: 32 }, (_, i) => ({
   111→        id: i,
   112→        left: Math.random() * 100,
   113→        delay: Math.random() * 1.2,
   114→        duration: 2.4 + Math.random() * 1.6,
   115→        size: 6 + Math.random() * 6,
   116→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   117→        drift: Math.round((Math.random() - 0.5) * 120),
   118→        round: Math.random() > 0.5,
   119→      })),
   120→    [],
   121→  );
   122→  return (
   123→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   124→      {pieces.map((p) => (
   125→        <span
   126→          key={p.id}
   127→          className="confetti-piece absolute top-[-14px] block"
   128→          style={{
   129→            left: `${p.left}%`,
   130→            width: p.size,
   131→            height: p.size * (p.round ? 1 : 0.45),
   132→            background: p.color,
   133→            borderRadius: p.round ? "50%" : 2,
   134→            animationDelay: `${p.delay}s`,
   135→            animationDuration: `${p.duration}s`,
   136→            "--confetti-drift": `${p.drift}px`,
   137→          } as React.CSSProperties}
   138→        />
   139→      ))}
   140→    </div>
   141→  );
   142→}

   449→  useEffect(() => {

   451→    function onKey(e: KeyboardEvent) {
   452→      const target = e.target as HTMLElement | null;
   453→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA")) return;
   454→      if (e.key === "Enter" && selected !== null) {
   455→        e.preventDefault();
   456→        nextQuestion();
   457→        return;
   458→      }
   459→      if (selected !== null) return;
   460→      const fa = "۱۲۳۴".indexOf(e.key);
   461→      const en = "1234".indexOf(e.key);
   462→      const digit = fa >= 0 ? fa : en;
   463→      if (digit >= 0 && digit < (current?.options.length ?? 0)) {
   464→        e.preventDefault();
   465→        chooseOption(digit);
   466→      }
   467→    }
   468→    window.addEventListener("keydown", onKey);
   469→    return () => window.removeEventListener("keydown", onKey);
   470→  }, [phase, selected, index, questions]);

   481→    const detail = questions.map((q, i) => ({ id: q.id, chosen: answers[i], answer: q.answer }));
   482→    fetch("/api/attempts", {
   483→      method: "POST",
   484→      headers: { "Content-Type": "application/json" },
   485→      body: JSON.stringify({
   486→        category:
   487→          mode === "review"
   488→            ? "review"
   489→            : mode === "daily"
   490→              ? "daily"
   491→              : mode === "weekly"
   492→                ? "weekly"
   493→                : mode === "starred"
   494→                  ? "starred"
   495→                  : mode === "combined"

   524→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   525→  async function exportMistakesCsv() {
   526→    if (!mistakeIds.length || exporting) return;
   527→    setExporting(true);
   528→    try {
   529→      const res = await fetch(`/api/questions?count=${mistakeIds.length}&ids=${mistakeIds.join(",")}`);
   530→      const data = await res.json();
   531→      const qs = (data.questions ?? []) as Question[];
   532→      if (!qs.length) {
   533→        toast.error("سؤالی برای خروجی یافت نشد.");
   534→        return;
   535→      }
   536→      const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   537→      const rows = [
   538→        ["ردیف", "متن سؤال", "پاسخ صحیح", "دسته", "مبحث", "سطح"].join(","),
   539→        ...qs.map((q, i) =>
   540→          [
   541→            i + 1,
   542→            esc(q.text),
   543→            esc(q.options[q.answer] ?? ""),
   544→            q.category === "grammar" ? "گرامر" : "واژگان",
   545→            esc(q.scope),
   546→            esc(q.level),
   547→          ].join(","),
   548→        ),
   549→      ];
   550→      const blob = new Blob(["\uFEFF" + rows.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   551→      const url = URL.createObjectURL(blob);
   552→      const a = document.createElement("a");
   553→      a.href = url;
   554→      a.download = "vision-one-mistakes.csv";
   555→      document.body.appendChild(a);
   556→      a.click();
   557→      a.remove();
   558→      URL.revokeObjectURL(url);
   559→      toast.success(`فایل CSV با ${qs.length.toLocaleString("fa-IR")} سؤال دانلود شد.`);
   560→    } catch {
   561→      toast.error("خطا در ساخت فایل CSV");
   562→    } finally {
   563→      setExporting(false);
   564→    }
   565→  }
