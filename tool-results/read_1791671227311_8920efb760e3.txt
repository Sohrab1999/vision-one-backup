   138→/** واژه هدف بدون پرانتز توضیح — «order (n)» → «order» */
   139→function toBase(word: string): string {
   140→  return word.replace(/\s*\([^)]*\)\s*/g, " ").trim();
   141→}
   142→
   143→function escapeRe(s: string): string {
   144→  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   145→}
   146→
   147→function wordCount(s: string): number {
   148→  return s.trim().split(/\s+/).length;
   149→}

   151→/** همه جمله‌های کوتاه کتاب: متن‌ها (پاراگراف‌ها) + مکالمه‌ها (خط‌ها) */
   152→function collectSentences(data: BookData): SourceSentence[] {
   153→  const out: SourceSentence[] = [];
   154→  for (const r of data.readings) {
   155→    for (const p of r.paragraphs) {
   156→      for (const s of p.en.split(/(?<=[.!?])\s+/)) {
   157→        const t = s.trim();
   158→        if (t) out.push({ en: t, fa: p.fa });
   159→      }
   160→    }
   161→  }
   162→  for (const c of data.conversations) {
   163→    for (const line of c.lines) {
   164→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   165→        const t = s.trim();
   166→        if (t) out.push({ en: t, fa: line.fa });
   167→      }
   168→    }
   169→  }
   170→  return out;
   171→}

   173→/**
   174→ * هسته ساخت پازل — تابع خالص: برای هر واژه (بدون تکرار) اولین جمله مناسب کتاب را
   175→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   176→ */
   177→function buildPool(data: BookData, lesson: number): ClozeItem[] {
   178→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   179→  const sentences = collectSentences(data);
   180→
   181→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   182→  const seenBase = new Set<string>();
   183→  const words: { base: string; entry: VocabEntry }[] = [];
   184→  for (const entry of vocabPool) {
   185→    const base = toBase(entry.word);
   186→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   187→    seenBase.add(base);
   188→    words.push({ base, entry });
   189→  }
   190→
   191→  const items: ClozeItem[] = [];
   192→  for (const { base, entry } of words) {
   193→    if (items.length >= MAX_ITEMS) break;
   194→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   195→    const found = sentences.find((s) => {
   196→      if (s.en.length < 30 || s.en.length > 160) return false;
   197→      const wc = wordCount(s.en);
   198→      if (wc < 6 || wc > 35) return false;
   199→      return re.test(s.en);
   200→    });
   201→    if (!found) continue;
   202→
   203→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   204→    const distractorPool = vocabPool.filter((w) => {
   205→      const b = toBase(w.word);
   206→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   207→    });
   208→    const distractors = pickUnique(distractorPool, 2, (w) => `${toBase(w.word)}|${w.meaning}`);
   209→    if (distractors.length < 2) continue;
   210→
   211→    items.push({
   212→      id: `${lesson}:${base}`,
   213→      sentence: found.en.replace(re, BLANK),
   214→      fa: found.fa,
   215→      answer: base,
   216→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   217→      meaning: entry.meaning,
   218→      lessonLabel: entry.lessonLabel,
   219→    });
   220→  }
   221→  return shuffle(items);
   222→}

   224→// ─── اسنپ‌شات پایدار رکورد (JSON.parse هر بار آبجکت تازه می‌سازد — باید کَش شود) ───
   225→
   226→let bestCacheKey = "";
   227→let bestCacheValue: BestRecord | null = null;
   228→
   229→function readBest(bookId: string, scope: string): BestRecord | null {
   230→  let raw: string | null = null;
   231→  try {
   232→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   233→  } catch {
   234→    return null;
   235→  }
   236→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   237→  if (cacheKey === bestCacheKey) return bestCacheValue;
   238→  let value: BestRecord | null = null;
   239→  if (raw) {
   240→    try {
   241→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   242→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   243→        value = { score: parsed.score, total: parsed.total };
   244→      }
   245→    } catch {
   246→      value = null;
   247→    }
   248→  }
   249→  bestCacheKey = cacheKey;
   250→  bestCacheValue = value;
   251→  return value;
   252→}

   444→      const acc = Math.round((finalCorrect / Math.max(deck.length, 1)) * 100);

   454→      setPhase("done");
   455→    },
   456→    [bookId, deck, lesson, scope],
   457→  );
   458→
   459→  const answer = useCallback(
   460→    (opt: string, viaKey?: number) => {
   461→      if (busy.current || picked !== null || paused) return;
   462→      const item = deck[idx];

   467→      // فاز G2-b — اعلام aria: با کیبورد «گزینه N» هم ابتدا اعلام می‌شود (کار اختصاصی G2-b)

   486→        buzz("error");

   490→      advanceRef.current = setTimeout(() => {

   494→          finish(finalCorrect);
   495→        } else {
   496→          setIdx(nextIdx);

   500→    [picked, deck, idx, correctCount, speed, finish, paused],

   506→  // پنجرهٔ باز نگهبان خروج (سطح effect) و قفل busy/پاسخ‌داده‌شده (داخل answer).

   511→      if (!item) return;
   512→      const opt = item.options[n - 1];

   518→  useEffect(() => {

   538→    } else {
   539→      clearAdvance();
   540→      // دور بدون پیشرفت هم یک جلسه است؛ همین‌جا رهاشده ثبت می‌شود
   541→      sessionRef.current?.abandon("exit");
   542→      resume();
   543→      setPhase("setup");

   546→
   547→  // ── Setup ──
   548→  if (phase === "setup") {

   555→                <PencilLine size={22} />

   567→                مرور هوشمند می‌رود.
   568→              </p>
   569→            </div>
   570→
   571→            <div>
   572→              <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>

   581→                      disabled={count === 0}

   613→            <div>
   614→              <p className="mb-2 text-[13px] font-bold">چند پرسش؟</p>

   616→                {ROUND_OPTIONS.map((n) => (
   617→                  <button
   618→                    key={n}
   619→                    type="button"

   642→            </div>

   711→  const revealed = picked !== null;
   712→  const answered = idx + (revealed ? 1 : 0);

   724→              <PencilLine size={11} />
   725→              {faNum(idx + 1)} / {faNum(deck.length)}
   726→            </Badge>
   727→            <Badge variant="secondary" className="gap-1 text-[11px]">
   728→              <Timer size={11} />
   729→              {fmtTime(secs)}
   730→            </Badge>
   731→            <Badge variant="secondary" className="gap-1 text-[11px]">

   738→        progress={
   739→          <div className="h-1.5 overflow-hidden rounded-full bg-muted">
   740→            <div
   741→              className="h-full rounded-full bg-gradient-to-l from-primary to-emerald-400 transition-all duration-500"
   742→              style={{ width: `${pct}%` }}
   743→            />
   744→          </div>
   745→        }
   746→        paused={paused}
   747→        onPauseToggle={togglePause}
   748→        muted={muted}
   749→        onMuteToggle={() => setMuted(toggleSoundMuted())}
   750→        onHelp={handleHelp}
   751→        onExit={handleExitClick}
   752→      >
   753→        <div className="relative">
   754→          <div className={cn("space-y-4", paused && "pointer-events-none blur-sm")} aria-hidden={paused}>
   755→            {item && (
   756→              <div className="space-y-4 rounded-2xl border bg-card p-6">
   757→                <div>
   758→                  <Badge variant="outline" className="border-teal-500/40 text-[10px] text-teal-600 dark:text-teal-400">

   760→                  </Badge>
   761→                </div>
   762→
   763→                <p dir="ltr" className="font-en text-[15px] leading-8 sm:text-base">
   764→                  {splitAt >= 0 ? (
   765→                    <>
   766→                      {before}
   767→                      <span className="mx-1 inline-block min-w-16 rounded-md border-b-2 border-amber-500/70 bg-amber-500/10 px-2 text-center font-black text-amber-600 dark:text-amber-400">
   768→                        {revealed ? item.answer : "؟"}
   769→                      </span>
   770→                      {after}
   771→                    </>
   772→                  ) : (
   773→                    item.sentence
   774→                  )}
   775→                </p>
   776→
   777→                <p className="border-r-2 border-primary/30 pr-3 text-[12px] leading-6 text-muted-foreground">
   778→                  ترجمه: {item.fa}
   779→                </p>
   780→
   781→                <div className="grid gap-2 sm:grid-cols-3">
   782→                  {item.options.map((opt, i) => {
   783→                    const isAnswer = opt === item.answer;
   784→                    const isPicked = picked === opt;
   785→                    return (

   788→                        type="button"
   789→                        dir="ltr"

   792→                        aria-keyshortcuts={String(i + 1)}
   793→                        className={cn(

   827→          {paused && (
