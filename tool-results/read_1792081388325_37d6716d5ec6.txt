    26→// ── نگاشت‌های جست‌وجو — از دادهٔ کتاب فعال ساخته می‌شوند ──
    27→interface LookupMaps {
    28→  dict: Map<string, DictEntry>;
    29→  vocab: Map<string, VocabWord>;
    30→  phrases: Set<string>;
    31→}
    32→
    33→function normalize(raw: string): string {
    34→  return raw
    35→    .replace(/[’‘]/g, "'")
    36→    .replace(/[^A-Za-z' \-]/g, "")
    37→    .trim()
    38→    .toLowerCase();
    39→}
    40→
    41→function buildMaps(dictionary: DictEntry[], vocabList: VocabEntry[]): LookupMaps {
    42→  const dict = new Map<string, DictEntry>();
    43→  const vocab = new Map<string, VocabWord>();
    44→  const phrases = new Set<string>();
    45→  for (const d of dictionary) {
    46→    const key = normalize(d.word);
    47→    if (key && !dict.has(key)) dict.set(key, d);
    48→    if (key.includes(" ")) phrases.add(key);
    49→  }
    50→  for (const v of vocabList) {
    51→    const key = normalize(v.word);
    52→    if (key && !vocab.has(key)) vocab.set(key, v);
    53→    if (key.includes(" ")) phrases.add(key);
    54→  }
    55→  return { dict, vocab, phrases };
    56→}

    59→let mapsCache: { dictionary: DictEntry[]; vocab: VocabEntry[]; maps: LookupMaps } | null = null;

    61→function getLookupMaps(dictionary: DictEntry[], vocab: VocabEntry[]): LookupMaps {
    62→  if (mapsCache && mapsCache.dictionary === dictionary && mapsCache.vocab === vocab) {
    63→    return mapsCache.maps;
    64→  }
    65→  const maps = buildMaps(dictionary, vocab);
    66→  mapsCache = { dictionary, vocab, maps };
    67→  return maps;
    68→}

   121→/** با رویداد باز شدن پاپ‌آپ صدا زده می‌شود — مهمان fetch نمی‌گیرد، فقط یک‌بار (یا پس از TTL) */
   122→function ensureNotes(bookId: string, authenticated: boolean) {
   123→  if (!bookId) return;
   124→  const st = notesCache.get(bookId);
   125→  if (!authenticated) {
   126→    if (!st || st.status !== "guest") {
   127→      notesCache.set(bookId, { rows: new Map(), status: "guest", fetchedAt: Date.now() });
   128→      notifyNotes();
   129→    }
   130→    return;
   131→  }
   132→  if (st && (st.status === "ready" || st.status === "loading") && Date.now() - st.fetchedAt < NOTES_TTL_MS) return;
   133→  notesCache.set(bookId, { rows: st?.rows ?? new Map(), status: "loading", fetchedAt: Date.now() });
   134→  notifyNotes();
   135→  void fetchNotes(bookId);
   136→}

   138→async function fetchNotes(bookId: string) {
   139→  const put = (rows: Map<string, NoteRow>, status: BookNotesState["status"]) => {
   140→    notesCache.set(bookId, { rows, status, fetchedAt: Date.now() });
   141→    notifyNotes();
   142→  };
   143→  try {
   144→    const res = await fetch(`/api/notes?book=${encodeURIComponent(bookId)}`, { cache: "no-store" });
   145→    const json = (await res.json().catch(() => null)) as { notes?: NoteRow[] } | null;
   146→    if (res.status === 401) {
   147→      put(new Map(), "guest");
   148→      return;
   149→    }
   150→    if (!res.ok || !json) {
   151→      // ۴۰۴ (سرویس هنوز آماده نیست) یا خطای سرور — محترمانه و بدون کرش
   152→      put(new Map(), "error");
   153→      toast.error("یادداشت‌ها بارگیری نشد — کمی بعد دوباره امتحان کن");
   154→      return;
   155→    }
   156→    const rows = new Map<string, NoteRow>();
   157→    for (const r of Array.isArray(json.notes) ? json.notes : []) {
   158→      if (rows.size >= NOTES_ROW_CAP) break;
   159→      if (r && typeof r.id === "string" && typeof r.word === "string" && typeof r.note === "string") {
   160→        rows.set(r.word, r);
   161→      }
   162→    }
   163→    put(rows, "ready");
   164→  } catch {
   165→    put(new Map(), "error");
   166→  }
   167→}

   169→interface LookupHit {
   170→  word: string;
   171→  dict?: DictEntry;
   172→  vocab?: VocabWord;
   173→  /** کالوکیشن‌های مرتبط — وقتی واژهٔ تکی در بانک نبود (کتاب‌های عبارت‌محور مثل کالوکیشن‌ها) */
   174→  related?: VocabWord[];
   175→}
   176→
   177→function lookup(maps: LookupMaps, norm: string): LookupHit {
   178→  const candidates = [norm];
   179→  // ساده‌سازی صرفی: جمع، فعل ساده، «'s»
   180→  if (norm.endsWith("'s")) candidates.push(norm.slice(0, -2));
   181→  if (norm.endsWith("ies")) candidates.push(norm.slice(0, -3) + "y");
   182→  if (norm.endsWith("es")) candidates.push(norm.slice(0, -2));
   183→  if (norm.endsWith("s") && !norm.endsWith("ss")) candidates.push(norm.slice(0, -1));
   184→  if (norm.endsWith("ed")) candidates.push(norm.slice(0, -2), norm.slice(0, -1));
   185→  if (norm.endsWith("ing")) candidates.push(norm.slice(0, -3), norm.slice(0, -3) + "e");
   186→  for (const c of candidates) {
   187→    if (!c) continue;
   188→    const hit = maps.dict.get(c);
   189→    const vocab = maps.vocab.get(c);
   190→    if (hit || vocab) return { word: c, dict: hit, vocab };
   191→  }
   192→  // فال‌بک عبارت‌محور: کالوکیشن‌های کتاب که همین واژه را دارند (سقف ۵) —
   193→  // برای کتاب‌هایی مثل «کالوکیشن‌ها» که بانک واژگانشان عبارت‌های چندواژه‌ای است
   194→  if (norm.length >= 3) {
   195→    const re = new RegExp(`(^|\\s|[^a-z])${norm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")}($|\\s|[^a-z])`);
   196→    const related: VocabWord[] = [];
   197→    for (const v of maps.vocab.values()) {
   198→      if (related.length >= 5) break;
   199→      if (re.test(normalize(v.word))) related.push(v);
   200→    }
   201→    if (related.length > 0) return { word: norm, related };
   202→  }
   203→  return { word: norm };
   204→}

   206→// ── توکنایز ایندکسمحور: واژه‌ها از regex + عبارت‌های دوتایی مثل "based on" ──
   207→interface WordSpan {
   208→  raw: string;
   209→  norm: string;
   210→  start: number;
   211→  end: number;
   212→}
   213→
   214→const WORD_RE = /[A-Za-z]+(?:['’-][A-Za-z]+)*/g;
   215→
   216→function extractWords(text: string, maps: LookupMaps): WordSpan[] {
   217→  const words: WordSpan[] = [];
   218→  for (const m of text.matchAll(WORD_RE)) {
   219→    const start = m.index ?? 0;
   220→    const raw = m[0];
   221→    words.push({ raw, norm: normalize(raw), start, end: start + raw.length });
   222→  }
   223→  // ادغام عبارت دوتایی شناخته‌شده (فقط وقتی بین دو واژه فقط فاصله باشد)
   224→  const merged: WordSpan[] = [];
   225→  let i = 0;
   226→  while (i < words.length) {
   227→    const a = words[i];
   228→    const b = words[i + 1];
   229→    if (
   230→      b &&
   231→      maps.phrases.has(`${a.norm} ${b.norm}`) &&
   232→      /^\s+$/.test(text.slice(a.end, b.start))
   233→    ) {
   234→      merged.push({
   235→        raw: text.slice(a.start, b.end),
   236→        norm: `${a.norm} ${b.norm}`,
   237→        start: a.start,
   238→        end: b.end,
   239→      });
   240→      i += 2;
   241→    } else {
   242→      merged.push(a);
   243→      i += 1;
   244→    }
   245→  }
   246→  return merged;
   247→}

   249→interface Segment {
   250→  kind: "text" | "word";
   251→  raw: string;
   252→  norm?: string;
   253→}
   254→
   255→function segmentText(text: string, maps: LookupMaps): Segment[] {
   256→  const words = extractWords(text, maps);
   257→  const segs: Segment[] = [];
   258→  let pos = 0;
   259→  for (const w of words) {
   260→    if (w.start > pos) segs.push({ kind: "text", raw: text.slice(pos, w.start) });
   261→    segs.push({ kind: "word", raw: w.raw, norm: w.norm });
   262→    pos = w.end;
   263→  }
   264→  if (pos < text.length) segs.push({ kind: "text", raw: text.slice(pos) });
   265→  return segs;
   266→}

   379→function LookupPopover({ sel, onClose }: { sel: { hit: LookupHit; x: number; y: number }; onClose: () => void }) {
   380→  const ref = useRef<HTMLDivElement>(null);
   381→  const { bookId } = useBook();
   382→  const starred = useStarredWords();
   383→  const srs = useSRSData();
   384→  const { learned, toggle: toggleLearnedWord } = useLearnedWords();
   385→  const { hit } = sel;
   386→
   387→  // تصویر حافظه (Dual Coding) — کاملاً event-driven؛ بدون effect جدید
   388→  const [memImg, setMemImg] = useState<{ word: string; data: string; loading: boolean }>({
   389→    word: "",
   390→    data: "",
   391→    loading: false,
   392→  });
   393→
   394→  // بستن با کلیک بیرون، Escape و اسکرول
   395→  useEffect(() => {
   396→    function onDown(e: MouseEvent) {
   397→      if (ref.current && !ref.current.contains(e.target as Node)) onClose();
   398→    }
   399→    function onKey(e: KeyboardEvent) {
   400→      if (e.key === "Escape") onClose();
   401→    }
   402→    window.addEventListener("mousedown", onDown);
   403→    window.addEventListener("keydown", onKey);
   404→    window.addEventListener("scroll", onClose, { once: true, passive: true });
   405→    return () => {
   406→      window.removeEventListener("mousedown", onDown);
   407→      window.removeEventListener("keydown", onKey);
   408→      window.removeEventListener("scroll", onClose);
   409→    };
   410→  }, [onClose]);
   411→
   412→  // موقعیت clamp شده به دید
   413→  const W = 300;
   414→  const vw = typeof window !== "undefined" ? window.innerWidth : 400;
   415→  const vh = typeof window !== "undefined" ? window.innerHeight : 600;
   416→  const left = Math.min(Math.max(sel.x - W / 2, 8), vw - W - 8);
   417→  const top = Math.min(sel.y + 8, vh - 200);
   418→
   419→  const isStarred = starred.has(hit.word);
   420→  const srsItem = srs[hit.word];
   421→  const srsBox = srsItem?.box ?? 0;
   422→  const isLearned = learned.has(hit.word);
   423→
   424→  // تولید تصویر حافظه برای واژه — ابتدا کش ماژولی، سپس POST به /api/word-image
   425→  async function loadMemoryImage() {
   426→    const word = hit.word;
   427→    if (!word || memImg.loading) return;
   428→    const hitCached = memoryImageCache.get(word);
   429→    if (hitCached) {
   430→      setMemImg({ word, data: hitCached, loading: false });
   431→      return;
   432→    }
   433→    setMemImg({ word, data: "", loading: true });
   434→    try {
   435→      const guest = readUserName();
   436→      const res = await fetch("/api/word-image", {
   437→        method: "POST",
   438→        headers: {
   439→          "Content-Type": "application/json",
   440→          ...(guest ? { "x-visionone-name": guest } : {}),
   441→        },
   442→        body: JSON.stringify({ word, meaning: hit.vocab?.meaning ?? "" }),
   443→      });
   444→      const payload: { image?: string; error?: string } = await res.json();
   445→      if (!res.ok || !payload.image) {
   446→        toast.error(payload.error ?? "تصویر حافظه ساخته نشد — دوباره تلاش کن");
   447→        setMemImg({ word: "", data: "", loading: false });
   448→        return;
   449→      }
   450→      memoryImageCache.set(word, payload.image);

   665→// ── بخش «📝 یادداشت من» — با key={word} مونت می‌شود تا هر واژه state تازه بگیرد ──
   666→// داده از کش ماژولی خوانده می‌شود (بدون fetch در هر باز شدن) و با notifyNotes تازه می‌شود.
   667→
   668→function WordNoteSection({ bookId, word }: { bookId: string; word: string }) {
   669→  const { status: sessionStatus } = useSession();
   670→  useSyncExternalStore(subscribeNotes, getNotesVersion, getNotesVersion);
   671→
   672→  const state = notesCache.get(bookId);
   673→  const row = state?.status === "ready" ? (state.rows.get(word) ?? undefined) : undefined;
   674→  const [draft, setDraft] = useState<string | null>(null);
   675→  const [busy, setBusy] = useState(false);
   676→
   677→  const value = (draft ?? row?.note ?? "").slice(0, NOTE_MAX_LEN);
   678→  const guest = sessionStatus !== "authenticated" || state?.status === "guest";
   679→
   680→  async function saveNote() {
   681→    const note = value.trim();
   682→    if (!note) {
   683→      toast.error("یادداشت خالی است — چیزی بنویس تا ذخیره شود");
   684→      return;
   685→    }
   686→    setBusy(true);
   687→    try {
   688→      const res = await fetch("/api/notes", {
   689→        method: "PUT",
   690→        headers: { "Content-Type": "application/json" },
   691→        body: JSON.stringify({ book: bookId, word, note }),
   692→      });
   693→      const json = (await res.json().catch(() => null)) as { note?: NoteRow; error?: string } | null;
   694→      if (res.status === 401) {
   695→        notesCache.set(bookId, { rows: new Map(), status: "guest", fetchedAt: Date.now() });
   696→        notifyNotes();
   697→        return;
   698→      }
   699→      if (!res.ok || !json?.note) {
   700→        toast.error(json?.error ?? "ذخیرهٔ یادداشت نشد — دوباره تلاش کن");
   701→        return;
   702→      }
   703→      const st = notesCache.get(bookId);
   704→      if (st) {
   705→        const rows = new Map(st.rows);
   706→        rows.set(word, json.note);
   707→        notesCache.set(bookId, { ...st, rows, status: "ready", fetchedAt: Date.now() });
   708→        notifyNotes();
   709→      }
   710→      setDraft(null);
   711→      toast.success("یادداشت ذخیره شد ✓");
   712→    } catch {
   713→      toast.error("خطا در ارتباط با سرور — اتصال اینترنتت را بررسی کن");
   714→    } finally {
   715→      setBusy(false);
   716→    }
   717→  }
   718→
   719→  async function removeNote() {
   720→    if (!row || busy) return;
   721→    setBusy(true);
   722→    try {
   723→      const res = await fetch(`/api/notes?id=${encodeURIComponent(row.id)}`, { method: "DELETE" });
   724→      if (res.status === 401) {
   725→        notesCache.set(bookId, { rows: new Map(), status: "guest", fetchedAt: Date.now() });
   726→        notifyNotes();
   727→        return;
   728→      }
   729→      if (!res.ok) {
   730→        toast.error("حذف یادداشت نشد — دوباره تلاش کن");
   731→        return;
   732→      }
   733→      const st = notesCache.get(bookId);
   734→      if (st) {
   735→        const rows = new Map(st.rows);
   736→        rows.delete(word);
   737→        notesCache.set(bookId, { ...st, rows, fetchedAt: Date.now() });
   738→        notifyNotes();
   739→      }
   740→      setDraft(null);
   741→      toast.success("یادداشت حذف شد");
   742→    } catch {
   743→      toast.error("خطا در ارتباط با سرور — اتصال اینترنتت را بررسی کن");
   744→    } finally {
   745→      setBusy(false);
   746→    }
   747→  }
   748→
   749→  // مهمان (بدون سشن یا 401) — کارت ورود کوچک، بدون خطا
   750→  if (guest) {
   751→    return (
   752→      <div>
   753→        <p className="text-[11.5px] font-black">📝 یادداشت من</p>
   754→        <div className="mt-2 rounded-xl border border-dashed border-primary/40 bg-primary/5 p-3 text-center">
   755→          <p className="text-[11.5px] leading-6 text-muted-foreground">
   756→            برای همگام‌شدن یادداشت با حسابت وارد شو 🔑
   757→          </p>
   758→          <button
   759→            type="button"
   760→            onClick={openProfileLogin}
   761→            className="mx-auto mt-2 flex min-h-9 items-center justify-center gap-1.5 rounded-full bg-primary px-4 text-[11.5px] font-black text-primary-foreground shadow-sm transition-all hover:scale-105 active:scale-95"
   762→          >
   763→            <LogIn size={13} />
   764→            ورود
   765→          </button>
   766→        </div>
   767→      </div>
   768→    );
   769→  }
   770→
   771→  return (
   772→    <div>
   773→      <div className="flex items-center justify-between gap-2">
   774→        <p className="text-[11.5px] font-black">📝 یادداشت من</p>
   775→        <span dir="ltr" className="text-[9.5px] font-bold text-muted-foreground">
   776→          {value.length.toLocaleString("fa-IR")} / {NOTE_MAX_LEN.toLocaleString("fa-IR")}
   777→        </span>
   778→      </div>
   779→      {state?.status === "loading" && (
   780→        <p className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   781→          <Loader2 size={10} className="animate-spin" />
   782→          در حال همگام‌سازی یادداشت‌ها…
   783→        </p>
   784→      )}
   785→      <Textarea
   786→        value={value}
   787→        onChange={(e) => setDraft(e.target.value.slice(0, NOTE_MAX_LEN))}
   788→        placeholder="نکتهٔ خودت را دربارهٔ این واژه بنویس… (مثلاً ترفند حفظش)"
   789→        maxLength={NOTE_MAX_LEN}
   790→        aria-label="یادداشت شخصی واژه"
   791→        className="mt-2 min-h-[64px] resize-none bg-muted/30 text-[12px] leading-6"
   792→      />
   793→      <div className="mt-2 flex items-center gap-2">
   794→        <button
   795→          type="button"
   796→          onClick={saveNote}
   797→          disabled={busy}
   798→          className="flex min-h-9 items-center justify-center gap-1.5 rounded-full bg-primary px-4 text-[11.5px] font-black text-primary-foreground shadow-sm transition-all hover:scale-105 active:scale-95 disabled:pointer-events-none disabled:opacity-60"
   799→        >
   800→          {busy ? <Loader2 size={13} className="animate-spin" /> : <Save size={13} />}
   801→          ذخیره
   802→        </button>
   803→        {row && (
   804→          <button
   805→            type="button"
   806→            onClick={removeNote}
   807→            disabled={busy}
   808→            aria-label="حذف یادداشت"
   809→            title="حذف یادداشت"
   810→            className="flex min-h-9 items-center justify-center gap-1.5 rounded-full border border-destructive/40 px-3.5 text-[11.5px] font-black text-destructive transition-all hover:bg-destructive/10 active:scale-95 disabled:pointer-events-none disabled:opacity-60"
   811→          >
   812→            <Trash2 size={13} />
   813→            حذف
   814→          </button>
   815→        )}
   816→      </div>
   817→    </div>
   818→  );
   819→}
