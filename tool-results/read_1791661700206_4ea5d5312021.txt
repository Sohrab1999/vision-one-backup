   286→   286→        at += 1;

   288→   288→        setPositions([...posRef.current]);
   289→   289→        later(step, STEP_MS);
   290→   290→      } else {
   291→   291→        land(p, d);
   292→   292→      }
   293→   293→    }
   294→   294→    step();

   306→   306→        setPositions([...posRef.current]);
   307→   307→        afterJump(p, d);
   308→   308→      }, 550);
   309→   309→    } else if (ladderTo !== undefined) {
   310→   310→      toast.success(`پله! رفتی به خانهٔ ${faNum(ladderTo)}`);
   311→   311→      later(() => {
   312→   312→        posRef.current[p] = ladderTo;
   313→   313→        setPositions([...posRef.current]);
   314→   314→        afterJump(p, d);
   315→   315→      }, 550);
   316→   316→    } else {
   317→   317→      afterJump(p, d);
   318→   318→    }
   319→   319→  }

   331→   331→      beginTurn((p + 1) % playersRef.current.length);
   332→   332→    }
   333→   333→  }

   336→   336→  function botAnswer(p: number): void {

   344→   344→      later(() => proceedAfterCorrect(p), 600);
   345→   345→    } else {
   346→   346→      toast.error(`${playersRef.current[p]?.name ?? "ربات"} بند شد!`);
   347→   347→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   348→   348→    }
   349→   349→  }

   359→   359→    if (ok) {
   360→   360→      later(() => proceedAfterCorrect(p), 350);
   361→   361→    } else {
   362→   362→      toast.error("بند شدی!");
   363→   363→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   364→   364→    }
