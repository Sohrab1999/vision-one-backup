   230→function fmtDuration(sec: number): string {
   231→  if (!sec) return "—";
   232→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   233→  const m = Math.floor(sec / 60);
   234→  const s = sec % 60;
   235→  if (m === 0) return `${fa(s)} ثانیه`;
   236→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   237→}
   238→
   239→function fmtDate(iso: string): string {
   240→  try {
   241→    return new Date(iso).toLocaleDateString("fa-IR", { month: "short", day: "numeric" });
   242→  } catch {
   243→    return "";
   244→  }
   245→}
