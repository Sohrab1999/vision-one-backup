   279→   279→   279→function scheduleBotAnswer(room: Room): void {
   280→   280→   280→  if (room.botTimer) {

   285→   285→   285→  const bot = [...room.players.values()].find((p) => p.isBot);
   286→   286→   286→  if (!bot || bot.botSkill === undefined) return;
   287→   287→   287→  const think = BOT_THINK_MIN_MS + Math.floor(Math.random() * (BOT_THINK_MAX_MS - BOT_THINK_MIN_MS + 1));
   288→   288→   288→  room.botTimer = setTimeout(() => {
   289→   289→   289→    room.botTimer = null;
   290→   290→   290→    botAnswer(room, bot);
   291→   291→   291→  }, think);
   292→   292→   292→}
   293→   293→   293→
   294→   294→   294→/** پاسخ ربات داخلی به سؤال جاری — با احتمال skill درست، وگرنه یک گزینهٔ غلط تصادفی */
   295→   295→   295→function botAnswer(room: Room, bot: Player): void {
   296→   296→   296→  if (room.state !== "playing" || room.phase !== "question" || !room.current) return;
   297→   297→   297→  if (room.answers.has(bot.id)) return;
   298→   298→   298→  const q = room.pack[room.currentIndex];
   299→   299→   299→  const skill = bot.botSkill ?? BOT_SKILL_BASE;
   300→   300→   300→  const correct = Math.random() < skill;
   301→   301→   301→  const wrongs = q.options.map((_, i) => i).filter((i) => i !== q.answer);
   302→   302→   302→  const choice = correct ? q.answer : (wrongs[Math.floor(Math.random() * wrongs.length)] ?? q.answer);
   303→   303→   303→  registerAnswer(room, bot, choice);
   304→   304→   304→}
   305→   305→   305→
   306→   306→   306→/**
   307→   307→   307→ * فاز G5-03 — ثبت پاسخ یک بازیکن (انسان یا ربات) — منطق امتیازدهی مشترک؛
   308→   308→   308→ * بدنه از هندلر `answer` استخراج شد تا ربات دقیقاً همان مسیر را طی کند
   309→   309→   309→ * (score → بستن زودهنگام اگر همه پاسخ دادند → reveal).
   310→   310→   310→ */
   311→   311→   311→function registerAnswer(room: Room, p: Player, choice: number): void {
   312→   312→   312→  if (room.phase !== "question" || !room.current) return;
   313→   313→   313→  const q = room.pack[room.currentIndex];
   314→   314→   314→  const remaining = Math.max(0, room.current.endsAt - now());
   315→   315→   315→  const bonus = Math.round((MAX_BONUS * remaining) / QUESTION_MS);
   316→   316→   316→  const correct = choice === q.answer;
   317→   317→   317→  const delta = correct ? BASE_SCORE + bonus : 0;
   318→   318→   318→  p.score += delta;
   319→   319→   319→  if (correct) p.correct += 1;
   320→   320→   320→  p.lastDelta = delta;
   321→   321→   321→  p.lastCorrect = correct;
   322→   322→   322→  room.answers.set(p.id, { choice, correct, delta });
   323→   323→   323→  touch(room);
   324→   324→   324→  emitToAll(room, "score", { questionIndex: room.currentIndex, scores: scoresOf(room) });
   325→   325→   325→
   326→   326→   326→  // همهٔ بازیکنان آنلاین پاسخ دادند؟ بستن زودهنگام سؤال
   327→   327→   327→  const connected = [...room.players.values()].filter((pl) => pl.connected);
   328→   328→   328→  if (connected.length > 0 && room.answers.size > 0 && room.answers.size >= connected.length) {
   329→   329→   329→    closeQuestion(room);
   330→   330→   330→  }
   331→   331→   331→}

   333→   333→   333→function closeQuestion(room: Room): void {
   334→   334→   334→  if (room.phase !== "question") return;
   335→   335→   335→  if (room.questionTimer) clearTimeout(room.questionTimer);
   336→   336→   336→  room.questionTimer = null;
   337→   337→   337→  room.phase = "reveal";
   338→   338→   338→  const q = room.pack[room.currentIndex];
   339→   339→   339→  room.revealAnswer = q.answer;
   340→   340→   340→  touch(room);
   341→   341→   341→  // طبق قرارداد: بعد از هر پاسخ یا انقضا رویداد score می‌رود
   342→   342→   342→  emitToAll(room, "score", { questionIndex: room.currentIndex, scores: scoresOf(room) });
   343→   343→   343→  emitToAll(room, "reveal", { index: room.currentIndex, answer: q.answer, scores: scoresOf(room) });
   344→   344→   344→  room.revealTimer = setTimeout(() => {
   345→   345→   345→    room.currentIndex += 1;
   346→   346→   346→    startQuestion(room);
   347→   347→   347→  }, REVEAL_MS);
   348→   348→   348→}

   350→   350→   350→function finishGame(room: Room): void {
   351→   351→   351→  clearTimers(room);
   352→   352→   352→  room.state = "ended";
   353→   353→   353→  room.phase = "reveal";
   354→   354→   354→  room.current = null;
   355→   355→   355→  room.revealAnswer = null;
   356→   356→   356→  touch(room);
   357→   357→   357→  const duration = room.startedAt ? Math.max(1, Math.round((now() - room.startedAt) / 1000)) : 0;
   358→   358→   358→  const finalScores = [...room.players.values()]
   359→   359→   359→    .map((p) => ({ id: p.id, name: p.name, score: p.score, correct: p.correct }))
   360→   360→   360→    .sort((a, b) => b.score - a.score || b.correct - a.correct);
   361→   361→   361→  emitToAll(room, "finished", { finalScores, packSummary: room.pack, duration });
   362→   362→   362→  console.log(`[duel] room ${room.code} finished (${duration}s)`);
   363→   363→   363→}

   367→   367→   367→function armGrace(room: Room): void {
   368→   368→   368→  if (room.graceTimer) return;
   369→   369→   369→  const earliest = Math.min(
   370→   370→   370→    ...[...room.players.values()].filter((p) => !p.connected && p.disconnectedAt).map((p) => p.disconnectedAt!),
   371→   371→   371→  );
   372→   372→   372→  if (!Number.isFinite(earliest)) return;
   373→   373→   373→  const wait = Math.max(1_000, earliest + GRACE_MS - now());
   374→   374→   374→  room.graceTimer = setTimeout(() => {
   375→   375→   375→    room.graceTimer = null;
   376→   376→   376→    const t = now();
   377→   377→   377→    let removed = false;
   378→   378→   378→    for (const p of [...room.players.values()]) {
   379→   379→   379→      if (!p.connected && p.disconnectedAt && t - p.disconnectedAt >= GRACE_MS - 250) {
   380→   380→   380→        room.players.delete(p.id);
   381→   381→   381→        removed = true;
   382→   382→   382→      }
   383→   383→   383→    }
   384→   384→   384→    if (room.players.size === 0 || allBots(room)) {
   385→   385→   385→      deleteRoom(room.code);
   386→   386→   386→      return;
   387→   387→   387→    }
   388→   388→   388→    if (removed) broadcastRoom(room);
   389→   389→   389→    // اگر هنوز بازیکن قطعی دارد، تایمر بعدی
   390→   390→   390→    if ([...room.players.values()].some((p) => !p.connected)) armGrace(room);
   391→   391→   391→  }, wait);
   392→   392→   392→}
