     4→     4→ * بازی «دوئل دونفره» (Same-device Duel) — فاز ۴ نقشه راه ویژن‌وان.

     9→     9→ * - نیمهٔ بالا = بازیکن ۱ (emerald)، نیمهٔ پایین = بازیکن ۲ (amber)؛ وسط، نوار سؤال
    10→    10→ *   مشترک با Badge شمارهٔ دور و تایمر ۱۰ ثانیه‌ای هر دور (نوار width).
    11→    11→ * - هر نیمه ۴ دکمهٔ گزینه دارد؛ ترتیب گزینه‌ها بین دو نیمه shuffle متفاوت است تا
    12→    12→ *   رقیب نتواند از جای انگشت دیگری تقلب کند.
    13→    13→ * - هر بازیکن یک‌بار پاسخ می‌دهد؛ دور وقتی داوری می‌شود که هر دو پاسخ بدهند یا وقت
    14→    14→ *   تمام شود (بی‌پاسخ = غلط). هر دو درست → سریع‌تر برنده؛ یکی درست → همان؛ هیچ‌کدام
    15→    15→ *   → تساوی بدون امتیاز. نتیجهٔ دور ~۱.۲ ثانیه نمایش داده می‌شود و دور بعد می‌آید.
    16→    16→ * - واژه‌های غلط‌زدهٔ واژگانی (فقط وقتی بازیکن واقعاً گزینه‌ای زده باشد) با
    17→    17→ *   seedReviewFromQuiz — متن گزینهٔ درست = خود واژهٔ انگلیسی — به صف مرور برمی‌گردند.
    18→    18→ * - پایان دورها: صفحهٔ برنده + جدول امتیاز (برد دور / پاسخ درست / دقت) + «دوئل جدید» و «تنظیمات».

    20→    20→ *   correct = دورهای بردهٔ بازیکن ۱، detail {p1,p2,p1Name,p2Name,winner}) + saveBest
    21→    21→ *   {score} = بیشترین برد دور بازیکن ۱ با اسکوپ `${درس|all}:${تعداد دور}`.
    22→    22→ *

    24→    24→ * ریست می‌شود) و setTimeLeft فقط نمایش را همگام نگه می‌دارد. داوریِ هر دور دقیقاً

    26→    26→ * در انصراف/unmount کامل پاک می‌شود (الگوی react-hooks v6: صفر setState در بدنهٔ effect).

   272→   272→    const c2 = a2 !== null && a2 === nq.answer;
   273→   273→    let w: RoundWinner;
   274→   274→    if (c1 && c2) w = p1MsRef.current <= p2MsRef.current ? 0 : 1;
   275→   275→    else if (c1) w = 0;
   276→   276→    else if (c2) w = 1;
   277→   277→    else w = -1;
   278→   278→    // واژه‌های غلط‌زدهٔ واژگانی به صف مرور برمی‌گردند (گزینهٔ درست = خود واژهٔ انگلیسی)

   280→   280→      if (a1 !== null && !c1) seedReviewFromQuiz(nq.options[nq.answer]);
   281→   281→      if (a2 !== null && !c2) seedReviewFromQuiz(nq.options[nq.answer]);
   282→   282→    }
   283→   283→    if (w === 0) wins1Ref.current += 1;
   284→   284→    else if (w === 1) wins2Ref.current += 1;
   285→   285→    if (c1) corr1Ref.current += 1;
   286→   286→    if (c2) corr2Ref.current += 1;
   287→   287→    setWins1(wins1Ref.current);
   288→   288→    setWins2(wins2Ref.current);
   289→   289→    setCorrect1(corr1Ref.current);
   290→   290→    setCorrect2(corr2Ref.current);

   294→   294→      if (overRef.current) return;
   295→   295→      if (qiRef.current + 1 >= configRef.current.rounds) finish();
   296→   296→      else startRound(qiRef.current + 1);
   297→   297→    }, RESULT_MS);
   298→   298→  }

   302→   302→    if (overRef.current || judgedRef.current || !qRef.current) return;
   303→   303→    const order = player === 0 ? p1OrderRef.current : p2OrderRef.current;

   308→   308→      p1PickRef.current = orig;
   309→   309→      p1MsRef.current = Math.max(0, Date.now() - roundStartRef.current);
   310→   310→      setP1Pick(orig);
   311→   311→    } else {
   312→   312→      if (p2PickRef.current !== null) return;
   313→   313→      p2PickRef.current = orig;
   314→   314→      p2MsRef.current = Math.max(0, Date.now() - roundStartRef.current);
   315→   315→      setP2Pick(orig);
   316→   316→    }
   317→   317→    // هر دو پاسخ دادند؟ دور همان‌جا داوری می‌شود
   318→   318→    if (p1PickRef.current !== null && p2PickRef.current !== null) judge();
   319→   319→  }

   331→   331→    const spent = Math.max(1, Math.round((Date.now() - startAtRef.current) / 1000));
