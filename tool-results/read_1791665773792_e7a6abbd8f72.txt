   100→   100→
   101→   101→export function HangmanGame() {
   102→   102→  const data = useBookData();
   103→   103→  const { bookId } = useBook();
   104→   104→  // بازی‌ها فقط داخل کتاب فعال رندر می‌شوند (useBookData بالا گارد است) — bookId عملاً هرگز null نیست
   105→   105→  const activeBookId: string = bookId ?? "";
   106→   106→
   107→   107→  const [phase, setPhase] = useState<Phase>("setup");
   108→   108→  const [lesson, setLesson] = useState(ALL_LESSONS);
   109→   109→  const [count, setCount] = useState(5);
   110→   110→
   111→   111→  const [deck, setDeck] = useState<WordItem[]>([]);
   112→   112→  const [idx, setIdx] = useState(0);
   113→   113→  /** حروف حدس‌زده‌شدهٔ واژهٔ جاری — ترتیب درج حفظ می‌شود (بادکنک‌ها با همین ترتیب می‌ترکند) */
   114→   114→  const [used, setUsed] = useState<string[]>([]);
   115→   115→  /** جزئیات واژه‌های تمام‌شده — مبنای آمار، Attempt و فهرست باخت‌ها */
   116→   116→  const [detail, setDetail] = useState<DetailRow[]>([]);
   117→   117→  const [secs, setSecs] = useState(0);
   118→   118→  const [isRecord, setIsRecord] = useState(false);
   119→   119→  /** پیام باکس aria-live — اعلام برد/باخت/حرف غلط برای فناوری‌های کمکی */
   120→   120→  const [liveMsg, setLiveMsg] = useState("");
   121→   121→  /** وضعیت واژهٔ جاری: null = در حال بازی، true = برد، false = باخت */
   122→   122→  const [result, setResult] = useState<boolean | null>(null);
   123→   123→
   124→   124→  const { speed } = useVoiceSpeed();
   125→   125→  /** قفل double-guess — از حدس هم‌زمان (کلیک + کیبورد) و حین مکث برد/باخت جلوگیری می‌کند */
   126→   126→  const locked = useRef(false);
   127→   127→  /** نگاشت همگامِ used — گارد سنکرون مقابل keydownهای پشت‌سرهم قبل از رندر بعدی */
   128→   128→  const usedRef = useRef<string[]>([]);
   129→   129→  /** ایندکس سنکرون واژهٔ جاری — گارد کلوزر کهنه حین تعویض واژه */
   130→   130→  const idxRef = useRef(0);
   131→   131→  const timer = useRef<number | undefined>(undefined);
   132→   132→  const secsRef = useRef(0);
   133→   133→
   134→   134→  // رکورد شخصی این اسکوپ — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   135→   135→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   136→   136→  const bestScore = useSyncExternalStore(
   137→   137→    subscribeGameData,
   138→   138→    () => {
   139→   139→      const b = loadBest<BestRecord>(activeBookId, GAME_KEY, scope);
   140→   140→      return b ? b.score : null;
   141→   141→    },
   142→   142→    () => null,
   143→   143→  );
   144→   144→
   145→   145→  // ── تایمر ۱ثانیه‌ای — functional setState + شمارندهٔ دقیق برای Attempt ──
   146→   146→  useEffect(() => {
   147→   147→    if (phase !== "playing") return;
   148→   148→    const t = window.setInterval(() => {
   149→   149→      secsRef.current += 1;
   150→   150→      setSecs((s) => s + 1);
   151→   151→    }, 1000);
   152→   152→    return () => window.clearInterval(t);
   153→   153→  }, [phase]);
   154→   154→
   155→   155→  // ── پاک‌سازی نهایی: تلفظ + تایمر معلق ──
   156→   156→  useEffect(() => {
   157→   157→    return () => {
   158→   158→      stopSpeaking();
   159→   159→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   160→   160→    };
   161→   161→  }, []);
   162→   162→
   163→   163→  /** پایان دور — استریک/جلسه/Attempt/رکورد (بیشترین برد) */
   164→   164→  const finish = useCallback(
   165→   165→    (finalDetail: DetailRow[]) => {
   166→   166→      const correct = finalDetail.filter((d) => d.ok).length;
   167→   167→      touchStreak();
   168→   168→      recordStudySession();
   169→   169→      bumpPlays(activeBookId, GAME_KEY);
   170→   170→      saveAttempt(activeBookId, GAME_KEY, {
   171→   171→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   172→   172→        total: deck.length,
   173→   173→        correct,
   174→   174→        duration: secsRef.current,
   175→   175→        detail: { words: finalDetail },
   176→   176→      });
   177→   177→      const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   178→   178→      const prev = loadBest<BestRecord>(activeBookId, GAME_KEY, recScope);
   179→   179→      if (!prev || correct > prev.score) {
   180→   180→        saveBest<BestRecord>(activeBookId, GAME_KEY, recScope, { score: correct });
   181→   181→        setIsRecord(true);
   182→   182→        toast.success(`رکورد جدید! ${faNum(correct)} برد از ${faNum(deck.length)} 🏆`);
   183→   183→      } else {
   184→   184→        toast.success(`دور تمام شد — ${faNum(correct)} برد از ${faNum(deck.length)}`);
   185→   185→      }
   186→   186→      setPhase("done");
   187→   187→    },
   188→   188→    [activeBookId, lesson, count, deck.length],
   189→   189→  );
   190→   190→
   191→   191→  /** ادامه به واژهٔ بعدی یا پایان دور — ریست کامل وضعیت واژه */
   192→   192→  const advance = useCallback(
   193→   193→    (nextIdx: number, finalDetail: DetailRow[]) => {
   194→   194→      if (nextIdx >= deck.length) {
   195→   195→        finish(finalDetail);
   196→   196→        return;
   197→   197→      }
   198→   198→      setIdx(nextIdx);
   199→   199→      idxRef.current = nextIdx;
   200→   200→      usedRef.current = [];
   201→   201→      setUsed([]);
   202→   202→      setResult(null);
   203→   203→    },
   204→   204→    [deck.length, finish],
   205→   205→  );
   206→   206→
   207→   207→  /** حدس یک حرف — هستهٔ بازی؛ هم برای کیبورد on-screen و هم فیزیکی */
   208→   208→  const guessLetter = useCallback(
   209→   209→    (raw: string) => {
   210→   210→      const letter = raw.toUpperCase();
   211→   211→      if (!/^[A-Z]$/.test(letter)) return;
   212→   212→      if (locked.current) return;
   213→   213→      const current = deck[idx];
   214→   214→      if (!current) return;
   215→   215→      if (idxRef.current !== idx) return;
   216→   216→      // گارد سنکرون — keydown پشت‌سرهم قبل از رندر بعدی نباید دوبار ثبت شود
   217→   217→      if (usedRef.current.includes(letter)) return;
   218→   218→      const nextUsed = [...usedRef.current, letter];
   219→   219→      usedRef.current = nextUsed;
   220→   220→      setUsed(nextUsed);
   221→   221→
   222→   222→      if (current.word.includes(letter)) {
   223→   223→        // حرف درست — حروف تکراری همه‌باهم آشکار می‌شوند (چیپ‌ها از used مشتق می‌شوند)
   224→   224→        if (current.word.split("").every((ch) => nextUsed.includes(ch))) {
   225→   225→          setLiveMsg(`درست بود! واژه «${current.word}» کامل شد 🎉`);
   226→   226→          locked.current = true;
   227→   227→          recordSRS(current.word, true);
   228→   228→          speak(current.word, { speed });
   229→   229→          const nextDetail = [...detail, { word: current.word, ok: true }];
   230→   230→          setDetail(nextDetail);
   231→   231→          setResult(true);
   232→   232→          timer.current = window.setTimeout(() => {
   233→   233→            locked.current = false;
   234→   234→            advance(idx + 1, nextDetail);
   235→   235→          }, WIN_LOCK_MS);
   236→   236→        } else {
   237→   237→          setLiveMsg(`حرف ${letter} درست بود — ادامه بده`);
   238→   238→        }
   239→   239→      } else {
   240→   240→        // حرف غلط — یک بادکنک می‌ترکد؛ با ششمین خطا واژه می‌بازد
   241→   241→        const wrongCount = nextUsed.filter((l) => !current.word.includes(l)).length;
   242→   242→        if (wrongCount >= MAX_WRONG) {
   243→   243→          setLiveMsg(`واژه «${current.word}» بود — به مرور اضافه شد`);
   244→   244→          locked.current = true;
   245→   245→          recordSRS(current.word, false);
   246→   246→          const nextDetail = [...detail, { word: current.word, ok: false }];
   247→   247→          setDetail(nextDetail);
   248→   248→          setResult(false);
   249→   249→          timer.current = window.setTimeout(() => {
   250→   250→            locked.current = false;
   251→   251→            advance(idx + 1, nextDetail);
   252→   252→          }, LOSS_LOCK_MS);
   253→   253→        } else {
   254→   254→          setLiveMsg(`حرف ${letter} در واژه نیست — یک بادکنک ترکید`);
   255→   255→        }
   256→   256→      }
   257→   257→    },
   258→   258→    [deck, idx, detail, speed, advance],
   259→   259→  );
   260→   260→
   261→   261→  // ── latest-ref — آخرین نسخهٔ guessLetter برای listener پایدار کیبورد ──
   262→   262→  const guessRef = useRef<(letter: string) => void>(() => {});
   263→   263→  useEffect(() => {
   264→   264→    guessRef.current = guessLetter;
   265→   265→  });
   266→   266→
   267→   267→  // ── کیبورد فیزیکی — فقط در فاز بازی؛ listener با cleanup ──
   268→   268→  useEffect(() => {
   269→   269→    if (phase !== "playing") return;
   270→   270→    const onKey = (e: KeyboardEvent) => {
   271→   271→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   272→   272→      if (/^[a-zA-Z]$/.test(e.key)) {
   273→   273→        guessRef.current(e.key);
   274→   274→      }
   275→   275→    };
   276→   276→    window.addEventListener("keydown", onKey);
   277→   277→    return () => window.removeEventListener("keydown", onKey);
   278→   278→  }, [phase]);
   279→   279→
   280→   280→  /** انصراف — پاک‌سازی تایمر معلق و صدا و برگشت به تنظیمات */
   281→   281→  const cancel = useCallback(() => {
   282→   282→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   283→   283→    locked.current = false;
   284→   284→    stopSpeaking();
   285→   285→    setPhase("setup");
   286→   286→  }, []);
   287→   287→
   288→   288→  /** شروع دور — استخر فقط A-Za-z طول ۳..۱۰، یکتا با pickUnique */
   289→   289→  const start = useCallback(() => {
   290→   290→    const pool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   291→   291→    const eligible = pool.filter((w) => WORD_RE.test(baseWord(w.word)));
   292→   292→    const chosen = pickUnique(eligible, count, (w) => baseWord(w.word).toUpperCase());
   293→   293→    if (chosen.length < count) {
   294→   294→      toast.error(
   295→   295→        `واژهٔ کافی برای ${faNum(count)} واژه پیدا نشد — درس دیگری را امتحان کن یا تعداد را کمتر کن`,
   296→   296→      );
   297→   297→      return;
   298→   298→    }
   299→   299→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   300→   300→    locked.current = false;
   301→   301→    usedRef.current = [];
   302→   302→    idxRef.current = 0;
   303→   303→    stopSpeaking();
   304→   304→    setDeck(
   305→   305→      chosen.map((w) => ({
   306→   306→        word: baseWord(w.word).toUpperCase(),
   307→   307→        meaning: w.meaning,
   308→   308→        pos: w.pos,
   309→   309→        lessonLabel: w.lessonLabel,
   310→   310→      })),
   311→   311→    );
   312→   312→    setIdx(0);
   313→   313→    setUsed([]);
   314→   314→    setDetail([]);
   315→   315→    setResult(null);
   316→   316→    setSecs(0);
   317→   317→    secsRef.current = 0;
   318→   318→    setIsRecord(false);
   319→   319→    setPhase("playing");
   320→   320→  }, [data.vocab, lesson, count]);
   321→   321→
   322→   322→  // ── Setup ──
   323→   323→  if (phase === "setup") {
   324→   324→    return (
   325→   325→      <Card className="mx-auto max-w-xl rise-in">
   326→   326→        <CardContent className="space-y-6 p-6">
   327→   327→          <div className="flex items-center gap-3">
   328→   328→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   329→   329→              <SpellCheck size={22} />
   330→   330→            </span>
   331→   331→            <div>
   332→   332→              <h2 className="text-lg font-black">دارباز</h2>
   333→   333→              <p className="text-xs text-muted-foreground">
   334→   334→                با سرنخ معنی فارسی، حرف‌به‌حرف واژه را پیدا کن — بادکنک‌هایت را نگه دار!
   335→   335→              </p>
   336→   336→            </div>
   337→   337→          </div>
   338→   338→
   339→   339→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   340→   340→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   341→   341→            <p>
   342→   342→              ۶ بادکنک داری؛ هر حرف غلط یک بادکنک می‌ترکد. واژهٔ کامل = تلفظ + ثبت در مرور هوشمند،
   343→   343→              اما اگر ۶ بادکنک ترکید، واژه را می‌بازی و به صف مرور اضافه می‌شود.
   344→   344→            </p>
   345→   345→          </div>
   346→   346→
   347→   347→          <div>
   348→   348→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   349→   349→            <div className="flex flex-wrap gap-2" role="group" aria-label="انتخاب درس">
   350→   350→              <button
   351→   351→                type="button"
   352→   352→                onClick={() => setLesson(ALL_LESSONS)}
   353→   353→                aria-pressed={lesson === ALL_LESSONS}
   354→   354→                className={cn(
   355→   355→                  "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   356→   356→                  lesson === ALL_LESSONS
   357→   357→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   358→   358→                    : "text-muted-foreground hover:bg-muted",
   359→   359→                )}
   360→   360→              >
   361→   361→                کل کتاب ({faNum(data.vocab.all.length)} واژه)
   362→   362→              </button>
   363→   363→              {data.vocab.lessons.filter((l) => l.lesson > 0).map((l) => (
   364→   364→                <button
   365→   365→                  key={l.lesson}
   366→   366→                  type="button"
   367→   367→                  onClick={() => setLesson(l.lesson)}
   368→   368→                  aria-pressed={lesson === l.lesson}
   369→   369→                  className={cn(
   370→   370→                    "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   371→   371→                    lesson === l.lesson
   372→   372→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   373→   373→                      : "text-muted-foreground hover:bg-muted",
   374→   374→                  )}
   375→   375→                >
