    81→function LevelRing({ level, pct }: { level: number; pct: number }) {
    82→  const R = 52;
    83→  const C = 2 * Math.PI * R;
    84→  const safe = Math.min(100, Math.max(0, pct));
    85→  const offset = C - (safe / 100) * C;
    86→  return (
    87→    <div
    88→      className="relative h-36 w-36 shrink-0"
    89→      role="img"
    90→      aria-label={`سطح ${faNum(level)} — پیشرفت ${faNum(safe)} درصد`}
    91→    >
    92→      <svg viewBox="0 0 120 120" className="h-full w-full -rotate-90" aria-hidden>
    93→        <circle cx="60" cy="60" r={R} fill="none" strokeWidth="10" className="stroke-muted" />
    94→        <circle
    95→          cx="60"
    96→          cy="60"
    97→          r={R}
    98→          fill="none"
    99→          strokeWidth="10"
   100→          strokeLinecap="round"
   101→          strokeDasharray={C}
   102→          strokeDashoffset={offset}
   103→          className="stroke-primary transition-[stroke-dashoffset] duration-700 ease-out"
   104→        />
   105→      </svg>
   106→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   107→        <span className="text-4xl font-black tabular-nums text-foreground">{faNum(level)}</span>
   108→        <span className="text-[11px] text-muted-foreground">سطح</span>
   109→      </div>
   110→    </div>

   330→// ─── بخش رده‌بندی هفتگی (fetch + skeleton) ───
   331→
   332→function LeaderboardSection() {
   333→  const { bookId } = useBook();
   334→  const [loaded, setLoaded] = useState<{ bookId: string; data: BoardData } | null>(null);
   335→  const [failedFor, setFailedFor] = useState<string | null>(null);
   336→
   337→  useEffect(() => {
   338→    if (!bookId) return;
   339→    let cancelled = false;
   340→    const ctrl = new AbortController();
   341→    fetch(`/api/leaderboard?book=${bookId}`, { signal: ctrl.signal })
   342→      .then((r) => (r.ok ? r.json() : null))
   343→      .then((raw: unknown) => {
   344→        if (cancelled || !raw) return;
   345→        const d = raw as Partial<BoardData>;
   346→        setLoaded({
   347→          bookId,
   348→          data: {
   349→            weekStart: typeof d.weekStart === "string" ? d.weekStart : "",
   350→            rows: Array.isArray(d.rows) ? d.rows : [],
   351→            me: d.me ?? null,
   352→          },
   353→        });
   354→        setFailedFor(null);
   355→      })
   356→      .catch(() => {
   357→        if (!cancelled) setFailedFor(bookId);
   358→      });
   359→    return () => {
   360→      cancelled = true;
   361→      ctrl.abort();
   362→    };
   363→  }, [bookId]);
   364→
   365→  const board = loaded && loaded.bookId === bookId ? loaded.data : null;
   366→  const isFailed = bookId !== null && failedFor === bookId;
   367→
   368→  if (!bookId) {
   369→    return (
   370→      <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center text-sm text-muted-foreground">
   371→        📚 برای دیدن رده‌بندی هفتگی، یک جزوه را از قفسه باز کن.
   372→      </div>
   373→    );
   374→  }
   375→
   376→  if (!board) {
   377→    if (isFailed) {
   378→      return (
   379→        <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center text-sm text-muted-foreground">
   380→          🌱 رده‌بندی همین حالا در دسترس نیست — بعداً دوباره امتحان کن.
   381→        </div>
   382→      );
   383→    }
   384→    return (
   385→      <div className="space-y-3" aria-label="در حال بارگذاری رده‌بندی" aria-busy="true">
   386→        {Array.from({ length: 5 }).map((_, i) => (
   387→          <Skeleton key={i} className="h-14 w-full rounded-2xl" />
   388→        ))}
   389→      </div>
   390→    );
   391→  }
   392→
   393→  if (board.rows.length === 0) {
   394→    return (
   395→      <div className="rounded-3xl border border-dashed bg-card/40 p-8 text-center">
   396→        <div className="text-3xl" aria-hidden>
   397→          🌱
   398→        </div>
   399→        <p className="mt-2 text-sm font-bold text-foreground">
   400→          هنوز این هفته کسی امتیاز نگرفته — تو اولین باش!
   401→        </p>
   402→        <p className="mt-1 text-xs text-muted-foreground">
   403→          با آزمون و بازی امتیاز جمع کن و از شنبه بالای جدول بایست.
   404→        </p>
   405→      </div>
   406→    );
   407→  }
   408→
   409→  return (
   410→    <div className="space-y-3">
   411→      {board.me && (
   412→        <div className="flex items-center gap-2 rounded-2xl border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-xs font-bold text-amber-700 dark:text-amber-400">
   413→          <Trophy className="size-3.5 shrink-0" aria-hidden />
   414→          <span>
   415→            جای تو در این هفته: ردهٔ {faNum(board.me.rank)} با {faNum(board.me.xp)} XP
   416→          </span>
   417→        </div>
   418→      )}
   419→      <ul className="space-y-2.5" aria-label="جدول رده‌بندی هفتگی">
   420→        {board.rows.map((row, i) => {
   421→          const rank = i + 1;
   422→          return (
   423→            <li
   424→              key={`${row.name}-${rank}`}
   425→              className={cn(
   426→                "flex flex-wrap items-center gap-3 rounded-2xl border bg-background/50 px-3 py-2.5",
   427→                rank <= 3 && "border-amber-500/30 bg-amber-500/5",
   428→              )}
   429→            >
   430→              <span
   431→                className="w-8 shrink-0 text-center text-sm font-extrabold tabular-nums"
   432→                aria-label={`ردهٔ ${faNum(rank)}`}
   433→              >
   434→                {rank <= 3 ? (RANK_MEDALS[rank - 1] ?? "") : faNum(rank)}
   435→              </span>
   436→              {row.avatar ? (
   437→                <span className="text-2xl leading-none" aria-hidden>
   438→                  {row.avatar}
   439→                </span>
   440→              ) : (
   441→                <span
   442→                  className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-muted text-muted-foreground"
   443→                  aria-hidden
   444→                >
   445→                  <UserRound className="size-4" />
   446→                </span>
   447→              )}
   448→              <span className="min-w-0 flex-1 truncate text-sm font-bold text-foreground">
   449→                {row.name}
   450→              </span>
   451→              <span className="flex flex-wrap items-center justify-end gap-x-3 gap-y-0.5 text-[11px] text-muted-foreground">
   452→                <span className="font-extrabold text-amber-600 dark:text-amber-400">
   453→                  {faNum(row.xp)} XP
   454→                </span>
   455→                <span>دقت {faNum(Math.round(row.accuracy))}٪</span>
   456→                <span>{faNum(row.quizzes)} آزمون</span>
   457→              </span>
   458→            </li>
   459→          );
   460→        })}
   461→      </ul>
   462→      <p className="flex items-start gap-1.5 pt-1 text-[11px] leading-5 text-muted-foreground">
   463→        <Info className="mt-0.5 size-3.5 shrink-0" aria-hidden />
   464→        رده‌بندی دوستانهٔ هفتهٔ جاری (از شنبه) — XP از آزمون‌ها و بازی‌های همین جزوه.
   465→      </p>
   466→    </div>
   467→  );
   468→}

   472→export function RewardsView() {
   473→  const xp = useXp();

   475→  const streak = useStreak();
   476→  const missions = useMissions();
   477→  const badges = useBadges();
   478→  const { accent, shop } = useAccent();

   480→  const allClaimed = missions.quests.length > 0 && missions.quests.every((q) => q.claimed);
   481→  const dayPct = Math.min(100, Math.round((xp.dayXp / DAILY_CAP) * 100));
   482→
   483→  // ── handlerهای فروشگاه (فقط callback رویدادی) ──
   484→
   485→  const buyTheme = (id: string, price: number) => {
   486→    const ok = spendCoins(price, `theme:${id}`);
   487→    if (ok) {
   488→      markOwned(`theme:${id}`);
   489→      applyAccent(id);
   490→      toast.success("تم تازه فعال شد!");
   491→    } else {
   492→      toast.error("سکه کافی نداری");
   493→    }
   494→  };
   495→
   496→  const buyAvatar = (id: string, price: number) => {
   497→    const ok = spendCoins(price, id);
   498→    if (ok) {
   499→      markOwned(id);
   500→      toast.success("آواتار تازه به کمدت اضافه شد!", {
   501→        description: "حالا می‌توانی انتخابش کنی",
   502→      });
   503→    } else {
   504→      toast.error("سکه کافی نداری");
   505→    }
   506→  };
   507→
   508→  const buyFreeze = () => {
   509→    if (streak.freezes >= FREEZE_MAX) return;
   510→    const ok = spendCoins(FREEZE_PRICE, "streak-freeze");
   511→    if (ok) {
   512→      grantFreeze();
   513→      toast.success("❄ یخ به انبار زنجیره اضافه شد", {
   514→        description: "یک روز غیبت را جبران می‌کند",
   515→      });
   516→    } else {
   517→      toast.error("سکه کافی نداری");
   518→    }
   519→  };
