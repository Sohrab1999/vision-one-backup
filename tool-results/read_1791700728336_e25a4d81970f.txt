    71→function TeacherTopBar({
    72→  state,
    73→  session,
    74→  lastSync,
    75→}: {
    76→  state: ClassState | null;
    77→  session: NonNullable<ReturnType<typeof useClassSession>>;
    78→  lastSync: number | null;
    79→}) {
    80→  const [confirmLeave, setConfirmLeave] = useState(false);
    81→  const [leaving, setLeaving] = useState(false);
    82→
    83→  async function copyCode() {
    84→    try {
    85→      await navigator.clipboard.writeText(session.code);
    86→      toast.success("کد کپی شد");
    87→    } catch {
    88→      toast.error("کپی نشد — کد را دستی بردار");
    89→    }
    90→  }
    91→
    92→  async function leave() {
    93→    if (!confirmLeave) {
    94→      setConfirmLeave(true);
    95→      window.setTimeout(() => setConfirmLeave(false), 3500);
    96→      return;
    97→    }
    98→    setLeaving(true);
    99→    try {
   100→      const res = await fetch(
   101→        `/api/class/${encodeURIComponent(session.code)}?key=${encodeURIComponent(session.key)}`,
   102→        { method: "DELETE" },
   103→      );
   104→      if (!res.ok) throw new Error();
   105→      setClassSession(null);
   106→      toast.success("کلاس حذف شد");
   107→    } catch {
   108→      toast.error("حذف کلاس ممکن نشد — دوباره تلاش کن");
   109→      setLeaving(false);
   110→      setConfirmLeave(false);
   111→    }
   112→  }
   113→
   114→  return (
   115→    <div className="rise-in flex flex-wrap items-center justify-between gap-3 rounded-2xl border bg-card p-4">
   116→      <div className="min-w-0 space-y-0.5">
   117→        <h1 className="truncate text-lg font-black">{state?.classroom.name ?? session.name ?? "کلاس من"}</h1>
   118→        <p className="text-[11.5px] text-muted-foreground">
   119→          داشبورد معلم
   120→          {lastSync !== null && (
   121→            <>
   122→              <span aria-hidden> · </span>
   123→              <span>به‌روزرسانی: {fmtClock(lastSync)}</span>
   124→            </>
   125→          )}
   126→        </p>
   127→      </div>
   128→
   129→      <div className="flex flex-wrap items-center gap-2">
   130→        {/* کد ۶رقمی بزرگ */}
   131→        <div className="flex items-center gap-1.5 rounded-2xl border-2 border-dashed border-primary/40 bg-primary/5 px-3 py-1.5">
   132→          <span
   133→            dir="ltr"
   134→            className="font-en text-2xl font-black tracking-[0.3em] text-primary sm:text-3xl"
   135→            aria-label={`کد کلاس ${session.code}`}
   136→          >
   137→            {session.code}
   138→          </span>
   139→          <Button
   140→            variant="ghost"
   141→            size="sm"
   142→            onClick={() => void copyCode()}
   143→            className="min-h-9 gap-1 px-2 text-[12px] font-bold text-primary"
   144→            aria-label="کپی کد کلاس"
   145→          >
   146→            <Copy size={14} aria-hidden />
   147→            کپی کد
   148→          </Button>
   149→        </div>
   150→
   151→        {/* خروج دوکلیکی */}
   152→        <Button
   153→          variant="ghost"
   154→          size="sm"
   155→          disabled={leaving}
   156→          onClick={() => void leave()}
   157→          className={cn(
   158→            "min-h-11 gap-1.5 px-3 text-[12px] font-bold",
   159→            confirmLeave
   160→              ? "bg-amber-500/15 text-amber-700 hover:bg-amber-500/25 dark:text-amber-400"
   161→              : "text-muted-foreground hover:bg-muted",
   162→          )}
   163→          aria-label={confirmLeave ? "تأیید حذف کلاس" : "خروج از کلاس"}
   164→        >
   165→          {leaving ? <Loader2 size={15} className="animate-spin" aria-hidden /> : <LogOut size={15} aria-hidden />}
   166→          {confirmLeave ? "مطمئنی؟ (کلیک دوم)" : "خروج از کلاس"}
   167→        </Button>
   168→      </div>
   169→    </div>
   170→  );
   171→}

   282→type AssignKind = "review" | "quiz";
   283→
   284→function AssignmentsCard({
   285→  state,
   286→  session,
   287→  refresh,
   288→}: {
   289→  state: ClassState | null;
   290→  session: NonNullable<ReturnType<typeof useClassSession>>;
   291→  refresh: () => void;
   292→}) {
   293→  const [title, setTitle] = useState("");
   294→  const [kind, setKind] = useState<AssignKind>("review");
   295→  const [lesson, setLesson] = useState(1);
   296→  const [questionCount, setQuestionCount] = useState(10);
   297→  const [busy, setBusy] = useState(false);
   298→
   299→  const assignments = state?.assignments ?? [];
   300→  const memberCount = state?.members?.length ?? 0;
   301→
   302→  async function submit() {
   303→    const t = title.trim();
   304→    if (t.length < 2) {
   305→      toast.error("عنوان تکلیف را بنویس");
   306→      return;
   307→    }
   308→    const qc = Math.min(50, Math.max(5, Math.round(questionCount) || 10));
   309→    const les = Math.min(12, Math.max(0, Math.round(lesson) || 0));
   310→    setBusy(true);
   311→    try {
   312→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/assignments`, {
   313→        method: "POST",
   314→        headers: { "Content-Type": "application/json" },
   315→        body: JSON.stringify({
   316→          teacherKey: session.key,
   317→          title: t,
   318→          kind,
   319→          lesson: kind === "review" ? les : undefined,
   320→          questionCount: kind === "quiz" ? qc : undefined,
   321→        }),
   322→      });
   323→      if (res.status === 201) {
   324→        toast.success("تکلیف اعلام شد 📣");
   325→        setTitle("");
   326→        refresh();
   327→      } else {
   328→        toast.error(await readApiError(res, "اعلام تکلیف ممکن نشد"));
   329→      }
   330→    } catch {
   331→      toast.error("ارتباط با سرور برقرار نشد");
   332→    } finally {
   333→      setBusy(false);
   334→    }
   335→  }
   336→
   337→  async function remind(a: ClassAssignment) {
   338→    try {
   339→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/notifications`, {
   340→        method: "POST",
   341→        headers: { "Content-Type": "application/json" },
   342→        body: JSON.stringify({ teacherKey: session.key, text: `⏰ تکلیف «${a.title}» را فراموش نکن!` }),
   343→      });
   344→      if (res.status === 201) toast.success("یادآوری ارسال شد ⏰");
   345→      else toast.error(await readApiError(res, "ارسال یادآوری ممکن نشد"));
   346→    } catch {
   347→      toast.error("ارتباط با سرور برقرار نشد");
   348→    }
   349→  }
   350→
   351→  return (
   352→    <Card className="rise-in">
   353→      <CardContent className="space-y-4 p-4">
   354→        <h2 className="flex items-center gap-1.5 text-sm font-black">
   355→          <ClipboardList size={16} className="text-primary" aria-hidden />
   356→          تکالیف
   357→        </h2>
   358→
   359→        {/* فرم فشرده اعلام تکلیف */}
   360→        <div className="space-y-2.5 rounded-xl border bg-muted/20 p-3">
   361→          <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-[1fr_auto]">
   362→            <Input
   363→              value={title}
   364→              onChange={(e) => setTitle(e.target.value)}
   365→              placeholder="عنوان تکلیف — مثلاً واژگان درس ۳ را مرور کن"
   366→              maxLength={80}
   367→              className="min-h-11"
   368→              aria-label="عنوان تکلیف"
   369→            />
   370→            <div className="flex gap-1 self-start rounded-xl border bg-card p-1" role="group" aria-label="نوع تکلیف">
   371→              <Button
   372→                type="button"
   373→                variant={kind === "review" ? "secondary" : "ghost"}
   374→                size="sm"
   375→                className="min-h-9"
   376→                onClick={() => setKind("review")}
   377→                aria-pressed={kind === "review"}
   378→              >
   379→                مرور واژگان
   380→              </Button>
   381→              <Button
   382→                type="button"
   383→                variant={kind === "quiz" ? "secondary" : "ghost"}
   384→                size="sm"
   385→                className="min-h-9"
   386→                onClick={() => setKind("quiz")}
   387→                aria-pressed={kind === "quiz"}
   388→              >
   389→                آزمون
   390→              </Button>
   391→            </div>
   392→          </div>
   393→
   394→          <div className="grid grid-cols-1 items-end gap-2.5 sm:grid-cols-[1fr_auto]">
   395→            {kind === "review" ? (
   396→              <div className="space-y-1">
   397→                <Label htmlFor="assign-lesson" className="text-[11.5px] text-muted-foreground">
   398→                  شماره درس (۰ تا ۱۲)
   399→                </Label>
   400→                <Input
   401→                  id="assign-lesson"
   402→                  type="number"
   403→                  inputMode="numeric"
   404→                  min={0}
   405→                  max={12}
   406→                  value={lesson}
   407→                  onChange={(e) => setLesson(Number(e.target.value))}
   408→                  className="min-h-11"
   409→                  aria-label="شماره درس مرور"
   410→                />
   411→              </div>
   412→            ) : (
   413→              <div className="space-y-1">
   414→                <Label htmlFor="assign-qc" className="text-[11.5px] text-muted-foreground">
   415→                  تعداد سؤال (۵ تا ۵۰)
   416→                </Label>
   417→                <Input
   418→                  id="assign-qc"
   419→                  type="number"
   420→                  inputMode="numeric"
   421→                  min={5}
   422→                  max={50}
   423→                  value={questionCount}
   424→                  onChange={(e) => setQuestionCount(Number(e.target.value))}
   425→                  className="min-h-11"
   426→                  aria-label="تعداد سؤال آزمون"
   427→                />
   428→              </div>
   429→            )}
   430→            <Button type="button" disabled={busy} onClick={() => void submit()} className="min-h-11 gap-1.5 font-bold">
   431→              {busy ? <Loader2 size={15} className="animate-spin" aria-hidden /> : <Plus size={15} aria-hidden />}
   432→              اعلام تکلیف
   433→            </Button>
   434→          </div>
   435→        </div>
   436→
   437→        {/* فهرست تکالیف */}
   438→        {assignments.length === 0 ? (
   439→          <p className="rounded-xl bg-muted/50 p-3 text-center text-[12px] text-muted-foreground">
   440→            هنوز تکلیفی ندادی — اولین تکلیف را اعلام کن! 📋
   441→          </p>
   442→        ) : (
   443→          <ul className="max-h-96 space-y-2 overflow-y-auto pe-1" aria-label="فهرست تکالیف">
   444→            {assignments.map((a) => {
   445→              const pct = memberCount > 0 ? Math.round((a.doneCount / memberCount) * 100) : 0;
   446→              return (
   447→                <li key={a.id} className="space-y-2 rounded-xl border bg-card px-3 py-2.5">
   448→                  <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   449→                    <span className="min-w-0 flex-1 truncate text-[13px] font-bold">{a.title}</span>
   450→                    <Badge
   451→                      variant="outline"
   452→                      className={cn(
   453→                        "text-[10.5px]",
   454→                        a.kind === "quiz"
   455→                          ? "border-teal-500/30 bg-teal-500/10 text-teal-700 dark:text-teal-400"
   456→                          : "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400",
   457→                      )}
   458→                    >
   459→                      {a.kind === "quiz"
   460→                        ? `آزمون ${faNum(a.questionCount ?? 0)} سؤالی`
   461→                        : `مرور${a.lesson != null ? ` — درس ${faNum(a.lesson)}` : ""}`}
   462→                    </Badge>
   463→                    {a.dueAt && (
   464→                      <span className="text-[10.5px] text-muted-foreground" aria-label="مهلت تکلیف">
   465→                        مهلت: {fmtJalaliDate(a.dueAt)}
   466→                      </span>
   467→                    )}
   468→                  </div>
   469→                  <div className="flex items-center gap-2">
   470→                    <Progress value={pct} className="h-2 flex-1" aria-label={`پیشرفت تکلیف ${a.title}`} />
   471→                    <span className="w-16 shrink-0 text-[11px] tabular-nums text-muted-foreground">
   472→                      {faNum(a.doneCount)}/{faNum(memberCount)}
   473→                    </span>
   474→                    <Button
   475→                      variant="ghost"
   476→                      size="sm"
   477→                      className="min-h-9 gap-1 px-2 text-[11.5px] font-bold text-amber-700 dark:text-amber-400"
   478→                      onClick={() => void remind(a)}
   479→                      aria-label={`یادآوری تکلیف ${a.title}`}
   480→                    >
   481→                      <Bell size={13} aria-hidden />
   482→                      یادآوری
   483→                    </Button>
   484→                  </div>
   485→                </li>
   486→              );
   487→            })}
   488→          </ul>
   489→        )}
   490→      </CardContent>
   491→    </Card>
   492→  );
   493→}

   496→
   497→function TeacherNotificationsCard({
   498→  session,
   499→  refresh,
   500→}: {
   501→  session: NonNullable<ReturnType<typeof useClassSession>>;
   502→  refresh: () => void;
   503→}) {
   504→  const [text, setText] = useState("");
   505→  const [busy, setBusy] = useState(false);
   506→
   507→  async function send() {
   508→    const t = text.trim();
   509→    if (!t) {
   510→      toast.error("متن اعلان را بنویس");
   511→      return;
   512→    }
   513→    setBusy(true);
   514→    try {
   515→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/notifications`, {
   516→        method: "POST",
   517→        headers: { "Content-Type": "application/json" },
   518→        body: JSON.stringify({ teacherKey: session.key, text: t }),
   519→      });
   520→      if (res.status === 201) {
   521→        toast.success("اعلان برای همه ارسال شد 📣");
   522→        setText("");
   523→        refresh();
   524→      } else {
   525→        toast.error(await readApiError(res, "ارسال اعلان ممکن نشد"));
   526→      }
   527→    } catch {
   528→      toast.error("ارتباط با سرور برقرار نشد");
   529→    } finally {
   530→      setBusy(false);
   531→    }
   532→  }
   533→
   534→  return (
   535→    <Card className="rise-in">
   536→      <CardContent className="space-y-2.5 p-4">
   537→        <h2 className="flex items-center gap-1.5 text-sm font-black">
   538→          <Megaphone size={16} className="text-primary" aria-hidden />
   539→          اعلان به کلاس
   540→        </h2>
   541→        <Textarea
   542→          value={text}
   543→          onChange={(e) => setText(e.target.value)}
   544→          placeholder="مثلاً فردا آزمون درس ۳ داریم — آماده باشید! 📚"
   545→          rows={2}
   546→          maxLength={300}
   547→          className="min-h-11 resize-none"
   548→          aria-label="متن اعلان"
   549→        />
   550→        <div className="flex justify-end">
   551→          <Button size="sm" disabled={busy} onClick={() => void send()} className="min-h-11 gap-1.5 px-4 font-bold">
   552→            {busy ? <Loader2 size={14} className="animate-spin" aria-hidden /> : <Send size={14} aria-hidden />}
   553→            ارسال به همه
   554→          </Button>
   555→        </div>
   556→      </CardContent>
   557→    </Card>
   558→  );

   563→const MEDALS = ["🥇", "🥈", "🥉"];
   564→
   565→function TeacherLiveCard({
   566→  session,
   567→  state,
   568→  refresh,
   569→}: {
   570→  session: NonNullable<ReturnType<typeof useClassSession>>;
   571→  state: ClassState | null;
   572→  refresh: () => void;
   573→}) {
   574→  const [liveData, setLiveData] = useState<TeacherLivePayload | null>(null);
   575→  const [count, setCount] = useState(5);
   576→  const [starting, setStarting] = useState(false);
   577→  const [patching, setPatching] = useState(false);
   578→
   579→  // ── polling دادهٔ زنده هر ۳ ثانیه (بدون setState در بدنهٔ effect) ──
   580→  // نکته: سرور پس از «پایان آزمون»، live را null می‌کند؛ گزارش پایانی فقط در
   581→  // پاسخ PATCH finish است — پس گزارش تمام‌شده را تا «بستن گزارش» نگه می‌داریم.
   582→  const fetchLive = useCallback(async () => {
   583→    try {
   584→      const res = await fetch(
   585→        `/api/class/${encodeURIComponent(session.code)}/live?key=${encodeURIComponent(session.key)}&role=teacher`,
   586→        { cache: "no-store" },
   587→      );
   588→      if (!res.ok) throw new Error();
   589→      const d = (await res.json()) as TeacherLivePayload;
   590→      setLiveData((prev) => (prev?.live?.status === "finished" && !d.live ? prev : d));
   591→    } catch {
   592→      /* ساکت — دور بعد تلاش می‌شود */
   593→    }
   594→  }, [session.code, session.key]);
   595→
   596→  const fetchLiveRef = useRef(fetchLive);
   597→  useEffect(() => {
   598→    fetchLiveRef.current = fetchLive;
   599→  }, [fetchLive]);
   600→
   601→  useEffect(() => {
   602→    const t0 = window.setTimeout(() => void fetchLiveRef.current(), 0);
   603→    const t = window.setInterval(() => void fetchLiveRef.current(), 3000);
   604→    return () => {
   605→      window.clearTimeout(t0);
   606→      window.clearInterval(t);
   607→    };
   608→  }, []);
   609→
   610→  const live = liveData?.live ?? null;
   611→  const isActive = live !== null && live.status !== "finished";
   612→  const isFinished = live !== null && live.status === "finished";
   613→
   614→  async function start() {
   615→    setStarting(true);
   616→    try {
   617→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/live`, {
   618→        method: "POST",
   619→        headers: { "Content-Type": "application/json" },
   620→        body: JSON.stringify({ teacherKey: session.key, count }),
   621→      });
   622→      if (res.status === 201) {
   623→        toast.success("آزمون زنده شروع شد — بچه‌ها هم‌زمان می‌بینند! 🔴");
   624→        await fetchLiveRef.current();
   625→        refresh();
   626→      } else {
   627→        toast.error(await readApiError(res, "شروع آزمون ممکن نشد"));
   628→      }
   629→    } catch {
   630→      toast.error("ارتباط با سرور برقرار نشد");
   631→    } finally {
   632→      setStarting(false);
   633→    }
   634→  }
   635→
   636→  async function patch(action: "next" | "finish") {
   637→    setPatching(true);
   638→    try {
   639→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/live`, {
   640→        method: "PATCH",
   641→        headers: { "Content-Type": "application/json" },
   642→        body: JSON.stringify({ teacherKey: session.key, action }),
   643→      });
   644→      if (!res.ok) {
   645→        toast.error(await readApiError(res, "انجام نشد — دوباره تلاش کن"));
   646→      } else {
   647→        // پاسخ PATCH همان shape GET معلم است — برای «پایان» همین را گزارش نهایی کن
   648→        const j = (await res.json().catch(() => null)) as TeacherLivePayload | null;
   649→        if (j?.live) setLiveData(j);
   650→        else await fetchLiveRef.current();
   651→      }
   652→      if (action === "finish" && res.ok) {
   653→        refresh();
   654→        toast.success("آزمون زنده تمام شد 🏁");
   655→      }
   656→    } catch {
   657→      toast.error("ارتباط با سرور برقرار نشد");
   658→    } finally {
   659→      setPatching(false);
   660→    }
   661→  }
   662→
   663→  const q = liveData?.question ?? null;
   664→  const leaderboard = liveData?.leaderboard ?? [];
   665→
   666→  return (
   667→    <Card className="rise-in border-primary/25 bg-gradient-to-bl from-primary/5 to-transparent">
   668→      <CardContent className="space-y-3.5 p-4">
   669→        <div className="flex flex-wrap items-center justify-between gap-2">
   670→          <h2 className="flex items-center gap-1.5 text-sm font-black">
   671→            <Trophy size={16} className="text-amber-500" aria-hidden />
   672→            آزمون زنده
   673→          </h2>
   674→          {isActive && live && (
   675→            <Badge className="bg-primary text-primary-foreground">
   676→              در جریان — سؤال {faNum(live.currentIndex + 1)} از {faNum(live.questionCount)}
   677→            </Badge>
   678→          )}
   679→        </div>
   680→
   681→        {/* ── آزمون در جریان ── */}
   682→        {isActive && (
   683→          <div className="space-y-3">
   684→            {q ? (
   685→              <div className="space-y-2 rounded-xl border bg-card p-3.5" aria-label="سؤال جاری آزمون زنده">
   686→                <p dir="ltr" className="font-en text-[15px] font-bold leading-relaxed">
   687→                  {q.text}
   688→                </p>
   689→                <div className="grid gap-1.5">
   690→                  {q.options.map((o, i) => (
   691→                    <div
   692→                      key={i}
   693→                      className={cn(
   694→                        "flex items-center gap-2 rounded-lg border px-3 py-2 text-[13px]",
   695→                        i === q.answer
   696→                          ? "border-emerald-500/50 bg-emerald-500/10 font-bold text-emerald-700 dark:text-emerald-400"
   697→                          : "bg-muted/30 text-foreground/80",
   698→                      )}
   699→                    >
   700→                      <span dir="ltr" className="font-en text-[11px] text-muted-foreground">
   701→                        {String.fromCharCode(65 + i)})
   702→                      </span>
   703→                      <span dir="ltr" className="font-en">
   704→                        {o}
   705→                      </span>
   706→                      {i === q.answer && <CheckCircle2 size={15} className="ms-auto shrink-0" aria-hidden />}
   707→                    </div>
   708→                  ))}
   709→                </div>
   710→              </div>
   711→            ) : (
   712→              <div className="flex items-center gap-2 rounded-xl border bg-card p-3.5 text-[12.5px] text-muted-foreground">
   713→                <Loader2 size={14} className="animate-spin" aria-hidden />
   714→                در حال آماده‌سازی سؤال اول…
   715→              </div>
   716→            )}
   717→
   718→            <p className="text-[12px] text-muted-foreground">
   719→              پاسخ‌دهندگان تا این لحظه: <b>{faNum(liveData?.answeredCount ?? 0)}</b> نفر
   720→            </p>
   721→
   722→            <LeaderboardRows rows={leaderboard} />
   723→
   724→            <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   725→              <Button disabled={patching} onClick={() => void patch("next")} className="min-h-11 gap-1.5 font-bold">
   726→                {patching ? <Loader2 size={15} className="animate-spin" aria-hidden /> : null}
   727→                سؤال بعدی
   728→              </Button>
   729→              <Button
   730→                variant="outline"
   731→                disabled={patching}
   732→                onClick={() => void patch("finish")}
   733→                className="min-h-11 gap-1.5 border-amber-500/40 font-bold text-amber-700 hover:bg-amber-500/10 dark:text-amber-400"
   734→              >
   735→                پایان آزمون
   736→              </Button>
   737→            </div>
   738→          </div>
   739→        )}
   740→
   741→        {/* ── آزمون تمام‌شده (گزارش نهایی) ── */}
   742→        {isFinished && (
   743→          <div className="space-y-3">
   744→            <p className="text-[13px] font-bold">🏁 آزمون تمام شد — جدول برنده‌های نهایی:</p>
   745→            <LeaderboardRows rows={leaderboard} />
   746→            <div className="flex justify-end">
   747→              <Button
   748→                variant="ghost"
   749→                size="sm"
   750→                className="min-h-9 text-[12px]"
   751→                onClick={() => setLiveData(null)}
   752→              >
   753→                بستن گزارش
   754→              </Button>
   755→            </div>
   756→          </div>
   757→        )}
   758→
   759→        {/* ── شروع آزمون ── */}
   760→        {!live && (
   761→          <div className="space-y-3">
   762→            <div className="flex flex-wrap items-center gap-2" role="group" aria-label="تعداد سؤال آزمون زنده">
   763→              <span className="text-[12px] text-muted-foreground">تعداد سؤال:</span>
   764→              {[3, 5, 10].map((c) => (
   765→                <Button
   766→                  key={c}
   767→                  variant={count === c ? "secondary" : "outline"}
   768→                  size="sm"
   769→                  className="min-h-9 min-w-11 font-black"
   770→                  onClick={() => setCount(c)}
   771→                  aria-pressed={count === c}
   772→                >
   773→                  {faNum(c)}
   774→                </Button>
   775→              ))}
   776→            </div>
   777→            <Button
   778→              disabled={starting || (state?.members?.length ?? 0) === 0}
   779→              onClick={() => void start()}
   780→              className="min-h-12 w-full gap-2 text-[15px] font-black"
   781→            >
   782→              {starting ? <Loader2 size={16} className="animate-spin" aria-hidden /> : <span aria-hidden>🔴</span>}
   783→              شروع آزمون زنده
   784→            </Button>
   785→            <p className="text-center text-[11.5px] text-muted-foreground">
   786→              همه هم‌زمان می‌بینند؛ جدول برنده‌ها لحظه‌ای پر می‌شود
   787→              {(state?.members?.length ?? 0) === 0 && " — اول صبر کن بچه‌ها بیایند 🎒"}
   788→            </p>
   789→          </div>
   790→        )}
   791→      </CardContent>
   792→    </Card>
   793→  );
   794→}

   796→/** جدول برنده‌ها — ردیف‌های 🥇🥈🥉 + نام + تعداد درست + زمان */
   797→function LeaderboardRows({ rows }: { rows: TeacherLivePayload["leaderboard"] }) {
   798→  if (rows.length === 0) {
   799→    return (
   800→      <p className="rounded-xl bg-muted/50 p-3 text-center text-[12px] text-muted-foreground">
   801→        هنوز کسی پاسخ نداده…
   802→      </p>
   803→    );
   804→  }
   805→  return (
   806→    <ol className="max-h-72 space-y-1.5 overflow-y-auto pe-1" aria-label="جدول برنده‌ها">
   807→      {rows.map((r, i) => (
   808→        <li
   809→          key={`${r.displayName}-${i}`}
   810→          className={cn(
   811→            "flex items-center gap-2 rounded-xl border px-3 py-2",
   812→            i === 0 && "border-amber-500/40 bg-amber-500/10",
   813→            i === 1 && "bg-muted/40",
   814→            i === 2 && "border-amber-500/20 bg-amber-500/5",
   815→            i > 2 && "bg-card",
   816→          )}
   817→        >
   818→          <span className="w-7 shrink-0 text-center text-base" aria-hidden>
   819→            {i < 3 ? MEDALS[i] : faNum(i + 1)}
   820→          </span>
   821→          <span className="min-w-0 flex-1 truncate text-[13px] font-bold">{r.displayName}</span>
   822→          <Badge
   823→            variant="outline"
   824→            className="border-emerald-500/30 bg-emerald-500/10 text-[10.5px] text-emerald-700 dark:text-emerald-400"
   825→          >
   826→            {faNum(r.correct)} درست
   827→          </Badge>
   828→          <span className="shrink-0 text-[11px] tabular-nums text-muted-foreground">
   829→            {faNum(Number((r.totalMs / 1000).toFixed(1)))} ثانیه
   830→          </span>
   831→        </li>
   832→      ))}
   833→    </ol>
   834→  );
   835→}
