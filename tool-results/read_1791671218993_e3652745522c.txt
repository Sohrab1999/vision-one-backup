   134→   134→/** چیدمان عمودی k دهانه در باند بازی‌پذیر — ارتفاع دهانه سقف‌دار، مازاد به دیواره‌ها می‌رود */
   135→   135→function layoutGaps(k: number): Array<{ y0: number; y1: number }> {
   136→   136→  const top = 34;
   137→   137→  const bottom = H - GROUND_H - 12;
   138→   138→  const avail = bottom - top;
   139→   139→  const barMin = 34;
   140→   140→  const gapCap = 150;
   141→   141→  let gapH = (avail - (k - 1) * barMin) / k;
   142→   142→  let barH = barMin;
   143→   143→  if (gapH > gapCap) {
   144→   144→    barH = barMin + ((gapH - gapCap) * k) / (k - 1);
   145→   145→    gapH = gapCap;
   146→   146→  }
   147→   147→  const out: Array<{ y0: number; y1: number }> = [];
   148→   148→  let y = top;
   149→   149→  for (let i = 0; i < k; i++) {
   150→   150→    out.push({ y0: y, y1: y + gapH });
   151→   151→    y += gapH + barH;
   152→   152→  }
   153→   153→  return out;
   154→   154→}

   156→   156→function roundRectPath(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   157→   157→  c.beginPath();
   158→   158→  c.moveTo(x + r, y);
   159→   159→  c.arcTo(x + w, y, x + w, y + h, r);
   160→   160→  c.arcTo(x + w, y + h, x, y + h, r);
   161→   161→  c.arcTo(x, y + h, x, y, r);
   162→   162→  c.arcTo(x, y, x + w, y, r);
   163→   163→  c.closePath();
   164→   164→}

   194→   194→function drawWall(c: CanvasRenderingContext2D, wall: FlappyWall, now: number) {
   195→   195→  const segs: Array<[number, number]> = [];
   196→   196→  let y = 0;
   197→   197→  for (const g of wall.gaps) {
   198→   198→    segs.push([y, g.y0]);
   199→   199→    y = g.y1;
   200→   200→  }
   201→   201→  segs.push([y, H - GROUND_H]);
   202→   202→  c.fillStyle = "#059669";
   203→   203→  for (const [a, b] of segs) if (b > a) c.fillRect(wall.x, a, WALL_W, b - a);
   204→   204→  c.fillStyle = "#047857";
   205→   205→  for (const g of wall.gaps) {
   206→   206→    c.fillRect(wall.x - 4, g.y0 - 10, WALL_W + 8, 10);
   207→   207→    c.fillRect(wall.x - 4, g.y1, WALL_W + 8, 10);
   208→   208→  }
   209→   209→  c.textAlign = "center";
   210→   210→  c.textBaseline = "middle";
   211→   211→  for (const g of wall.gaps) {
   212→   212→    const cy = (g.y0 + g.y1) / 2;
   213→   213→    let fs = 15;
   214→   214→    c.font = `800 ${fs}px system-ui, sans-serif`;
   215→   215→    while (c.measureText(g.text).width > WALL_W - 16 && fs > 8) {
   216→   216→      fs -= 1;
   217→   217→      c.font = `800 ${fs}px system-ui, sans-serif`;
   218→   218→    }
   219→   219→    const tw = Math.min(c.measureText(g.text).width, WALL_W - 16);
   220→   220→    c.fillStyle = "rgba(255, 255, 255, 0.92)";
   221→   221→    roundRectPath(c, wall.x + WALL_W / 2 - tw / 2 - 8, cy - 11, tw + 16, 22, 8);
   222→   222→    c.fill();
   223→   223→    c.fillStyle = "#065f46";
   224→   224→    c.fillText(g.text, wall.x + WALL_W / 2, cy + 1);
   225→   225→  }
   226→   226→  if (wall.flashUntil > now) {
   227→   227→    if (wall.ok) {
   228→   228→      const g = wall.gaps.find((gp) => gp.correct);
   229→   229→      if (g) {
   230→   230→        c.fillStyle = "rgba(16, 185, 129, 0.30)";
   231→   231→        c.fillRect(wall.x - 4, g.y0, WALL_W + 8, g.y1 - g.y0);
   232→   232→      }
   233→   233→    } else {
   234→   234→      c.fillStyle = "rgba(239, 68, 68, 0.22)";
   235→   235→      c.fillRect(wall.x - 4, 0, WALL_W + 8, H - GROUND_H);
   236→   236→    }
   237→   237→  }
   238→   238→}
