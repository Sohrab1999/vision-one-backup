   179→   179→function collectSentences(data: BookData): SourceSentence[] {
   180→   180→  const out: SourceSentence[] = [];
   181→   181→  for (const r of data.readings) {
   182→   182→    for (const p of r.paragraphs) {

   186→   186→      }
   187→   187→    }
   188→   188→  }
   189→   189→  for (const c of data.conversations) {
   190→   190→    for (const line of c.lines) {
   191→   191→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   192→   192→        const t = s.trim();
   193→   193→        if (t) out.push({ en: t, fa: line.fa });
   194→   194→      }
   195→   195→    }
   196→   196→  }
   197→   197→  return out;
   198→   198→}

   202→   202→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   203→   203→ */
   204→   204→function buildPool(data: BookData, lesson: number, distractorCount = 2): ClozeItem[] {
   205→   205→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   206→   206→  const sentences = collectSentences(data);
   207→   207→
   208→   208→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   209→   209→  const seenBase = new Set<string>();
   210→   210→  const words: { base: string; entry: VocabEntry }[] = [];
   211→   211→  for (const entry of vocabPool) {
   212→   212→    const base = toBase(entry.word);
   213→   213→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   214→   214→    seenBase.add(base);
   215→   215→    words.push({ base, entry });
   216→   216→  }
   217→   217→
   218→   218→  const items: ClozeItem[] = [];
   219→   219→  for (const { base, entry } of words) {
   220→   220→    if (items.length >= MAX_ITEMS) break;
   221→   221→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   222→   222→    const found = sentences.find((s) => {
   223→   223→      if (s.en.length < 30 || s.en.length > 160) return false;
   224→   224→      const wc = wordCount(s.en);
   225→   225→      if (wc < 6 || wc > 35) return false;
   226→   226→      return re.test(s.en);
   227→   227→    });
   228→   228→    if (!found) continue;
   229→   229→
   230→   230→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   231→   231→    // (فاز G3-01 — تعداد حواس‌پرتی سطح‌دار؛ پیش‌فرض ۲ = همان رفتار قبلی)
   232→   232→    const distractorPool = vocabPool.filter((w) => {
   233→   233→      const b = toBase(w.word);
   234→   234→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   235→   235→    });
   236→   236→    const distractors = pickUnique(distractorPool, distractorCount, (w) => `${toBase(w.word)}|${w.meaning}`);
   237→   237→    if (distractors.length < distractorCount) continue;
   238→   238→
   239→   239→    items.push({
   240→   240→      id: `${lesson}:${base}`,
   241→   241→      sentence: found.en.replace(re, BLANK),
   242→   242→      fa: found.fa,
   243→   243→      answer: base,
   244→   244→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   245→   245→      meaning: entry.meaning,
   246→   246→      lessonLabel: entry.lessonLabel,
   247→   247→    });
   248→   248→  }
   249→   249→  return shuffle(items);
   250→   250→}

   257→   257→function readBest(bookId: string, scope: string): BestRecord | null {
   258→   258→  let raw: string | null = null;
   259→   259→  try {
   260→   260→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   261→   261→  } catch {
   262→   262→    return null;
   263→   263→  }
   264→   264→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   265→   265→  if (cacheKey === bestCacheKey) return bestCacheValue;
   266→   266→  let value: BestRecord | null = null;
   267→   267→  if (raw) {
   268→   268→    try {
   269→   269→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   270→   270→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   271→   271→        value = { score: parsed.score, total: parsed.total };
   272→   272→      }
   273→   273→    } catch {
   274→   274→      value = null;
   275→   275→    }
   276→   276→  }
   277→   277→  bestCacheKey = cacheKey;
   278→   278→  bestCacheValue = value;
   279→   279→  return value;
   280→   280→}
