   327→  function chooseOption(optIdx: number) {

   333→    setAnswers((prev) => {
   334→      const next = [...prev];
   335→      next[index] = optIdx;
   336→      return next;

   339→      if (optIdx === current.answer) {
   340→        // کیت بازی‌ها — بازخورد صوتی/لرزشی پاسخ درست
   341→        playSound("correct");
   342→        buzz("success");

   383→  /** ساخت یک تست مشابه از سؤال جاری با هوش مصنوعی — نمایش به‌صورت کارت بونس */
   384→  async function generateSimilar(fromBonus = false) {
   385→    if (!current || bonusLoading) return;
   386→    // «سؤال مشابه بعدی» از روی سؤالِ مشابهٔ فعلی زنجیر می‌شود (تولید بی‌نهایت)؛
   387→    // دکمهٔ اصلی همیشه از سؤال اصلی منبع می‌گیرد (تکرار همان سؤال → پاسخ کش‌شده).
   388→    const src = fromBonus && bonus ? bonus : current;
   389→    setBonusLoading(true);
   390→    try {
   391→      const res = await fetch("/api/similar-question", {
   392→        method: "POST",
   393→        headers: { "Content-Type": "application/json" },
   394→        body: JSON.stringify({
   395→          text: src.text,
   396→          options: src.options,
   397→          answer: src.answer,
   398→          category: current.category,
   399→          book: {
   400→            id: bookId,
   401→            title: bookData?.info.title ?? "",
   402→            grade: meta?.gradeLabel ?? "",
   403→          },
   404→        }),
   405→      });
   406→      const data = await res.json();
   407→      if (!res.ok || !data.question) {
   408→        toast.error(data.error ?? "سؤال مشابه ساخته نشد — دوباره تلاش کن");
   409→        return;
   410→      }
   411→      setBonus(data.question as SimilarBonus);
   412→      setBonusSelected(null);
   413→      setBonusRegen((n) => n + 1);
   414→    } catch {
   415→      toast.error("خطا در ارتباط با هوش مصنوعی");
   416→    } finally {
   417→      setBonusLoading(false);
   418→    }
   419→  }

   421→  function nextQuestion() {
   422→    if (advanceRef.current) clearTimeout(advanceRef.current);
   423→    if (index + 1 >= questions.length) {

   427→    setIndex((i) => i + 1);
   428→    setSelected(answers[index + 1]);

   550→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   551→  async function exportMistakesCsv() {
   552→    if (!mistakeIds.length || exporting) return;
   553→    setExporting(true);
   554→    try {
   555→      const res = await fetch(`/api/questions?count=${mistakeIds.length}&ids=${mistakeIds.join(",")}&book=${bookId}`);
   556→      const data = await res.json();
   557→      const qs = (data.questions ?? []) as Question[];
   558→      if (!qs.length) {
   559→        toast.error("سؤالی برای خروجی یافت نشد.");
   560→        return;
   561→      }
   562→      const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   563→      const rows = [
   564→        ["ردیف", "متن سؤال", "پاسخ صحیح", "دسته", "مبحث", "سطح"].join(","),
   565→        ...qs.map((q, i) =>
   566→          [
   567→            i + 1,
   568→            esc(q.text),
   569→            esc(q.options[q.answer] ?? ""),
   570→            q.category === "grammar" ? "گرامر" : "واژگان",
   571→            esc(q.scope),
   572→            esc(q.level),
   573→          ].join(","),
   574→        ),
   575→      ];
   576→      const blob = new Blob(["\uFEFF" + rows.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   577→      const url = URL.createObjectURL(blob);
   578→      const a = document.createElement("a");
   579→      a.href = url;
   580→      a.download = "vision-one-mistakes.csv";
   581→      document.body.appendChild(a);
   582→      a.click();
   583→      a.remove();
   584→      URL.revokeObjectURL(url);
   585→      toast.success(`فایل CSV با ${qs.length.toLocaleString("fa-IR")} سؤال دانلود شد.`);
   586→    } catch {
   587→      toast.error("خطا در ساخت فایل CSV");
   588→    } finally {
   589→      setExporting(false);
   590→    }
   591→  }
