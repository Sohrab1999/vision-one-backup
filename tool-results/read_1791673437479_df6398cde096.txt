    98→function pad2(n: number): string {
    99→  return n < 10 ? `0${n}` : String(n);
   100→}

   102→/** تاریخ امروز به شکل YYYY-MM-DD (تقویم محلی — پایدار بین رفرش‌ها) */
   103→function todayKey(): string {
   104→  const d = new Date();
   105→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}`;
   106→}
   107→
   108→/** جابه‌جایی تقویمی تاریخ YYYY-MM-DD (بدون وابستگی به منطقهٔ زمانی) */
   109→function shiftDate(date: string, days: number): string {
   110→  const parts = date.split("-").map(Number);
   111→  const t = new Date(Date.UTC(parts[0], (parts[1] ?? 1) - 1, parts[2] ?? 1) + days * 86400000);
   112→  return `${t.getUTCFullYear()}-${pad2(t.getUTCMonth() + 1)}-${pad2(t.getUTCDate())}`;
   113→}
   114→
   115→/** هش ساده و پایدار (djb2) برای انتخاب واژهٔ روز */
   116→function hashStr(s: string): number {
   117→  let h = 5381;
   118→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   119→  return Math.abs(h);
   120→}

   122→/**
   123→ * رنگ‌بندی دو-گذرهٔ وردل — گذر اول فقط سبزها را می‌شمارد، گذر دوم
   124→ * زرد/خاکستری را با شمارش حروف باقی‌ماندهٔ پاسخ تعیین می‌کند
   125→ * (پاسخ کلاسیک: answer "speed" × guess "erase" → Eسبز، Rخاکستری، Aخاکستری، Sسبز، Eزرد).
   126→ */
   127→function judgeVerdicts(guess: string, answer: string): Verdict[] {
   128→  const res: Verdict[] = Array<Verdict>(WORD_LEN).fill("absent");
   129→  const remaining: Record<string, number> = {};
   130→  for (let i = 0; i < WORD_LEN; i++) {
   131→    if (guess[i] === answer[i]) {
   132→      res[i] = "correct";
   133→    } else {
   134→      remaining[answer[i]] = (remaining[answer[i]] ?? 0) + 1;
   135→    }
   136→  }
   137→  for (let i = 0; i < WORD_LEN; i++) {
   138→    if (res[i] === "correct") continue;
   139→    const ch = guess[i];
   140→    if ((remaining[ch] ?? 0) > 0) {
   141→      res[i] = "present";
   142→      remaining[ch] -= 1;
   143→    }
   144→  }
   145→  return res;
   146→}

   181→function loadDailyState(bookId: string): DailyState | null {
   182→  try {
   183→    const raw = localStorage.getItem(dailyKeyOf(bookId));
   184→    if (!raw) return null;
   185→    const p = JSON.parse(raw) as Partial<DailyState> | null;
   186→    if (!p || typeof p.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(p.date)) return null;
   187→    if (p.status !== "playing" && p.status !== "won" && p.status !== "lost") return null;
   188→    if (!Array.isArray(p.guesses)) return null;
   189→    const guesses = p.guesses
   190→      .filter((g): g is string => typeof g === "string" && WORD_RE.test(g))
   191→      .slice(0, MAX_ROWS);
   192→    return { date: p.date, guesses, status: p.status };
   193→  } catch {
   194→    return null;
   195→  }
   196→}
   197→
   198→function writeDailyState(bookId: string, s: DailyState): void {
   199→  try {
   200→    localStorage.setItem(dailyKeyOf(bookId), JSON.stringify(s));
   201→  } catch {
   202→    /* noop */
   203→  }
   204→  notifyGameData();
   205→}

   207→function loadDailyStreak(bookId: string): DailyStreakState | null {
   208→  try {
   209→    const raw = localStorage.getItem(streakKeyOf(bookId));
   210→    if (!raw) return null;
   211→    const p = JSON.parse(raw) as Partial<DailyStreakState> | null;
   212→    if (!p || typeof p.date !== "string" || typeof p.streak !== "number") return null;
   213→    return { date: p.date, streak: p.streak };
   214→  } catch {
   215→    return null;
   216→  }
   217→}

   225→  notifyGameData();

   254→  const [dailyClosed, setDailyClosed] = useState(false);
   255→
   256→  const { speed } = useVoiceSpeed();
   257→  const locked = useRef(false);
   258→  const timers = useRef<number[]>([]);
   259→  const startedAt = useRef(0);
   260→
   261→  // ── استخر واژگان ۵حرفی — یکتا (lowercase) و مرتب (seed پایدار روزانه) ──
   262→  const pool = useMemo<VocabEntry[]>(() => {
   263→    const seen = new Set<string>();
   264→    const out: VocabEntry[] = [];
   265→    for (const w of data.vocab.all) {
   266→      if (!WORD_RE.test(w.word)) continue;
   267→      const key = w.word.toLowerCase();
   268→      if (seen.has(key)) continue;
   269→      seen.add(key);
   270→      out.push(w);
   271→    }
   272→    out.sort((a, b) => {
   273→      const x = a.word.toLowerCase();
   274→      const y = b.word.toLowerCase();
   275→      return x < y ? -1 : x > y ? 1 : 0;
   276→    });
   277→    return out;
   278→  }, [data.vocab]);
   279→
   280→  const poolSet = useMemo(() => new Set(pool.map((w) => w.word.toLowerCase())), [pool]);
   281→
   282→  /** واژهٔ روز — hash پایدار از `${bookId}:${YYYY-MM-DD}` روی استخر مرتب‌شده */
   283→  const dailyAnswer = useMemo(() => {
   284→    if (pool.length === 0) return "";
   285→    return pool[hashStr(`${activeBookId}:${todayKey()}`) % pool.length].word.toLowerCase();
   286→  }, [pool, activeBookId]);
   287→
   288→  // ── وضعیت روزانه از localStorage — استور بیرونی (بازیابی mount بدون setState در effect) ──

   291→    () => {
   292→      const s = loadDailyState(activeBookId);

   294→    },
   295→    () => "null",
   296→  );

   298→    () => (dailyJson === "null" ? null : (JSON.parse(dailyJson) as DailyState)),

   303→  const dailyToday = useMemo<DailyState | null>(() => {
   304→    if (!dailyRaw || dailyRaw.date !== todayKey()) return null;
   305→    if (pool.length === 0 || !dailyAnswer) return null;
   306→    if (dailyRaw.status === "playing") {
   307→      if (dailyRaw.guesses.some((g) => g === dailyAnswer)) return { ...dailyRaw, status: "won" };
   308→      if (dailyRaw.guesses.length >= MAX_ROWS) return { ...dailyRaw, status: "lost" };
   309→    }
   310→    return dailyRaw;
   311→  }, [dailyRaw, pool.length, dailyAnswer]);
   312→
   313→  // ── رکوردها و زنجیره — اسنپ‌شات رشته‌ای پایدار (الگوی match-game) ──
   314→  const metaJson = useSyncExternalStore(
   315→    subscribeGameData,
   316→    () =>
   317→      JSON.stringify({
   318→        infBest: loadBest<BestMoves>(activeBookId, GAME_KEY, "infinite"),
   319→        dailyBest: loadBest<BestScore>(activeBookId, GAME_KEY, "daily-streak"),
   320→        streak: loadDailyStreak(activeBookId),
   321→      }),
   322→    () => "null",
   323→  );
   324→  const meta = useMemo<{ infBest: BestMoves | null; dailyBest: BestScore | null; streak: DailyStreakState | null }>(
   325→    () =>
   326→      metaJson === "null"
   327→        ? { infBest: null, dailyBest: null, streak: null }
   328→        : (JSON.parse(metaJson) as { infBest: BestMoves | null; dailyBest: BestScore | null; streak: DailyStreakState | null }),
   329→    [metaJson],
   330→  );
   331→
   332→  // ── تفکیک نماها ──
   333→  /** روزانهٔ امروز فعال است (بازیابی mount یا شروع دستی) */
   334→  const dailyLive = mode === "daily" && !dailyClosed && dailyToday !== null;
   335→  /** نمای بازی روزانه — حتی حین انیمیشن حدس آخر (وضعیت استور هنوز «playing» یا فلیپ تمام‌نشده) */
   336→  const dailyPlayingView = dailyLive && (dailyToday.status === "playing" || revealRow !== -1);
   337→  const dailyDoneView = dailyLive && !dailyPlayingView;
   338→  const infinitePlaying = phase === "playing" && !dailyLive;
   339→
   340→  const activeAnswer = dailyLive ? dailyAnswer : answer;
   341→  const activeCommitted = dailyPlayingView ? dailyToday.guesses : guesses;
   342→
   343→  const answerEntry = useMemo<VocabEntry | null>(
   344→    () => pool.find((w) => w.word.toLowerCase() === activeAnswer) ?? null,
   345→    [pool, activeAnswer],
   346→  );
   347→
   348→  /** بهترین وضعیت هر حرف روی کیبورد — از همهٔ حدس‌های ثبت‌شده */
   349→  const keyStatus = useMemo(() => {
   350→    const map = new Map<string, Verdict>();
   351→    for (const g of activeCommitted) {
   352→      const vs = judgeVerdicts(g, activeAnswer);
   353→      for (let i = 0; i < WORD_LEN; i++) {
   354→        const prev = map.get(g[i]);
   355→        if (!prev || VERDICT_RANK[vs[i]] > VERDICT_RANK[prev]) map.set(g[i], vs[i]);
   356→      }
   357→    }
   358→    return map;
   359→  }, [activeCommitted, activeAnswer]);
   360→
   361→  // ── تایمرهای پله‌ای فلیپ — همیشه پاک‌شونده ──
   362→  const clearTimers = useCallback(() => {
   363→    for (const t of timers.current) window.clearTimeout(t);
   364→    timers.current = [];
   365→  }, []);
   366→
   367→  useEffect(() => {
   368→    return () => {
   369→      stopSpeaking();
   370→      for (const t of timers.current) window.clearTimeout(t);
   371→    };
   372→  }, []);
   373→
   374→  // ── پایان دور — استریک/جلسه/Attempt/SRS/رکورد ──
   375→  const finishGame = useCallback(
   376→    (win: boolean, rowsUsed: number, gameMode: Mode, word: string) => {
   377→      touchStreak();
   378→      recordStudySession();
   379→      bumpPlays(activeBookId, GAME_KEY);
   380→      const started = startedAt.current || Date.now();
   381→      startedAt.current = 0;
   382→      const duration = Math.max(1, Math.round((Date.now() - started) / 1000));
   383→      saveAttempt(activeBookId, GAME_KEY, {
   384→        lesson: 0, // وردل کتاب‌محور است (بدون انتخاب درس)
   385→        total: 1,
   386→        correct: win ? 1 : 0,
   387→        duration,
   388→        detail: { word, rows: rowsUsed, mode: gameMode },
   389→      });
   390→      recordSRS(word, win);
   391→      if (win) {
   392→        speak(word, { speed });
   393→        if (gameMode === "infinite") {
   394→          const prev = loadBest<BestMoves>(activeBookId, GAME_KEY, "infinite");
   395→          if (!prev || rowsUsed < prev.moves) {
   396→            saveBest<BestMoves>(activeBookId, GAME_KEY, "infinite", { moves: rowsUsed });
   397→            setIsRecord(true);
   398→            toast.success(`رکورد جدید! برد با ${faNum(rowsUsed)} حدس 🏆`);
   399→          } else {
   400→            toast.success(`واژه را با ${faNum(rowsUsed)} حدس پیدا کردی! 🎉`);
   401→          }
   402→        } else {
   403→          // زنجیرهٔ برد پیاپی روزانه — از localStorage خود بازی
   404→          const today = todayKey();
   405→          const prevStreak = loadDailyStreak(activeBookId);
   406→          const streak = prevStreak && prevStreak.date === shiftDate(today, -1) ? prevStreak.streak + 1 : 1;
   407→          saveDailyStreak(activeBookId, { date: today, streak });
   408→          const prevBest = loadBest<BestScore>(activeBookId, GAME_KEY, "daily-streak");
   409→          if (!prevBest || streak > prevBest.score) {
   410→            saveBest<BestScore>(activeBookId, GAME_KEY, "daily-streak", { score: streak });
   411→            setIsRecord(true);
   412→            toast.success(`رکورد زنجیره! ${faNum(streak)} برد پیاپی 🏆`);
   413→          } else {
   414→            toast.success(`واژهٔ امروز پیدا شد — زنجیرهٔ ${faNum(streak)} روزه! 🎉`);
   415→          }
   416→        }
   417→      } else {
   418→        toast.error("این‌بار پیدا نشد — واژهٔ درست را ببین");
   419→      }
   420→      if (gameMode === "infinite") setPhase("done"); // روزانه: نما از استور مشتق می‌شود
   421→    },
   422→    [activeBookId, speed],
   423→  );
   424→
   425→  // ── ارسال حدس ──
   426→  const submitGuess = useCallback(() => {
   427→    if (locked.current || !activeAnswer) return;
   428→    const rowIdx = activeCommitted.length;
