   134→/** چیدمان عمودی k دهانه در باند بازی‌پذیر — ارتفاع دهانه سقف‌دار، مازاد به دیواره‌ها می‌رود */
   135→function layoutGaps(k: number): Array<{ y0: number; y1: number }> {
   136→  const top = 34;
   137→  const bottom = H - GROUND_H - 12;
   138→  const avail = bottom - top;
   139→  const barMin = 34;
   140→  const gapCap = 150;
   141→  let gapH = (avail - (k - 1) * barMin) / k;
   142→  let barH = barMin;
   143→  if (gapH > gapCap) {
   144→    barH = barMin + ((gapH - gapCap) * k) / (k - 1);
   145→    gapH = gapCap;
   146→  }
   147→  const out: Array<{ y0: number; y1: number }> = [];
   148→  let y = top;
   149→  for (let i = 0; i < k; i++) {
   150→    out.push({ y0: y, y1: y + gapH });
   151→    y += gapH + barH;
   152→  }
   153→  return out;
   154→}

   156→function roundRectPath(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   157→  c.beginPath();
   158→  c.moveTo(x + r, y);
   159→  c.arcTo(x + w, y, x + w, y + h, r);
   160→  c.arcTo(x + w, y + h, x, y + h, r);
   161→  c.arcTo(x, y + h, x, y, r);
   162→  c.arcTo(x, y, x + w, y, r);
   163→  c.closePath();
   164→}

   167→function drawSky(c: CanvasRenderingContext2D, now: number, w: number) {
   168→  c.fillStyle = "rgba(45, 212, 191, 0.16)";
   169→  for (let i = 0; i < 3; i++) {
   170→    const cx = ((i * 270 + now * (0.012 + i * 0.005)) % (w + 220)) - 110;
   171→    const cy = 54 + i * 64;
   172→    c.beginPath();
   173→    c.arc(cx, cy, 22, 0, Math.PI * 2);
   174→    c.arc(cx + 25, cy + 6, 15, 0, Math.PI * 2);
   175→    c.arc(cx - 23, cy + 8, 13, 0, Math.PI * 2);
   176→    c.fill();
   177→  }
   178→  c.fillStyle = "#059669";
   179→  c.fillRect(0, 0, w, 4);
   180→}

   182→function drawGround(c: CanvasRenderingContext2D, now: number, w: number, speed: number) {
   183→  const gy = H - GROUND_H;

   193→/** دیوار emerald با کلاهک + گزینه‌ها با auto-shrink تا در دهانه جا شوند */
   194→function drawWall(c: CanvasRenderingContext2D, wall: FlappyWall, now: number) {
   195→  const segs: Array<[number, number]> = [];
   196→  let y = 0;
   197→  for (const g of wall.gaps) {
   198→    segs.push([y, g.y0]);
   199→    y = g.y1;
   200→  }
   201→  segs.push([y, H - GROUND_H]);
   202→  c.fillStyle = "#059669";
   203→  for (const [a, b] of segs) if (b > a) c.fillRect(wall.x, a, WALL_W, b - a);
   204→  c.fillStyle = "#047857";
   205→  for (const g of wall.gaps) {
   206→    c.fillRect(wall.x - 4, g.y0 - 10, WALL_W + 8, 10);
   207→    c.fillRect(wall.x - 4, g.y1, WALL_W + 8, 10);
   208→  }
   209→  c.textAlign = "center";
   210→  c.textBaseline = "middle";
   211→  for (const g of wall.gaps) {
   212→    const cy = (g.y0 + g.y1) / 2;
   213→    let fs = 15;
   214→    c.font = `800 ${fs}px system-ui, sans-serif`;
   215→    while (c.measureText(g.text).width > WALL_W - 16 && fs > 8) {
   216→      fs -= 1;
   217→      c.font = `800 ${fs}px system-ui, sans-serif`;
   218→    }
   219→    const tw = Math.min(c.measureText(g.text).width, WALL_W - 16);
   220→    c.fillStyle = "rgba(255, 255, 255, 0.92)";
   221→    roundRectPath(c, wall.x + WALL_W / 2 - tw / 2 - 8, cy - 11, tw + 16, 22, 8);
   222→    c.fill();
   223→    c.fillStyle = "#065f46";
   224→    c.fillText(g.text, wall.x + WALL_W / 2, cy + 1);
   225→  }
   226→  if (wall.flashUntil > now) {
   227→    if (wall.ok) {
   228→      const g = wall.gaps.find((gp) => gp.correct);
   229→      if (g) {
   230→        c.fillStyle = "rgba(16, 185, 129, 0.30)";
   231→        c.fillRect(wall.x - 4, g.y0, WALL_W + 8, g.y1 - g.y0);
   232→      }
   233→    } else {
   234→      c.fillStyle = "rgba(239, 68, 68, 0.22)";
   235→      c.fillRect(wall.x - 4, 0, WALL_W + 8, H - GROUND_H);
   236→    }
   237→  }
   238→}

   240→/** پرنده = دایره amber + بال/چشم/منقار ساده + چرخش بر اساس سرعت عمودی + چشمک مصونیت */
   241→function drawBird(
   242→  c: CanvasRenderingContext2D,
   243→  now: number,
   244→  y: number,
   245→  vy: number,
   246→  bx: number,
   247→  invulnUntil: number,
   248→  lastFlapAt: number,
   249→) {
   250→  c.save();
   251→  if (now < invulnUntil) c.globalAlpha = 0.45 + 0.3 * Math.abs(Math.sin(now / 60));
   252→  c.translate(bx, y);
   253→  c.rotate(Math.max(-0.38, Math.min(0.6, vy / 650)));
   254→  c.beginPath();
   255→  c.arc(0, 0, BIRD_R, 0, Math.PI * 2);
   256→  c.fillStyle = "#f59e0b";
   257→  c.fill();
   258→  c.lineWidth = 2.5;
   259→  c.strokeStyle = "#b45309";
   260→  c.stroke();
   261→  const flapT = Math.max(0, 1 - (now - lastFlapAt) / 350);
   262→  const wy = 3 + Math.sin(now / (flapT > 0 ? 45 : 110)) * (2.5 + flapT * 3);
   263→  c.beginPath();
   264→  c.ellipse(-4, wy, 6.5, 4.2, -0.5, 0, Math.PI * 2);
   265→  c.fillStyle = "#fcd34d";
   266→  c.fill();
   267→  c.stroke();
   268→  c.beginPath();
   269→  c.arc(5, -4.5, 3.4, 0, Math.PI * 2);
   270→  c.fillStyle = "#ffffff";
   271→  c.fill();
   272→  c.beginPath();
   273→  c.arc(6.3, -4.5, 1.6, 0, Math.PI * 2);
   274→  c.fillStyle = "#1c1917";
   275→  c.fill();
   276→  c.beginPath();
   277→  c.moveTo(11, -1);
   278→  c.lineTo(18.5, 2);
   279→  c.lineTo(11, 5.5);
   280→  c.closePath();
   281→  c.fillStyle = "#d97706";
   282→  c.fill();
   283→  c.restore();
   284→}

   366→  /** سؤال بعدی از دستهٔ یکتا — اتمام استخر → reshuffle و ادامه */
   367→  const nextQuestion = useCallback((): QuizQuestionSeed => {
   368→    if (deckIdxRef.current >= deckRef.current.length) {
   369→      deckRef.current = shuffle(deckRef.current);
   370→      deckIdxRef.current = 0;

   375→  }, []);

   378→  const makeWall = useCallback(
   379→    (x: number): FlappyWall => {
   380→      const q = nextQuestion();
   381→      const correctText = q.options[q.answer];
   382→      const k = 2 + Math.floor(Math.random() * 3);
   383→      const distractors = shuffle(q.options.filter((_, i) => i !== q.answer)).slice(0, k - 1);
   384→      const texts = shuffle([correctText, ...distractors]);
   385→      const slots = layoutGaps(k);

   387→        x,
   388→        gaps: slots.map((s, i) => ({
   389→          y0: s.y0,
   390→          y1: s.y1,
   391→          text: texts[i],
   392→          correct: texts[i] === correctText,
   393→        })),

   410→    speedRef.current = cfg.base;
   411→    birdRef.current = { y: H / 2, vy: 0 };
   412→    wallsRef.current = [];
   413→    popupsRef.current = [];
   414→    hudWallRef.current = null;
   415→    comboRef.current = 0;
   416→    scoreRef.current = 0;
   417→    heartsRef.current = HEARTS;
   418→    totalRef.current = 0;
   419→    correctRef.current = 0;
   420→    maxComboRef.current = 0;
   421→    wrongRef.current = [];
   422→    invulnUntilRef.current = 0;
   423→    flashUntilRef.current = 0;
   424→    lastFlapAtRef.current = 0;
   425→    startedAtRef.current = performance.now();
   426→    overRef.current = false;
   427→    const deck = pickUnique(pool, pool.length, (q) => q.text);
   428→    deckRef.current = deck;
   429→    deckIdxRef.current = 0;
   430→    // سؤال دیوار اول — دیوار واقعی در effect «playing» با nextQuestion از همین سرِ دسته زده می‌شود
   431→    setHudQ(deck.length > 0 ? deck[0] : null);
   432→    setScore(0);
   433→    setCombo(0);
   434→    setHearts(HEARTS);
   435→    setIsRecord(false);
   436→    setMaxCombo(0);
   437→    setCorrectCount(0);
   438→    setWrongCount(0);
   439→    setWrongWords([]);
   440→    setPhase("playing");
   441→  }, [pool, easy]);
   442→
   443→  /** پایان دور — فقط یک‌بار (overRef قفل)؛ ثبت streak/session/plays/attempt/best */
   444→  const finish = useCallback(() => {
   445→    if (overRef.current) return;
   446→    overRef.current = true;
   447→    const total = totalRef.current;
   448→    const finalScore = scoreRef.current;
   449→    const elapsed = Math.max(1, Math.round((performance.now() - startedAtRef.current) / 1000));
