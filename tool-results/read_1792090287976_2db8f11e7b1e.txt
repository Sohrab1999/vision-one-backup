   279→   * همان زمان باقی‌مانده دوباره مسلح می‌شود — بستن ریسک ثبت‌شدهٔ G0-05.

   281→  function scheduleChain(): void {
   282→    if (chainTimerRef.current !== null) {
   283→      window.clearTimeout(chainTimerRef.current);
   284→      chainTimerRef.current = null;
   285→    }
   286→    if (pausedRef.current) return;
   287→    const pend = chainPendingRef.current;
   288→    if (!pend) return;

   290→      chainTimerRef.current = null;
   291→      const cur = chainPendingRef.current;
   292→      chainPendingRef.current = null;
   293→      cur?.fn();
   294→    }, Math.max(0, pend.due - gameNow()));
   295→  }

   407→  /** شروع یک دور: سؤال بعدی + دو ترتیب گزینهٔ متفاوت + ریست تایمر و قفل‌ها */
   408→  function startRound(idx: number): void {
   409→    let deck = deckRef.current;
   410→    let i = idx;
   411→    if (i >= deck.length) {
   412→      // استخر تمام شد → از نو شافل و ادامه (چرخهٔ بی‌پایان مثل speed-quiz)
   413→      deck = shuffle(deck);

   415→      i = 0;
   416→    }

   418→    const nq = deck[i];
   419→    qRef.current = nq;
   420→    // ترتیب گزینه‌ها بین دو نیمه باید متفاوت باشد
   421→    const o1 = shuffle(BASE_ORDER);
   422→    let o2 = shuffle(BASE_ORDER);
   423→    let tries = 0;
   424→    while (o2.join(",") === o1.join(",") && tries < 8) {
   425→      o2 = shuffle(BASE_ORDER);
   426→      tries += 1;
   427→    }
   428→    p1OrderRef.current = o1;
   429→    p2OrderRef.current = o2;
   430→    p1PickRef.current = null;
   431→    p2PickRef.current = null;
   432→    p1MsRef.current = 0;
   433→    p2MsRef.current = 0;
   434→    judgedRef.current = false;
   435→    timeRef.current = ROUND_TIME;
   436→    roundStartRef.current = Date.now();
   437→    roundNoRef.current += 1;
   438→    setQ(nq);
   439→    setP1Order(o1);
   440→    setP2Order(o2);
   441→    setP1Pick(null);
   442→    setP2Pick(null);
   443→    setResult(null);
   444→    setTimeLeft(ROUND_TIME);
   445→    setRoundNum(roundNoRef.current);
   446→    // اعلام شروع دور برای فناوری‌های کمکی (فاز G1-05 — aria-live مرکزی)
   447→    setLiveMsg(
   448→      `دور ${faNum(roundNoRef.current)} از ${faNum(configRef.current.rounds)} شروع شد — هر دو بازیکن جواب بدهید`,
   449→    );
   450→  }

   452→  /** داوری دور — دقیقاً یک‌بار (قفل judgedRef): هر دو درست → سریع‌تر، یکی درست → همان، هیچ‌کدام → تساوی */
   453→  function judge(): void {
   454→    if (judgedRef.current || !qRef.current) return;
   455→    judgedRef.current = true;
   456→    const nq = qRef.current;
   457→    const a1 = p1PickRef.current;
   458→    const a2 = p2PickRef.current;
   459→    const c1 = a1 !== null && a1 === nq.answer;
   460→    const c2 = a2 !== null && a2 === nq.answer;
   461→    let w: RoundWinner;
   462→    if (c1 && c2) w = p1MsRef.current <= p2MsRef.current ? 0 : 1;
   463→    else if (c1) w = 0;
   464→    else if (c2) w = 1;
   465→    else w = -1;
   466→    // واژه‌های غلط‌زدهٔ واژگانی به صف مرور برمی‌گردند (گزینهٔ درست = خود واژهٔ انگلیسی)
   467→    if (nq.category === "vocabulary") {
   468→      if (a1 !== null && !c1) seedReviewFromQuiz(nq.options[nq.answer]);
   469→      if (a2 !== null && !c2) seedReviewFromQuiz(nq.options[nq.answer]);
   470→    }
   471→    if (w === 0) wins1Ref.current += 1;
   472→    else if (w === 1) wins2Ref.current += 1;
   473→    setWins1(wins1Ref.current);
   474→    setWins2(wins2Ref.current);
   475→    setResult({ c1, c2, winner: w });
   476→    // فاز G2-d — بازخورد صوتی/لرزشی per-بازیکن (طرح تفکیک شنیداری در سربرگ فایل):
   477→    // بازیکن ۱: درست=correct/غلط=wrong · بازیکن ۲: درست=win/غلط=shoot — بی‌پاسخ = غلط
   478→    playSound(c1 ? "correct" : "wrong");
   479→    buzz(c1 ? "success" : "error");
   480→    playSound(c2 ? "win" : "shoot");
   481→    buzz(c2 ? "success" : "error");
   482→    // اعلام نتیجهٔ دور برای فناوری‌های کمکی (فاز G1-05 — قانون سه‌کاناله)
   483→    setLiveMsg(
   484→      w === -1
   485→        ? `دور ${faNum(roundNoRef.current)}: تساوی — هیچ‌کدام درست جواب نداد`
   486→        : `دور ${faNum(roundNoRef.current)}: ${w === 0 ? configRef.current.p1 : configRef.current.p2} برندهٔ دور شد`,
   487→    );
   488→    // نمایش نتیجه ~۱.۲ ثانیه → دور بعد یا پایان مسابقه (زنجیرهٔ بی‌توقف — فریز حین توقف)

   490→      if (overRef.current) return;
   491→      if (qiRef.current + 1 >= configRef.current.rounds) finish();

   628→  // در بدنهٔ effect (react-hooks v6) — شمارش واقعی در timeRef که با شروع هر دور ریست می‌شود.
   629→  // فاز G0-05 — گارد visibilitychange/pageshow: در تب مخفی interval پاک می‌شود و
   630→  // بازگشت از bfcache هم پوشش داده می‌شود؛ unmount همیشه پاک می‌کند.
   631→  // فاز G2-d — گیت روی paused (توقف دستی یا auto-pause هوک): حین توقف interval وجود
   632→  // ندارد و بج زمان/نوار ثابت می‌مانند؛ این شنونده فقط «کمربند» است — توقف اصلی از
