   178→   178→/** خواندن پیشرفت از localStorage با اعتبارسنجی کامل شکل داده */
   179→   179→function readProgress(key: string): AdventureProgress | null {
   180→   180→  try {
   181→   181→    const raw = localStorage.getItem(key);
   182→   182→    if (!raw) return null;
   183→   183→    const p = JSON.parse(raw) as Partial<AdventureProgress>;
   184→   184→    if (
   185→   185→      !Array.isArray(p.stars) ||
   186→   186→      p.stars.length !== TOTAL_STAGES ||
   187→   187→      !p.stars.every((s) => typeof s === "number")
   188→   188→    ) {
   189→   189→      return null;
   190→   190→    }
   191→   191→    if (
   192→   192→      !Array.isArray(p.done) ||
   193→   193→      p.done.length !== TOTAL_STAGES ||
   194→   194→      !p.done.every((d) => typeof d === "boolean")
   195→   195→    ) {
   196→   196→      return null;
   197→   197→    }
   198→   198→    const order =
   199→   199→      Array.isArray(p.order) &&
   200→   200→      p.order.length === TOTAL_STAGES &&
   201→   201→      p.order.every((n) => typeof n === "number" && n >= 0 && n <= 3)
   202→   202→        ? (p.order as StageType[])
   203→   203→        : null;
   204→   204→    return { stars: p.stars, done: p.done, order };
   205→   205→  } catch {
   206→   206→    return null;
   207→   207→  }
   208→   208→}

   214→   214→    /* noop */
   215→   215→  }

   217→   217→}

   220→   220→function orderOf(p: AdventureProgress): StageType[] {

   270→   270→    },
   271→   271→    () => null,
   272→   272→  );
   273→   273→  const best = useMemo<BestRec | null>(
   274→   274→    () => (bestRaw ? (JSON.parse(bestRaw) as BestRec) : null),

   290→   290→

   295→   295→      timers.forEach((id) => window.clearTimeout(id));

   299→   299→
   300→   300→  /** setTimeout با ثبت در ref — همیشه در unmount پاک می‌شود */
   301→   301→  const later = useCallback((fn: () => void, ms: number) => {

   303→   303→      timersRef.current.delete(id);
   304→   304→      fn();

   322→   322→          done: Array<boolean>(TOTAL_STAGES).fill(false),
   323→   323→          order: [...cyc, ...cyc],
   324→   324→        });
   325→   325→        usedWords.current.clear();
   326→   326→        secsRef.current = 0;
   327→   327→        setIsRecord(false);
   328→   328→      }
   329→   329→      stopSpeaking();
   330→   330→      setChallenge(null);
   331→   331→      setPhase("map");
   332→   332→    },
   333→   333→    [data.vocab, lesson, progressKey],
   334→   334→  );
   335→   335→
   336→   336→  // ── ساخت و شروع یک مرحله ──
   337→   337→
   338→   338→  const startStage = useCallback(
   339→   339→    (idx: number) => {
   340→   340→      if (!progress) return;
   341→   341→      const pool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   342→   342→      const distinctMeanings = new Set(pool.map((w) => w.meaning));
   343→   343→      const type = orderOf(progress)[idx];
   344→   344→      const used = usedWords.current;
   345→   345→      // واژه‌های مصرف‌نشدهٔ این دور در اولویت‌اند؛ اگر تمام شد از کل استخر
   346→   346→      const fresh = pool.filter((w) => !used.has(baseWord(w.word)));
   347→   347→      const src = fresh.length > 0 ? fresh : pool;
   348→   348→      const draw = (n: number) => {
   349→   349→        const got = pickUnique(src, n, (w) => w.meaning);
   350→   350→        return got.length >= n ? got : pickUnique(pool, n, (w) => w.meaning);
   351→   351→      };
   352→   352→
   353→   353→      if (type === 0) {
   354→   354→        // آزمون واژه — ۳ سؤال چهارگزینه‌ای معنی
   355→   355→        if (distinctMeanings.size < 4) {
   356→   356→          toast.error("برای «آزمون واژه» حداقل ۴ معنی متمایز لازم است — درس دیگری را انتخاب کن");
   357→   357→          return;
   358→   358→        }
   359→   359→        const picks = draw(3);
   360→   360→        if (picks.length < 3) {
   361→   361→          toast.error("برای «آزمون واژه» حداقل ۴ معنی متمایز لازم است — درس دیگری را انتخاب کن");
   362→   362→          return;
   363→   363→        }
   364→   364→        const questions: QuizQ[] = picks.map((w) => {
   365→   365→          const distractors = pickUnique(
   366→   366→            pool.filter((d) => d.meaning !== w.meaning),
   367→   367→            3,
   368→   368→            (d) => d.meaning,
   369→   369→          ).map((d) => d.meaning);
   370→   370→          const options = shuffle([w.meaning, ...distractors]);
   371→   371→          return { word: baseWord(w.word), meaning: w.meaning, options, answer: options.indexOf(w.meaning) };
   372→   372→        });
   373→   373→        picks.forEach((w) => used.add(baseWord(w.word)));
   374→   374→        setChallenge({ ...blankChallenge(idx, type), questions });
   375→   375→        setPhase("challenge");
   376→   376→        return;
   377→   377→      }
   378→   378→
   379→   379→      if (type === 1) {
   380→   380→        // حروف جابه‌جا — ۱ واژه با tiles لمسی
   381→   381→        const latin = src.filter((w) => WORD_RE.test(baseWord(w.word)));
   382→   382→        const source = latin.length > 0 ? latin : pool.filter((w) => WORD_RE.test(baseWord(w.word)));
   383→   383→        const pick = pickUnique(source, 1, (w) => baseWord(w.word))[0];
   384→   384→        if (!pick) {
   385→   385→          toast.error("واژهٔ انگلیسی مناسبی برای «حروف جابه‌جا» پیدا نشد");
   386→   386→          return;
   387→   387→        }
   388→   388→        const base = baseWord(pick.word);
   389→   389→        used.add(base);
   390→   390→        const tiles = shuffle(base.split("").map((letter, i) => ({ id: i, letter })));
   391→   391→        // جلوگیری از چیدمان اولیهٔ تصادفیِ درست
   392→   392→        if (base.length > 1 && tiles.map((t) => t.letter).join("") === base) {
   393→   393→          const first = tiles[0];
   394→   394→          tiles[0] = tiles[1];
   395→   395→          tiles[1] = first;
   396→   396→        }
   397→   397→        setChallenge({
   398→   398→          ...blankChallenge(idx, type),
   399→   399→          word: base,
   400→   400→          meaning: pick.meaning,
   401→   401→          tiles,
   402→   402→          slots: Array.from({ length: base.length }, () => null as number | null),
   403→   403→        });
   404→   404→        setPhase("challenge");
   405→   405→        return;
   406→   406→      }
   407→   407→
   408→   408→      if (type === 2) {
