    69→function useWordOfDay(vocab: VocabEntry[]) {
    70→  return useMemo(() => {
    71→    const day = Math.floor(Date.now() / 86400000);
    72→    return vocab[day % vocab.length];
    73→  }, [vocab]);
    74→}
