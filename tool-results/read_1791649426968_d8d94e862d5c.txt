   258→   258→function fmtDuration(sec: number): string {
   259→   259→  if (!sec) return "—";
   260→   260→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   261→   261→  const m = Math.floor(sec / 60);
   262→   262→  const s = sec % 60;
   263→   263→  if (m === 0) return `${fa(s)} ثانیه`;
   264→   264→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   265→   265→}

   270→   270→  } catch {
   271→   271→    return "";
   272→   272→  }
