   244→function fmtDuration(sec: number): string {
   245→  if (!sec) return "—";
   246→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   247→  const m = Math.floor(sec / 60);
   248→  const s = sec % 60;
   249→  if (m === 0) return `${fa(s)} ثانیه`;
   250→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   251→}
   252→
   253→function fmtDate(iso: string): string {
   254→  try {
   255→    return new Date(iso).toLocaleDateString("fa-IR", { month: "short", day: "numeric" });
   256→  } catch {
   257→    return "";
   258→  }
   259→}
