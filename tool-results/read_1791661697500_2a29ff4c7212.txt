   109→/**
   110→ * موقعیت شبکهٔ هر خانه — شماره‌گذاری مارپیچی کلاسیک:
   111→ * ردیف پایین (b = 0) از چپ بالا می‌رود، ردیف‌های فرد از راست (زیگزاگ).
   112→ * چیدمان RTL است، پس ستون شبکه = ۱۰ − فاصله از چپ؛
   113→ * مثلاً خانهٔ ۱ → col 10/row 10 (پایین-چپ) و خانهٔ ۱۰۰ → col 10/row 1 (بالا-چپ).
   114→ */
   115→const CELL_POS: { col: number; row: number }[] = Array.from({ length: 100 }, (_, i) => {
   116→  const n = i + 1;
   117→  const b = Math.floor((n - 1) / 10);
   118→  const fromLeft = b % 2 === 0 ? (n - 1) % 10 : 9 - ((n - 1) % 10);
   119→  return { col: 10 - fromLeft, row: 9 - b + 1 };
   120→});

   126→/** یک گزینهٔ غلط تصادفی برای جواب ربات */
   127→function wrongIndexOf(q: QuizQuestionSeed): number {
   128→  const wrongs = [0, 1, 2, 3].filter((i) => i !== q.answer);
   129→  return wrongs[Math.floor(Math.random() * wrongs.length)] ?? q.answer;
   130→}

   153→  /** بازخورد پاسخ — گزینهٔ انتخابی + درست/غلط (ربات یا انسان) */

   162→
   163→  // ── refs — منبع حقیقت منطق بازی (همهٔ callbackهای تایمر از این‌ها می‌خوانند) ──

   172→  /** قفل نوبت تا پایان کامل جریان (پاسخ → تاس → حرکت → مار/پله) */

   174→  /** زنجیرهٔ همهٔ setTimeoutهای جریان نوبت — در انصراف/unmount پاک می‌شود */
   175→  const timersRef = useRef<number[]>([]);

   209→  /** زمان‌بندی داخل زنجیرهٔ ref‌شده — بعد از اجرا خودش را از فهرست حذف می‌کند */
   210→  function later(fn: () => void, ms: number): void {
   211→    const t = window.setTimeout(() => {
   212→      timersRef.current = timersRef.current.filter((x) => x !== t);
   213→      fn();
   214→    }, ms);
   215→    timersRef.current.push(t);
   216→  }
   217→
   218→  function clearTimers(): void {
   219→    for (const t of timersRef.current) window.clearTimeout(t);
   220→    timersRef.current = [];
   221→  }
   222→
   223→  // ── جریان بازی ──
   224→
   225→  function drawNext(): void {
   226→    let deck = deckRef.current;
   227→    let next = qiRef.current + 1;
   228→    if (next >= deck.length) {
   229→      // استخر سؤال‌ها تمام شد → از نو شافل و ادامه
   230→      deck = shuffle(deck);
   231→      deckRef.current = deck;
   232→      next = 0;
   233→    }
   234→    qiRef.current = next;
   235→    qRef.current = deck[next];
   236→    setQ(deck[next]);
   237→    setQNum((n) => n + 1);
   238→  }
   239→
   240→  function bumpStats(p: number, ok: boolean): void {
   241→    const s = statsRef.current[p];
   242→    s.asked += 1;
   243→    if (ok) s.correct += 1;
   244→    else s.wrong += 1;
   245→    setStats(statsRef.current.map((x) => ({ ...x })));
   246→  }
   247→
   248→  function beginTurn(p: number): void {
   249→    curRef.current = p;
   250→    setCur(p);
   251→    setReveal(null);
   252→    setDiceFace(null);
   253→    turnLockRef.current = false;
   254→    drawNext();
   255→    if (playersRef.current[p]?.isBot) {
   256→      later(() => botAnswer(p), BOT_DELAY);
   257→    }
   258→  }
   259→
   260→  /** تاس: چند چرخش نمایشی → عدد قطعی → شروع حرکت مهره */
   261→  function proceedAfterCorrect(p: number): void {
   262→    const d = 1 + Math.floor(Math.random() * 6);
   263→    let flips = 0;
   264→    function cycle(): void {
   265→      flips += 1;
   266→      if (flips <= DICE_FLIPS) {
   267→        setDiceFace(1 + Math.floor(Math.random() * 6));
   268→        setDicePop((k) => k + 1);
   269→        later(cycle, 90);
   270→      } else {
   271→        setDiceFace(d);
   272→        setDicePop((k) => k + 1);
   273→        later(() => hop(p, d), 420);
   274→      }
   275→    }
   276→    later(cycle, 350);
   277→  }
   278→
   279→  /** حرکت مهره خانه‌به‌خانه با زنجیرهٔ setTimeout پله‌ای (~۱۲۰ms) */
   280→  function hop(p: number, d: number): void {
   281→    const from = posRef.current[p];
   282→    const to = Math.min(100, from + d);
   283→    let at = from;
   284→    function step(): void {
   285→      if (at < to) {
   286→        at += 1;
   287→        posRef.current[p] = at;
   288→        setPositions([...posRef.current]);
   289→        later(step, STEP_MS);
   290→      } else {
   291→        land(p, d);
   292→      }
   293→    }
   294→    step();
   295→  }
   296→
   297→  /** بررسی مار/پله در خانهٔ فرود و اعلام انتقال با toast */
   298→  function land(p: number, d: number): void {
   299→    const pos = posRef.current[p];
   300→    const snakeTo = SNAKES[pos];
   301→    const ladderTo = LADDERS[pos];
   302→    if (snakeTo !== undefined) {
   303→      toast.error(`مار! برگشتی به خانهٔ ${faNum(snakeTo)}`);
   304→      later(() => {
   305→        posRef.current[p] = snakeTo;
   306→        setPositions([...posRef.current]);
   307→        afterJump(p, d);
   308→      }, 550);
   309→    } else if (ladderTo !== undefined) {
   310→      toast.success(`پله! رفتی به خانهٔ ${faNum(ladderTo)}`);
   311→      later(() => {
   312→        posRef.current[p] = ladderTo;
   313→        setPositions([...posRef.current]);
   314→        afterJump(p, d);
   315→      }, 550);
   316→    } else {
   317→      afterJump(p, d);
   318→    }
   319→  }
   320→
   321→  /** بعد از مار/پله: پایان بازی یا نوبت بعدی (تاس ۶ = نوبت اضافه) */
   322→  function afterJump(p: number, d: number): void {
   323→    if (posRef.current[p] >= 100) {
   324→      finish(p);
   325→      return;
   326→    }
   327→    if (d === 6) {
   328→      toast.success(`تاس ${faNum(6)} — نوبت اضافه برای ${playersRef.current[p]?.name ?? ""}!`);
   329→      beginTurn(p);
   330→    } else {
   331→      beginTurn((p + 1) % playersRef.current.length);
   332→    }
   333→  }
   334→
   335→  /** پاسخ ربات — نمایش گزینهٔ انتخابی‌اش و طی همان جریان بازیکن انسانی */
   336→  function botAnswer(p: number): void {
   337→    const cq = qRef.current;
   338→    if (!cq) return;
   339→    const ok = Math.random() < BOT_SKILL;
   340→    const idx = ok ? cq.answer : wrongIndexOf(cq);
   341→    setReveal({ picked: idx, ok, byBot: true });
   342→    bumpStats(p, ok);
   343→    if (ok) {
   344→      later(() => proceedAfterCorrect(p), 600);
   345→    } else {
   346→      toast.error(`${playersRef.current[p]?.name ?? "ربات"} بند شد!`);
   347→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   348→    }
   349→  }
   350→
   351→  function answerHuman(idx: number): void {
   352→    if (turnLockRef.current || !q) return;
   353→    const p = curRef.current;
   354→    if (playersRef.current[p]?.isBot) return;
   355→    turnLockRef.current = true;
   356→    const ok = idx === q.answer;
   357→    setReveal({ picked: idx, ok, byBot: false });
   358→    bumpStats(p, ok);
   359→    if (ok) {
   360→      later(() => proceedAfterCorrect(p), 350);
   361→    } else {
   362→      toast.error("بند شدی!");
   363→      later(() => beginTurn((p + 1) % playersRef.current.length), 1100);
   364→    }
   365→  }
   366→
   367→  function buildPlayers(): Player[] {
   368→    if (mode === "bot") {
   369→      return [
   370→        { name: names[0]?.trim() || "تو", isBot: false, ci: 0 },
   371→        { name: "ربات", isBot: true, ci: 1 },
   372→      ];
   373→    }
   374→    return Array.from({ length: count }, (_, i) => ({
   375→      name: names[i]?.trim() || `بازیکن ${faNum(i + 1)}`,
   376→      isBot: false,
   377→      ci: i,
   378→    }));
   379→  }

   381→  function start(): void {
   382→    if (pool.length < MIN_POOL) {
   383→      toast.error("سؤال کافی در این محدوده نیست — محدودهٔ دیگری را انتخاب کن");
   384→      return;
   385→    }
   386→    clearTimers();
   387→    const ps = buildPlayers();
   388→    playersRef.current = ps;
   389→    setPlayers(ps);
   390→    posRef.current = ps.map(() => 0);
   391→    setPositions([...posRef.current]);
   392→    statsRef.current = ps.map(() => ({ asked: 0, correct: 0, wrong: 0 }));
   393→    setStats(statsRef.current.map((x) => ({ ...x })));
   394→    // سؤال یکتا روی متن — کل استخر شافل‌شده؛ تمام شد؟ در drawNext از نو شافل می‌شود
   395→    deckRef.current = pickUnique(pool, pool.length, (item) => item.text);
   396→    qiRef.current = -1;
   397→    curRef.current = 0;
   398→    secsRef.current = 0;
   399→    setSecs(0);
   400→    setQNum(0);
   401→    setReveal(null);
   402→    setDiceFace(null);
   403→    setDicePop(0);
   404→    setWinner(null);
   405→    setIsRecord(false);
   406→    setPhase("playing");
   407→    beginTurn(0);
   408→  }
   409→
   410→  function finish(winnerIdx: number): void {
   411→    clearTimers();
   412→    turnLockRef.current = true;
   413→    const spent = secsRef.current;
   414→    setSecs(spent);
   415→    touchStreak();
   416→    recordStudySession();
   417→    bumpPlays(activeBookId, GAME_KEY);
   418→    const ps = playersRef.current;
   419→    // آمار Attempt فقط از بازیکن‌های انسانی (total = سؤال‌های پرسیده‌شده از انسان‌ها)
   420→    const humanIdxs = ps.map((_, i) => i).filter((i) => !ps[i]?.isBot);
   421→    const total = humanIdxs.reduce((a, i) => a + (statsRef.current[i]?.asked ?? 0), 0);
   422→    const correct = humanIdxs.reduce((a, i) => a + (statsRef.current[i]?.correct ?? 0), 0);
   423→    if (total > 0) {
   424→      saveAttempt(activeBookId, GAME_KEY, {
   425→        lesson: lesson === ALL_LESSON ? 0 : lesson,
   426→        total,
   427→        correct,
   428→        duration: spent,
   429→        detail: {
   430→          winner: ps[winnerIdx]?.name ?? "",
   431→          players: ps.map((pl, i) => ({
   432→            name: pl.name,
   433→            isBot: pl.isBot,
   434→            asked: statsRef.current[i]?.asked ?? 0,
   435→            correct: statsRef.current[i]?.correct ?? 0,
   436→            wrong: statsRef.current[i]?.wrong ?? 0,
   437→            position: posRef.current[i] ?? 0,
   438→          })),
   439→        },
   440→      });
   441→    }
   442→    // رکورد سریع‌ترین پیروزی — فقط برندهٔ انسانی رکورد می‌سازد
   443→    if (!ps[winnerIdx]?.isBot) {
   444→      const prev = loadBest<BestRecord>(activeBookId, GAME_KEY, modeScope);
   445→      if (!prev || spent < prev.time) {
   446→        saveBest(activeBookId, GAME_KEY, modeScope, { time: spent });
   447→        setIsRecord(true);
   448→      }
