   138→   138→/** واژه هدف بدون پرانتز توضیح — «order (n)» → «order» */

   151→   151→/** همه جمله‌های کوتاه کتاب: متن‌ها (پاراگراف‌ها) + مکالمه‌ها (خط‌ها) */
   152→   152→function collectSentences(data: BookData): SourceSentence[] {
   153→   153→  const out: SourceSentence[] = [];
   154→   154→  for (const r of data.readings) {
   155→   155→    for (const p of r.paragraphs) {
   156→   156→      for (const s of p.en.split(/(?<=[.!?])\s+/)) {
   157→   157→        const t = s.trim();
   158→   158→        if (t) out.push({ en: t, fa: p.fa });
   159→   159→      }
   160→   160→    }
   161→   161→  }
   162→   162→  for (const c of data.conversations) {
   163→   163→    for (const line of c.lines) {
   164→   164→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   165→   165→        const t = s.trim();
   166→   166→        if (t) out.push({ en: t, fa: line.fa });
   167→   167→      }
   168→   168→    }
   169→   169→  }
   170→   170→  return out;
   171→   171→}

   175→   175→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   176→   176→ */
   177→   177→function buildPool(data: BookData, lesson: number): ClozeItem[] {
   178→   178→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   179→   179→  const sentences = collectSentences(data);
   180→   180→
   181→   181→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   182→   182→  const seenBase = new Set<string>();
   183→   183→  const words: { base: string; entry: VocabEntry }[] = [];
   184→   184→  for (const entry of vocabPool) {
   185→   185→    const base = toBase(entry.word);
   186→   186→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   187→   187→    seenBase.add(base);
   188→   188→    words.push({ base, entry });
   189→   189→  }
   190→   190→
   191→   191→  const items: ClozeItem[] = [];
   192→   192→  for (const { base, entry } of words) {
   193→   193→    if (items.length >= MAX_ITEMS) break;
   194→   194→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   195→   195→    const found = sentences.find((s) => {
   196→   196→      if (s.en.length < 30 || s.en.length > 160) return false;
   197→   197→      const wc = wordCount(s.en);
   198→   198→      if (wc < 6 || wc > 35) return false;
   199→   199→      return re.test(s.en);
   200→   200→    });
   201→   201→    if (!found) continue;
   202→   202→
   203→   203→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   204→   204→    const distractorPool = vocabPool.filter((w) => {
   205→   205→      const b = toBase(w.word);
   206→   206→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   207→   207→    });
   208→   208→    const distractors = pickUnique(distractorPool, 2, (w) => `${toBase(w.word)}|${w.meaning}`);
   209→   209→    if (distractors.length < 2) continue;
   210→   210→
   211→   211→    items.push({
   212→   212→      id: `${lesson}:${base}`,
   213→   213→      sentence: found.en.replace(re, BLANK),
   214→   214→      fa: found.fa,
   215→   215→      answer: base,
   216→   216→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   217→   217→      meaning: entry.meaning,
   218→   218→      lessonLabel: entry.lessonLabel,
   219→   219→    });
   220→   220→  }
   221→   221→  return shuffle(items);
   222→   222→}

   229→   229→function readBest(bookId: string, scope: string): BestRecord | null {
   230→   230→  let raw: string | null = null;
   231→   231→  try {
   232→   232→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   233→   233→  } catch {
   234→   234→    return null;
   235→   235→  }
   236→   236→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   237→   237→  if (cacheKey === bestCacheKey) return bestCacheValue;
   238→   238→  let value: BestRecord | null = null;
   239→   239→  if (raw) {
   240→   240→    try {
   241→   241→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   242→   242→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   243→   243→        value = { score: parsed.score, total: parsed.total };
   244→   244→      }
   245→   245→    } catch {
   246→   246→      value = null;
   247→   247→    }
   248→   248→  }
   249→   249→  bestCacheKey = cacheKey;
   250→   250→  bestCacheValue = value;
   251→   251→  return value;
   252→   252→}
