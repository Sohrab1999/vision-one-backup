   181→/** چیدمان عمودی k دهانه در باند بازی‌پذیر — ارتفاع دهانه سقف‌دار، مازاد به دیواره‌ها می‌رود */
   182→function layoutGaps(k: number): Array<{ y0: number; y1: number }> {
   183→  const top = 34;
   184→  const bottom = H - GROUND_H - 12;
   185→  const avail = bottom - top;
   186→  const barMin = 34;
   187→  const gapCap = 150;
   188→  let gapH = (avail - (k - 1) * barMin) / k;
   189→  let barH = barMin;
   190→  if (gapH > gapCap) {
   191→    barH = barMin + ((gapH - gapCap) * k) / (k - 1);
   192→    gapH = gapCap;
   193→  }
   194→  const out: Array<{ y0: number; y1: number }> = [];
   195→  let y = top;
   196→  for (let i = 0; i < k; i++) {
   197→    out.push({ y0: y, y1: y + gapH });
   198→    y += gapH + barH;
   199→  }
   200→  return out;
   201→}

   203→function roundRectPath(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   204→  c.beginPath();
   205→  c.moveTo(x + r, y);
   206→  c.arcTo(x + w, y, x + w, y + h, r);
   207→  c.arcTo(x + w, y + h, x, y + h, r);
   208→  c.arcTo(x, y + h, x, y, r);
   209→  c.arcTo(x, y, x + w, y, r);
   210→  c.closePath();
   211→}

   240→/** دیوار emerald با کلاهک + گزینه‌ها با auto-shrink تا در دهانه جا شوند */
   241→function drawWall(c: CanvasRenderingContext2D, wall: FlappyWall, now: number) {
   242→  const segs: Array<[number, number]> = [];
   243→  let y = 0;
   244→  for (const g of wall.gaps) {
   245→    segs.push([y, g.y0]);
   246→    y = g.y1;
   247→  }
   248→  segs.push([y, H - GROUND_H]);
   249→  c.fillStyle = "#059669";
   250→  for (const [a, b] of segs) if (b > a) c.fillRect(wall.x, a, WALL_W, b - a);
   251→  c.fillStyle = "#047857";
   252→  for (const g of wall.gaps) {
   253→    c.fillRect(wall.x - 4, g.y0 - 10, WALL_W + 8, 10);
   254→    c.fillRect(wall.x - 4, g.y1, WALL_W + 8, 10);
   255→  }
   256→  c.textAlign = "center";
   257→  c.textBaseline = "middle";
   258→  for (const g of wall.gaps) {
   259→    const cy = (g.y0 + g.y1) / 2;
   260→    let fs = 15;
   261→    c.font = `800 ${fs}px system-ui, sans-serif`;
   262→    while (c.measureText(g.text).width > WALL_W - 16 && fs > 8) {
   263→      fs -= 1;
   264→      c.font = `800 ${fs}px system-ui, sans-serif`;
   265→    }
   266→    const tw = Math.min(c.measureText(g.text).width, WALL_W - 16);
   267→    c.fillStyle = "rgba(255, 255, 255, 0.92)";
   268→    roundRectPath(c, wall.x + WALL_W / 2 - tw / 2 - 8, cy - 11, tw + 16, 22, 8);
   269→    c.fill();
   270→    c.fillStyle = "#065f46";
   271→    c.fillText(g.text, wall.x + WALL_W / 2, cy + 1);
   272→  }
   273→  if (wall.flashUntil > now) {
   274→    if (wall.ok) {
   275→      const g = wall.gaps.find((gp) => gp.correct);
   276→      if (g) {
   277→        c.fillStyle = "rgba(16, 185, 129, 0.30)";
   278→        c.fillRect(wall.x - 4, g.y0, WALL_W + 8, g.y1 - g.y0);
   279→      }
   280→    } else {
   281→      c.fillStyle = "rgba(239, 68, 68, 0.22)";
   282→      c.fillRect(wall.x - 4, 0, WALL_W + 8, H - GROUND_H);
   283→    }
   284→  }
   285→}
