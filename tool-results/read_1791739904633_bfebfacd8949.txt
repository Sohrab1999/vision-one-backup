   176→/** واژه هدف بدون پرانتز توضیح — «order (n)» → «order» */
   177→function toBase(word: string): string {
   178→  return word.replace(/\s*\([^)]*\)\s*/g, " ").trim();
   179→}
   180→
   181→function escapeRe(s: string): string {
   182→  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   183→}
   184→
   185→function wordCount(s: string): number {
   186→  return s.trim().split(/\s+/).length;
   187→}

   189→/** همه جمله‌های کوتاه کتاب: متن‌ها (پاراگراف‌ها) + مکالمه‌ها (خط‌ها) */
   190→function collectSentences(data: BookData): SourceSentence[] {
   191→  const out: SourceSentence[] = [];
   192→  for (const r of data.readings) {
   193→    for (const p of r.paragraphs) {
   194→      for (const s of p.en.split(/(?<=[.!?])\s+/)) {
   195→        const t = s.trim();
   196→        if (t) out.push({ en: t, fa: p.fa });
   197→      }
   198→    }
   199→  }
   200→  for (const c of data.conversations) {
   201→    for (const line of c.lines) {
   202→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   203→        const t = s.trim();
   204→        if (t) out.push({ en: t, fa: line.fa });
   205→      }
   206→    }
   207→  }
   208→  return out;
   209→}

   211→/**
   212→ * هسته ساخت پازل — تابع خالص: برای هر واژه (بدون تکرار) اولین جمله مناسب کتاب را
   213→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   214→ */
   215→function buildPool(data: BookData, lesson: number, distractorCount = 2): ClozeItem[] {
   216→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   217→  const sentences = collectSentences(data);
   218→
   219→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   220→  const seenBase = new Set<string>();
   221→  const words: { base: string; entry: VocabEntry }[] = [];
   222→  for (const entry of vocabPool) {
   223→    const base = toBase(entry.word);
   224→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   225→    seenBase.add(base);
   226→    words.push({ base, entry });
   227→  }
   228→
   229→  const items: ClozeItem[] = [];
   230→  for (const { base, entry } of words) {
   231→    if (items.length >= MAX_ITEMS) break;
   232→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   233→    const found = sentences.find((s) => {
   234→      if (s.en.length < 30 || s.en.length > 160) return false;
   235→      const wc = wordCount(s.en);
   236→      if (wc < 6 || wc > 35) return false;
   237→      return re.test(s.en);
   238→    });
   239→    if (!found) continue;
   240→
   241→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   242→    // (فاز G3-01 — تعداد حواس‌پرتی سطح‌دار؛ پیش‌فرض ۲ = همان رفتار قبلی)
   243→    const distractorPool = vocabPool.filter((w) => {
   244→      const b = toBase(w.word);
   245→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   246→    });
   247→    const distractors = pickUnique(distractorPool, distractorCount, (w) => `${toBase(w.word)}|${w.meaning}`);
   248→    if (distractors.length < distractorCount) continue;
   249→
   250→    items.push({
   251→      id: `${lesson}:${base}`,
   252→      sentence: found.en.replace(re, BLANK),
   253→      fa: found.fa,
   254→      answer: base,
   255→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   256→      meaning: entry.meaning,
   257→      lessonLabel: entry.lessonLabel,
   258→    });
   259→  }
   260→  return shuffle(items);
   261→}

   282→/** گارد شکل سؤال تولیدی — دفاع دوم روی والیدیشن سرور (بدون any) */
   283→function isAiQuestionShape(v: unknown): v is AiQuestion {
   284→  if (typeof v !== "object" || v === null) return false;
   285→  const o = v as Record<string, unknown>;
   286→  return (
   287→    typeof o.word === "string" &&
   288→    o.word.trim().length > 0 &&
   289→    typeof o.sentence === "string" &&
   290→    o.sentence.trim().length > 0 &&
   291→    Array.isArray(o.options) &&
   292→    o.options.length >= 2 &&
   293→    o.options.length <= 6 &&
   294→    o.options.every((x) => typeof x === "string" && x.trim().length > 0) &&
   295→    typeof o.answerIndex === "number" &&
   296→    Number.isInteger(o.answerIndex) &&
   297→    o.answerIndex >= 0 &&
   298→    o.answerIndex < o.options.length
   299→  );
   300→}

   302→/**
   303→ * صدا زدن سؤال‌ساز با مهلت ۸ ثانیه (AbortController + Promise.race با تایمر).
   304→ * هر خطا/تایم‌اوت/پاسخ بد → null (بدون پرتاب استثنا؛ دور عادی ادامه می‌یابد).
   305→ */
   306→async function fetchAiQuestions(
   307→  game: "cloze" | "speed-quiz",
   308→  pool: { word: string; meaning: string }[],
   309→  ctrl: AbortController,
   310→): Promise<AiQuestion[] | null> {
   311→  let timerId: number | null = null;
   312→  const timeout = new Promise<null>((resolve) => {
   313→    timerId = window.setTimeout(() => {
   314→      ctrl.abort();
   315→      resolve(null);
   316→    }, AI_TIMEOUT_MS);
   317→  });
   318→  try {
   319→    const resPromise = fetch("/api/game-questions", {
   320→      method: "POST",
   321→      headers: { "Content-Type": "application/json" },
   322→      body: JSON.stringify({ game, pool, count: AI_COUNT }),
   323→      signal: ctrl.signal,
   324→    })
   325→      .then(async (res) => (res.ok ? ((await res.json()) as { questions?: unknown }) : null))
   326→      .catch(() => null); // لغو/شبکه → بی‌صدا (مسابقه منتظر نمی‌ماند)
   327→    const res = await Promise.race([resPromise, timeout]);
   328→    const raw = res && typeof res === "object" ? (res as { questions?: unknown }).questions : null;
   329→    if (!Array.isArray(raw)) return null;
   330→    const valid = raw.filter(isAiQuestionShape);
   331→    return valid.length > 0 ? valid : null;
   332→  } catch {
   333→    return null;
   334→  } finally {
   335→    if (timerId !== null) window.clearTimeout(timerId);
   336→  }
   337→}

   339→/**
   340→ * تبدیل سؤال‌های تولیدی به پرسش‌های بازی جای خالی — دفاع دوم سند: واژه باید
   341→ * دقیقاً عضو استخر همین دور باشد و گزینهٔ درست عین همان واژه؛ جای خالیِ «___»
   342→ * به قالب خود بازی (BLANK) تبدیل و گزینه‌ها هم‌تعداد سطح دور می‌شوند. سؤال‌های
   343→ * مشکوک بی‌صدا رد می‌شوند (دور عادی جبران می‌کند).
   344→ */
   345→function aiQuestionsToCloze(
   346→  questions: AiQuestion[],
   347→  pool: ClozeItem[],
   348→  optCount: number,
   349→): ClozeItem[] {
   350→  const byWord = new Map(pool.map((it) => [it.answer.toLowerCase(), it]));
   351→  const out: ClozeItem[] = [];
   352→  for (const q of questions) {
   353→    const entry = byWord.get(q.word.trim().toLowerCase());
   354→    if (!entry) continue; // واژه بیرون از استخر این دور → رد
   355→    const answerOpt = q.options[q.answerIndex]?.trim().toLowerCase();
   356→    if (answerOpt !== entry.answer.toLowerCase()) continue; // پاسخِ درست‌نما → رد
   357→    const wrongs = q.options
   358→      .map((o) => o.trim())
   359→      .filter((o) => o.toLowerCase() !== entry.answer.toLowerCase());
   360→    out.push({
   361→      id: `ai:${entry.answer}`,
   362→      sentence: q.sentence.replace(/_{3,}/g, BLANK),
   363→      // جملهٔ AI ترجمهٔ کتاب ندارد — معنی واژهٔ هدف همان راهنمای فارسی پرسش است
   364→      fa: `معنی واژهٔ گم‌شده: ${entry.meaning}`,
   365→      answer: entry.answer,
   366→      options: shuffle([entry.answer, ...shuffle(wrongs).slice(0, Math.max(0, optCount - 1))]),
   367→      meaning: entry.meaning,
   368→      lessonLabel: entry.lessonLabel,
   369→      source: "ai",
   370→    });
   371→  }
   372→  return out;
   373→}

   377→let bestCacheKey = "";
   378→let bestCacheValue: BestRecord | null = null;

   380→function readBest(bookId: string, scope: string): BestRecord | null {
   381→  let raw: string | null = null;
   382→  try {
   383→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   384→  } catch {
   385→    return null;
   386→  }
   387→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   388→  if (cacheKey === bestCacheKey) return bestCacheValue;
   389→  let value: BestRecord | null = null;
   390→  if (raw) {
   391→    try {
   392→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   393→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   394→        value = { score: parsed.score, total: parsed.total };
   395→      }
   396→    } catch {
   397→      value = null;
   398→    }
   399→  }
   400→  bestCacheKey = cacheKey;
   401→  bestCacheValue = value;
   402→  return value;
   403→}

   737→  // الگوی latest-ref (دارباز): listener پایدار روی window فقط حین playing — کلیدهای
   738→  // ۱/۲/۳ لاتین یا فارسی همان گزینهٔ N را پاسخ می‌دهند. گاردها: فاز playing + توقف +
   739→  // پنجرهٔ باز نگهبان خروج (سطح effect) و قفل busy/پاسخ‌داده‌شده (داخل answer).
   740→  const pickByKeyRef = useRef<(n: number) => void>(() => {});

   764→
   765→  // انصراف حین دور — فاز G2-b (نگهبان خروج): با حداقل یک پاسخ، GameExitGuard؛
   766→  // بدون هیچ پاسخی، خروج مستقیم + تله‌متری abandon
   767→  const handleExitClick = useCallback(() => {

   770→      setExitAsk(true);
   771→    } else {
   772→      clearAdvance();

   774→      sessionRef.current?.abandon("exit");
   775→      resume();

   777→    }
   778→  }, [idx, picked, resume, clearAdvance]);
   779→
   780→  // ── Setup ──
   781→  if (phase === "setup") {
   782→    // فاز G3-02 — شمارش کارت‌های ضعیف/سررسید/نو استخر (صفر = تمرکز هوشمند غیرفعال)
   783→    const weakCount = countWeak(pools.get(lesson) ?? [], (it) => it.answer);
   784→    return (
   785→      <>
   786→        <Card className="mx-auto max-w-xl rise-in">
