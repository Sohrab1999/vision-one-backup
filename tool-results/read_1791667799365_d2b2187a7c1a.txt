    99→    99→ * واژهٔ مجاز: فقط حروف/رقم/آندرلاین + آپاستروف، با حداکثر یک نقطه/کاما/علامت
   100→   100→ * سؤال پایانی — جمله‌های دارای dash خاص یا گیومهٔ تایپوگرافی حذف می‌شوند.
   101→   101→ */
   102→   102→const WORD_OK = /^[\w']+[.,?]?$/;
   103→   103→
   104→   104→/** استخر جمله‌های یک محدوده — با سنجش بازهٔ واژه و یکتاسازی متن */
   105→   105→function buildPool(
   106→   106→  data: BookData,
   107→   107→  lesson: number,
   108→   108→  minW: number,
   109→   109→  maxW: number,
   110→   110→): SentenceItem[] {
   111→   111→  const seen = new Set<string>();
   112→   112→  const items: SentenceItem[] = [];
   113→   113→  const push = (en: string, fa: string, lessonNo: number, kind: SentenceKind, src?: string) => {
   114→   114→    const text = en.trim();
   115→   115→    if (!text) return;
   116→   116→    const words = text.split(/\s+/);
   117→   117→    if (words.length < minW || words.length > maxW) return;
   118→   118→    for (const w of words) {
   119→   119→      if (!WORD_OK.test(w)) return;
   120→   120→    }
   121→   121→    const key = text.toLowerCase();
   122→   122→    if (seen.has(key)) return;
   123→   123→    seen.add(key);
   124→   124→    items.push({ en: text, fa: fa.trim(), lesson: lessonNo, kind, src });
   125→   125→  };
   126→   126→  const inScope = (l: number) => lesson === ALL_LESSONS || l === lesson;
   127→   127→
   128→   128→  for (const r of data.readings) {
   129→   129→    if (!inScope(r.lesson)) continue;
   130→   130→    for (const p of r.paragraphs) push(p.en, p.fa, r.lesson, "reading");
   131→   131→  }
   132→   132→  for (const c of data.conversations) {
   133→   133→    if (!inScope(c.lesson)) continue;
   134→   134→    for (const line of c.lines) push(line.en, line.fa, c.lesson, "conversation");
   135→   135→  }
   136→   136→  for (const d of data.dictionary) {
   137→   137→    if (!inScope(d.lesson)) continue;
   138→   138→    for (const ex of d.examples) push(ex, "", d.lesson, "dict", d.word);
   139→   139→  }
   140→   140→  return items;
   141→   141→}
   142→   142→
   143→   143→export function SentenceBuilderGame() {
   144→   144→  const data = useBookData();
   145→   145→  const { bookId } = useBook();
   146→   146→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا تضمینش می‌کند) — در عمل هرگز تهی نیست
   147→   147→  const activeBookId: string = bookId ?? "";
   148→   148→  const { speed } = useVoiceSpeed();
   149→   149→
   150→   150→  const [phase, setPhase] = useState<Phase>("setup");
   151→   151→  const [lesson, setLesson] = useState<number>(ALL_LESSONS);
   152→   152→  const [count, setCount] = useState<number>(3);
   153→   153→
   154→   154→  const [deck, setDeck] = useState<SentenceItem[]>([]);
   155→   155→  const [idx, setIdx] = useState(0);
   156→   156→  const [words, setWords] = useState<string[]>([]);
   157→   157→  const [tiles, setTiles] = useState<Tile[]>([]);
   158→   158→  const [slots, setSlots] = useState<(number | null)[]>([]);
   159→   159→
   160→   160→  const [solved, setSolved] = useState(0);
   161→   161→  const [skipped, setSkipped] = useState(0);
   162→   162→  const [hinted, setHinted] = useState(0);
   163→   163→  const [secs, setSecs] = useState(0);
   164→   164→  const [isRecord, setIsRecord] = useState(false);
   165→   165→
   166→   166→  /** انیمیشن match-pop روی ردیف جمله هنگام چیدن کامل درست */
   167→   167→  const [pop, setPop] = useState(false);
   168→   168→
   169→   169→  const locked = useRef(false);
   170→   170→  const timer = useRef<number | undefined>(undefined);
   171→   171→  const secsRef = useRef(0);
   172→   172→
   173→   173→  const busy = pop;
   174→   174→
   175→   175→  // استخر جملهٔ محدودهٔ انتخابی — اگر بازهٔ عادی کم آمد، بازهٔ واژه گسترده می‌شود
   176→   176→  const poolInfo = useMemo<{ items: SentenceItem[]; widened: boolean }>(() => {
   177→   177→    const base = buildPool(data, lesson, MIN_WORDS, MAX_WORDS);
   178→   178→    if (base.length >= WIDEN_THRESHOLD) return { items: base, widened: false };
   179→   179→    return { items: buildPool(data, lesson, MIN_WORDS_WIDE, MAX_WORDS_WIDE), widened: true };
   180→   180→  }, [data, lesson]);
   181→   181→
   182→   182→  // رکورد شخصی — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   183→   183→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   184→   184→  const best = useSyncExternalStore(
   185→   185→    subscribeGameData,
   186→   186→    () => {
   187→   187→      const b = loadBest<BestRec>(activeBookId, GAME_KEY, scope);
   188→   188→      return b ? b.time : null;
   189→   189→    },
   190→   190→    () => null,
   191→   191→  );
   192→   192→
   193→   193→  // ── تایمر (functional) + شمارندهٔ دقیق برای رکورد ──
   194→   194→  useEffect(() => {
   195→   195→    if (phase !== "playing") return;
   196→   196→    const t = window.setInterval(() => {
   197→   197→      secsRef.current += 1;
   198→   198→      setSecs((s) => s + 1);
   199→   199→    }, 1000);
   200→   200→    return () => window.clearInterval(t);
   201→   201→  }, [phase]);
   202→   202→
   203→   203→  // ── پاک‌سازی نهایی: تلفظ + تایمر معلق ──
   204→   204→  useEffect(() => {
   205→   205→    return () => {
   206→   206→      stopSpeaking();
   207→   207→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   208→   208→    };
   209→   209→  }, []);
   210→   210→
   211→   211→  /** بارگذاری جملهٔ بعدی — واژه‌های به‌هم‌ریخته + خانه‌های خالی */
   212→   212→  const loadSentence = useCallback((item: SentenceItem) => {
   213→   213→    const ws = item.en.trim().split(/\s+/);
   214→   214→    setWords(ws);
   215→   215→    setTiles(shuffle(ws.map((word, i) => ({ id: i, word }))));
   216→   216→    setSlots(Array.from({ length: ws.length }, () => null as number | null));
   217→   217→    setPop(false);
   218→   218→  }, []);
   219→   219→
   220→   220→  /** پایان دور — استریک/جلسه/Attempt/رکورد */
   221→   221→  const finishRun = useCallback(
   222→   222→    (finalSolved: number) => {
   223→   223→      touchStreak();
   224→   224→      recordStudySession();
   225→   225→      bumpPlays(activeBookId, GAME_KEY);
   226→   226→      saveAttempt(activeBookId, GAME_KEY, {
   227→   227→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   228→   228→        total: deck.length,
   229→   229→        correct: finalSolved,
   230→   230→        duration: secsRef.current,
   231→   231→        detail: { hints: hinted, skipped },
   232→   232→      });
   233→   233→      // رکورد فقط برای دورِ کاملِ بدون رد کردن و بدون راهنما
   234→   234→      if (skipped === 0 && hinted === 0) {
   235→   235→        const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   236→   236→        const prev = loadBest<BestRec>(activeBookId, GAME_KEY, recScope);
   237→   237→        if (!prev || secsRef.current < prev.time) {
   238→   238→          saveBest<BestRec>(activeBookId, GAME_KEY, recScope, { time: secsRef.current });
   239→   239→          setIsRecord(true);
   240→   240→        }
   241→   241→      }
   242→   242→      setSecs(secsRef.current);
   243→   243→      setPhase("done");
   244→   244→    },
   245→   245→    [activeBookId, lesson, count, deck.length, hinted, skipped],
   246→   246→  );
   247→   247→
   248→   248→  /** ادامه به جملهٔ بعد یا پایان دور */
   249→   249→  const advanceOrFinish = useCallback(
   250→   250→    (nextIdx: number, finalSolved: number) => {
   251→   251→      if (nextIdx >= deck.length) {
   252→   252→        finishRun(finalSolved);
   253→   253→        return;
   254→   254→      }
   255→   255→      setIdx(nextIdx);
   256→   256→      loadSentence(deck[nextIdx]);
   257→   257→    },
   258→   258→    [deck, loadSentence, finishRun],
   259→   259→  );
   260→   260→
   261→   261→  /** چیدن کامل درست — پاپ + تلفظ جملهٔ کامل و بعد جملهٔ بعد */
   262→   262→  const solveSentence = useCallback(
   263→   263→    (item: SentenceItem, finalSolved: number) => {
   264→   264→      locked.current = true;
   265→   265→      setPop(true);
   266→   266→      speak(item.en, { speed });
   267→   267→      timer.current = window.setTimeout(() => {
   268→   268→        locked.current = false;
   269→   269→        setPop(false);
   270→   270→        advanceOrFinish(idx + 1, finalSolved);
   271→   271→      }, SOLVE_LOCK_MS);
   272→   272→    },
   273→   273→    [speed, idx, advanceOrFinish],
   274→   274→  );
   275→   275→
   276→   276→  /** گذاشتن یک واژه از استخر در اولین خانهٔ خالی + بررسی چیدن کامل */
   277→   277→  const placeTile = useCallback(
   278→   278→    (tileId: number) => {
   279→   279→      if (locked.current) return;
   280→   280→      const current = deck[idx];
   281→   281→      if (!current) return;
   282→   282→      const empty = slots.indexOf(null);
   283→   283→      if (empty === -1) return;
   284→   284→      const tile = tiles.find((t) => t.id === tileId);
   285→   285→      if (!tile) return;
   286→   286→      const nextSlots = [...slots];
   287→   287→      nextSlots[empty] = tileId;
   288→   288→      setSlots(nextSlots);
   289→   289→      // همهٔ خانه‌ها پر شد؟ ترتیب چیده‌شده را با جملهٔ پایه بسنجیم
   290→   290→      if (nextSlots.every((s) => s !== null)) {
   291→   291→        const attempt = nextSlots.map((s) => tiles.find((t) => t.id === s)?.word ?? "");
   292→   292→        const ok = attempt.every((w, i) => w === words[i]);
   293→   293→        if (ok) {
   294→   294→          const finalSolved = solved + 1;
   295→   295→          setSolved(finalSolved);
   296→   296→          solveSentence(current, finalSolved);
   297→   297→        }
   298→   298→      }
   299→   299→    },
   300→   300→    [deck, idx, slots, tiles, words, solved, solveSentence],
   301→   301→  );
   302→   302→
   303→   303→  /** برگرداندن واژهٔ یک خانه به استخر (کلیک روی خانهٔ پر) */
   304→   304→  const returnAt = useCallback((index: number) => {
   305→   305→    if (locked.current) return;
   306→   306→    setSlots((prev) => {
   307→   307→      if (prev[index] === null) return prev;
   308→   308→      const next = [...prev];
   309→   309→      next[index] = null;
   310→   310→      return next;
   311→   311→    });
   312→   312→  }, []);
   313→   313→
   314→   314→  /** پاک کردن آخرین واژه (دکمه یا Backspace) */
   315→   315→  const returnLast = useCallback(() => {
   316→   316→    if (locked.current) return;
   317→   317→    setSlots((prev) => {
   318→   318→      for (let i = prev.length - 1; i >= 0; i--) {
   319→   319→        if (prev[i] !== null) {
   320→   320→          const next = [...prev];
   321→   321→          next[i] = null;
   322→   322→          return next;
   323→   323→        }
   324→   324→      }
   325→   325→      return prev;
   326→   326→    });
   327→   327→  }, []);
   328→   328→
   329→   329→  /** پاک کردن همه — همهٔ واژه‌ها به استخر برمی‌گردند */
   330→   330→  const clearAll = useCallback(() => {
   331→   331→    if (locked.current) return;
   332→   332→    setSlots((prev) => (prev.every((s) => s === null) ? prev : prev.map(() => null)));
   333→   333→  }, []);
   334→   334→
   335→   335→  /** راهنمای دوم — یک واژهٔ درستِ بعدی را جای‌گذاری می‌کند (رکورد دور می‌سوزد) */
   336→   336→  const applyHint = useCallback(() => {
   337→   337→    if (locked.current) return;
   338→   338→    const empty = slots.indexOf(null);
   339→   339→    if (empty === -1) return;
   340→   340→    const used = new Set(slots.filter((s): s is number => s !== null));
   341→   341→    const tile = tiles.find((t) => !used.has(t.id) && t.word === words[empty]);
   342→   342→    if (!tile) return;
   343→   343→    setHinted((h) => h + 1);
   344→   344→    placeTile(tile.id);
   345→   345→  }, [slots, tiles, words, placeTile]);
   346→   346→
   347→   347→  /** رد کردن — بدون SRS، فقط شمارش و جملهٔ بعد (رکورد دور می‌سوزاند) */
   348→   348→  const skipSentence = useCallback(() => {
   349→   349→    if (locked.current) return;
   350→   350→    const current = deck[idx];
   351→   351→    if (!current) return;
   352→   352→    locked.current = true;
   353→   353→    setSkipped((s) => s + 1);
   354→   354→    timer.current = window.setTimeout(() => {
   355→   355→      locked.current = false;
   356→   356→      advanceOrFinish(idx + 1, solved);
   357→   357→    }, SKIP_LOCK_MS);
   358→   358→  }, [deck, idx, solved, advanceOrFinish]);
   359→   359→
   360→   360→  /** انصراف — تایمر معلق پاک و به تنظیمات برگردیم */
   361→   361→  const cancel = useCallback(() => {
   362→   362→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   363→   363→    locked.current = false;
   364→   364→    stopSpeaking();
   365→   365→    setPhase("setup");
   366→   366→  }, []);
   367→   367→
   368→   368→  /** شروع دور — انتخاب جمله‌ها از استخر (اگر کم بود، تکرار با shuffle مجاز است) */
   369→   369→  const start = useCallback(() => {
   370→   370→    const items = poolInfo.items;
   371→   371→    if (items.length === 0) {
   372→   372→      toast.error("جملهٔ قابل بازی در این محدوده پیدا نشد — درس دیگری را امتحان کن");
   373→   373→      return;
   374→   374→    }
   375→   375→    const picked: SentenceItem[] = [];
   376→   376→    while (picked.length < count) {
   377→   377→      const batch = shuffle(items).slice(0, Math.min(count - picked.length, items.length));
   378→   378→      picked.push(...batch);
   379→   379→    }
   380→   380→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   381→   381→    locked.current = false;
   382→   382→    secsRef.current = 0;
   383→   383→    setDeck(picked);
   384→   384→    setIdx(0);
   385→   385→    setSolved(0);
   386→   386→    setSkipped(0);
   387→   387→    setHinted(0);
   388→   388→    setSecs(0);
   389→   389→    setIsRecord(false);
   390→   390→    loadSentence(picked[0]);
   391→   391→    setPhase("playing");
   392→   392→  }, [poolInfo, count, loadSentence]);
   393→   393→
