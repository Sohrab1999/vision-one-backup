   518→   518→    [puzzle, bid, lesson, scope],
   519→   519→  );
   520→   520→
   521→   521→  /**
   522→   522→   * فاز G2-b — هستهٔ مشترک تأیید انتخاب (هر دو مسیر کلیکِ دو-مرحله‌ای و درگ لمسی):
   523→   523→   * مسیر هم‌خط را با واژه‌های پیدانشده می‌سنجد — درست: pop + SRS + تلفظ + صدا؛
   524→   524→   * غلط: wrong/error + لرزش دو سر مسیر + شمارش خطا + نگاشت به واژهٔ مرور.
   525→   525→   */
   526→   526→  const confirmPath = useCallback(
   527→   527→    (cells: Cell[], viaDrag: boolean) => {
   528→   528→      if (!puzzle || paused) return;
   529→   529→      const { grid, placements } = puzzle;
   530→   530→      if (found.size >= placements.length) return; // دور در حال بستن است (پنجرهٔ تأخیر پایان)

   532→   532→      if (shakeTimer.current) {
   533→   533→        clearTimeout(shakeTimer.current);
   534→   534→        shakeTimer.current = null;
   535→   535→        setWrongKeys(null);
   536→   536→      }
   537→   537→      const str = cells.map((k) => grid[k.row][k.col]).join("");
   538→   538→      const rev = [...str].reverse().join("");
   539→   539→      const hit = placements.find((p) => !found.has(p.base) && (p.base === str || p.base === rev));
   540→   540→      if (hit) {
   541→   541→        const next = new Set(found);
   542→   542→        next.add(hit.base);
   543→   543→        const isLast = next.size >= placements.length;
   544→   544→        setFound(next);
   545→   545→        setSel(null);
   546→   546→        const keys = cells.map((k) => `${k.row}:${k.col}`);
   547→   547→        setPopKeys(keys);
   548→   548→        if (popTimer.current) clearTimeout(popTimer.current);
   549→   549→        popTimer.current = setTimeout(() => setPopKeys(null), 500);

   551→   551→        speak(hit.base.toLowerCase(), { speed });
   552→   552→        // فاز G2-b — سه‌کانالهٔ یافتن واژه (صدا + لرزش + اعلام aria)
   553→   553→        if (isLast) {
   554→   554→          // همه یافت شدند — فانفار برد؛ پایان دور با تأخیر کوتاه (پس از انیمیشن pop)
   555→   555→          playSound("win");
   556→   556→          buzz("record");
   557→   557→          setLiveMsg(
   558→   558→            viaDrag
   559→   559→              ? `انتخاب: ${hit.base} — همه شکار شدند! 🎉`
   560→   560→              : `«${hit.base}» پیدا شد — همه شکار شدند! 🎉`,
   561→   561→          );

   565→   565→        } else {
   566→   566→          playSound("correct");
   567→   567→          buzz("success");
   568→   568→          setLiveMsg(
   569→   569→            viaDrag
   570→   570→              ? `انتخاب: ${hit.base} — واژه پیدا شد! 🎉`
   571→   571→              : `«${hit.base}» پیدا شد! 🎉`,
   572→   572→          );
   573→   573→        }
   574→   574→      } else {

   576→   576→        playSound("wrong");
   577→   577→        buzz("error");
   578→   578→        setWrongCount((n) => n + 1);
   579→   579→        setLiveMsg(
   580→   580→          viaDrag
   581→   581→            ? `انتخاب: ${str} — این واژه در فهرست نیست`
   582→   582→            : `«${str}» در فهرست واژه‌ها نیست — دوباره تلاش کن`,
   583→   583→        );
   584→   584→        // فاز G2-b — نگاشت خطا به واژهٔ مرور: اگر مسیر اشتباه بخشی از یک واژهٔ
   585→   585→        // پیدانشده بود (کاربر آن را ناقص به یاد داشته)، همان واژه برای «مرور خطاها»
   586→   586→        // صفحهٔ نتیجه ثبت می‌شود؛ وگرنه فقط شمارندهٔ خطا بالا می‌رود.
   587→   587→        const guess = [str, rev].find((s) => s.length >= 3);
   588→   588→        const target = guess
   589→   589→          ? placements.find((p) => !found.has(p.base) && p.base.includes(guess))

   599→   599→        const first = cells[0];
   600→   600→        const last = cells[cells.length - 1];
   601→   601→        setWrongKeys([`${first.row}:${first.col}`, `${last.row}:${last.col}`]);
   602→   602→        setSel(null);
   603→   603→        shakeTimer.current = setTimeout(() => setWrongKeys(null), 600);
   604→   604→      }
   605→   605→    },
   606→   606→    [puzzle, paused, found, speed, finish],
   607→   607→  );
   608→   608→

   610→   610→  const tapCell = useCallback(
   611→   611→    (row: number, col: number) => {
   612→   612→      if (!puzzle || paused) return;
   613→   613→      if (found.size >= puzzle.placements.length) return; // دور در حال بستن است
   614→   614→      // لمس جدید هر لرزش قبلی را تمام می‌کند
   615→   615→      if (shakeTimer.current) {
   616→   616→        clearTimeout(shakeTimer.current);
   617→   617→        shakeTimer.current = null;
   618→   618→        setWrongKeys(null);
   619→   619→      }
   620→   620→      // tap اول → شروع انتخاب
   621→   621→      if (!sel) {
   622→   622→        setSel({ row, col });
   623→   623→        return;
   624→   624→      }
   625→   625→      // tap روی همان خانه → لغو انتخاب
   626→   626→      if (sel.row === row && sel.col === col) {
   627→   627→        setSel(null);
   628→   628→        return;
   629→   629→      }

   631→   631→      const cells = pathBetween(sel, { row, col });
   632→   632→      if (!cells) {
   633→   633→        setSel({ row, col });
   634→   634→        return;
   635→   635→      }
   636→   636→      confirmPath(cells, false);
   637→   637→    },
   638→   638→    [puzzle, paused, found.size, sel, confirmPath],
   639→   639→  );

   643→   643→  // جابه‌جایی، confirm نمی‌شود و onClick خودش هندل می‌کند)؛ حرکت با elementFromPoint
   644→   644→  // روی container ردیابی و با اولین حرکتِ واقعی، اشاره‌گر به container کپچر می‌شود تا
   645→   645→  // رهاکردن بیرون شبکه هم دریافت شود؛ رهاکردن = ثبت مسیر snap‌شده با confirmPath مشترک.

   651→   651→      capturedRef.current = false;
   652→   652→      suppressClickUntilRef.current = 0;
   653→   653→      // فاز G2-b — شروع درگ: بازخورد کلیک (صدا + لرزش)

   655→   655→      buzz("light");
   656→   656→      setDrag({ from: { row, col }, to: { row, col } });

   658→   658→    [paused, puzzle, found.size],
   659→   659→  );

   661→   661→  const onGridPointerMove = useCallback(

   666→   666→      // اولین حرکت واقعی → کپچر به container تا حرکت/رهاکردن بیرون شبکه هم برسد
   667→   667→      if (!capturedRef.current) {
   668→   668→        try {
   669→   669→          gridRef.current?.setPointerCapture(e.pointerId);
   670→   670→          capturedRef.current = true;
   671→   671→        } catch {
   672→   672→          /* کپچر اختیاری است — مسیر داخل شبکه بدون آن هم کار می‌کند */
   673→   673→        }
   674→   674→      }
   675→   675→      setDrag((d) => (d ? { ...d, to: cell } : d));
   676→   676→    },
   677→   677→    [drag],
   678→   678→  );

   684→   684→      const { from, to } = drag;
   685→   685→      activePointerRef.current = null;
   686→   686→      if (capturedRef.current) {
   687→   687→        try {
   688→   688→          gridRef.current?.releasePointerCapture(e.pointerId);
   689→   689→        } catch {
   690→   690→          /* noop */
   691→   691→        }
   692→   692→        capturedRef.current = false;
   693→   693→      }
   694→   694→      setDrag(null);
   695→   695→      if (mode !== "confirm") return;
   696→   696→      // تپ ساده (بدون ورود به سلول دیگر) — مسیر کلیک/کیبورد خودش هندل می‌کند
   697→   697→      if (from.row === to.row && from.col === to.col) return;

   723→   723→    }
   724→   724→    return s;
   725→   725→  }, [drag, puzzle, size]);
   726→   726→

   728→   728→    cn(

   731→   731→        ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"

   734→   734→
   735→   735→  // انصراف حین دور — فاز G2-b نگهبان خروج: با ≥۱ واژه یافته (یا انتخاب آغازشده)،
   736→   736→  // GameExitGuard؛ بدون هیچ پیشرفتی، خروج مستقیم + تله‌متری abandon
