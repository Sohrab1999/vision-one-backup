   160→/** واژهٔ کتاب → واژهٔ جدول: حذف پرانتزها + حروف بزرگ (مثل «commit (a crime)» → «COMMIT») */
   161→function baseWordOf(word: string): string {
   162→  return word.replace(/\s*\([^)]*\)\s*/g, " ").trim().toUpperCase();

   165→/** استخر معتبر: پایه ۳..۹ حرف لاتینِ خالص (بدون فاصله) + حذف تکراری بر اساس پایه */
   166→function validWordList(pool: VocabEntry[]): VocabEntry[] {
   167→  const seen = new Set<string>();
   168→  return pool.filter((w) => {
   169→    const base = baseWordOf(w.word);
   170→    if (!/^[A-Z]{3,9}$/.test(base)) return false;
   171→    if (seen.has(base)) return false;
   172→    seen.add(base);
   173→    return true;
   174→  });
   175→}

   177→/** خانه‌های بین دو خانهٔ هم‌خط (سطر/ستون/قطر) — غیرهم‌خط = null */
   178→function pathBetween(a: Cell, b: Cell): Cell[] | null {
   179→  const dr = Math.sign(b.row - a.row);
   180→  const dc = Math.sign(b.col - a.col);
   181→  const aligned = a.row === b.row || a.col === b.col || Math.abs(b.row - a.row) === Math.abs(b.col - a.col);
   182→  if (!aligned) return null;
   183→  const cells: Cell[] = [{ row: a.row, col: a.col }];
   184→  let r = a.row;
   185→  let c = a.col;
   186→  while (r !== b.row || c !== b.col) {
   187→    r += dr;
   188→    c += dc;
   189→    cells.push({ row: r, col: c });
   190→  }
   191→  return cells;
   192→}

   194→/**
   195→ * فاز G2-b — مسیر موقت درگ با snap به نزدیک‌ترین جهت مجاز (۸ جهت):
   196→ * بردار حرکت (سطر/ستون) زاویه‌اش به نزدیک‌ترین ضریب ۴۵° گرد می‌شود و طول مسیر
   197→ * تصویر بردار روی همان جهت است (حداقل ۱) — همیشه داخل جدول clamp می‌شود.
   198→ * نتیجه هرگز null نیست؛ انحراف دست کاربر به نزدیک‌ترین جهت درست snap می‌شود.
   199→ */
   200→function dragPathBetween(from: Cell, to: Cell, size: number): Cell[] {
   201→  const dr = to.row - from.row;
   202→  const dc = to.col - from.col;
   203→  if (dr === 0 && dc === 0) return [{ row: from.row, col: from.col }];
   204→  const step = Math.PI / 4;
   205→  const snapIdx = Math.round(Math.atan2(dr, dc) / step); // ‎-4..4
   206→  const dirR = Math.round(Math.sin(snapIdx * step));
   207→  const dirC = Math.round(Math.cos(snapIdx * step));
   208→  // طول بر حسب «گام» — تصویر بردار روی جهت snap بر طول همان بردار گام (۱ یا √۲)
   209→  let len = Math.round((dr * dirR + dc * dirC) / (dirR * dirR + dirC * dirC));
   210→  if (len < 1) len = 1;
   211→  const roomR = dirR > 0 ? size - 1 - from.row : dirR < 0 ? from.row : size - 1;
   212→  const roomC = dirC > 0 ? size - 1 - from.col : dirC < 0 ? from.col : size - 1;
   213→  len = Math.min(len, dirR === 0 ? roomC : dirC === 0 ? roomR : Math.min(roomR, roomC));
   214→  const cells: Cell[] = [{ row: from.row, col: from.col }];
   215→  for (let i = 1; i <= len; i++) {
   216→    cells.push({ row: from.row + i * dirR, col: from.col + i * dirC });
   217→  }
   218→  return cells;
   219→}

   234→/** ساخت پازل — تابع خالص؛ اگر کمتر از ۴ واژه جا شد از نو می‌سازد (تا ۱۰ بار) */
   235→function generatePuzzle(words: PuzzleWord[], size: number): Puzzle {
   236→  let last: Puzzle = { grid: [], placements: [] };
   237→  for (let round = 0; round < 10; round++) {
   238→    const p = buildOnce(words, size);
   239→    last = p;
   240→    if (p.placements.length >= 4) break;
   241→  }
   242→  return last;
   243→}

   245→/** یک دور ساخت جدول — واژه‌های جا نشده حذف می‌شوند */
   246→function buildOnce(words: PuzzleWord[], size: number): Puzzle {
   247→  const grid: (string | null)[][] = Array.from({ length: size }, () => Array<string | null>(size).fill(null));
   248→  const placements: Placement[] = [];
   249→  const sorted = words
   250→    .filter((w) => w.base.length <= size) // واژه بلندتر از جدول هرگز جا نمی‌شود
   251→    .sort((a, b) => b.base.length - a.base.length); // بلندها اول — شانس جا شدن بیشتر
   252→  for (const w of sorted) {
   253→    const L = w.base.length;
   254→    for (let t = 0; t < 200; t++) {
   255→      const [dr, dc] = DIRS[Math.floor(Math.random() * DIRS.length)];
   256→      // بازهٔ مجاز نقطهٔ شروع برای این جهت — اگر خالی باشد واژه در این جهت جا نمی‌شود
   257→      const rMin = dr === -1 ? L - 1 : 0;
   258→      const rMax = dr === 1 ? size - L : size - 1;
   259→      const cMin = dc === -1 ? L - 1 : 0;
   260→      const cMax = dc === 1 ? size - L : size - 1;
   261→      if (rMin > rMax || cMin > cMax) continue;
   262→      const r0 = rMin + Math.floor(Math.random() * (rMax - rMin + 1));
   263→      const c0 = cMin + Math.floor(Math.random() * (cMax - cMin + 1));
   264→      // برخورد فقط با همان حرف مجاز است
   265→      let ok = true;
   266→      for (let i = 0; i < L; i++) {
   267→        const cell = grid[r0 + i * dr][c0 + i * dc];
   268→        if (cell !== null && cell !== w.base[i]) {
   269→          ok = false;
   270→          break;
   271→        }
   272→      }
   273→      if (!ok) continue;
   274→      const cells: [number, number][] = [];
   275→      for (let i = 0; i < L; i++) {
   276→        const r = r0 + i * dr;
   277→        const c = c0 + i * dc;
   278→        grid[r][c] = w.base[i];
   279→        cells.push([r, c]);
   280→      }
   281→      placements.push({ base: w.base, raw: w.raw, meaning: w.meaning, cells });
   282→      break; // همین واژه جا شد — سراغ بعدی
   283→    }
   284→  }
   285→  // خانه‌های خالی: ۵۰٪ حرف از خود واژه‌های پازل (استتار) + ۵۰٪ A-Z تصادفی
   286→  const bag: string[] = [];
   287→  for (const p of placements) {
   288→    for (const ch of p.base) bag.push(ch);
   289→  }
   290→  const finalGrid: string[][] = grid.map((row) =>
   291→    row.map((cell) => {
   292→      if (cell !== null) return cell;
   293→      if (bag.length > 0 && Math.random() < 0.5) return bag[Math.floor(Math.random() * bag.length)];
   294→      return String.fromCharCode(65 + Math.floor(Math.random() * 26));
   295→    }),
   296→  );
   297→  return { grid: finalGrid, placements };
   298→}

   386→  // استخر معتبر درس انتخابی + شمارندهٔ قابل‌شکار کل کتاب (برچسب چیپ)
   387→  const candidates = useMemo(
   388→    () => validWordList(lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson)),
   389→    [data.vocab, lesson],
   390→  );
   391→  const allValidCount = useMemo(() => validWordList(data.vocab.all).length, [data.vocab]);
   392→
   393→  // رکورد شخصی — اسنپ‌شات عددیِ پایدار (primitive) با الگوی match-game
   394→  const best = useSyncExternalStore(
   395→    subscribeGameData,
   396→    () => {
   397→      const b = loadBest<BestRecord>(bid, GAME_KEY, scope);
   398→      return b && typeof b.time === "number" ? b.time : null;
   399→    },
   400→    () => null,
   401→  );
   402→
   403→  // تایمر — functional setState (بدون setState مستقیم در effect)
   404→  // فاز G2-b — گیت روی paused: هنگام توقف (دستی یا خودکارِ تب مخفی) هیچ تیکی جلو نمی‌رود
   405→  useEffect(() => {
   406→    if (phase !== "playing" || paused) return;
   407→    const t = window.setInterval(() => {
   408→      secsRef.current += 1;
   409→      setSecs((s) => s + 1);
   410→    }, 1000);
   411→    return () => window.clearInterval(t);
   412→  }, [phase, paused]);
   413→
   414→  /** پاک‌سازی تایمرهای معلق دور (انیمیشن‌ها + پایان با تأخیر) */
   415→  const clearRoundTimers = useCallback(() => {
   416→    if (popTimer.current) clearTimeout(popTimer.current);
   417→    if (shakeTimer.current) clearTimeout(shakeTimer.current);
   418→    if (finishTimer.current) clearTimeout(finishTimer.current);
   419→  }, []);
   420→
   421→  // cleanup: قطع تلفظ + تایمرهای معلق + ثبت «رهاشده» تله‌متری (فاز G2-b)
   422→  useEffect(() => {
   423→    return () => {
   424→      sessionRef.current?.abandon("unmount");
   425→      stopSpeaking();
   426→      clearRoundTimers();
   427→    };
   428→  }, [clearRoundTimers]);
   429→
   430→  /** شروع دور — ساخت پازل + ریست کامل وضعیت + جلسهٔ تازهٔ تله‌متری */
   431→  const start = useCallback(() => {
   432→    // اگر واژه‌های جاگیرنده به تعداد خواستی کافی بود از همان‌ها بگیر (برای ۸×۸ واژهٔ ۹حرفی بی‌فایده است)
   433→    const fitting = candidates.filter((w) => baseWordOf(w.word).length <= size);
   434→    const pool = fitting.length >= wordCount ? fitting : candidates;
   435→    const chosen = pickUnique(pool, wordCount, (w) => baseWordOf(w.word)).map((w) => ({
   436→      base: baseWordOf(w.word),
   437→      raw: w.word,
   438→      meaning: w.meaning,
   439→    }));
   440→    if (chosen.length < 4) {
   441→      toast.error("واژهٔ قابل‌شکار کافی در این محدوده نیست — درس دیگری را امتحان کن");
   442→      return;
   443→    }
   444→    const p = generatePuzzle(chosen, size);
   445→    if (p.placements.length < 4) {
   446→      toast.error("جدول با این تنظیمات ساخته نشد — اندازهٔ جدول یا درس را عوض کن");
   447→      return;
   448→    }
   449→    clearRoundTimers();
   450→    setPuzzle(p);
   451→    setClues(shuffle(p.placements)); // ترتیب سرنخ‌ها بی‌ربط به ترتیب جای‌گذاری
   452→    setFound(new Set());
   453→    setSel(null);
   454→    setPopKeys(null);
   455→    setWrongKeys(null);
   456→    setShowWords(false);
   457→    setSecs(0);
   458→    secsRef.current = 0;
   459→    setFinalSecs(null);
   460→    setIsRecord(false);
   461→    setLiveMsg("");
   462→    setWrongCount(0);
   463→    setMissedWords([]);
   464→    setDrag(null);
   465→    activePointerRef.current = null;
   466→    capturedRef.current = false;
   467→    suppressClickUntilRef.current = 0;
   468→    endedAtRef.current = 0;
   469→    startedAtRef.current = Date.now();
   470→    resume();
   471→    setExitAsk(false);
   472→    // فاز G2-b — جلسهٔ تازهٔ تله‌متری برای این دور
   473→    sessionRef.current = createGameSession({
   474→      game: GAME_KEY,
   475→      level: scope,
   476→    });
   477→    sessionRef.current.start();
   478→    setPhase("playing");
   479→  }, [candidates, wordCount, size, scope, resume, clearRoundTimers]);
   480→
   481→  /** پایان دور — استریک/جلسه/Attempt/رکورد (کمترین زمان) — الگوی کیت */
   482→  const finish = useCallback(
   483→    (foundCount: number) => {
   484→      const total = puzzle?.placements.length ?? foundCount;
   485→      // فاز G2-b — زمان واقعی دور (لحظهٔ شکار آخرین واژه، نه لحظهٔ فراخوانی با تأخیر)
   486→      const finalTime = Math.max(
   487→        1,
   488→        Math.round(((endedAtRef.current || Date.now()) - (startedAtRef.current || Date.now())) / 1000),
   489→      );
   490→      setFinalSecs(finalTime);
   491→      touchStreak();
   492→      recordStudySession();
   493→      bumpPlays(bid, GAME_KEY);
   494→      // فاز G2-b — پایان عادی جلسهٔ تله‌متری
   495→      sessionRef.current?.end("finish");
   496→      // فاز G2-b — بازخورد سه‌کانالهٔ پایان دور (الگوی پایلوت memory)
   497→      playSound("finish");
   498→      buzz("success");
   499→      saveAttempt(bid, GAME_KEY, {
   500→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   501→        total,
   502→        correct: foundCount,
   503→        duration: finalTime,
   504→        detail: (puzzle?.placements ?? []).map((p) => ({ word: p.raw, ok: true })),
   505→      });
   506→      const prev = loadBest<BestRecord>(bid, GAME_KEY, scope);
   507→      if (!prev || typeof prev.time !== "number" || finalTime < prev.time) {
   508→        saveBest(bid, GAME_KEY, scope, { time: finalTime });
   509→        setIsRecord(true);
   510→        playSound("record");
   511→        buzz("record");
   512→        toast.success(`رکورد جدید! ${faNum(total)} واژه در ${fmtTime(finalTime)} 🏆`);
   513→      } else {
   514→        toast.success(`همه شکار شدند — ${faNum(total)} واژه در ${fmtTime(finalTime)}! 🎯`);
   515→      }
   516→      setPhase("done");
   517→    },
   518→    [puzzle, bid, lesson, scope],
   519→  );
   520→
   521→  /**
   522→   * فاز G2-b — هستهٔ مشترک تأیید انتخاب (هر دو مسیر کلیکِ دو-مرحله‌ای و درگ لمسی):
   523→   * مسیر هم‌خط را با واژه‌های پیدانشده می‌سنجد — درست: pop + SRS + تلفظ + صدا؛
   524→   * غلط: wrong/error + لرزش دو سر مسیر + شمارش خطا + نگاشت به واژهٔ مرور.
   525→   */
   526→  const confirmPath = useCallback(
   527→    (cells: Cell[], viaDrag: boolean) => {
   528→      if (!puzzle || paused) return;
   529→      const { grid, placements } = puzzle;
   530→      if (found.size >= placements.length) return; // دور در حال بستن است (پنجرهٔ تأخیر پایان)
   531→      // انتخاب جدید هر لرزش قبلی را تمام می‌کند
   532→      if (shakeTimer.current) {
   533→        clearTimeout(shakeTimer.current);
   534→        shakeTimer.current = null;
   535→        setWrongKeys(null);
   536→      }
   537→      const str = cells.map((k) => grid[k.row][k.col]).join("");
   538→      const rev = [...str].reverse().join("");
   539→      const hit = placements.find((p) => !found.has(p.base) && (p.base === str || p.base === rev));
   540→      if (hit) {
   541→        const next = new Set(found);
   542→        next.add(hit.base);
   543→        const isLast = next.size >= placements.length;
   544→        setFound(next);
   545→        setSel(null);
   546→        const keys = cells.map((k) => `${k.row}:${k.col}`);
   547→        setPopKeys(keys);
   548→        if (popTimer.current) clearTimeout(popTimer.current);
   549→        popTimer.current = setTimeout(() => setPopKeys(null), 500);
   550→        recordSRS(hit.raw, true); // کلید SRS = واژهٔ خام کتاب (سازگار با بقیهٔ اپ)
   551→        speak(hit.base.toLowerCase(), { speed });
   552→        // فاز G2-b — سه‌کانالهٔ یافتن واژه (صدا + لرزش + اعلام aria)
   553→        if (isLast) {
   554→          // همه یافت شدند — فانفار برد؛ پایان دور با تأخیر کوتاه (پس از انیمیشن pop)
   555→          playSound("win");
   556→          buzz("record");
   557→          setLiveMsg(
   558→            viaDrag
   559→              ? `انتخاب: ${hit.base} — همه شکار شدند! 🎉`
   560→              : `«${hit.base}» پیدا شد — همه شکار شدند! 🎉`,
   561→          );
   562→          endedAtRef.current = Date.now();
   563→          if (finishTimer.current) clearTimeout(finishTimer.current);
   564→          finishTimer.current = setTimeout(() => finish(next.size), FINISH_DELAY_MS);
   565→        } else {
   566→          playSound("correct");
   567→          buzz("success");
   568→          setLiveMsg(
   569→            viaDrag
   570→              ? `انتخاب: ${hit.base} — واژه پیدا شد! 🎉`
   571→              : `«${hit.base}» پیدا شد! 🎉`,
   572→          );
   573→        }
   574→      } else {
   575→        // انتخاب غلط — فاز G2-b: صدا + لرزش + اعلام (سه‌کاناله)
   576→        playSound("wrong");
   577→        buzz("error");
   578→        setWrongCount((n) => n + 1);
   579→        setLiveMsg(
   580→          viaDrag
   581→            ? `انتخاب: ${str} — این واژه در فهرست نیست`
   582→            : `«${str}» در فهرست واژه‌ها نیست — دوباره تلاش کن`,
   583→        );
   584→        // فاز G2-b — نگاشت خطا به واژهٔ مرور: اگر مسیر اشتباه بخشی از یک واژهٔ
   585→        // پیدانشده بود (کاربر آن را ناقص به یاد داشته)، همان واژه برای «مرور خطاها»
   586→        // صفحهٔ نتیجه ثبت می‌شود؛ وگرنه فقط شمارندهٔ خطا بالا می‌رود.
   587→        const guess = [str, rev].find((s) => s.length >= 3);
   588→        const target = guess
   589→          ? placements.find((p) => !found.has(p.base) && p.base.includes(guess))
   590→          : undefined;
   591→        if (target) {
   592→          setMissedWords((prev) =>
   593→            prev.some((m) => m.front === target.raw)
   594→              ? prev
   595→              : [...prev, { front: target.raw, back: target.meaning }],
   596→          );
   597→        }
   598→        // لرزش بصری روی دو سر مسیر غلط
   599→        const first = cells[0];
   600→        const last = cells[cells.length - 1];
   601→        setWrongKeys([`${first.row}:${first.col}`, `${last.row}:${last.col}`]);
   602→        setSel(null);
   603→        shakeTimer.current = setTimeout(() => setWrongKeys(null), 600);
   604→      }
   605→    },
   606→    [puzzle, paused, found, speed, finish],
   607→  );
   608→
   609→  /** مسیر کلیک دو-مرحله‌ای (موجود فاز ۳) — تأیید نهایی از confirmPath مشترک */
   610→  const tapCell = useCallback(
   611→    (row: number, col: number) => {
   612→      if (!puzzle || paused) return;
   613→      if (found.size >= puzzle.placements.length) return; // دور در حال بستن است
   614→      // لمس جدید هر لرزش قبلی را تمام می‌کند
   615→      if (shakeTimer.current) {
   616→        clearTimeout(shakeTimer.current);
   617→        shakeTimer.current = null;
   618→        setWrongKeys(null);
   619→      }
   620→      // tap اول → شروع انتخاب
   621→      if (!sel) {
   622→        setSel({ row, col });
   623→        return;
   624→      }
   625→      // tap روی همان خانه → لغو انتخاب
   626→      if (sel.row === row && sel.col === col) {
   627→        setSel(null);
   628→        return;
   629→      }
   630→      // tap دوم غیرهم‌خط → تغییر شروع انتخاب (تجربهٔ روان)
   631→      const cells = pathBetween(sel, { row, col });
   632→      if (!cells) {
   633→        setSel({ row, col });
   634→        return;
   635→      }
   636→      confirmPath(cells, false);
   637→    },
   638→    [puzzle, paused, found.size, sel, confirmPath],
   639→  );
   640→
   641→  // ── فاز G2-b — درگ لمسی/ماوس (Pointer Events) روی شبکه ──
   642→  // pointerDown روی سلول = شروع درگ (کلیک دو-مرحله‌ای دست نمی‌خورد: تپ ساده بدون
   643→  // جابه‌جایی، confirm نمی‌شود و onClick خودش هندل می‌کند)؛ حرکت با elementFromPoint
   644→  // روی container ردیابی و با اولین حرکتِ واقعی، اشاره‌گر به container کپچر می‌شود تا
   645→  // رهاکردن بیرون شبکه هم دریافت شود؛ رهاکردن = ثبت مسیر snap‌شده با confirmPath مشترک.
   646→  const onCellPointerDown = useCallback(
   647→    (row: number, col: number, e: ReactPointerEvent<HTMLButtonElement>) => {
   648→      if (paused || !puzzle) return;
   649→      if (found.size >= puzzle.placements.length) return; // دور در حال بستن است
   650→      activePointerRef.current = e.pointerId;
   651→      capturedRef.current = false;
   652→      suppressClickUntilRef.current = 0;
   653→      // فاز G2-b — شروع درگ: بازخورد کلیک (صدا + لرزش)
   654→      playSound("click");
   655→      buzz("light");
   656→      setDrag({ from: { row, col }, to: { row, col } });
   657→    },
   658→    [paused, puzzle, found.size],
   659→  );
   660→
   661→  const onGridPointerMove = useCallback(
   662→    (e: ReactPointerEvent<HTMLDivElement>) => {
   663→      if (!drag || activePointerRef.current !== e.pointerId) return;
   664→      const cell = cellFromPoint(e.clientX, e.clientY);
   665→      if (!cell || (cell.row === drag.to.row && cell.col === drag.to.col)) return;
   666→      // اولین حرکت واقعی → کپچر به container تا حرکت/رهاکردن بیرون شبکه هم برسد
   667→      if (!capturedRef.current) {
   668→        try {
   669→          gridRef.current?.setPointerCapture(e.pointerId);
   670→          capturedRef.current = true;
   671→        } catch {
   672→          /* کپچر اختیاری است — مسیر داخل شبکه بدون آن هم کار می‌کند */
   673→        }
   674→      }
   675→      setDrag((d) => (d ? { ...d, to: cell } : d));
   676→    },
   677→    [drag],
   678→  );
   679→
   680→  /** پایان درگ — confirm برای pointerUp/leave، cancel برای pointercancel */
   681→  const endDrag = useCallback(
   682→    (e: ReactPointerEvent<HTMLDivElement>, mode: "confirm" | "cancel") => {
   683→      if (!drag || activePointerRef.current !== e.pointerId) return;
   684→      const { from, to } = drag;
   685→      activePointerRef.current = null;
   686→      if (capturedRef.current) {
   687→        try {
   688→          gridRef.current?.releasePointerCapture(e.pointerId);
   689→        } catch {
   690→          /* noop */
   691→        }
   692→        capturedRef.current = false;
   693→      }
   694→      setDrag(null);
   695→      if (mode !== "confirm") return;
   696→      // تپ ساده (بدون ورود به سلول دیگر) — مسیر کلیک/کیبورد خودش هندل می‌کند
   697→      if (from.row === to.row && from.col === to.col) return;
   698→      // کلیکِ همان ژستِ درگ نباید دوباره confirm کند (هدف click بعد از pointerUp)
   699→      suppressClickUntilRef.current = Date.now() + 600;
   700→      confirmPath(dragPathBetween(from, to, size), true);
   701→    },
   702→    [drag, size, confirmPath],
   703→  );
   704→
   705→  // خانه‌های واژه‌های پیدا‌شده — برای رنگ‌آمیزی دائمی جدول
   706→  const foundCellKeys = useMemo(() => {
   707→    const s = new Set<string>();
   708→    if (puzzle) {
   709→      for (const p of puzzle.placements) {
   710→        if (found.has(p.base)) {
   711→          for (const [r, c] of p.cells) s.add(`${r}:${c}`);
   712→        }
   713→      }
   714→    }
   715→    return s;
   716→  }, [puzzle, found]);
   717→
   718→  // فاز G2-b — خانه‌های مسیر موقت درگ (هایلایت کهربایی تا رهاکردن)
   719→  const dragCellKeys = useMemo(() => {
   720→    const s = new Set<string>();
   721→    if (drag && puzzle) {
   722→      for (const c of dragPathBetween(drag.from, drag.to, size)) s.add(`${c.row}:${c.col}`);
   723→    }
   724→    return s;
   725→  }, [drag, puzzle, size]);
   726→
   727→  const chipCls = (active: boolean) =>
   728→    cn(
   729→      "flex min-h-11 items-center rounded-full border px-3.5 text-[12px] font-bold transition-all",
   730→      active
   731→        ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   732→        : "text-muted-foreground hover:bg-muted",
   733→    );
   734→
   735→  // انصراف حین دور — فاز G2-b نگهبان خروج: با ≥۱ واژه یافته (یا انتخاب آغازشده)،
   736→  // GameExitGuard؛ بدون هیچ پیشرفتی، خروج مستقیم + تله‌متری abandon
   737→  const handleExitClick = useCallback(() => {
   738→    stopSpeaking();
   739→    if (found.size > 0 || sel !== null) {
   740→      setExitAsk(true);
   741→    } else {
   742→      clearRoundTimers();
   743→      // دور بدون پیشرفت هم یک جلسه است؛ همین‌جا رهاشده ثبت می‌شود
   744→      sessionRef.current?.abandon("exit");
   745→      resume();
   746→      setPhase("setup");
   747→    }
   748→  }, [found.size, sel, clearRoundTimers, resume]);
   749→
   750→  // ── Setup ──
   751→  if (phase === "setup") {
   752→    return (
   753→      <>
   754→        <Card className="mx-auto max-w-xl rise-in">
   755→          <CardContent className="space-y-6 p-6">
   756→            <div className="flex items-center gap-3">
   757→              <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   758→                <Search size={22} />
   759→              </span>
   760→              <div>
   761→                <h2 className="text-lg font-black">شکار کلمه</h2>
   762→                <p className="text-xs text-muted-foreground">
   763→                  واژه‌های درس در جدول پنهان‌اند — با معنی فارسی شکارشان کن!
   764→                </p>
   765→              </div>
   766→            </div>
   767→
   768→            <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   769→              <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   770→              <p>
   771→                انگشت یا ماوس را از حرف اول تا آخر واژه بکش و رها کن؛ یا اول حرف اول و بعد حرف آخر را
   772→                بزن — افقی، عمودی یا مورب. سرنخ‌ها معنی فارسی‌اند.
   773→              </p>
   774→            </div>
   775→
   776→            <div>
   777→              <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   778→              <div className="flex flex-wrap gap-2" role="group" aria-label="انتخاب درس">
   779→                <button
   780→                  type="button"
   781→                  aria-pressed={lesson === ALL_LESSONS}
   782→                  onClick={() => setLesson(ALL_LESSONS)}
   783→                  className={chipCls(lesson === ALL_LESSONS)}
   784→                >
   785→                  کل کتاب ({faNum(allValidCount)} واژه)
   786→                </button>
   787→                {data.vocab.lessons
   788→                  .filter((l) => l.lesson > 0)
   789→                  .map((l) => (
   790→                    <button
   791→                      key={l.lesson}
   792→                      type="button"
   793→                      aria-pressed={lesson === l.lesson}
   794→                      onClick={() => setLesson(l.lesson)}
   795→                      className={chipCls(lesson === l.lesson)}
   796→                    >
   797→                      {l.label}
   798→                    </button>
   799→                  ))}
   800→              </div>
