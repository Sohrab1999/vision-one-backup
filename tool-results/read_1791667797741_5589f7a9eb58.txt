    83→/** اگر جمله‌های بازهٔ عادی کمتر از این شد، بازه گسترده می‌شود */
    84→const WIDEN_THRESHOLD = 10;
    85→const MIN_WORDS_WIDE = 3;
    86→const MAX_WORDS_WIDE = 16;
    87→/** مکث پس از چیدن کامل جمله (ms) */

    89→/** مکث پس از رد کردن (ms) */
    90→const SKIP_LOCK_MS = 400;

    92→const KIND_LABEL: Record<SentenceKind, string> = {

    94→  conversation: "مکالمه",
    95→  dict: "دیکشنری",

    98→/**
    99→ * واژهٔ مجاز: فقط حروف/رقم/آندرلاین + آپاستروف، با حداکثر یک نقطه/کاما/علامت
   100→ * سؤال پایانی — جمله‌های دارای dash خاص یا گیومهٔ تایپوگرافی حذف می‌شوند.
   101→ */
   102→const WORD_OK = /^[\w']+[.,?]?$/;
   103→
   104→/** استخر جمله‌های یک محدوده — با سنجش بازهٔ واژه و یکتاسازی متن */
   105→function buildPool(
   106→  data: BookData,
   107→  lesson: number,
   108→  minW: number,
   109→  maxW: number,
   110→): SentenceItem[] {
   111→  const seen = new Set<string>();
   112→  const items: SentenceItem[] = [];
   113→  const push = (en: string, fa: string, lessonNo: number, kind: SentenceKind, src?: string) => {
   114→    const text = en.trim();
   115→    if (!text) return;
   116→    const words = text.split(/\s+/);
   117→    if (words.length < minW || words.length > maxW) return;
   118→    for (const w of words) {
   119→      if (!WORD_OK.test(w)) return;
   120→    }
   121→    const key = text.toLowerCase();
   122→    if (seen.has(key)) return;
   123→    seen.add(key);
   124→    items.push({ en: text, fa: fa.trim(), lesson: lessonNo, kind, src });
   125→  };
   126→  const inScope = (l: number) => lesson === ALL_LESSONS || l === lesson;
   127→
   128→  for (const r of data.readings) {
   129→    if (!inScope(r.lesson)) continue;
   130→    for (const p of r.paragraphs) push(p.en, p.fa, r.lesson, "reading");
   131→  }
   132→  for (const c of data.conversations) {
   133→    if (!inScope(c.lesson)) continue;
   134→    for (const line of c.lines) push(line.en, line.fa, c.lesson, "conversation");
   135→  }
   136→  for (const d of data.dictionary) {
   137→    if (!inScope(d.lesson)) continue;
   138→    for (const ex of d.examples) push(ex, "", d.lesson, "dict", d.word);
   139→  }
   140→  return items;
   141→}

   143→export function SentenceBuilderGame() {
   144→  const data = useBookData();
   145→  const { bookId } = useBook();
   146→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا تضمینش می‌کند) — در عمل هرگز تهی نیست
   147→  const activeBookId: string = bookId ?? "";
   148→  const { speed } = useVoiceSpeed();
   149→
   150→  const [phase, setPhase] = useState<Phase>("setup");
   151→  const [lesson, setLesson] = useState<number>(ALL_LESSONS);
   152→  const [count, setCount] = useState<number>(3);
   153→
   154→  const [deck, setDeck] = useState<SentenceItem[]>([]);
   155→  const [idx, setIdx] = useState(0);
   156→  const [words, setWords] = useState<string[]>([]);
   157→  const [tiles, setTiles] = useState<Tile[]>([]);
   158→  const [slots, setSlots] = useState<(number | null)[]>([]);
   159→
   160→  const [solved, setSolved] = useState(0);
   161→  const [skipped, setSkipped] = useState(0);
   162→  const [hinted, setHinted] = useState(0);
   163→  const [secs, setSecs] = useState(0);
   164→  const [isRecord, setIsRecord] = useState(false);
   165→
   166→  /** انیمیشن match-pop روی ردیف جمله هنگام چیدن کامل درست */
   167→  const [pop, setPop] = useState(false);
   168→
   169→  const locked = useRef(false);
   170→  const timer = useRef<number | undefined>(undefined);
   171→  const secsRef = useRef(0);
   172→
   173→  const busy = pop;
   174→
   175→  // استخر جملهٔ محدودهٔ انتخابی — اگر بازهٔ عادی کم آمد، بازهٔ واژه گسترده می‌شود
   176→  const poolInfo = useMemo<{ items: SentenceItem[]; widened: boolean }>(() => {
   177→    const base = buildPool(data, lesson, MIN_WORDS, MAX_WORDS);
   178→    if (base.length >= WIDEN_THRESHOLD) return { items: base, widened: false };
   179→    return { items: buildPool(data, lesson, MIN_WORDS_WIDE, MAX_WORDS_WIDE), widened: true };
   180→  }, [data, lesson]);
   181→
   182→  // رکورد شخصی — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   183→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   184→  const best = useSyncExternalStore(
   185→    subscribeGameData,
   186→    () => {
   187→      const b = loadBest<BestRec>(activeBookId, GAME_KEY, scope);
   188→      return b ? b.time : null;
   189→    },
   190→    () => null,
   191→  );
   192→
   193→  // ── تایمر (functional) + شمارندهٔ دقیق برای رکورد ──
   194→  useEffect(() => {
   195→    if (phase !== "playing") return;
   196→    const t = window.setInterval(() => {
   197→      secsRef.current += 1;
   198→      setSecs((s) => s + 1);
   199→    }, 1000);
   200→    return () => window.clearInterval(t);
   201→  }, [phase]);
   202→
   203→  // ── پاک‌سازی نهایی: تلفظ + تایمر معلق ──
   204→  useEffect(() => {
   205→    return () => {
   206→      stopSpeaking();
   207→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   208→    };
   209→  }, []);
   210→
   211→  /** بارگذاری جملهٔ بعدی — واژه‌های به‌هم‌ریخته + خانه‌های خالی */
   212→  const loadSentence = useCallback((item: SentenceItem) => {
   213→    const ws = item.en.trim().split(/\s+/);
   214→    setWords(ws);
   215→    setTiles(shuffle(ws.map((word, i) => ({ id: i, word }))));
   216→    setSlots(Array.from({ length: ws.length }, () => null as number | null));
   217→    setPop(false);
   218→  }, []);
   219→
   220→  /** پایان دور — استریک/جلسه/Attempt/رکورد */
   221→  const finishRun = useCallback(
   222→    (finalSolved: number) => {
   223→      touchStreak();
   224→      recordStudySession();
   225→      bumpPlays(activeBookId, GAME_KEY);
   226→      saveAttempt(activeBookId, GAME_KEY, {
   227→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   228→        total: deck.length,
   229→        correct: finalSolved,
   230→        duration: secsRef.current,
   231→        detail: { hints: hinted, skipped },
   232→      });
   233→      // رکورد فقط برای دورِ کاملِ بدون رد کردن و بدون راهنما
   234→      if (skipped === 0 && hinted === 0) {
   235→        const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${count}`;
   236→        const prev = loadBest<BestRec>(activeBookId, GAME_KEY, recScope);
   237→        if (!prev || secsRef.current < prev.time) {
   238→          saveBest<BestRec>(activeBookId, GAME_KEY, recScope, { time: secsRef.current });
   239→          setIsRecord(true);
   240→        }
   241→      }
   242→      setSecs(secsRef.current);
   243→      setPhase("done");
   244→    },
   245→    [activeBookId, lesson, count, deck.length, hinted, skipped],
   246→  );
   247→
   248→  /** ادامه به جملهٔ بعد یا پایان دور */
   249→  const advanceOrFinish = useCallback(
   250→    (nextIdx: number, finalSolved: number) => {
   251→      if (nextIdx >= deck.length) {
   252→        finishRun(finalSolved);
   253→        return;
   254→      }
   255→      setIdx(nextIdx);
   256→      loadSentence(deck[nextIdx]);
   257→    },
   258→    [deck, loadSentence, finishRun],
   259→  );
   260→
   261→  /** چیدن کامل درست — پاپ + تلفظ جملهٔ کامل و بعد جملهٔ بعد */
   262→  const solveSentence = useCallback(
   263→    (item: SentenceItem, finalSolved: number) => {
   264→      locked.current = true;
   265→      setPop(true);
   266→      speak(item.en, { speed });
   267→      timer.current = window.setTimeout(() => {
   268→        locked.current = false;
   269→        setPop(false);
   270→        advanceOrFinish(idx + 1, finalSolved);
   271→      }, SOLVE_LOCK_MS);
   272→    },
   273→    [speed, idx, advanceOrFinish],
   274→  );
   275→
   276→  /** گذاشتن یک واژه از استخر در اولین خانهٔ خالی + بررسی چیدن کامل */
   277→  const placeTile = useCallback(
   278→    (tileId: number) => {
   279→      if (locked.current) return;
   280→      const current = deck[idx];
   281→      if (!current) return;
   282→      const empty = slots.indexOf(null);
   283→      if (empty === -1) return;
   284→      const tile = tiles.find((t) => t.id === tileId);
   285→      if (!tile) return;
   286→      const nextSlots = [...slots];
   287→      nextSlots[empty] = tileId;
   288→      setSlots(nextSlots);
   289→      // همهٔ خانه‌ها پر شد؟ ترتیب چیده‌شده را با جملهٔ پایه بسنجیم
   290→      if (nextSlots.every((s) => s !== null)) {
   291→        const attempt = nextSlots.map((s) => tiles.find((t) => t.id === s)?.word ?? "");
   292→        const ok = attempt.every((w, i) => w === words[i]);
   293→        if (ok) {
   294→          const finalSolved = solved + 1;
   295→          setSolved(finalSolved);
   296→          solveSentence(current, finalSolved);
   297→        }
   298→      }
   299→    },
   300→    [deck, idx, slots, tiles, words, solved, solveSentence],
   301→  );
   302→
   303→  /** برگرداندن واژهٔ یک خانه به استخر (کلیک روی خانهٔ پر) */
   304→  const returnAt = useCallback((index: number) => {
   305→    if (locked.current) return;
   306→    setSlots((prev) => {
   307→      if (prev[index] === null) return prev;
   308→      const next = [...prev];
   309→      next[index] = null;
   310→      return next;
   311→    });
   312→  }, []);
   313→
   314→  /** پاک کردن آخرین واژه (دکمه یا Backspace) */
   315→  const returnLast = useCallback(() => {
   316→    if (locked.current) return;
   317→    setSlots((prev) => {
   318→      for (let i = prev.length - 1; i >= 0; i--) {
   319→        if (prev[i] !== null) {
   320→          const next = [...prev];
   321→          next[i] = null;
   322→          return next;
   323→        }
   324→      }
   325→      return prev;
   326→    });
   327→  }, []);
   328→
   329→  /** پاک کردن همه — همهٔ واژه‌ها به استخر برمی‌گردند */
   330→  const clearAll = useCallback(() => {
   331→    if (locked.current) return;
   332→    setSlots((prev) => (prev.every((s) => s === null) ? prev : prev.map(() => null)));
   333→  }, []);
   334→
   335→  /** راهنمای دوم — یک واژهٔ درستِ بعدی را جای‌گذاری می‌کند (رکورد دور می‌سوزد) */
   336→  const applyHint = useCallback(() => {
   337→    if (locked.current) return;
   338→    const empty = slots.indexOf(null);
   339→    if (empty === -1) return;
   340→    const used = new Set(slots.filter((s): s is number => s !== null));
   341→    const tile = tiles.find((t) => !used.has(t.id) && t.word === words[empty]);
   342→    if (!tile) return;
   343→    setHinted((h) => h + 1);
   344→    placeTile(tile.id);
   345→  }, [slots, tiles, words, placeTile]);
   346→
   347→  /** رد کردن — بدون SRS، فقط شمارش و جملهٔ بعد (رکورد دور می‌سوزاند) */
   348→  const skipSentence = useCallback(() => {
   349→    if (locked.current) return;
   350→    const current = deck[idx];
   351→    if (!current) return;
   352→    locked.current = true;
   353→    setSkipped((s) => s + 1);
   354→    timer.current = window.setTimeout(() => {
   355→      locked.current = false;
   356→      advanceOrFinish(idx + 1, solved);
   357→    }, SKIP_LOCK_MS);
   358→  }, [deck, idx, solved, advanceOrFinish]);
   359→
   360→  /** انصراف — تایمر معلق پاک و به تنظیمات برگردیم */
   361→  const cancel = useCallback(() => {
   362→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   363→    locked.current = false;
   364→    stopSpeaking();
   365→    setPhase("setup");
   366→  }, []);
   367→
   368→  /** شروع دور — انتخاب جمله‌ها از استخر (اگر کم بود، تکرار با shuffle مجاز است) */
   369→  const start = useCallback(() => {
   370→    const items = poolInfo.items;
   371→    if (items.length === 0) {
   372→      toast.error("جملهٔ قابل بازی در این محدوده پیدا نشد — درس دیگری را امتحان کن");
   373→      return;
   374→    }
   375→    const picked: SentenceItem[] = [];
   376→    while (picked.length < count) {
   377→      const batch = shuffle(items).slice(0, Math.min(count - picked.length, items.length));
   378→      picked.push(...batch);
   379→    }
   380→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   381→    locked.current = false;
   382→    secsRef.current = 0;
   383→    setDeck(picked);
   384→    setIdx(0);
   385→    setSolved(0);
   386→    setSkipped(0);
   387→    setHinted(0);
   388→    setSecs(0);
   389→    setIsRecord(false);
   390→    loadSentence(picked[0]);
   391→    setPhase("playing");
   392→  }, [poolInfo, count, loadSentence]);
   393→
   394→  // ── کیبورد (latest-ref): Backspace آخرین واژهٔ چیده‌شده را برمی‌گرداند ──
   395→  const returnLastRef = useRef<() => void>(() => {});
   396→  useEffect(() => {
   397→    returnLastRef.current = returnLast;
   398→  });
   399→  useEffect(() => {
   400→    if (phase !== "playing") return;
   401→    const onKey = (e: KeyboardEvent) => {
   402→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   403→      if (e.key === "Backspace") {
   404→        e.preventDefault();
   405→        returnLastRef.current();
   406→      }
   407→    };
   408→    window.addEventListener("keydown", onKey);
   409→    return () => window.removeEventListener("keydown", onKey);
   410→  }, [phase]);
   411→
   412→  const current = deck[idx];
   413→  const usedIds = new Set(slots.filter((s): s is number => s !== null));
   414→  const wordOf = (id: number): string => tiles.find((t) => t.id === id)?.word ?? "";
   415→  const pct = Math.round(((solved + skipped) / Math.max(deck.length, 1)) * 100);
   416→
   417→  // ── Setup ──
   418→  if (phase === "setup") {
   419→    return (
   420→      <Card className="mx-auto max-w-xl rise-in">
   421→        <CardContent className="space-y-6 p-6">
   422→          <div className="flex items-center gap-3">
   423→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-teal-500/15 text-teal-600 dark:text-teal-400">
   424→              <Blocks size={22} />
   425→            </span>
   426→            <div>
   427→              <h2 className="text-lg font-black">سازنده جمله</h2>
   428→              <p className="text-xs text-muted-foreground">
   429→                واژه‌های جمله به‌هم ریخته‌اند — از روی ترجمه، جمله را درست بچین!
   430→              </p>
   431→            </div>
   432→          </div>
   433→
   434→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   435→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   436→            <p>
   437→              روی واژه‌ها به ترتیب بزن تا جملهٔ انگلیسی ساخته شود؛ ترجمهٔ فارسی از ابتدا نمایان است.
   438→              رد کردن یا راهنما رکورد را می‌سوزاند — رکورد: سریع‌ترین دور کامل!
   439→            </p>
   440→          </div>
   441→
   442→          <div>
   443→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   444→            <div className="flex flex-wrap gap-2" role="group" aria-label="انتخاب درس">
   445→              <button
   446→                type="button"
   447→                onClick={() => setLesson(ALL_LESSONS)}
   448→                aria-pressed={lesson === ALL_LESSONS}
   449→                className={cn(
   450→                  "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   451→                  lesson === ALL_LESSONS
   452→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   453→                    : "text-muted-foreground hover:bg-muted",
   454→                )}
   455→              >
