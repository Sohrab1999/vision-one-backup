     1→"use client";
     2→
     3→/**
     4→ * بازی «حروف جابه‌جا» (Word Scramble) — فاز ۳ نقشه راه جامع ۲۰۲۶.
     5→ *
     6→ * حروف واژه‌های انگلیسی به‌هم ریخته می‌شوند؛ بازیکن با معنی فارسی حروف را
     7→ * با کلیک (یا کیبورد) سر جای خودش می‌چیند.
     8→ * - حل درست: recordSRS(✓) + پخش تلفظ
     9→ * - رد کردن: recordSRS(✗) + نمایش واژهٔ درست برای ۱ ثانیه
    10→ * - رکورد: سریع‌ترین اجرای کامل (بدون رد/راهنما) به تفکیک درس و تعداد واژه
    11→ */
    12→
    13→import { useCallback, useEffect, useMemo, useRef, useState, useSyncExternalStore } from "react";
    14→import {
    15→  Shuffle,
    16→  Play,
    17→  RotateCcw,
    18→  Trophy,
    19→  Timer,
    20→  Target,
    21→  CheckCircle2,
    22→  XCircle,
    23→  Eraser,
    24→  Lightbulb,
    25→  SkipForward,
    26→  Volume2,
    27→  Sparkles,
    28→  PartyPopper,
    29→} from "lucide-react";
    30→import { Button } from "@/components/ui/button";
    31→import { Card, CardContent } from "@/components/ui/card";
    32→import { Badge } from "@/components/ui/badge";
    33→import { cn } from "@/lib/utils";
    34→import { toast } from "sonner";
    35→import type { VocabEntry } from "@/data/types";
    36→import { useBook, useBookData } from "@/lib/book-context";
    37→import { recordSRS } from "@/lib/srs-store";
    38→import { touchStreak } from "@/lib/streak-store";
    39→import { recordStudySession } from "@/lib/goal-store";
    40→import { speak, stopSpeaking } from "@/lib/speak";
    41→import { useVoiceSpeed } from "@/lib/voice-store";
    42→import {
    43→  shuffle,
    44→  faNum,
    45→  fmtTime,
    46→  pickUnique,
    47→  loadBest,
    48→  saveBest,
    49→  bumpPlays,
    50→  saveAttempt,
    51→  subscribeGameData,
    52→} from "./shared";
    53→
    54→type Phase = "setup" | "playing" | "done";
    55→
    56→/** حرف با شناسه یکتا — حروف تکراری مستقل مدیریت می‌شوند (اینکس‌محور) */
    57→interface Tile {
    58→  id: number;
    59→  letter: string;
    60→}
    61→
    62→/** شکل رکورد شخصی این بازی */
    63→interface BestRec {
    64→  time: number;
    65→}
    66→
    67→/** ردیف جزئیات دور — برای Attempt سروری و فهرست ردشده‌ها */
    68→interface DetailRow {
    69→  word: string;
    70→  ok: boolean;
    71→}
    72→
    73→const ALL_LESSONS = -1;
    74→const ROUND_OPTIONS = [5, 8, 12];
    75→
    76→/** فقط واژه‌های تک‌کلمه‌ای لاتینِ ۳ تا ۱۲ حرفی وارد بازی می‌شوند */
    77→const WORD_RE = /^[A-Za-z]{3,12}$/;
    78→
    79→/** واژهٔ پایه — حذف نشانه‌های دستوری داخل پرانتز: "dry (adj.)" → "dry" */
    80→function baseWord(w: string): string {
    81→  return w.replace(/\s*\([^)]*\)\s*/g, " ").trim();
    82→}
    83→
    84→export function ScrambleGame() {
    85→  const data = useBookData();
    86→  const { bookId: activeBookId } = useBook();
    87→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا تضمینش می‌کند) — در عمل هرگز تهی نیست
    88→  const bookId: string = activeBookId ?? "";
    89→  const { speed } = useVoiceSpeed();
    90→
    91→  const [phase, setPhase] = useState<Phase>("setup");
    92→  const [lesson, setLesson] = useState<number>(ALL_LESSONS);
    93→  const [rounds, setRounds] = useState<number>(8);
    94→
    95→  const [deck, setDeck] = useState<VocabEntry[]>([]);
    96→  const [idx, setIdx] = useState(0);
    97→  const [tiles, setTiles] = useState<Tile[]>([]);
    98→  const [slots, setSlots] = useState<(number | null)[]>([]);
    99→
   100→  const [solved, setSolved] = useState(0);
   101→  const [skipped, setSkipped] = useState(0);
   102→  const [hinted, setHinted] = useState(0);
   103→  const [detail, setDetail] = useState<DetailRow[]>([]);
   104→  const [secs, setSecs] = useState(0);
   105→  const [isRecord, setIsRecord] = useState(false);
   106→
   107→  /** انیمیشن match-pop روی ردیف پاسخ هنگام حل درست */
   108→  const [pop, setPop] = useState(false);
   109→  /** واژهٔ درستی که بعد از رد کردن برای ۱ ثانیه نشان داده می‌شود */
   110→  const [reveal, setReveal] = useState<string | null>(null);
   111→
   112→  const locked = useRef(false);
   113→  const timer = useRef<number | undefined>(undefined);
   114→  const secsRef = useRef(0);
   115→
   116→  const busy = pop || reveal !== null;
   117→
   118→  // استخر قابل بازی — فقط واژه‌های تک‌کلمه‌ای ۳ تا ۱۲ حرفی
   119→  const pool = useMemo(() => {
   120→    const base = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   121→    return base.filter((w) => WORD_RE.test(baseWord(w.word)));
   122→  }, [data.vocab, lesson]);
   123→
   124→  // رکورد شخصی — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   125→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   126→  const best = useSyncExternalStore(
   127→    subscribeGameData,
   128→    () => {
   129→      const b = loadBest<BestRec>(bookId, "scramble", scope);
   130→      return b ? b.time : null;
   131→    },
   132→    () => null,
   133→  );
   134→
   135→  // ── تایمر (functional) + شمارندهٔ دقیق برای رکورد ──
   136→  useEffect(() => {
   137→    if (phase !== "playing") return;
   138→    const t = window.setInterval(() => {
   139→      secsRef.current += 1;
   140→      setSecs((s) => s + 1);
   141→    }, 1000);
   142→    return () => window.clearInterval(t);
   143→  }, [phase]);
   144→
   145→  // ── پاک‌سازی نهایی: تلفظ + تایمرهای معلق ──
   146→  useEffect(() => {
   147→    return () => {
   148→      stopSpeaking();
   149→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   150→    };
   151→  }, []);
   152→
   153→  /** بارگذاری واژهٔ بعدی — حروف به‌هم‌ریخته + خانه‌های خالی */
   154→  const loadWord = useCallback((entry: VocabEntry) => {
   155→    const base = baseWord(entry.word);
   156→    setTiles(shuffle(base.split("").map((letter, i) => ({ id: i, letter }))));
   157→    setSlots(Array.from({ length: base.length }, () => null as number | null));
   158→    setPop(false);
   159→  }, []);
   160→
   161→  /** پایان دور — استریک/جلسه/Attempt/رکورد */
   162→  const finishRun = useCallback(
   163→    (finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   164→      touchStreak();
   165→      recordStudySession();
   166→      bumpPlays(bookId, "scramble");
   167→      saveAttempt(bookId, "scramble", {
   168→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   169→        total: deck.length,
   170→        correct: finalSolved,
   171→        duration: secsRef.current,
   172→        detail: finalDetail,
   173→      });
   174→      // رکورد فقط برای اجرای کاملِ بدون رد کردن و بدون راهنما
   175→      if (finalSkipped === 0 && hinted === 0) {
   176→        const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   177→        const prev = loadBest<BestRec>(bookId, "scramble", recScope);
   178→        if (!prev || secsRef.current < prev.time) {
   179→          saveBest<BestRec>(bookId, "scramble", recScope, { time: secsRef.current });
   180→          setIsRecord(true);
   181→        }
   182→      }
   183→      setPhase("done");
   184→    },
   185→    [bookId, lesson, rounds, deck.length, hinted],
   186→  );
   187→
   188→  /** ادامه به واژهٔ بعد یا پایان دور */
   189→  const advanceOrFinish = useCallback(
   190→    (nextIdx: number, finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   191→      if (nextIdx >= deck.length) {
   192→        finishRun(finalSolved, finalSkipped, finalDetail);
   193→        return;
   194→      }
   195→      setIdx(nextIdx);
   196→      loadWord(deck[nextIdx]);
   197→    },
   198→    [deck, loadWord, finishRun],
   199→  );
   200→
   201→  /** حل درست — پاپ + تلفظ + SRS(✓) و بعد واژهٔ بعد */
   202→  const solveWord = useCallback(
   203→    (entry: VocabEntry, finalSolved: number, finalDetail: DetailRow[]) => {
   204→      locked.current = true;
   205→      setPop(true);
   206→      const base = baseWord(entry.word);
   207→      recordSRS(base, true);
   208→      speak(base, { speed });
   209→      timer.current = window.setTimeout(() => {
   210→        locked.current = false;
   211→        setPop(false);
   212→        advanceOrFinish(idx + 1, finalSolved, skipped, finalDetail);
   213→      }, 650);
   214→    },
   215→    [speed, idx, skipped, advanceOrFinish],
   216→  );
   217→
   218→  /** گذاشتن یک حرف از استخر در اولین خانهٔ خالی + بررسی حل شدن */
   219→  const placeTile = useCallback(
   220→    (tileId: number) => {
   221→      if (locked.current) return;
   222→      const current = deck[idx];
   223→      if (!current) return;
   224→      const empty = slots.indexOf(null);
   225→      if (empty === -1) return;
   226→      const tile = tiles.find((t) => t.id === tileId);
   227→      if (!tile) return;
   228→      const nextSlots = [...slots];
   229→      nextSlots[empty] = tileId;
   230→      setSlots(nextSlots);
   231→      // همهٔ خانه‌ها پر شد؟ تطبیق با واژهٔ پایه را بسنجیم
   232→      if (nextSlots.every((s) => s !== null)) {
   233→        const base = baseWord(current.word);
   234→        const attempt = nextSlots
   235→          .map((s) => tiles.find((t) => t.id === s)?.letter ?? "")
   236→          .join("");
   237→        if (attempt === base) {
   238→          const finalSolved = solved + 1;
   239→          const nextDetail = [...detail, { word: base, ok: true }];
   240→          setSolved(finalSolved);
   241→          setDetail(nextDetail);
   242→          solveWord(current, finalSolved, nextDetail);
   243→        }
   244→      }
   245→    },
   246→    [deck, idx, slots, tiles, solved, detail, solveWord],
   247→  );
   248→
   249→  /** برگرداندن حرفِ یک خانه به استخر (کلیک روی خانهٔ پر) */
   250→  const returnAt = useCallback((index: number) => {
   251→    if (locked.current) return;
   252→    setSlots((prev) => {
   253→      if (prev[index] === null) return prev;
   254→      const next = [...prev];
   255→      next[index] = null;
   256→      return next;
   257→    });
   258→  }, []);
   259→
   260→  /** پاک کردن آخرین حرف (دکمه یا Backspace) */
   261→  const returnLast = useCallback(() => {
   262→    if (locked.current) return;
   263→    setSlots((prev) => {
   264→      for (let i = prev.length - 1; i >= 0; i--) {
   265→        if (prev[i] !== null) {
   266→          const next = [...prev];
   267→          next[i] = null;
   268→          return next;
   269→        }
   270→      }
   271→      return prev;
   272→    });
   273→  }, []);
   274→
   275→  /** راهنما — یک حرف درستِ بعدی را جای‌گذاری می‌کند (رکورد دور می‌سوزد) */
   276→  const applyHint = useCallback(() => {
   277→    if (locked.current) return;
   278→    const current = deck[idx];
   279→    if (!current) return;
   280→    const empty = slots.indexOf(null);
   281→    if (empty === -1) return;
   282→    const base = baseWord(current.word);
   283→    const used = new Set(slots.filter((s): s is number => s !== null));
   284→    const tile = tiles.find((t) => !used.has(t.id) && t.letter === base[empty]);
   285→    if (!tile) return;
   286→    setHinted((h) => h + 1);
   287→    placeTile(tile.id);
   288→  }, [deck, idx, slots, tiles, placeTile]);
   289→
   290→  /** رد کردن — SRS(✗) + نمایش واژهٔ درست برای ۱ ثانیه (رکورد دور می‌سوزد) */
   291→  const skipWord = useCallback(() => {
   292→    if (locked.current) return;
   293→    const current = deck[idx];
   294→    if (!current) return;
   295→    locked.current = true;
   296→    const base = baseWord(current.word);
   297→    recordSRS(base, false);
   298→    const nextSkipped = skipped + 1;
   299→    const nextDetail = [...detail, { word: base, ok: false }];
   300→    setSkipped(nextSkipped);
   301→    setDetail(nextDetail);
   302→    setReveal(base);
   303→    timer.current = window.setTimeout(() => {
   304→      locked.current = false;
   305→      setReveal(null);
   306→      advanceOrFinish(idx + 1, solved, nextSkipped, nextDetail);
   307→    }, 1000);
   308→  }, [deck, idx, skipped, detail, solved, advanceOrFinish]);
   309→
   310→  /** انصراف — تایمرهای معلق پاک و به تنظیمات برگردیم */
   311→  const cancel = useCallback(() => {
   312→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   313→    locked.current = false;
   314→    stopSpeaking();
   315→    setPhase("setup");
   316→  }, []);
   317→
   318→  /** شروع دور — انتخاب واژه‌های یکتا از استخر قابل بازی */
   319→  const start = useCallback(() => {
   320→    const chosen = pickUnique(pool, rounds, (w) => baseWord(w.word));
   321→    if (chosen.length < 3) {
   322→      toast.error("واژه قابل بازی در این محدوده پیدا نشد");
   323→      return;
   324→    }
   325→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   326→    locked.current = false;
   327→    secsRef.current = 0;
   328→    setDeck(chosen);
   329→    setIdx(0);
   330→    setSolved(0);
   331→    setSkipped(0);
   332→    setHinted(0);
   333→    setDetail([]);
   334→    setSecs(0);
   335→    setIsRecord(false);
   336→    setReveal(null);
   337→    loadWord(chosen[0]);
   338→    setPhase("playing");
   339→  }, [pool, rounds, loadWord]);
   340→
   341→  // ── کیبورد: فقط در فاز بازی؛ حروف → جای‌گذاری، Backspace → برگرداندن ──
   342→  useEffect(() => {
   343→    if (phase !== "playing") return;
   344→    const onKey = (e: KeyboardEvent) => {
   345→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   346→      if (e.key === "Backspace") {
   347→        e.preventDefault();
   348→        returnLast();
   349→        return;
   350→      }
   351→      if (e.key === "Enter") return; // حل خودکار است — اینتر کاری نمی‌کند
   352→      if (/^[a-zA-Z]$/.test(e.key)) {
   353→        const letter = e.key.toLowerCase();
   354→        const used = new Set(slots.filter((s): s is number => s !== null));
   355→        const tile = tiles.find((t) => !used.has(t.id) && t.letter.toLowerCase() === letter);
   356→        if (tile) placeTile(tile.id);
   357→      }
   358→    };
   359→    window.addEventListener("keydown", onKey);
   360→    return () => window.removeEventListener("keydown", onKey);
   361→  }, [phase, slots, tiles, placeTile, returnLast]);
   362→
   363→  const current = deck[idx];
   364→  const usedIds = new Set(slots.filter((s): s is number => s !== null));
   365→  const letterOf = (id: number): string => tiles.find((t) => t.id === id)?.letter ?? "";
   366→  const pct = Math.round(((solved + skipped) / Math.max(deck.length, 1)) * 100);
   367→
   368→  // ── Setup ──
   369→  if (phase === "setup") {
   370→    return (
   371→      <Card className="mx-auto max-w-xl rise-in">
   372→        <CardContent className="space-y-6 p-6">
   373→          <div className="flex items-center gap-3">
   374→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-amber-500/15 text-amber-600 dark:text-amber-400">
   375→              <Shuffle size={22} />
   376→            </span>
   377→            <div>
   378→              <h2 className="text-lg font-black">حروف جابه‌جا</h2>
   379→              <p className="text-xs text-muted-foreground">
   380→                حروف واژه‌ها ریخته شده — با معنی فارسی سرهمشان کن!
   381→              </p>
   382→            </div>
   383→          </div>
   384→
   385→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   386→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   387→            <p>
   388→              روی حروف به ترتیب بزن تا واژه ساخته شود؛ کیبورد هم کار می‌کند. واژه‌های سخت را می‌توانی رد کنی ولی رکورد می‌سوزد!
   389→            </p>
   390→          </div>
   391→
   392→          <div>
   393→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   394→            <div className="flex flex-wrap gap-2">
   395→              <button
   396→                type="button"
   397→                onClick={() => setLesson(ALL_LESSONS)}
   398→                className={cn(
   399→                  "rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   400→                  lesson === ALL_LESSONS
   401→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   402→                    : "text-muted-foreground hover:bg-muted",
   403→                )}
   404→              >
   405→                کل کتاب ({faNum(data.vocab.all.length)} واژه)
   406→              </button>
   407→              {data.vocab.lessons.filter((l) => l.lesson > 0).map((l) => (
   408→                <button
   409→                  key={l.lesson}
   410→                  type="button"
   411→                  onClick={() => setLesson(l.lesson)}
   412→                  className={cn(
   413→                    "rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   414→                    lesson === l.lesson
   415→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   416→                      : "text-muted-foreground hover:bg-muted",
   417→                  )}
   418→                >
   419→                  {l.label}
   420→                </button>
   421→              ))}
   422→            </div>
   423→          </div>
   424→
   425→          <div>
   426→            <p className="mb-2 text-[13px] font-bold">چند واژه؟</p>
   427→            <div className="flex flex-wrap gap-2">
   428→              {ROUND_OPTIONS.map((n) => (
   429→                <button
   430→                  key={n}
   431→                  type="button"
   432→                  onClick={() => setRounds(n)}
   433→                  className={cn(
   434→                    "rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   435→                    rounds === n
   436→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   437→                      : "text-muted-foreground hover:bg-muted",
   438→                  )}
   439→                >
   440→                  {faNum(n)} واژه
   441→                </button>
   442→              ))}
   443→            </div>
   444→          </div>
   445→
   446→          {best !== null && (
   447→            <Badge variant="outline" className="gap-1.5 border-amber-500/40 text-amber-600 dark:text-amber-400">
   448→              <Trophy size={12} />
   449→              سریع‌ترین اجرا: {fmtTime(best)}
   450→            </Badge>
   451→          )}
   452→
   453→          <Button size="lg" className="h-12 w-full gap-2 text-base" onClick={start}>
   454→            <Play size={18} />
   455→            شروع بازی
   456→          </Button>
   457→        </CardContent>
   458→      </Card>
   459→    );
   460→  }
   461→
   462→  // ── Done ──
   463→  if (phase === "done") {
   464→    const skippedRows = detail
   465→      .filter((d) => !d.ok)
   466→      .map((d) => ({
   467→        word: d.word,
   468→        meaning: deck.find((w) => baseWord(w.word) === d.word)?.meaning ?? "",
   469→      }));
   470→    return (
   471→      <Card className="mx-auto max-w-xl rise-in border-amber-500/30">
   472→        <CardContent className="flex flex-col items-center gap-4 p-8 text-center">
   473→          <span className="flex h-16 w-16 items-center justify-center rounded-full bg-amber-500/15 text-amber-600 dark:text-amber-400">
   474→            <PartyPopper size={30} />
   475→          </span>
   476→          <h2 className="text-xl font-black">تمام شد!</h2>
   477→          {isRecord && (
   478→            <Badge className="gap-1.5 bg-amber-500 text-amber-950 shadow-md shadow-amber-500/30">
   479→              <Trophy size={13} />
   480→              رکورد جدید! 🏆
   481→            </Badge>
