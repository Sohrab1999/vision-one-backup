    70→function useWordOfDay(vocab: VocabEntry[]) {
    71→  return useMemo(() => {
    72→    const day = Math.floor(Date.now() / 86400000);
    73→    return vocab[day % vocab.length];
    74→  }, [vocab]);
    75→}
