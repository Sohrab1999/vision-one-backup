   104→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
   105→function Confetti() {
   106→  const pieces = useMemo(
   107→    () =>
   108→      Array.from({ length: 32 }, (_, i) => ({
   109→        id: i,
   110→        left: Math.random() * 100,
   111→        delay: Math.random() * 1.2,
   112→        duration: 2.4 + Math.random() * 1.6,
   113→        size: 6 + Math.random() * 6,
   114→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   115→        drift: Math.round((Math.random() - 0.5) * 120),
   116→        round: Math.random() > 0.5,
   117→      })),
   118→    [],
   119→  );
   120→  return (
   121→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   122→      {pieces.map((p) => (
   123→        <span
   124→          key={p.id}
   125→          className="confetti-piece absolute top-[-14px] block"
   126→          style={{
   127→            left: `${p.left}%`,
   128→            width: p.size,
   129→            height: p.size * (p.round ? 1 : 0.45),
   130→            background: p.color,
   131→            borderRadius: p.round ? "50%" : 2,
   132→            animationDelay: `${p.delay}s`,
   133→            animationDuration: `${p.duration}s`,
   134→            "--confetti-drift": `${p.drift}px`,
   135→          } as React.CSSProperties}
   136→        />
   137→      ))}
   138→    </div>
   139→  );
   140→}

   403→  function chooseOption(optIdx: number) {
   404→    if (selected !== null) return;
   405→    if (timed > 0 && deadlineRef.current > 0) {
   406→      answeredLeftRef.current = Math.max(0, Math.ceil((deadlineRef.current - Date.now()) / 1000));
   407→    }

   409→    setAnswers((prev) => {
   410→      const next = [...prev];
   411→      next[index] = optIdx;
   412→      return next;
   413→    });
   414→    if (current) {
   415→      if (optIdx === current.answer) {
   416→        // در مرور اشتباهات و مرور ترکیبی، پاسخ درست سؤال را از دفتر اشتباهات حذف می‌کند
   417→        if (mode === "review" || mode === "combined") resolveMistake(current.id);
   418→      } else {
   419→        recordMistake(current.id);
   420→      }
   421→    }
   422→  }

   452→  function nextQuestion() {
   453→    if (advanceRef.current) clearTimeout(advanceRef.current);
   454→    if (index + 1 >= questions.length) {
   455→      finish();
   456→      return;
   457→    }
   458→    setIndex((i) => i + 1);
   459→    setSelected(answers[index + 1]);
   460→    setAiExplanation(null);
   461→    deadlineRef.current = mode === "daily" || mode === "weekly" ? 0 : Date.now() + timed * 1000;
   462→  }

   464→  // میان‌برهای کیبورد حین آزمون: ۱-۴ (فارسی/انگلیسی) انتخاب گزینه، Enter سؤال بعدی
   465→  useEffect(() => {
   466→    if (phase !== "playing") return;
   467→    function onKey(e: KeyboardEvent) {
   468→      const target = e.target as HTMLElement | null;
   469→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA")) return;
   470→      if (e.key === "Enter" && selected !== null) {
   471→        e.preventDefault();
   472→        nextQuestion();
   473→        return;
   474→      }
   475→      if (selected !== null) return;
   476→      const fa = "۱۲۳۴".indexOf(e.key);
   477→      const en = "1234".indexOf(e.key);
   478→      const digit = fa >= 0 ? fa : en;
   479→      if (digit >= 0 && digit < (current?.options.length ?? 0)) {
   480→        e.preventDefault();
   481→        chooseOption(digit);
   482→      }
   483→    }
   484→    window.addEventListener("keydown", onKey);
   485→    return () => window.removeEventListener("keydown", onKey);
   486→  }, [phase, selected, index, questions]);

   488→  function finish() {

   546→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   547→  async function exportMistakesCsv() {
   548→    if (!mistakeIds.length || exporting) return;
   549→    setExporting(true);
   550→    try {
   551→      const res = await fetch(`/api/questions?count=${mistakeIds.length}&ids=${mistakeIds.join(",")}&book=${bookId}`);
   552→      const data = await res.json();
   553→      const qs = (data.questions ?? []) as Question[];
   554→      if (!qs.length) {
   555→        toast.error("سؤالی برای خروجی یافت نشد.");
   556→        return;
   557→      }
   558→      const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   559→      const rows = [
   560→        ["ردیف", "متن سؤال", "پاسخ صحیح", "دسته", "مبحث", "سطح"].join(","),
   561→        ...qs.map((q, i) =>
   562→          [
   563→            i + 1,
   564→            esc(q.text),
   565→            esc(q.options[q.answer] ?? ""),
   566→            q.category === "grammar" ? "گرامر" : "واژگان",
   567→            esc(q.scope),
   568→            esc(q.level),
   569→          ].join(","),
   570→        ),
   571→      ];
   572→      const blob = new Blob(["\uFEFF" + rows.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   573→      const url = URL.createObjectURL(blob);
   574→      const a = document.createElement("a");
   575→      a.href = url;
   576→      a.download = "vision-one-mistakes.csv";
   577→      document.body.appendChild(a);
   578→      a.click();
   579→      a.remove();
   580→      URL.revokeObjectURL(url);
   581→      toast.success(`فایل CSV با ${qs.length.toLocaleString("fa-IR")} سؤال دانلود شد.`);
   582→    } catch {
   583→      toast.error("خطا در ساخت فایل CSV");
   584→    } finally {
   585→      setExporting(false);
   586→    }
   587→  }
