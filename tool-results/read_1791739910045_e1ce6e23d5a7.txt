   200→   200→  for (const c of data.conversations) {
   201→   201→    for (const line of c.lines) {
   202→   202→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   203→   203→        const t = s.trim();
   204→   204→        if (t) out.push({ en: t, fa: line.fa });

   206→   206→    }
   207→   207→  }
   208→   208→  return out;

   213→   213→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   214→   214→ */
   215→   215→function buildPool(data: BookData, lesson: number, distractorCount = 2): ClozeItem[] {
   216→   216→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   217→   217→  const sentences = collectSentences(data);
   218→   218→
   219→   219→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   220→   220→  const seenBase = new Set<string>();
   221→   221→  const words: { base: string; entry: VocabEntry }[] = [];
   222→   222→  for (const entry of vocabPool) {
   223→   223→    const base = toBase(entry.word);
   224→   224→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   225→   225→    seenBase.add(base);
   226→   226→    words.push({ base, entry });
   227→   227→  }
   228→   228→
   229→   229→  const items: ClozeItem[] = [];
   230→   230→  for (const { base, entry } of words) {
   231→   231→    if (items.length >= MAX_ITEMS) break;
   232→   232→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   233→   233→    const found = sentences.find((s) => {
   234→   234→      if (s.en.length < 30 || s.en.length > 160) return false;
   235→   235→      const wc = wordCount(s.en);
   236→   236→      if (wc < 6 || wc > 35) return false;
   237→   237→      return re.test(s.en);
   238→   238→    });
   239→   239→    if (!found) continue;
   240→   240→
   241→   241→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   242→   242→    // (فاز G3-01 — تعداد حواس‌پرتی سطح‌دار؛ پیش‌فرض ۲ = همان رفتار قبلی)
   243→   243→    const distractorPool = vocabPool.filter((w) => {
   244→   244→      const b = toBase(w.word);
   245→   245→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   246→   246→    });
   247→   247→    const distractors = pickUnique(distractorPool, distractorCount, (w) => `${toBase(w.word)}|${w.meaning}`);
   248→   248→    if (distractors.length < distractorCount) continue;
   249→   249→
   250→   250→    items.push({
   251→   251→      id: `${lesson}:${base}`,
   252→   252→      sentence: found.en.replace(re, BLANK),
   253→   253→      fa: found.fa,
   254→   254→      answer: base,
   255→   255→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   256→   256→      meaning: entry.meaning,
   257→   257→      lessonLabel: entry.lessonLabel,
   258→   258→    });
   259→   259→  }
   260→   260→  return shuffle(items);
   261→   261→}

   283→   283→function isAiQuestionShape(v: unknown): v is AiQuestion {
   284→   284→  if (typeof v !== "object" || v === null) return false;
   285→   285→  const o = v as Record<string, unknown>;
   286→   286→  return (
   287→   287→    typeof o.word === "string" &&
   288→   288→    o.word.trim().length > 0 &&
   289→   289→    typeof o.sentence === "string" &&
   290→   290→    o.sentence.trim().length > 0 &&
   291→   291→    Array.isArray(o.options) &&
   292→   292→    o.options.length >= 2 &&
   293→   293→    o.options.length <= 6 &&
   294→   294→    o.options.every((x) => typeof x === "string" && x.trim().length > 0) &&
   295→   295→    typeof o.answerIndex === "number" &&
   296→   296→    Number.isInteger(o.answerIndex) &&
   297→   297→    o.answerIndex >= 0 &&
   298→   298→    o.answerIndex < o.options.length
   299→   299→  );
   300→   300→}

   302→   302→/**
   303→   303→ * صدا زدن سؤال‌ساز با مهلت ۸ ثانیه (AbortController + Promise.race با تایمر).
   304→   304→ * هر خطا/تایم‌اوت/پاسخ بد → null (بدون پرتاب استثنا؛ دور عادی ادامه می‌یابد).
   305→   305→ */
   306→   306→async function fetchAiQuestions(
   307→   307→  game: "cloze" | "speed-quiz",
   308→   308→  pool: { word: string; meaning: string }[],
   309→   309→  ctrl: AbortController,
   310→   310→): Promise<AiQuestion[] | null> {
   311→   311→  let timerId: number | null = null;
   312→   312→  const timeout = new Promise<null>((resolve) => {
   313→   313→    timerId = window.setTimeout(() => {
   314→   314→      ctrl.abort();
   315→   315→      resolve(null);
   316→   316→    }, AI_TIMEOUT_MS);
   317→   317→  });
   318→   318→  try {
   319→   319→    const resPromise = fetch("/api/game-questions", {
   320→   320→      method: "POST",
   321→   321→      headers: { "Content-Type": "application/json" },
   322→   322→      body: JSON.stringify({ game, pool, count: AI_COUNT }),
   323→   323→      signal: ctrl.signal,
   324→   324→    })
   325→   325→      .then(async (res) => (res.ok ? ((await res.json()) as { questions?: unknown }) : null))
   326→   326→      .catch(() => null); // لغو/شبکه → بی‌صدا (مسابقه منتظر نمی‌ماند)
   327→   327→    const res = await Promise.race([resPromise, timeout]);
   328→   328→    const raw = res && typeof res === "object" ? (res as { questions?: unknown }).questions : null;
   329→   329→    if (!Array.isArray(raw)) return null;
   330→   330→    const valid = raw.filter(isAiQuestionShape);
   331→   331→    return valid.length > 0 ? valid : null;
   332→   332→  } catch {
   333→   333→    return null;
   334→   334→  } finally {
   335→   335→    if (timerId !== null) window.clearTimeout(timerId);
   336→   336→  }
   337→   337→}

   341→   341→ * دقیقاً عضو استخر همین دور باشد و گزینهٔ درست عین همان واژه؛ جای خالیِ «___»
   342→   342→ * به قالب خود بازی (BLANK) تبدیل و گزینه‌ها هم‌تعداد سطح دور می‌شوند. سؤال‌های
   343→   343→ * مشکوک بی‌صدا رد می‌شوند (دور عادی جبران می‌کند).
   344→   344→ */
   345→   345→function aiQuestionsToCloze(
   346→   346→  questions: AiQuestion[],
   347→   347→  pool: ClozeItem[],
   348→   348→  optCount: number,
   349→   349→): ClozeItem[] {
   350→   350→  const byWord = new Map(pool.map((it) => [it.answer.toLowerCase(), it]));
   351→   351→  const out: ClozeItem[] = [];
   352→   352→  for (const q of questions) {
   353→   353→    const entry = byWord.get(q.word.trim().toLowerCase());
   354→   354→    if (!entry) continue; // واژه بیرون از استخر این دور → رد
   355→   355→    const answerOpt = q.options[q.answerIndex]?.trim().toLowerCase();
   356→   356→    if (answerOpt !== entry.answer.toLowerCase()) continue; // پاسخِ درست‌نما → رد
   357→   357→    const wrongs = q.options
   358→   358→      .map((o) => o.trim())
   359→   359→      .filter((o) => o.toLowerCase() !== entry.answer.toLowerCase());
   360→   360→    out.push({
   361→   361→      id: `ai:${entry.answer}`,
   362→   362→      sentence: q.sentence.replace(/_{3,}/g, BLANK),
   363→   363→      // جملهٔ AI ترجمهٔ کتاب ندارد — معنی واژهٔ هدف همان راهنمای فارسی پرسش است
   364→   364→      fa: `معنی واژهٔ گم‌شده: ${entry.meaning}`,
   365→   365→      answer: entry.answer,
   366→   366→      options: shuffle([entry.answer, ...shuffle(wrongs).slice(0, Math.max(0, optCount - 1))]),
   367→   367→      meaning: entry.meaning,
   368→   368→      lessonLabel: entry.lessonLabel,
   369→   369→      source: "ai",
   370→   370→    });
   371→   371→  }
   372→   372→  return out;
   373→   373→}

   375→   375→// ─── اسنپ‌شات پایدار رکورد (JSON.parse هر بار آبجکت تازه می‌سازد — باید کَش شود) ───
   376→   376→
   377→   377→let bestCacheKey = "";
   378→   378→let bestCacheValue: BestRecord | null = null;
   379→   379→
   380→   380→function readBest(bookId: string, scope: string): BestRecord | null {
   381→   381→  let raw: string | null = null;
   382→   382→  try {
   383→   383→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   384→   384→  } catch {
   385→   385→    return null;
   386→   386→  }
   387→   387→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   388→   388→  if (cacheKey === bestCacheKey) return bestCacheValue;
   389→   389→  let value: BestRecord | null = null;
   390→   390→  if (raw) {
   391→   391→    try {
   392→   392→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   393→   393→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   394→   394→        value = { score: parsed.score, total: parsed.total };
   395→   395→      }
   396→   396→    } catch {
   397→   397→      value = null;
   398→   398→    }
   399→   399→  }
   400→   400→  bestCacheKey = cacheKey;
   401→   401→  bestCacheValue = value;
   402→   402→  return value;
   403→   403→}
