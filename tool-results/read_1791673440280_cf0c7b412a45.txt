   110→   110→export interface GenerateOptions {

   118→   118→
   119→   119→const cellKey = (r: number, c: number): string => `${r},${c}`;
   120→   120→
   121→   121→const parseKey = (k: string): [number, number] => {
   122→   122→  const i = k.indexOf(",");
   123→   123→  return [Number(k.slice(0, i)), Number(k.slice(i + 1))];
   124→   124→};
   125→   125→
   126→   126→/** خانهٔ jام یک جای‌گذاری روی بوم خام */
   127→   127→const draftCellKey = (p: { row: number; col: number; dir: CrosswordDir }, j: number): string =>
   128→   128→  p.dir === "across" ? cellKey(p.row, p.col + j) : cellKey(p.row + j, p.col);
   129→   129→
   130→   130→/**
   131→   131→ * سازگاری یک جای‌گذاری را بررسی می‌کند:
   132→   132→ * - حرف‌های از قبل اشغال‌شده باید دقیقاً برابر باشند (حداکثر یک تقاطع)
   133→   133→ * - سلول قبل از آغاز و بعد از پایان واژه خالی باشد (واژه به واژهٔ دیگر نمی‌چسبد)
   134→   134→ * - برای سلول‌های غیرتقاطع، هر دو همسایهٔ عمود بر جهت واژه خالی باشند
   135→   135→ */
   136→   136→function canPlace(
   137→   137→  word: string,
   138→   138→  r0: number,
   139→   139→  c0: number,
   140→   140→  dir: CrosswordDir,
   141→   141→  cells: Map<string, string>,
   142→   142→): boolean {
   143→   143→  const dr = dir === "down" ? 1 : 0;
   144→   144→  const dc = dir === "across" ? 1 : 0;
   145→   145→  const len = word.length;
   146→   146→  if (cells.has(cellKey(r0 - dr, c0 - dc))) return false;
   147→   147→  if (cells.has(cellKey(r0 + dr * len, c0 + dc * len))) return false;
   148→   148→  let crossings = 0;
   149→   149→  for (let j = 0; j < len; j++) {
   150→   150→    const r = r0 + dr * j;
   151→   151→    const c = c0 + dc * j;
   152→   152→    const cur = cells.get(cellKey(r, c));
   153→   153→    if (cur !== undefined) {
   154→   154→      if (cur !== word[j] || crossings > 0) return false;
   155→   155→      crossings += 1;
   156→   156→    } else if (cells.has(cellKey(r - dc, c - dr)) || cells.has(cellKey(r + dc, c + dr))) {
   157→   157→      return false;
   158→   158→    }
   159→   159→  }
   160→   160→  return crossings === 1;
   161→   161→}

   163→   163→/** یک تلاش خام ساخت — خروجی روی بوم نامحدود با مختصات ممکن منفی */
   164→   164→interface BuildDraft {
   165→   165→  placed: { entry: CrosswordEntry; row: number; col: number; dir: CrosswordDir }[];
   166→   166→  minR: number;
   167→   167→  minC: number;
   168→   168→  maxR: number;
   169→   169→  maxC: number;
   170→   170→}
   171→   171→
   172→   172→/** یک تلاش ساخت: واژهٔ اول افقی در مبدأ، هر واژهٔ بعدی در بهترین تقاطع یا حذف */
   173→   173→function tryBuild(entries: CrosswordEntry[], maxSize: number): BuildDraft | null {
   174→   174→  if (entries.length === 0) return null;
   175→   175→  const cells = new Map<string, string>();
   176→   176→  const placed: BuildDraft["placed"] = [];
   177→   177→
   178→   178→  // واژهٔ اول افقی وسط بوم — بقیه در چهار جهت دورش رشد می‌کنند و برش آخر وسطش می‌گذارد
   179→   179→  const first = entries[0];
   180→   180→  for (let j = 0; j < first.word.length; j++) cells.set(cellKey(0, j), first.word[j]);
   181→   181→  placed.push({ entry: first, row: 0, col: 0, dir: "across" });
   182→   182→  let minR = 0;
   183→   183→  let maxR = 0;
   184→   184→  let minC = 0;
   185→   185→  let maxC = first.word.length - 1;
   186→   186→
   187→   187→  for (let i = 1; i < entries.length; i++) {
   188→   188→    const word = entries[i].word;
   189→   189→    let best: BuildDraft["placed"][number] | null = null;
   190→   190→    let bestScore = Infinity;
   191→   191→    for (const [key, ch] of cells) {
   192→   192→      if (!word.includes(ch)) continue;
   193→   193→      const [r, c] = parseKey(key);
   194→   194→      for (let j = 0; j < word.length; j++) {
   195→   195→        if (word[j] !== ch) continue;
   196→   196→        for (const dir of ["across", "down"] as const) {
   197→   197→          const r0 = dir === "across" ? r : r - j;
   198→   198→          const c0 = dir === "across" ? c - j : c;
   199→   199→          if (!canPlace(word, r0, c0, dir, cells)) continue;
   200→   200→          const nMinR = Math.min(minR, r0);
   201→   201→          const nMinC = Math.min(minC, c0);
   202→   202→          const nMaxR = Math.max(maxR, dir === "across" ? r0 : r0 + word.length - 1);
   203→   203→          const nMaxC = Math.max(maxC, dir === "across" ? c0 + word.length - 1 : c0);
   204→   204→          if (nMaxR - nMinR + 1 > maxSize || nMaxC - nMinC + 1 > maxSize) continue;
   205→   205→          // کمترین ابعاد افزوده + کمی تصادف برای تنوع چیدمان
   206→   206→          const added = nMaxR - minR + nMaxC - minC;
   207→   207→          const score = added * 100 + Math.random() * 7;
   208→   208→          if (score < bestScore) {
   209→   209→            bestScore = score;
   210→   210→            best = { entry: entries[i], row: r0, col: c0, dir };
   211→   211→          }
   212→   212→        }
   213→   213→      }
   214→   214→    }
   215→   215→    if (best) {
   216→   216→      const w = best.entry.word;
   217→   217→      const dr = best.dir === "down" ? 1 : 0;
   218→   218→      const dc = best.dir === "across" ? 1 : 0;
   219→   219→      for (let j = 0; j < w.length; j++) {
   220→   220→        cells.set(cellKey(best.row + dr * j, best.col + dc * j), w[j]);
   221→   221→      }
   222→   222→      minR = Math.min(minR, best.row);
   223→   223→      minC = Math.min(minC, best.col);
   224→   224→      maxR = Math.max(maxR, best.dir === "down" ? best.row + w.length - 1 : best.row);
   225→   225→      maxC = Math.max(maxC, best.dir === "across" ? best.col + w.length - 1 : best.col);
   226→   226→      placed.push(best);
   227→   227→    }
   228→   228→    // واژه‌ای که هیچ تقاطع سازگاری ندارد حذف می‌شود
   229→   229→  }
   230→   230→  return { placed, minR, minC, maxR, maxC };
   231→   231→}

   233→   233→/** برش به bounding box + شماره‌گذاری استاندارد سرنخ‌ها (پیمایش ردیفی) */
   234→   234→function finalize(build: BuildDraft, maxSize: number): CrosswordPuzzle | null {
   235→   235→  const { placed, minR, minC, maxR, maxC } = build;
   236→   236→  const rows = maxR - minR + 1;
   237→   237→  const cols = maxC - minC + 1;
   238→   238→  if (rows > maxSize || cols > maxSize) return null;
   239→   239→  const grid: CrosswordCell[][] = [];
   240→   240→  for (let r = 0; r < rows; r++) {
   241→   241→    const row: CrosswordCell[] = [];
   242→   242→    for (let c = 0; c < cols; c++) row.push({ letter: "", num: null, acrossId: null, downId: null });
   243→   243→    grid.push(row);
   244→   244→  }
   245→   245→  const placements: CrosswordPlacement[] = placed.map((p, id) => ({
   246→   246→    id,
   247→   247→    word: p.entry.word,
   248→   248→    meaning: p.entry.meaning,
   249→   249→    lesson: p.entry.lesson,
   250→   250→    lessonLabel: p.entry.lessonLabel,
   251→   251→    row: p.row - minR,
   252→   252→    col: p.col - minC,
   253→   253→    dir: p.dir,
   254→   254→    num: 0,
   255→   255→  }));
   256→   256→  for (const p of placements) {
   257→   257→    for (let j = 0; j < p.word.length; j++) {
   258→   258→      const [r, c] = parseKey(draftCellKey(p, j));
   259→   259→      const cell = grid[r][c];
   260→   260→      cell.letter = p.word[j];
   261→   261→      if (p.dir === "across") cell.acrossId = p.id;
   262→   262→      else cell.downId = p.id;
   263→   263→    }
   264→   264→  }
   265→   265→  const startNums = new Map<string, number>();
   266→   266→  const starts = [...new Set(placements.map((p) => cellKey(p.row, p.col)))];
   267→   267→  starts.sort((a, b) => {
   268→   268→    const [r1, c1] = parseKey(a);
   269→   269→    const [r2, c2] = parseKey(b);
   270→   270→    return r1 - r2 || c1 - c2;
   271→   271→  });
   272→   272→  starts.forEach((k, i) => startNums.set(k, i + 1));
   273→   273→  for (const p of placements) {
   274→   274→    p.num = startNums.get(cellKey(p.row, p.col)) ?? 0;
   275→   275→    grid[p.row][p.col].num = p.num;
   276→   276→  }
   277→   277→  return { rows, cols, grid, placements };
   278→   278→}
   279→   279→
   280→   280→/**
   281→   281→ * تولید جدول متقاطع از استخر واژه:
   282→   282→ * ۱) واژهٔ اول افقی وسط بوم ۲) هر واژهٔ بعدی در بهترین تقاطع (حرف مشترک + سازگاری
   283→   283→ * اطراف + کمترین ابعاد افزوده) ۳) واژه‌های جا نشده حذف ۴) اگر کمتر از minWords
   284→   284→ * واژه جا شد، با seed جدید تا maxTries بار تلاش و بهترین نتیجه برگردانده می‌شود.
   285→   285→ * جدول بزرگ‌تر از maxSize در هر بُعد رد می‌شود.
   286→   286→ */
   287→   287→export function generateCrossword(
   288→   288→  entries: CrosswordEntry[],
   289→   289→  opts: GenerateOptions = {},
   290→   290→): CrosswordPuzzle | null {
   291→   291→  const minWords = Math.max(opts.minWords ?? 5, 1);
   292→   292→  const maxTries = Math.max(opts.maxTries ?? 60, 1);
   293→   293→  const maxSize = Math.max(opts.maxSize ?? 13, 3);
   294→   294→  if (entries.length < minWords) return null;
   295→   295→
   296→   296→  let bestPuzzle: CrosswordPuzzle | null = null;
   297→   297→  for (let t = 0; t < maxTries; t++) {
   298→   298→    const order = shuffle(entries);
   299→   299→    const build = tryBuild(order, maxSize);
   300→   300→    if (!build || build.placed.length < minWords) continue;
   301→   301→    const puzzle = finalize(build, maxSize);
   302→   302→    if (!puzzle) continue;
   303→   303→    if (
   304→   304→      !bestPuzzle ||
   305→   305→      puzzle.placements.length > bestPuzzle.placements.length ||
   306→   306→      (puzzle.placements.length === bestPuzzle.placements.length &&
   307→   307→        puzzle.rows * puzzle.cols < bestPuzzle.rows * bestPuzzle.cols)
   308→   308→    ) {
   309→   309→      bestPuzzle = puzzle;
   310→   310→    }
   311→   311→    if (puzzle.placements.length === entries.length) break; // همه واژه‌ها جا گرفتند
   312→   312→  }
   313→   313→  return bestPuzzle;
   314→   314→}

   368→   368→  };
   369→   369→
   370→   370→  const { speed } = useVoiceSpeed();
   371→   371→  const secsRef = useRef(0);
   372→   372→  /** آینهٔ سنکرون letters — ضد keydown پشت‌سرهم (حروف پشت‌هم گم نشوند) */
   373→   373→  const lettersRef = useRef<Record<string, string>>({});
   374→   374→  const timersRef = useRef<number[]>([]);
   375→   375→
   376→   376→  /** setTimeout ثبت‌شده — در شروع دور و unmount پاک می‌شود */
   377→   377→  const later = (fn: () => void, ms: number) => {
   378→   378→    const id = window.setTimeout(fn, ms);
   379→   379→    timersRef.current.push(id);
   380→   380→  };
   381→   381→
   382→   382→  // رکورد شخصی این اسکوپ — اسنپ‌شات رشته‌ای (الگوی memory-game)
   383→   383→  const bestScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${wordCount}`;
   384→   384→  const bestRaw = useSyncExternalStore(
   385→   385→    subscribeGameData,
   386→   386→    () => {
   387→   387→      const b = loadBest<BestRecord>(activeBookId, GAME_KEY, bestScope);
   388→   388→      return b ? JSON.stringify(b) : null;
   389→   389→    },
   390→   390→    () => null,
   391→   391→  );
   392→   392→  const best = useMemo<BestRecord | null>(
   393→   393→    () => (bestRaw ? (JSON.parse(bestRaw) as BestRecord) : null),
   394→   394→    [bestRaw],
   395→   395→  );
   396→   396→
   397→   397→  const cellsOf = (p: CrosswordPlacement): string[] => {
   398→   398→    const out: string[] = [];
   399→   399→    for (let j = 0; j < p.word.length; j++) out.push(draftCellKey(p, j));
   400→   400→    return out;
   401→   401→  };
   402→   402→
   403→   403→  /** واژه‌هایی که با این حروف کامل و درست‌اند (از بین حل‌نشده‌ها) */
   404→   404→  const completionsIn = (lts: Record<string, string>, already: Set<number>): number[] => {
   405→   405→    if (!puzzle) return [];
   406→   406→    const out: number[] = [];
   407→   407→    for (const p of puzzle.placements) {
   408→   408→      if (already.has(p.id)) continue;
   409→   409→      let ok = true;
   410→   410→      for (let j = 0; j < p.word.length; j++) {
   411→   411→        if ((lts[draftCellKey(p, j)] ?? "") !== p.word[j]) {
   412→   412→          ok = false;
   413→   413→          break;
   414→   414→        }
   415→   415→      }
   416→   416→      if (ok) out.push(p.id);
   417→   417→    }
   418→   418→    return out;
   419→   419→  };
   420→   420→
   421→   421→  const finish = (finalSecs: number, finalHints: number) => {
   422→   422→    const placed = puzzle?.placements.length ?? 0;
   423→   423→    touchStreak();
   424→   424→    recordStudySession();
   425→   425→    bumpPlays(activeBookId, GAME_KEY);
   426→   426→    saveAttempt(activeBookId, GAME_KEY, {
   427→   427→      lesson: lesson === ALL_LESSONS ? 0 : lesson,
   428→   428→      total: placed,
   429→   429→      correct: placed,
   430→   430→      duration: finalSecs,
   431→   431→      detail: { hints: finalHints },
   432→   432→    });
   433→   433→    setSecs(finalSecs);
   434→   434→    if (finalHints === 0) {
   435→   435→      const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${wordCount}`;
   436→   436→      const prev = loadBest<BestRecord>(activeBookId, GAME_KEY, scope);
   437→   437→      if (!prev || finalSecs < prev.time) {
   438→   438→        saveBest(activeBookId, GAME_KEY, scope, { time: finalSecs });
   439→   439→        setIsRecord(true);
   440→   440→        toast.success(`رکورد جدید! جدول در ${fmtTime(finalSecs)} حل شد 🏆`);
   441→   441→      } else {
   442→   442→        toast.success(`جدول کامل شد — ${fmtTime(finalSecs)} 🎉`);
   443→   443→      }
   444→   444→    } else {
   445→   445→      toast.success(`جدول کامل شد — ${fmtTime(finalSecs)} با ${faNum(finalHints)} راهنما 🎉`);
   446→   446→    }
   447→   447→    setPhase("done");
   448→   448→  };
   449→   449→
   450→   450→  /** ثبت واژه‌های تازه‌حل‌شده: قفل سلول‌ها + pop + تلفظ + پایان دور */
   451→   451→  const applySolved = (ids: number[], hCount: number) => {
   452→   452→    if (!puzzle) return;
   453→   453→    const cells: string[] = [];
   454→   454→    ids.forEach((id) => cells.push(...cellsOf(puzzle.placements[id])));
   455→   455→    setSolved((prev) => {
   456→   456→      const next = new Set(prev);
   457→   457→      ids.forEach((id) => next.add(id));
   458→   458→      return next;
   459→   459→    });
   460→   460→    setLocked((prev) => {
   461→   461→      const next = new Set(prev);
   462→   462→      cells.forEach((k) => next.add(k));
   463→   463→      return next;
   464→   464→    });
   465→   465→    setPopKeys(cells);
   466→   466→    later(() => setPopKeys([]), POP_MS);
   467→   467→    speak(puzzle.placements[ids[0]].word, { speed });
   468→   468→    if (solved.size + ids.length >= puzzle.placements.length) {
   469→   469→      later(() => finish(secsRef.current, hCount), FINISH_MS);
   470→   470→    }
   471→   471→  };
   472→   472→
   473→   473→  const start = () => {
   474→   474→    const pool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   475→   475→    // گارد واژهٔ جدول‌پذیر: فقط حروف انگلیسی، طول ۳ تا ۹
   476→   476→    const valid = pool.filter((w) => /^[A-Za-z]{3,9}$/.test(w.word));
   477→   477→    const normalized = valid.map((w) => ({ ...w, word: w.word.toUpperCase() }));
   478→   478→    const chosen = pickUnique(normalized, wordCount, (w) => w.word);
   479→   479→    if (chosen.length < wordCount) {
   480→   480→      toast.error(
   481→   481→        `واژهٔ انگلیسی ۳ تا ۹ حرفیِ کافی پیدا نشد (${faNum(chosen.length)} از ${faNum(wordCount)}) — درس یا تعداد دیگری را امتحان کن`,
   482→   482→      );
   483→   483→      return;
   484→   484→    }
   485→   485→    const entries: CrosswordEntry[] = chosen.map((w) => ({
   486→   486→      word: w.word,
   487→   487→      meaning: w.meaning,
   488→   488→      lesson: w.lesson,
   489→   489→      lessonLabel: w.lessonLabel,
   490→   490→    }));
   491→   491→    const built = generateCrossword(entries, { minWords: MIN_PLACED, maxTries: 60, maxSize: 13 });
   492→   492→    if (!built || built.placements.length < MIN_PLACED) {
   493→   493→      toast.error("ساخت جدول متقاطع با این واژه‌ها ممکن نشد — دوباره شروع کن یا درس دیگری را انتخاب کن");
   494→   494→      return;
   495→   495→    }
   496→   496→    setPuzzle(built);
   497→   497→    lettersRef.current = {};
   498→   498→    setLetters({});
   499→   499→    setLocked(new Set());
   500→   500→    setPopKeys([]);
   501→   501→    setShakeKeys([]);
   502→   502→    setSolved(new Set());
   503→   503→    setHints(0);
   504→   504→    setSecs(0);
   505→   505→    secsRef.current = 0;
   506→   506→    setIsRecord(false);
   507→   507→    setDir("across");
   508→   508→    const startP = built.placements.find((p) => p.dir === "across") ?? built.placements[0];
   509→   509→    setActive({ r: startP.row, c: startP.col });
   510→   510→    setPhase("playing");
   511→   511→  };
   512→   512→
   513→   513→  /** تایپ یک حرف در سلول فعال + پیشروی + تشخیص خودکار واژهٔ کامل درست */
   514→   514→  const typeLetter = (ch: string) => {
   515→   515→    if (!puzzle || !active || !activeWord) return;
   516→   516→    const up = ch.toUpperCase();
   517→   517→    if (!/^[A-Z]$/.test(up)) return;
   518→   518→    const key = cellKey(active.r, active.c);
   519→   519→    if (locked.has(key)) return;
   520→   520→    const next = { ...lettersRef.current, [key]: up };
   521→   521→    lettersRef.current = next;
   522→   522→    setLetters(next);
   523→   523→
   524→   524→    const completed = completionsIn(next, solved);
   525→   525→    // قفل‌های پس از این تایپ — سلول‌های واژه‌های تازه‌حل‌شده هم رد می‌شوند
   526→   526→    const lockPlus = new Set(locked);
   527→   527→    completed.forEach((id) => cellsOf(puzzle.placements[id]).forEach((k) => lockPlus.add(k)));
   528→   528→
   529→   529→    // پیشروی به بعدیِ آزاد در همان واژه
   530→   530→    const cells = cellsOf(activeWord);
   531→   531→    const idx = cells.indexOf(key);
   532→   532→    let target: string | null = null;
   533→   533→    for (let j = idx + 1; j < cells.length; j++) {
   534→   534→      if (!lockPlus.has(cells[j])) {
   535→   535→        target = cells[j];
   536→   536→        break;
   537→   537→      }
   538→   538→    }
   539→   539→    let jumpDir: CrosswordDir | null = null;
   540→   540→    if (completed.length > 0) {
   541→   541→      applySolved(completed, hints);
   542→   542→      // اگر مقصد داخل واژهٔ تازه‌حل‌شده قفل شد → اولین سلول آزادِ واژهٔ حل‌نشدهٔ بعدی
   543→   543→      if (!target || lockPlus.has(target)) {
   544→   544→        const unsolved = puzzle.placements.find(
   545→   545→          (p) => !solved.has(p.id) && !completed.includes(p.id),
   546→   546→        );
   547→   547→        const free = unsolved ? cellsOf(unsolved).find((k) => !lockPlus.has(k)) : undefined;
   548→   548→        if (free && unsolved) {
   549→   549→          target = free;
   550→   550→          jumpDir = unsolved.dir;
   551→   551→        }
   552→   552→      }
   553→   553→    }
   554→   554→    if (!target) return;
   555→   555→    const [tr, tc] = parseKey(target);
   556→   556→    if (jumpDir) {
   557→   557→      setDir(jumpDir);
   558→   558→    } else {
   559→   559→      const tcell = puzzle.grid[tr][tc];
   560→   560→      const okDir = dir === "across" ? tcell.acrossId !== null : tcell.downId !== null;
   561→   561→      if (!okDir) setDir(dir === "across" ? "down" : "across");
   562→   562→    }
   563→   563→    setActive({ r: tr, c: tc });
   564→   564→  };
   565→   565→
   566→   566→  /** Backspace: پاک‌کردن سلول فعال، و اگر خالی بود پرش به سلول آزاد قبلی و پاک‌کردنش */
   567→   567→  const backspace = () => {
   568→   568→    if (!puzzle || !active || !activeWord) return;
   569→   569→    const key = cellKey(active.r, active.c);
   570→   570→    if (locked.has(key)) return;
   571→   571→    if (lettersRef.current[key]) {
   572→   572→      const next = { ...lettersRef.current };
   573→   573→      delete next[key];
   574→   574→      lettersRef.current = next;
   575→   575→      setLetters(next);
   576→   576→      return;
   577→   577→    }
   578→   578→    const cells = cellsOf(activeWord);
   579→   579→    const idx = cells.indexOf(key);
   580→   580→    for (let j = idx - 1; j >= 0; j--) {
   581→   581→      if (!locked.has(cells[j])) {
   582→   582→        const pk = cells[j];
   583→   583→        const next = { ...lettersRef.current };
   584→   584→        delete next[pk];
   585→   585→        lettersRef.current = next;
   586→   586→        setLetters(next);
   587→   587→        const [pr, pc] = parseKey(pk);
   588→   588→        setActive({ r: pr, c: pc });
   589→   589→        return;
   590→   590→      }
   591→   591→    }
   592→   592→  };
   593→   593→
   594→   594→  /** راهنما: حرف سلول فعال (یا نزدیک‌ترین سلول نیازمند) آشکار می‌شود +۱ شمارش */
   595→   595→  const hint = () => {
   596→   596→    if (!puzzle) return;
   597→   597→    const isFree = (k: string) => !locked.has(k);
   598→   598→    const needsLetter = (k: string) => {
   599→   599→      const [r, c] = parseKey(k);
   600→   600→      return isFree(k) && lettersRef.current[k] !== puzzle.grid[r][c].letter;
   601→   601→    };
   602→   602→    let target: { key: string; dir: CrosswordDir } | null = null;
   603→   603→    if (active && activeWord) {
   604→   604→      const cur = cellKey(active.r, active.c);
   605→   605→      if (needsLetter(cur)) target = { key: cur, dir: activeWord.dir };
   606→   606→      else {
   607→   607→        const free = cellsOf(activeWord).find(needsLetter) ?? cellsOf(activeWord).find(isFree);
   608→   608→        if (free) target = { key: free, dir: activeWord.dir };
   609→   609→      }
   610→   610→    }
   611→   611→    if (!target) {
   612→   612→      const unsolved = puzzle.placements.find((p) => !solved.has(p.id));
   613→   613→      const free = unsolved ? cellsOf(unsolved).find(isFree) : undefined;
   614→   614→      if (free && unsolved) target = { key: free, dir: unsolved.dir };
   615→   615→    }
   616→   616→    if (!target) return;
   617→   617→    const [tr, tc] = parseKey(target.key);
   618→   618→    const next = { ...lettersRef.current, [target.key]: puzzle.grid[tr][tc].letter };
   619→   619→    lettersRef.current = next;
   620→   620→    setLetters(next);
   621→   621→    setHints((h) => h + 1);
   622→   622→    setDir(target.dir);
   623→   623→    setActive({ r: tr, c: tc });
   624→   624→    const completed = completionsIn(next, solved);
   625→   625→    if (completed.length > 0) applySolved(completed, hints + 1);
   626→   626→  };
   627→   627→
   628→   628→  /** بررسی: حروف درست قفل، غلط‌ها لرزش و خالی */
   629→   629→  const check = () => {
   630→   630→    if (!puzzle) return;
   631→   631→    const wrong: string[] = [];
   632→   632→    const right: string[] = [];
   633→   633→    for (const [key, ch] of Object.entries(lettersRef.current)) {
   634→   634→      if (locked.has(key)) continue;
   635→   635→      const [r, c] = parseKey(key);
   636→   636→      const cell = puzzle.grid[r]?.[c];
   637→   637→      if (cell && cell.letter === ch) right.push(key);
   638→   638→      else wrong.push(key);
   639→   639→    }
   640→   640→    if (right.length > 0) {
   641→   641→      setLocked((prev) => {
   642→   642→        const next = new Set(prev);
   643→   643→        right.forEach((k) => next.add(k));
   644→   644→        return next;
   645→   645→      });
   646→   646→    }
   647→   647→    if (wrong.length > 0) {
   648→   648→      setShakeKeys(wrong);
   649→   649→      later(() => {
   650→   650→        setShakeKeys([]);
   651→   651→        const next = { ...lettersRef.current };
   652→   652→        wrong.forEach((k) => delete next[k]);
   653→   653→        lettersRef.current = next;
   654→   654→        setLetters(next);
   655→   655→      }, SHAKE_MS);
   656→   656→      toast.error(`${faNum(wrong.length)} حرف غلط پاک شد${right.length > 0 ? " — درست‌ها قفل شدند" : ""}`);
   657→   657→    } else if (right.length > 0) {
   658→   658→      toast.success(`${faNum(right.length)} حرف درست قفل شد`);
   659→   659→    } else {
   660→   660→      toast("چیزی برای بررسی وارد نشده");
   661→   661→    }
   662→   662→  };
   663→   663→
   664→   664→  /** tap سلول: انتخاب؛ دوبار tap روی همان سلول = تغییر جهت */
   665→   665→  const onCellTap = (r: number, c: number) => {
   666→   666→    if (!puzzle) return;
   667→   667→    const cell = puzzle.grid[r][c];
   668→   668→    if (cell.acrossId === null && cell.downId === null) return;
   669→   669→    if (active && active.r === r && active.c === c) {
   670→   670→      if (cell.acrossId !== null && cell.downId !== null) {
   671→   671→        setDir((d) => (d === "across" ? "down" : "across"));
   672→   672→      }
   673→   673→      return;
   674→   674→    }
   675→   675→    setActive({ r, c });
   676→   676→    const curOk = dir === "across" ? cell.acrossId !== null : cell.downId !== null;
   677→   677→    if (!curOk) setDir(dir === "across" ? "down" : "across");
   678→   678→  };
   679→   679→
   680→   680→  /** tap سرنخ: فعال‌سازی و هایلایت مسیر واژه */
   681→   681→  const focusWord = (p: CrosswordPlacement) => {
   682→   682→    const cells = cellsOf(p);
   683→   683→    const target = cells.find((k) => !locked.has(k)) ?? cells[0];
   684→   684→    const [r, c] = parseKey(target);
   685→   685→    setActive({ r, c });
   686→   686→    setDir(p.dir);
   687→   687→  };
   688→   688→
   689→   689→  // واژهٔ فعال بر اساس سلول و جهت (اگر جهت در این سلول معتبر نباشد، جهت دیگر)
   690→   690→  // بدون useMemo دستی — React Compiler خودش memo می‌کند
   691→   691→  const activeWord: CrosswordPlacement | null = (() => {
   692→   692→    if (!puzzle || !active) return null;
   693→   693→    const cell = puzzle.grid[active.r]?.[active.c];
   694→   694→    if (!cell) return null;
   695→   695→    const primary = dir === "across" ? cell.acrossId : cell.downId;
   696→   696→    if (primary !== null) return puzzle.placements[primary];
   697→   697→    const other = dir === "across" ? cell.downId : cell.acrossId;
   698→   698→    return other !== null ? puzzle.placements[other] : null;
   699→   699→  })();
   700→   700→
   701→   701→  // تایمر — بدون setState مستقیم در effect (react-hooks v6)؛ secsRef برای زمان دقیق پایان
   702→   702→  useEffect(() => {
   703→   703→    if (phase !== "playing") return;
   704→   704→    const t = setInterval(() => {
   705→   705→      secsRef.current += 1;
   706→   706→      setSecs((s) => s + 1);
   707→   707→    }, 1000);
   708→   708→    return () => clearInterval(t);
   709→   709→  }, [phase]);
   710→   710→
   711→   711→  // cleanup کامل unmount: صدا + تایمرهای معلق
   712→   712→  useEffect(() => {
   713→   713→    return () => {
   714→   714→      stopSpeaking();
   715→   715→      timersRef.current.forEach((id) => window.clearTimeout(id));
   716→   716→    };
   717→   717→  }, []);
   718→   718→
   719→   719→  // کیبورد فیزیکی — الگوی latest-ref: شنونده فقط با [phase] ثبت می‌شود و بدنهٔ آن
   720→   720→  // همیشه آخرین نسخهٔ توابع را صدا می‌زند (صفر setState در بدنهٔ effect)
   721→   721→  const keyRef = useRef<(e: KeyboardEvent) => void>(() => {});
   722→   722→  useEffect(() => {
   723→   723→    keyRef.current = (e: KeyboardEvent) => {
   724→   724→      if (exitAskRef.current) return; // نگهبان خروج — تایپ حروف حین دیالوگ تأیید خروج
   725→   725→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   726→   726→      if (e.key === "Backspace") {
   727→   727→        e.preventDefault();
   728→   728→        backspace();
   729→   729→        return;
   730→   730→      }
   731→   731→      if (/^[a-zA-Z]$/.test(e.key)) typeLetter(e.key);
   732→   732→    };
   733→   733→  });
   734→   734→  useEffect(() => {
   735→   735→    if (phase !== "playing") return;
   736→   736→    const onKey = (e: KeyboardEvent) => keyRef.current(e);
   737→   737→    window.addEventListener("keydown", onKey);
   738→   738→    return () => window.removeEventListener("keydown", onKey);
   739→   739→  }, [phase]);
   740→   740→
   741→   741→  // ── Setup ──
   742→   742→  if (phase === "setup") {
   743→   743→    return (
   744→   744→      <Card className="mx-auto max-w-xl rise-in">
   745→   745→        <CardContent className="space-y-6 p-6">
   746→   746→          <div className="flex items-center gap-3">
   747→   747→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-teal-500/15 text-teal-600 dark:text-teal-400">
   748→   748→              <Puzzle size={22} />
   749→   749→            </span>
   750→   750→            <div>
   751→   751→              <h2 className="text-lg font-black">جدول کلمات متقاطع</h2>
