   116→   116→function Confetti() {
   117→   117→  const pieces = useMemo(
   118→   118→    () =>
   119→   119→      Array.from({ length: 32 }, (_, i) => ({
   120→   120→        id: i,
   121→   121→        left: Math.random() * 100,

   123→   123→        duration: 2.4 + Math.random() * 1.6,

   125→   125→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   126→   126→        drift: Math.round((Math.random() - 0.5) * 120),
   127→   127→        round: Math.random() > 0.5,
   128→   128→      })),
   129→   129→    [],
   130→   130→  );
   131→   131→  return (
   132→   132→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   133→   133→      {pieces.map((p) => (
   134→   134→        <span
   135→   135→          key={p.id}
   136→   136→          className="confetti-piece absolute top-[-14px] block"
   137→   137→          style={{
   138→   138→            left: `${p.left}%`,
   139→   139→            width: p.size,
   140→   140→            height: p.size * (p.round ? 1 : 0.45),
   141→   141→            background: p.color,
   142→   142→            borderRadius: p.round ? "50%" : 2,
   143→   143→            animationDelay: `${p.delay}s`,
   144→   144→            animationDuration: `${p.duration}s`,
   145→   145→            "--confetti-drift": `${p.drift}px`,
   146→   146→          } as React.CSSProperties}
   147→   147→        />
   148→   148→      ))}
   149→   149→    </div>
   150→   150→  );
   151→   151→}
