    69→/** کارت صف مرور — واژه یا فعل بی‌قاعده */
    70→interface QueueCard {
    71→  /** کلید SRS: متن واژه یا "v:base" */
    72→  key: string;
    73→  kind: "word" | "verb";
    74→  /** روی کارت — واژه یا شکل سادهٔ فعل */

    76→  /** پشت کارت — معنی فارسی */

    81→  past?: string;
    82→  participle?: string;
    83→  /** کارت نو (هرگز پاسخ داده نشده)؟ — برای شمارندهٔ روزانه */
    84→  isNewCard: boolean;
    85→  /** ترتیب کارت‌های سررسید در صف */

    87→  /** تعداد فراموشی — برای صف دردسرسازها */
    88→  lapses: number;

    91→function shuffle<T>(arr: T[]): T[] {
    92→  const a = [...arr];
    93→  for (let i = a.length - 1; i > 0; i--) {
    94→    const j = Math.floor(Math.random() * (i + 1));
    95→    [a[i], a[j]] = [a[j], a[i]];
    96→  }
    97→  return a;
    98→}

   100→/**
   101→ * صف استاندارد مرور: کارت‌های سررسید (صعودی dueAt) + کارت‌های نو تا سقف
   102→ * باقی‌ماندهٔ روز (به‌ترتیب درس، شافل درون‌درس؛ فعل‌ها در پایان) — سقف کل ۶۰.
   103→ */
   104→function buildQueue(
   105→  srs: Record<string, SrsItem>,
   106→  words: VocabEntry[],
   107→  verbs: IrregularVerb[],
   108→  now: number,
   109→  newLimit: number,
   110→  newDone: number,
   111→): QueueCard[] {
   112→  const due: QueueCard[] = [];
   113→  const newByLesson = new Map<number, QueueCard[]>();
   114→  const newVerbs: QueueCard[] = [];
   115→  let totalNew = 0;
   116→
   117→  for (const w of words) {
   118→    const st = srs[w.word];
   119→    if (!st || st.reps === 0) {
   120→      totalNew++;
   121→      const card: QueueCard = {
   122→        key: w.word,
   123→        kind: "word",
   124→        front: w.word,
   125→        meaning: w.meaning,
   126→        lessonLabel: w.lessonLabel,
   127→        isNewCard: true,
   128→        dueAt: Number.MAX_SAFE_INTEGER,
   129→        lapses: 0,
   130→      };
   131→      const list = newByLesson.get(w.lesson);
   132→      if (list) list.push(card);
   133→      else newByLesson.set(w.lesson, [card]);
   134→    } else if (st.dueAt <= now) {
   135→      due.push({
   136→        key: w.word,
   137→        kind: "word",
   138→        front: w.word,
   139→        meaning: w.meaning,
   140→        lessonLabel: w.lessonLabel,
   141→        isNewCard: false,
   142→        dueAt: st.dueAt,
   143→        lapses: st.lapses,
   144→      });
   145→    }
   146→  }
   147→
   148→  for (const v of verbs) {
   149→    const key = `v:${v.base}`;
   150→    const st = srs[key];
   151→    if (!st || st.reps === 0) {
   152→      totalNew++;
   153→      newVerbs.push({
   154→        key,
   155→        kind: "verb",
   156→        front: v.base,
   157→        meaning: v.meaning,
   158→        lessonLabel: "فعل",
   159→        past: v.past,
   160→        participle: v.participle,
   161→        isNewCard: true,
   162→        dueAt: Number.MAX_SAFE_INTEGER,
   163→        lapses: 0,
   164→      });
   165→    } else if (st.dueAt <= now) {
   166→      due.push({
   167→        key,
   168→        kind: "verb",
   169→        front: v.base,
   170→        meaning: v.meaning,
   171→        lessonLabel: "فعل",
   172→        past: v.past,
   173→        participle: v.participle,
   174→        isNewCard: false,
   175→        dueAt: st.dueAt,
   176→        lapses: st.lapses,
   177→      });
   178→    }
   179→  }
   180→
   181→  due.sort((a, b) => a.dueAt - b.dueAt);
   182→
   183→  // کارت‌های نو: ترتیب درس‌ها + شافل درون هر درس؛ فعل‌های نو در پایان
   184→  const newCards: QueueCard[] = [];
   185→  for (const lesson of [...newByLesson.keys()].sort((a, b) => a - b)) {
   186→    newCards.push(...shuffle(newByLesson.get(lesson)!));
   187→  }
   188→  newCards.push(...shuffle(newVerbs));
   189→
   190→  const remaining = Math.max(0, Math.min(newLimit - newDone, totalNew));
   191→  return [...due, ...newCards.slice(0, remaining)].slice(0, QUEUE_CAP);
   192→}

   194→/** فهرست کارت‌های دردسرساز — پرتکرارترین فراموشی‌ها اول */
   195→function collectLeeches(srs: Record<string, SrsItem>, words: VocabEntry[], verbs: IrregularVerb[]): QueueCard[] {
   196→  const out: QueueCard[] = [];
   197→  for (const w of words) {
   198→    const st = srs[w.word];
   199→    if (isLeech(st)) {
   200→      out.push({
   201→        key: w.word,
   202→        kind: "word",
   203→        front: w.word,
   204→        meaning: w.meaning,
   205→        lessonLabel: w.lessonLabel,
   206→        isNewCard: false,
   207→        dueAt: st.dueAt,
   208→        lapses: st.lapses,
   209→      });
   210→    }
   211→  }
   212→  for (const v of verbs) {
   213→    const key = `v:${v.base}`;
   214→    const st = srs[key];
   215→    if (isLeech(st)) {
   216→      out.push({
   217→        key,
   218→        kind: "verb",
   219→        front: v.base,
   220→        meaning: v.meaning,
   221→        lessonLabel: "فعل",
   222→        past: v.past,
   223→        participle: v.participle,
   224→        isNewCard: false,
   225→        dueAt: st.dueAt,
   226→        lapses: st.lapses,
   227→      });
   228→    }
   229→  }
   230→  return out.sort((a, b) => b.lapses - a.lapses);
   231→}

   233→/** شمارش کارت‌های نو (واژه + فعل) — هرگز وارد چرخه نشده‌اند */
   234→function countNew(srs: Record<string, SrsItem>, words: VocabEntry[], verbs: IrregularVerb[]): number {
   235→  let n = 0;
   236→  for (const w of words) {
   237→    const st = srs[w.word];
   238→    if (!st || st.reps === 0) n++;
   239→  }
   240→  for (const v of verbs) {
   241→    const st = srs[`v:${v.base}`];
   242→    if (!st || st.reps === 0) n++;
   243→  }
   244→  return n;
   245→}

   247→/** پیش‌بینی بار مرور ۷ روز آینده از کارت‌های در چرخه (سرریز گذشته → ستون «امروز») */
   248→function buildForecast(srs: Record<string, SrsItem>, now: number): { label: string; count: number }[] {
   249→  const start = new Date(now);
   250→  start.setHours(0, 0, 0, 0);
   251→  const day0 = start.getTime();
   252→  const fmt = new Intl.DateTimeFormat("fa-IR", { weekday: "short" });
   253→  const out: { label: string; count: number }[] = [];
   254→  for (let i = 0; i < 7; i++) {
   255→    const from = day0 + i * DAY_MS;
   256→    const to = from + DAY_MS;
   257→    let count = 0;
   258→    for (const st of Object.values(srs)) {
   259→      if (st.reps > 0 && (i === 0 ? st.dueAt < to : st.dueAt >= from && st.dueAt < to)) count++;
   260→    }
   261→    out.push({ label: i === 0 ? "امروز" : i === 1 ? "فردا" : fmt.format(new Date(from)), count });
   262→  }
   263→  return out;
   264→}

   266→/**
   267→ * نمای مرور هوشمند — onNavigate اختیاری است (بی‌استفاده هم امن).
   268→ */
   269→export function ReviewView({ onNavigate }: { onNavigate?: (v: string) => void }) {
   270→  const data = useBookData();
   271→  const srs = useSRSData();
   272→
   273→  const [phase, setPhase] = useState<Phase>("setup");
   274→  const [queue, setQueue] = useState<QueueCard[]>(EMPTY_QUEUE);
   275→  const [index, setIndex] = useState(0);
   276→  const [flipped, setFlipped] = useState(false);
   277→  const [againCount, setAgainCount] = useState(0);
   278→  const [hardCount, setHardCount] = useState(0);
   279→  const [goodCount, setGoodCount] = useState(0);
   280→
   281→  // سقف کارت نو و شمارندهٔ روزانه فقط یک‌بار در ابتدا خوانده می‌شوند و
   282→  // در event handlerها (شروع جلسه / بازگشت به setup) تازه‌سازی می‌شوند.
   283→  const [newLimit, setNewLimitUi] = useState(() => getDailyNewLimit());
   284→  const [newDone, setNewDone] = useState(() => getTodayReview().newDone);
   285→
   286→  const startedAtRef = useRef(0);
   287→  const againRef = useRef(0); // فاز ۶ — شمارش پاسخ‌های «باز هم» برای XP مرور
   288→
   289→  const words = data.vocab.all;
   290→  const verbs = data.verbs;
   291→  const now = Date.now();
   292→  const current = queue[index];
   293→
   294→  // ── KPIها — توابع خالص روی استور ری‌اکتیو (محاسبه در رندر) ──
   295→  let dueCount = 0;
   296→  let inCycle = 0;
   297→  for (const st of Object.values(srs)) {
   298→    if (st.reps > 0) {
   299→      inCycle++;
   300→      if (st.dueAt <= now) dueCount++;
   301→    }
   302→  }
   303→  const totalNew = countNew(srs, words, verbs);
   304→  const newRemaining = Math.max(0, Math.min(newLimit - newDone, totalNew));
   305→  const leeches = collectLeeches(srs, words, verbs);
   306→  const forecast = buildForecast(srs, now);
   307→  const forecastMax = Math.max(1, ...forecast.map((f) => f.count));
   308→
   309→  const preview = phase === "setup" ? buildQueue(srs, words, verbs, now, newLimit, newDone) : EMPTY_QUEUE;
   310→  const previewFor =
   311→    phase === "playing" && current ? previewIntervals(srs[current.key], now) : ([0, 0, 0] as [number, number, number]);
   312→
   313→  // جمع‌بندی فاز done — درصد موفقیت (بدون «نمی‌دونم») و مدت جلسه از ref زمان شروع
   314→  const successPct = Math.round(((goodCount + hardCount) / Math.max(queue.length, 1)) * 100);
   315→  const elapsedMs = Math.max(0, Date.now() - startedAtRef.current);
   316→  const elapsedMin = Math.floor(elapsedMs / 60_000);
   317→  const elapsedSec = Math.floor((elapsedMs % 60_000) / 1000);
   318→  const duration = elapsedMin > 0 ? `${fa(elapsedMin)} دقیقه و ${fa(elapsedSec)} ثانیه` : `${fa(elapsedSec)} ثانیه`;
   319→
   320→  // ── event handlerها — صف و شمارنده‌ها اینجا ساخته می‌شوند ──
   321→
   322→  function beginSession(cards: QueueCard[]) {
   323→    if (cards.length === 0) return;
   324→    startedAtRef.current = Date.now();
   325→    setQueue(cards);
   326→    setIndex(0);
   327→    setFlipped(false);
   328→    setAgainCount(0);
   329→    againRef.current = 0;
   330→    setHardCount(0);
   331→    setGoodCount(0);
   332→    setPhase("playing");
   333→  }
   334→
   335→  function startLeechSession() {
   336→    const q = collectLeeches(srs, words, verbs).slice(0, QUEUE_CAP);
   337→    if (q.length === 0) {
   338→      toast.error("الان کارت دردسرسازی برای تمرین ویژه نیست.");
   339→      return;
   340→    }
   341→    beginSession(q);
   342→  }
   343→
   344→  function changeLimit(n: number) {
   345→    setDailyNewLimit(n);
   346→    setNewLimitUi(n);
   347→    toast.info(`سقف کارت نو روزانه روی ${fa(n)} کارت تنظیم شد.`);
   348→  }
   349→
   350→  function backToSetup() {
   351→    // تازه‌سازی شمارنده‌ها پس از جلسه — صف در رندر setup بازمحاسبه می‌شود
   352→    setNewDone(getTodayReview().newDone);
   353→    setNewLimitUi(getDailyNewLimit());
   354→    setPhase("setup");
   355→  }
   356→
   357→  const answer = useCallback(
   358→    (grade: SrsGrade) => {
   359→      const card = queue[index];
   360→      if (!card) return;
   361→      gradeSrs(card.key, grade);
   362→      bumpTodayReview(card.isNewCard ? 1 : 0);
   363→      if (grade === 0) {
   364→        setAgainCount((c) => c + 1);
   365→        againRef.current += 1;
   366→      }
   367→      else if (grade === 1) setHardCount((c) => c + 1);
   368→      else setGoodCount((c) => c + 1);
   369→      if (index + 1 >= queue.length) {
   370→        awardReviewActivity(queue.length, queue.length - againRef.current); // فاز ۶ — XP مرور + زنجیره + ماموریت
   371→        setPhase("done");
   372→      } else {
   373→        setIndex((i) => i + 1);
   374→        setFlipped(false);
   375→      }
   376→    },
   377→    [queue, index],
   378→  );
   379→
   380→  // میان‌برهای کیبورد — الگوی flashcard-view (setState فقط در callback رویداد)
   381→  useEffect(() => {
   382→    if (phase !== "playing") return;
   383→    function onKey(e: KeyboardEvent) {
   384→      if (e.code === "Space" || e.key === "Enter") {
   385→        e.preventDefault();
   386→        setFlipped((f) => !f);
   387→      } else if (!flipped) {
   388→        return; // درجه‌ها فقط پس از دیدن پاسخ
   389→      } else if (e.key === "1" || e.key === "۱") {
   390→        answer(0);
   391→      } else if (e.key === "2" || e.key === "۲") {
   392→        answer(1);
   393→      } else if (e.key === "3" || e.key === "۳") {
   394→        answer(2);
   395→      }
   396→    }
   397→    window.addEventListener("keydown", onKey);
   398→    return () => window.removeEventListener("keydown", onKey);
   399→  }, [phase, flipped, answer]);
   400→
   401→  const announcement =
   402→    phase === "setup"
   403→      ? "نمای مرور هوشمند امروز آماده است"
   404→      : phase === "playing"
   405→        ? `کارت ${fa(index + 1)} از ${fa(queue.length)}`
   406→        : "جلسهٔ مرور به پایان رسید";
   407→
   408→  return (
   409→    <div className="space-y-4" role="region" aria-label="مرور هوشمند امروز">
   410→      <div className="sr-only" role="status" aria-live="polite">
   411→        {announcement}
   412→      </div>
   413→
   414→      {/* ── فاز setup ── */}
   415→      {phase === "setup" && (
   416→        <>
   417→          <Card className="mx-auto max-w-2xl rise-in">
   418→            <CardContent className="space-y-6 p-6">
   419→              {/* سربرگ */}
   420→              <div className="flex items-center gap-3">
   421→                <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-primary/10 text-primary">
   422→                  <Brain size={24} />
   423→                </span>
   424→                <div>
   425→                  <h1 className="text-lg font-black">مرور هوشمند امروز</h1>
   426→                  <p className="text-xs leading-5 text-muted-foreground">
   427→                    الگوریتم SM-2 — هر کارت دقیقاً لحظه‌ای تکرار می‌شود که نزدیک فراموشی‌ات باشی.
   428→                  </p>
   429→                </div>
   430→              </div>
   431→
   432→              {/* KPIها */}
   433→              <div className="grid grid-cols-2 gap-2 sm:grid-cols-4" role="list" aria-label="آمار مرور امروز">
   434→                <div role="listitem" className="rounded-2xl border border-amber-500/25 bg-amber-500/8 p-3 text-center">
   435→                  <p className="text-2xl font-black text-amber-600 dark:text-amber-400">{fa(dueCount)}</p>
   436→                  <p className="mt-0.5 text-[11px] text-muted-foreground">سررسید امروز</p>
   437→                </div>
   438→                <div role="listitem" className="rounded-2xl border border-emerald-500/25 bg-emerald-500/8 p-3 text-center">
   439→                  <p className="text-2xl font-black text-emerald-600 dark:text-emerald-400">{fa(newRemaining)}</p>
   440→                  <p className="mt-0.5 text-[11px] text-muted-foreground">نو باقی‌مانده امروز</p>
   441→                </div>
   442→                <div role="listitem" className="rounded-2xl border bg-muted/40 p-3 text-center">
   443→                  <p className="text-2xl font-black text-amber-600 dark:text-amber-400">{fa(leeches.length)}</p>

   448→                  <p className="mt-0.5 text-[11px] text-muted-foreground">کارت در چرخه</p>
   449→                </div>

   451→
   452→              {/* سقف کارت نو روزانه */}
   453→              <div>
   454→                <p className="mb-2 text-[13px] font-bold">سقف کارت نو روزانه</p>
   455→                <div className="flex flex-wrap gap-2" role="group" aria-label="سقف کارت نو روزانه">
   456→                  {NEW_LIMIT_OPTIONS.map((n) => (
