   137→/** تاریخ امروز به شکل YYYY-MM-DD (تقویم محلی — پایدار بین رفرش‌ها) */
   138→function todayKey(): string {

   143→/** جابه‌جایی تقویمی تاریخ YYYY-MM-DD (بدون وابستگی به منطقهٔ زمانی) */
   144→function shiftDate(date: string, days: number): string {
   145→  const parts = date.split("-").map(Number);
   146→  const t = new Date(Date.UTC(parts[0], (parts[1] ?? 1) - 1, parts[2] ?? 1) + days * 86400000);
   147→  return `${t.getUTCFullYear()}-${pad2(t.getUTCMonth() + 1)}-${pad2(t.getUTCDate())}`;
   148→}

   150→/** هش ساده و پایدار (djb2) برای انتخاب واژهٔ روز */
   151→function hashStr(s: string): number {
   152→  let h = 5381;
   153→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   154→  return Math.abs(h);
   155→}

   157→/**
   158→ * رنگ‌بندی دو-گذرهٔ وردل — گذر اول فقط سبزها را می‌شمارد، گذر دوم
   159→ * زرد/خاکستری را با شمارش حروف باقی‌ماندهٔ پاسخ تعیین می‌کند
   160→ * (پاسخ کلاسیک: answer "speed" × guess "erase" → Eسبز، Rخاکستری، Aخاکستری، Sسبز، Eزرد).
   161→ * فاز G2-d — طول پارامتری شد (۴/۵/۶ حرفی).

   163→function judgeVerdicts(guess: string, answer: string, len: number): Verdict[] {
   164→  const res: Verdict[] = Array<Verdict>(len).fill("absent");
   165→  const remaining: Record<string, number> = {};
   166→  for (let i = 0; i < len; i++) {
   167→    if (guess[i] === answer[i]) {
   168→      res[i] = "correct";
   169→    } else {
   170→      remaining[answer[i]] = (remaining[answer[i]] ?? 0) + 1;
   171→    }
   172→  }
   173→  for (let i = 0; i < len; i++) {
   174→    if (res[i] === "correct") continue;
   175→    const ch = guess[i];
   176→    if ((remaining[ch] ?? 0) > 0) {
   177→      res[i] = "present";
   178→      remaining[ch] -= 1;
   179→    }
   180→  }
   181→  return res;
   182→}

   223→function loadDailyState(bookId: string): DailyState | null {
   224→  try {
   225→    const raw = localStorage.getItem(dailyKeyOf(bookId));
   226→    if (!raw) return null;
   227→    const p = JSON.parse(raw) as Partial<DailyState> | null;
   228→    if (!p || typeof p.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(p.date)) return null;
   229→    if (p.status !== "playing" && p.status !== "won" && p.status !== "lost") return null;
   230→    if (!Array.isArray(p.guesses)) return null;
   231→    const guesses = p.guesses
   232→      .filter((g): g is string => typeof g === "string" && WORD_RE.test(g))
   233→      .slice(0, MAX_ROWS);
   234→    return { date: p.date, guesses, status: p.status };
   235→  } catch {
   236→    return null;
   237→  }
   238→}
