   307→/** مسیر مستطیل گرد (سازگار با همهٔ مرورگرها) */
   308→function rrPath(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
   309→  c.beginPath();
   310→  c.moveTo(x + r, y);
   311→  c.arcTo(x + w, y, x + w, y + h, r);
   312→  c.arcTo(x + w, y + h, x, y + h, r);
   313→  c.arcTo(x, y + h, x, y, r);
   314→  c.arcTo(x, y, x + w, y, r);
   315→  c.closePath();
   316→}

   318→/** بافت بیلبورد گزینه — قرص سفید با متن انگلیسی تیره (رویه‌ای، بدون فونت خارجی) */
   319→function makeLabelTexture(text: string): THREE.CanvasTexture {
   320→  const W = 640;
   321→  const Hc = 160;
   322→  const cv = document.createElement("canvas");
   323→  cv.width = W;
   324→  cv.height = Hc;
   325→  const c = cv.getContext("2d");
   326→  if (c) {
   327→    let fs = 64;
   328→    c.font = `800 ${fs}px system-ui, sans-serif`;
   329→    while (c.measureText(text).width > W - 96 && fs > 24) {
   330→      fs -= 3;
   331→      c.font = `800 ${fs}px system-ui, sans-serif`;
   332→    }
   333→    const tw = Math.min(c.measureText(text).width, W - 96);
   334→    const pw = Math.min(tw + 56, W - 24);
   335→    const ph = 102;
   336→    const px = (W - pw) / 2;
   337→    const py = (Hc - ph) / 2;
   338→    c.fillStyle = "rgba(255,255,255,0.95)";
   339→    rrPath(c, px, py, pw, ph, 34);
   340→    c.fill();
   341→    c.lineWidth = 6;
   342→    c.strokeStyle = "rgba(6,95,70,0.35)";
   343→    rrPath(c, px + 3, py + 3, pw - 6, ph - 6, 30);
   344→    c.stroke();
   345→    c.fillStyle = "#065f46";
   346→    c.textAlign = "center";
   347→    c.textBaseline = "middle";
   348→    c.fillText(text, W / 2, Hc / 2 + 4, W - 90);
   349→  }
   350→  const tex = new THREE.CanvasTexture(cv);
   351→  tex.colorSpace = THREE.SRGBColorSpace;
   352→  tex.anisotropy = 2;
   353→  return tex;
   354→}

   427→/** فاز F3 — انتگرال حرکت بازیکن از کلیدهای فشرده‌شده (جهش bridge فقط داخل توابع ماژول) */
   428→function stepPlayer(b: SceneBridge, dt: number): boolean {
   429→  let mx = 0;
   430→  let mz = 0;
   431→  if (b.keys.has("w")) mz += 1;
   432→  if (b.keys.has("s")) mz -= 1;
   433→  if (b.keys.has("d")) mx += 1;
   434→  if (b.keys.has("a")) mx -= 1;
   435→  const moving = mx !== 0 || mz !== 0;
   436→  if (moving) {
   437→    const yaw = b.aim.yaw;
   438→    const sin = Math.sin(yaw);
   439→    const cos = Math.cos(yaw);
   440→    // جلو = −z محلی؛ راست = +x محلی — بدون pitch (گام زمینی)
   441→    const vx = -sin * mz + cos * mx;
   442→    const vz = -cos * mz - sin * mx;
   443→    const len = Math.hypot(vx, vz) || 1;
   444→    const sp = MOVE_SPEED * dt;
   445→    b.pos.x = clamp(b.pos.x + (vx / len) * sp, -MOVE_X_LIM, MOVE_X_LIM);
   446→    b.pos.z = clamp(b.pos.z + (vz / len) * sp, MOVE_Z_MIN, MOVE_Z_MAX);
   447→  }
   448→  b.moveMag = moving ? 1 : 0;
   449→  return moving;
   450→}

   452→/** به‌روزرسانی ترنسفورم دوربین اول‌شخص — تابع ماژول‌سطح (جهش THREE فقط از اینجا) */
   453→function applyRig(cam: THREE.Camera, b: SceneBridge, aspect: number, bobY: number): void {
   454→  b.laneK = Math.max(0.55, Math.min(1, aspect / 1.5));
   455→  const baseZ = 7 + (1 - b.laneK) * 4.2;
   456→  cam.position.x = b.pos.x;
   457→  cam.position.z = baseZ + b.pos.z;
   458→  cam.position.y = 1.7 + bobY;
   459→  cam.rotation.y = b.aim.yaw;
   460→  cam.rotation.x = b.aim.pitch;
   461→}

   463→/** تنظیم اولیهٔ دوربین — تابع ماژول‌سطح */
   464→function initRig(cam: THREE.Camera): void {
   465→  cam.position.set(0, 1.7, 7);
   466→  cam.rotation.order = "YXZ";

   469→/** دوربین اول‌شخص — چرخش + حرکت WASD (فاز F3) + باب سر + تطبیق عرض صحنه در پرتره */
   470→function Rig({ bridge }: { bridge: React.RefObject<SceneBridge> }) {
   471→  const { camera, size } = useThree();
   472→  const bobT = useRef(0);
   473→  useEffect(() => {
   474→    initRig(camera);
   475→  }, [camera]);
   476→  useFrame((_state, delta) => {
   477→    const dt = Math.min(delta, 0.05); // سقف delta — پرش برگشت به تب
   478→    const moving = stepPlayer(bridge.current, dt);
   479→    bobT.current += dt * (moving ? 8.5 : 2.2);
   480→    const bobY = Math.sin(bobT.current) * (moving ? 0.042 : 0.012);
   481→    applyRig(camera, bridge.current, size.width / Math.max(1, size.height), bobY);
   482→  });
   483→  return null;
   484→}

   535→        <cylinderGeometry args={[0.028, 0.032, 0.42, 12]} />
   536→        <meshStandardMaterial color="#111827" roughness={0.35} metalness={0.7} />
   537→      </mesh>
   538→      {/* دستهٔ کمک‌رنگ */}

   642→function DecoShape({ spec }: { spec: DecoSpec }) {

   667→/** یک هدف متحرک: اوکتاهدر درخشان + بیلبورد برچسب انگلیسی */
   668→function Target({
   669→  spec,
   670→  registry,
   671→  bridge,
   672→}: {
   673→  spec: TargetSpec;
   674→  registry: React.RefObject<THREE.Mesh[]>;
   675→  bridge: React.RefObject<SceneBridge>;
   676→}) {
   677→  const groupRef = useRef<THREE.Group>(null);
   678→  const meshRef = useRef<THREE.Mesh>(null);
   679→  const tex = useMemo(() => makeLabelTexture(spec.text), [spec.text]);
   680→  const t0 = useMemo(() => Math.random() * 6, []);
   681→  useEffect(() => () => tex.dispose(), [tex]);
   682→
   683→  // ثبت/حذف مش برای raycast شلیک
   684→  useEffect(() => {
   685→    const m = meshRef.current;
   686→    const list = registry.current;
   687→    if (!m || !list) return;
   688→    m.userData.id = spec.id;
   689→    list.push(m);
   690→    return () => {
   691→      const i = list.indexOf(m);
   692→      if (i >= 0) list.splice(i, 1);
   693→    };
   694→  }, [registry, spec.id]);
   695→
   696→  useFrame((state) => {
   697→    const g = groupRef.current;
   698→    const m = meshRef.current;
   699→    if (!g || !m) return;
   700→    const t = state.clock.elapsedTime;
   701→    const b = bridge.current;
   702→    g.position.set(
   703→      spec.lane * b.laneK + Math.sin(t * 0.5 + t0) * spec.driftAmp * b.laneK,
   704→      spec.baseY + Math.sin(t * spec.bobSpeed + t0) * 0.4,
   705→      spec.baseZ,
   706→    );
   707→    m.rotation.y += 0.008 * spec.spin * 60 * 0.016;
   708→    m.rotation.x = Math.sin(t * 0.4 + t0) * 0.25;
   709→    // فلاش سبز/قرمز پس از پاسخ + تنفس ملایم
   710→    const mat = m.material as THREE.MeshStandardMaterial;
   711→    const now = t * 1000;
   712→    const f = b.flashes.find((fl) => fl.id === spec.id && now < fl.until);
   713→    if (f) {
   714→      mat.emissive.setHex(f.kind === "ok" ? 0x10b981 : 0xef4444);
   715→      mat.emissiveIntensity = 1.15;
   716→    } else {
   717→      mat.emissive.setHex(spec.colorHex);
   718→      mat.emissiveIntensity = 0.24 + Math.sin(now / 420 + t0) * 0.09;
   719→    }
   720→  });
   721→
   722→  return (
   723→    <group ref={groupRef} position={[spec.lane, spec.baseY, spec.baseZ]}>
   724→      <mesh ref={meshRef}>
   725→        <octahedronGeometry args={[TARGET_R, 0]} />
   726→        <meshStandardMaterial
   727→          color={spec.colorHex}
   728→          flatShading
   729→          roughness={0.35}
   730→          metalness={0.15}
   731→          emissive={spec.colorHex}
   732→          emissiveIntensity={0.25}
   733→        />
   734→      </mesh>
   735→      <sprite position={[0, 1.55, 0]} scale={[2.9, 0.73, 1]}>
   736→        <spriteMaterial map={tex} transparent depthWrite={false} />
   737→      </sprite>
   738→    </group>
   739→  );
   740→}

   742→/** استخر انفجار رنگین‌کمانی — InstancedMesh با رنگ هر ذره */
   743→function Burst({ apiRef, count }: { apiRef: React.RefObject<BurstApi | null>; count: number }) {
   744→  const meshRef = useRef<THREE.InstancedMesh>(null);
   745→  const parts = useMemo(() => {
   746→    const dirs = new Float32Array(count * 3);
   747→    const speeds = new Float32Array(count);
   748→    for (let i = 0; i < count; i++) {
   749→      const u = Math.random() * Math.PI * 2;
   750→      const v = Math.acos(2 * Math.random() - 1);
   751→      dirs[i * 3] = Math.sin(v) * Math.cos(u);
   752→      dirs[i * 3 + 1] = Math.cos(v);
   753→      dirs[i * 3 + 2] = Math.sin(v) * Math.sin(u);
   754→      speeds[i] = 1.7 + Math.random() * 2.8;
   755→    }
   756→    return { dirs, speeds };
   757→  }, [count]);
   758→  const st = useRef({ active: false, t0: 0, origin: new THREE.Vector3() });
   759→  const dummy = useMemo(() => new THREE.Object3D(), []);
   760→
   761→  useEffect(() => {
   762→    const mesh = meshRef.current;
   763→    if (!mesh) return;
   764→    // رنگ هر ذره — پالت رنگین‌کمان
   765→    const c = new THREE.Color();
   766→    const rainbow = [0xf43f5e, 0xf97316, 0xfacc15, 0x22c55e, 0x14b8a6, 0x38bdf8, 0xa78bfa, 0xf472b6];
   767→    for (let i = 0; i < count; i++) {
   768→      c.setHex(rainbow[i % rainbow.length]);
   769→      mesh.setColorAt(i, c);
   770→    }
   771→    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
   772→    // پنهان‌سازی اولیه
   773→    const m0 = new THREE.Matrix4().makeScale(0, 0, 0);
   774→    for (let i = 0; i < count; i++) mesh.setMatrixAt(i, m0);
   775→    mesh.instanceMatrix.needsUpdate = true;
   776→  }, [count]);
   777→
   778→  useEffect(() => {
   779→    apiRef.current = {
   780→      fire: (origin) => {
   781→        st.current.active = true;
   782→        st.current.t0 = performance.now();
   783→        st.current.origin.copy(origin);
   784→      },
   785→    };
   786→    return () => {
   787→      apiRef.current = null;
   788→    };
   789→  }, [apiRef]);
   790→
   791→  useFrame(() => {
   792→    const mesh = meshRef.current;
   793→    if (!mesh || !st.current.active) return;
   794→    const t = (performance.now() - st.current.t0) / 950;
   795→    if (t >= 1) {
   796→      st.current.active = false;
   797→      const m0 = new THREE.Matrix4().makeScale(0, 0, 0);
   798→      for (let i = 0; i < count; i++) mesh.setMatrixAt(i, m0);
   799→      mesh.instanceMatrix.needsUpdate = true;
   800→      return;
   801→    }
   802→    const e = easeOutCubic(t);
   803→    for (let i = 0; i < count; i++) {
   804→      const d = e * parts.speeds[i] * 2.1;
   805→      dummy.position.set(
   806→        st.current.origin.x + parts.dirs[i * 3] * d,
   807→        st.current.origin.y + parts.dirs[i * 3 + 1] * d - t * t * 1.5,
   808→        st.current.origin.z + parts.dirs[i * 3 + 2] * d,
   809→      );
   810→      const s = (1 - t) * 0.3;
   811→      dummy.scale.setScalar(Math.max(0.001, s));
   812→      dummy.rotation.set(t * 5 + i, t * 4, i * 0.7);
   813→      dummy.updateMatrix();
   814→      mesh.setMatrixAt(i, dummy.matrix);
   815→    }
   816→    mesh.instanceMatrix.needsUpdate = true;
   817→  });
   818→
   819→  return (
   820→    <instancedMesh ref={meshRef} args={[undefined, undefined, count]} frustumCulled={false}>
   821→      <tetrahedronGeometry args={[1, 0]} />
   822→      <meshBasicMaterial toneMapped={false} />
   823→    </instancedMesh>
   824→  );
   825→}
