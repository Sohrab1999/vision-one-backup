    71→    71→function TeacherTopBar({

    81→    81→  const [leaving, setLeaving] = useState(false);
    82→    82→
    83→    83→  async function copyCode() {

   137→   137→            {session.code}
   138→   138→          </span>
   139→   139→          <Button
   140→   140→            variant="ghost"
   141→   141→            size="sm"
   142→   142→            onClick={() => void copyCode()}

   155→   155→          disabled={leaving}

   157→   157→          className={cn(
   158→   158→            "min-h-11 gap-1.5 px-3 text-[12px] font-bold",
   159→   159→            confirmLeave

   169→   169→    </div>

   284→   284→function AssignmentsCard({
   285→   285→  state,
   286→   286→  session,
   287→   287→  refresh,
   288→   288→}: {
   289→   289→  state: ClassState | null;
   290→   290→  session: NonNullable<ReturnType<typeof useClassSession>>;
   291→   291→  refresh: () => void;
   292→   292→}) {
   293→   293→  const [title, setTitle] = useState("");
   294→   294→  const [kind, setKind] = useState<AssignKind>("review");
   295→   295→  const [lesson, setLesson] = useState(1);
   296→   296→  const [questionCount, setQuestionCount] = useState(10);
   297→   297→  const [busy, setBusy] = useState(false);
   298→   298→
   299→   299→  const assignments = state?.assignments ?? [];
   300→   300→  const memberCount = state?.members?.length ?? 0;
   301→   301→
   302→   302→  async function submit() {
   303→   303→    const t = title.trim();
   304→   304→    if (t.length < 2) {
   305→   305→      toast.error("عنوان تکلیف را بنویس");
   306→   306→      return;
   307→   307→    }
   308→   308→    const qc = Math.min(50, Math.max(5, Math.round(questionCount) || 10));
   309→   309→    const les = Math.min(12, Math.max(0, Math.round(lesson) || 0));
   310→   310→    setBusy(true);
   311→   311→    try {
   312→   312→      const res = await fetch(`/api/class/${encodeURIComponent(session.code)}/assignments`, {
   313→   313→        method: "POST",
   314→   314→        headers: { "Content-Type": "application/json" },
   315→   315→        body: JSON.stringify({
   316→   316→          teacherKey: session.key,
   317→   317→          title: t,
   318→   318→          kind,
   319→   319→          lesson: kind === "review" ? les : undefined,
   320→   320→          questionCount: kind === "quiz" ? qc : undefined,
   321→   321→        }),
   322→   322→      });
   323→   323→      if (res.status === 201) {
   324→   324→        toast.success("تکلیف اعلام شد 📣");
   325→   325→        setTitle("");
   326→   326→        refresh();
   327→   327→      } else {
   328→   328→        toast.error(await readApiError(res, "اعلام تکلیف ممکن نشد"));
   329→   329→      }
   330→   330→    } catch {
   331→   331→      toast.error("ارتباط با سرور برقرار نشد");
   332→   332→    } finally {
   333→   333→      setBusy(false);
   334→   334→    }
   335→   335→  }

   341→   341→        headers: { "Content-Type": "application/json" },
   342→   342→        body: JSON.stringify({ teacherKey: session.key, text: `⏰ تکلیف «${a.title}» را فراموش نکن!` }),
   343→   343→      });
   344→   344→      if (res.status === 201) toast.success("یادآوری ارسال شد ⏰");
   345→   345→      else toast.error(await readApiError(res, "ارسال یادآوری ممکن نشد"));
   346→   346→    } catch {
   347→   347→      toast.error("ارتباط با سرور برقرار نشد");
   348→   348→    }
   349→   349→  }
   350→   350→
   351→   351→  return (
   352→   352→    <Card className="rise-in">
   353→   353→      <CardContent className="space-y-4 p-4">
   354→   354→        <h2 className="flex items-center gap-1.5 text-sm font-black">
   355→   355→          <ClipboardList size={16} className="text-primary" aria-hidden />
   356→   356→          تکالیف
   357→   357→        </h2>
   358→   358→
   359→   359→        {/* فرم فشرده اعلام تکلیف */}
   360→   360→        <div className="space-y-2.5 rounded-xl border bg-muted/20 p-3">
   361→   361→          <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-[1fr_auto]">
   362→   362→            <Input
   363→   363→              value={title}
   364→   364→              onChange={(e) => setTitle(e.target.value)}
   365→   365→              placeholder="عنوان تکلیف — مثلاً واژگان درس ۳ را مرور کن"
   366→   366→              maxLength={80}
   367→   367→              className="min-h-11"
   368→   368→              aria-label="عنوان تکلیف"
   369→   369→            />
   370→   370→            <div className="flex gap-1 self-start rounded-xl border bg-card p-1" role="group" aria-label="نوع تکلیف">
   371→   371→              <Button
   372→   372→                type="button"
   373→   373→                variant={kind === "review" ? "secondary" : "ghost"}
   374→   374→                size="sm"
   375→   375→                className="min-h-9"
   376→   376→                onClick={() => setKind("review")}
   377→   377→                aria-pressed={kind === "review"}
   378→   378→              >
   379→   379→                مرور واژگان
   380→   380→              </Button>
   381→   381→              <Button
   382→   382→                type="button"
   383→   383→                variant={kind === "quiz" ? "secondary" : "ghost"}
   384→   384→                size="sm"
   385→   385→                className="min-h-9"
   386→   386→                onClick={() => setKind("quiz")}
   387→   387→                aria-pressed={kind === "quiz"}
   388→   388→              >
   389→   389→                آزمون
   390→   390→              </Button>
   391→   391→            </div>
   392→   392→          </div>

   394→   394→          <div className="grid grid-cols-1 items-end gap-2.5 sm:grid-cols-[1fr_auto]">
   395→   395→            {kind === "review" ? (
   396→   396→              <div className="space-y-1">
   397→   397→                <Label htmlFor="assign-lesson" className="text-[11.5px] text-muted-foreground">
   398→   398→                  شماره درس (۰ تا ۱۲)

   408→   408→                  className="min-h-11"
   409→   409→                  aria-label="شماره درس مرور"
   410→   410→                />
   411→   411→              </div>

   416→   416→                </Label>
   417→   417→                <Input

   419→   419→                  type="number"
   420→   420→                  inputMode="numeric"
   421→   421→                  min={5}
   422→   422→                  max={50}
   423→   423→                  value={questionCount}

   426→   426→                  aria-label="تعداد سؤال آزمون"
   427→   427→                />
   428→   428→              </div>
   429→   429→            )}
   430→   430→            <Button type="button" disabled={busy} onClick={() => void submit()} className="min-h-11 gap-1.5 font-bold">
   431→   431→              {busy ? <Loader2 size={15} className="animate-spin" aria-hidden /> : <Plus size={15} aria-hidden />}
   432→   432→              اعلام تکلیف
   433→   433→            </Button>
   434→   434→          </div>
