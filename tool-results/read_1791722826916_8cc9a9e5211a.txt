   122→function generateCode(): string {
   123→  for (let i = 0; i < 500; i++) {
   124→    const code = String(Math.floor(1000 + Math.random() * 9000));
   125→    if (!rooms.has(code)) return code;
   126→  }
   127→  return String(Date.now() % 10000).padStart(4, "0");
   128→}

   130→function sanitizeName(raw: unknown): string {
   131→  const s = typeof raw === "string" ? raw.trim().replace(/\s+/g, " ").slice(0, 24) : "";
   132→  return s.length > 0 ? s : "مهمان";
   133→}

   135→/** پالایش بستهٔ سؤال میزبان — خروجی تمیز و سالم؛ null = نامعتبر */
   136→function sanitizePack(raw: unknown): DuelQuestion[] | null {
   137→  if (!Array.isArray(raw)) return null;
   138→  const out: DuelQuestion[] = [];
   139→  for (const item of raw.slice(0, MAX_PACK)) {
   140→    if (!item || typeof item !== "object") continue;
   141→    const q = item as Record<string, unknown>;
   142→    const text = typeof q.text === "string" ? q.text.trim().slice(0, 400) : "";
   143→    const options = Array.isArray(q.options)
   144→      ? q.options.filter((o): o is string => typeof o === "string" && o.trim().length > 0).map((o) => o.slice(0, 120))
   145→      : [];
   146→    const answer = typeof q.answer === "number" && Number.isInteger(q.answer) ? q.answer : -1;
   147→    if (!text || options.length < 2 || answer < 0 || answer >= options.length) continue;
   148→    out.push({
   149→      id: typeof q.id === "string" ? q.id.slice(0, 40) : `q${out.length + 1}`,
   150→      text,
   151→      options,
   152→      answer,
   153→    });
   154→  }
   155→  return out.length >= 1 ? out : null;
   156→}

   158→function playersPublic(room: Room): PlayerPublic[] {
   159→  return [...room.players.values()].map((p) => ({
   160→    id: p.id,
   161→    name: p.name,
   162→    score: p.score,
   163→    correct: p.correct,
   164→    lastDelta: p.lastDelta,
   165→    lastCorrect: p.lastCorrect,
   166→    connected: p.connected,
   167→    ...(p.isBot ? { isBot: true } : {}),
   168→  }));
   169→}

   171→/** فاز G5-03 — آیا همهٔ بازیکنانِ باقی‌ماندهٔ اتاق ربات داخلی‌اند؟ (میزبان رفته) */
   172→function allBots(room: Room): boolean {
   173→  return [...room.players.values()].every((p) => p.isBot === true);
   174→}

   176→function scoresOf(room: Room): ScoresMap {
   177→  const out: ScoresMap = {};
   178→  for (const p of room.players.values()) {
   179→    out[p.id] = { name: p.name, score: p.score, lastDelta: p.lastDelta, lastCorrect: p.lastCorrect };
   180→  }
   181→  return out;
   182→}

   184→function emitToSocket(room: Room, playerId: string, event: string, payload: unknown): void {
   185→  const p = room.players.get(playerId);
   186→  // فاز G5-03 — ربات داخلی socket واقعی ندارد؛ ساکت رد شو (بدون لاگ نویز)
   187→  if (!p || p.isBot) return;
   188→  if (!p.socketId) {
   189→    console.log(`[duel][DBG] emitToSocket ${event} → ${playerId} SKIP (no socketId)`);
   190→    return;
   191→  }
   192→  const s = io.sockets.sockets.get(p.socketId);
   193→  if (!s) {
   194→    console.log(`[duel][DBG] emitToSocket ${event} → ${playerId} SKIP (socket ${p.socketId} not in io map)`);
   195→    return;
   196→  }
   197→  console.log(`[duel][DBG] emitToSocket ${event} → ${playerId} via ${p.socketId} OK`);
   198→  s.emit(event, payload);
   199→}

   201→function emitToAll(room: Room, event: string, payload: unknown): void {
   202→  for (const p of room.players.values()) {
   203→    emitToSocket(room, p.id, event, payload);
   204→  }
   205→}

   207→/** وضعیت اتاق را به تک‌تک سوکت‌ها به‌صورت شخصی می‌فرستد (هر کدام `you` خودشان را می‌گیرند) */
   208→function broadcastRoom(room: Room): void {
   209→  for (const p of room.players.values()) {
   210→    emitToSocket(room, p.id, "joined", {
   211→      code: room.code,
   212→      players: playersPublic(room),
   213→      hostId: room.hostId,
   214→      state: room.state,
   215→      you: p.id,
   216→    });
   217→  }
   218→}

   220→function clearTimers(room: Room): void {
   221→  if (room.questionTimer) clearTimeout(room.questionTimer);
   222→  if (room.revealTimer) clearTimeout(room.revealTimer);
   223→  if (room.botTimer) clearTimeout(room.botTimer);
   224→  room.questionTimer = null;
   225→  room.revealTimer = null;
   226→  room.botTimer = null;

   229→function deleteRoom(code: string): void {
   230→  const room = rooms.get(code);
   231→  if (!room) return;
   232→  clearTimers(room);
   233→  if (room.graceTimer) clearTimeout(room.graceTimer);
   234→  rooms.delete(code);
   235→  for (const [sid, ctx] of socketToRoom) {
   236→    if (ctx.code === code) socketToRoom.delete(sid);
   237→  }
   238→  console.log(`[duel] room ${code} deleted`);
   239→}

   247→function startQuestion(room: Room): void {
   248→  if (room.currentIndex >= room.pack.length) {
   249→    finishGame(room);
   250→    return;
   251→  }
   252→  clearTimers(room);
   253→  room.state = "playing";
   254→  room.phase = "question";
   255→  room.revealAnswer = null;
   256→  room.answers = new Map();
   257→  const q = room.pack[room.currentIndex];
   258→  const t = now();
   259→  room.current = { index: room.currentIndex, total: room.pack.length, text: q.text, options: q.options, endsAt: t + QUESTION_MS, serverNow: t };
   260→  touch(room);
   261→  emitToAll(room, "question", {
   262→    index: room.currentIndex,
   263→    total: room.pack.length,
   264→    text: q.text,
   265→    options: q.options,
   266→    endsAt: room.current.endsAt,
   267→    serverNow: t,
   268→  });
   269→  // ۱۵۰ms گاری تأخیر شبکه — قضاوت نهایی با زمان انقضا
   270→  room.questionTimer = setTimeout(() => closeQuestion(room), QUESTION_MS + 150);
   271→  // فاز G5-03 — ربات داخلی هم مثل حریف واقعی «فکر می‌کند» و جواب می‌دهد
   272→  scheduleBotAnswer(room);
   273→}
   274→
   275→/**
   276→ * فاز G5-03 — زمان‌بندی پاسخ ربات داخلی: زمان فکر تصادفی ۲..۸ ثانیه
   277→ * (همیشه کمتر از مهلت ۱۵ث) سپس همان مسیر registerAnswer پاسخ انسانی.
   278→ */
   279→function scheduleBotAnswer(room: Room): void {
   280→  if (room.botTimer) {
   281→    clearTimeout(room.botTimer);
   282→    room.botTimer = null;
   283→  }
   284→  if (room.state !== "playing" || room.phase !== "question") return;
   285→  const bot = [...room.players.values()].find((p) => p.isBot);
   286→  if (!bot || bot.botSkill === undefined) return;
   287→  const think = BOT_THINK_MIN_MS + Math.floor(Math.random() * (BOT_THINK_MAX_MS - BOT_THINK_MIN_MS + 1));
   288→  room.botTimer = setTimeout(() => {
   289→    room.botTimer = null;
   290→    botAnswer(room, bot);
   291→  }, think);
   292→}
   293→
   294→/** پاسخ ربات داخلی به سؤال جاری — با احتمال skill درست، وگرنه یک گزینهٔ غلط تصادفی */
   295→function botAnswer(room: Room, bot: Player): void {
   296→  if (room.state !== "playing" || room.phase !== "question" || !room.current) return;
   297→  if (room.answers.has(bot.id)) return;
   298→  const q = room.pack[room.currentIndex];
   299→  const skill = bot.botSkill ?? BOT_SKILL_BASE;
   300→  const correct = Math.random() < skill;
   301→  const wrongs = q.options.map((_, i) => i).filter((i) => i !== q.answer);
   302→  const choice = correct ? q.answer : (wrongs[Math.floor(Math.random() * wrongs.length)] ?? q.answer);
   303→  registerAnswer(room, bot, choice);
   304→}
   305→
   306→/**
   307→ * فاز G5-03 — ثبت پاسخ یک بازیکن (انسان یا ربات) — منطق امتیازدهی مشترک؛
   308→ * بدنه از هندلر `answer` استخراج شد تا ربات دقیقاً همان مسیر را طی کند
   309→ * (score → بستن زودهنگام اگر همه پاسخ دادند → reveal).
   310→ */
   311→function registerAnswer(room: Room, p: Player, choice: number): void {
   312→  if (room.phase !== "question" || !room.current) return;
   313→  const q = room.pack[room.currentIndex];
   314→  const remaining = Math.max(0, room.current.endsAt - now());
   315→  const bonus = Math.round((MAX_BONUS * remaining) / QUESTION_MS);
   316→  const correct = choice === q.answer;
   317→  const delta = correct ? BASE_SCORE + bonus : 0;
   318→  p.score += delta;
   319→  if (correct) p.correct += 1;
   320→  p.lastDelta = delta;
   321→  p.lastCorrect = correct;
   322→  room.answers.set(p.id, { choice, correct, delta });
   323→  touch(room);
   324→  emitToAll(room, "score", { questionIndex: room.currentIndex, scores: scoresOf(room) });
   325→
   326→  // همهٔ بازیکنان آنلاین پاسخ دادند؟ بستن زودهنگام سؤال
   327→  const connected = [...room.players.values()].filter((pl) => pl.connected);
   328→  if (connected.length > 0 && room.answers.size > 0 && room.answers.size >= connected.length) {
   329→    closeQuestion(room);
   330→  }
   331→}
   332→
   333→function closeQuestion(room: Room): void {
   334→  if (room.phase !== "question") return;
   335→  if (room.questionTimer) clearTimeout(room.questionTimer);
   336→  room.questionTimer = null;
   337→  room.phase = "reveal";
   338→  const q = room.pack[room.currentIndex];
   339→  room.revealAnswer = q.answer;
   340→  touch(room);
   341→  // طبق قرارداد: بعد از هر پاسخ یا انقضا رویداد score می‌رود
   342→  emitToAll(room, "score", { questionIndex: room.currentIndex, scores: scoresOf(room) });
   343→  emitToAll(room, "reveal", { index: room.currentIndex, answer: q.answer, scores: scoresOf(room) });
   344→  room.revealTimer = setTimeout(() => {
   345→    room.currentIndex += 1;
   346→    startQuestion(room);
   347→  }, REVEAL_MS);
   348→}
   349→
   350→function finishGame(room: Room): void {
   351→  clearTimers(room);
   352→  room.state = "ended";
   353→  room.phase = "reveal";
   354→  room.current = null;
   355→  room.revealAnswer = null;
   356→  touch(room);
   357→  const duration = room.startedAt ? Math.max(1, Math.round((now() - room.startedAt) / 1000)) : 0;
   358→  const finalScores = [...room.players.values()]
   359→    .map((p) => ({ id: p.id, name: p.name, score: p.score, correct: p.correct }))
   360→    .sort((a, b) => b.score - a.score || b.correct - a.correct);
   361→  emitToAll(room, "finished", { finalScores, packSummary: room.pack, duration });
   362→  console.log(`[duel] room ${room.code} finished (${duration}s)`);
   363→}

   365→// ─── مهلت قطعی (grace) ───
   366→
   367→function armGrace(room: Room): void {
   368→  if (room.graceTimer) return;
   369→  const earliest = Math.min(
   370→    ...[...room.players.values()].filter((p) => !p.connected && p.disconnectedAt).map((p) => p.disconnectedAt!),
   371→  );
   372→  if (!Number.isFinite(earliest)) return;
   373→  const wait = Math.max(1_000, earliest + GRACE_MS - now());
   374→  room.graceTimer = setTimeout(() => {
   375→    room.graceTimer = null;
   376→    const t = now();
   377→    let removed = false;
   378→    for (const p of [...room.players.values()]) {
   379→      if (!p.connected && p.disconnectedAt && t - p.disconnectedAt >= GRACE_MS - 250) {
   380→        room.players.delete(p.id);
   381→        removed = true;
   382→      }
   383→    }
   384→    if (room.players.size === 0 || allBots(room)) {
   385→      deleteRoom(room.code);
   386→      return;
   387→    }
   388→    if (removed) broadcastRoom(room);
   389→    // اگر هنوز بازیکن قطعی دارد، تایمر بعدی
   390→    if ([...room.players.values()].some((p) => !p.connected)) armGrace(room);
   391→  }, wait);
   392→}

   394→function clearGraceIfAllConnected(room: Room): void {
   395→  if (room.graceTimer && [...room.players.values()].every((p) => p.connected)) {
   396→    clearTimeout(room.graceTimer);
   397→    room.graceTimer = null;
   398→  }
   399→}

   403→const httpServer = createServer();
   404→const io = new Server(httpServer, {
   405→  // DO NOT change the path — Caddy forwards via XTransformPort

   412→io.on("connection", (socket: Socket) => {
   413→  console.log(`[duel] connected: ${socket.id}`);
   414→
   415→  // ── ساخت اتاق (میزبان؛ بستهٔ سؤال را خودش می‌فرستد) ──
   416→  socket.on("create", (raw: unknown) => {
   417→    const data = (raw ?? {}) as { name?: unknown; pack?: unknown };
   418→    if (rooms.size >= MAX_ROOMS) {
   419→      emitError(socket, "سرور شلوغ است — چند لحظه بعد تلاش کن");
   420→      return;
   421→    }
   422→    const pack = sanitizePack(data.pack);
   423→    if (!pack) {
   424→      emitError(socket, "بستهٔ سؤال نامعتبر است");
   425→      return;
   426→    }
   427→    const playerId = newPlayerId();
   428→    const room: Room = {
   429→      code: generateCode(),
   430→      hostId: playerId,
   431→      players: new Map(),
   432→      pack,
   433→      state: "lobby",
   434→      currentIndex: 0,
   435→      phase: "question",
   436→      answers: new Map(),
   437→      current: null,
   438→      revealAnswer: null,
   439→      questionTimer: null,
   440→      revealTimer: null,
   441→      graceTimer: null,
   442→      botTimer: null,
   443→      startedAt: null,
   444→      createdAt: now(),
   445→      lastActivity: now(),
   446→    };
   447→    room.players.set(playerId, {
   448→      id: playerId,
   449→      name: sanitizeName(data.name),
   450→      score: 0,
   451→      correct: 0,
   452→      lastDelta: 0,
   453→      lastCorrect: true,
   454→      connected: true,
   455→      socketId: socket.id,
   456→      disconnectedAt: null,
   457→    });
   458→    rooms.set(room.code, room);
   459→    socketToRoom.set(socket.id, { code: room.code, playerId });
   460→    socket.emit("created", { code: room.code, playerId, total: pack.length });
   461→    broadcastRoom(room);
   462→    console.log(`[duel] room ${room.code} created by ${playerId} (${pack.length} questions)`);
   463→  });

   465→  // ── فاز G5-03 — تمرین با ربات تطبیقی (میزبان + ربات داخلی بدون socket واقعی) ──
   466→  // ورودی: { name?, pack?, winRate? } — winRate ∈ [0,1] نرخ برد میزبان (پیش‌فرض ۰٫۵).
   467→  // مهارت ربات: s = clamp(0.35, 0.9, 0.65 + (winRate − 0.5) × 0.6) — کشش برد/باخت به ~۵۰٪.
   468→  // جریان بعدی همان اتاق کلاسیک است: start → question → answer/reveal → finished؛
   469→  // ربات همهٔ رخدادهای حریف واقعی را از سمت سرور برای socketId میزبان emit می‌کند.
   470→  socket.on("play_bot", (raw: unknown) => {
   471→    const data = (raw ?? {}) as { name?: unknown; pack?: unknown; winRate?: unknown };
   472→    if (rooms.size >= MAX_ROOMS) {
   473→      emitError(socket, "سرور شلوغ است — چند لحظه بعد تلاش کن");
   474→      return;
   475→    }
   476→    const pack = sanitizePack(data.pack);
   477→    if (!pack) {
   478→      emitError(socket, "بستهٔ سؤال نامعتبر است");
   479→      return;
   480→    }
   481→    const winRate =
   482→      typeof data.winRate === "number" && Number.isFinite(data.winRate)
   483→        ? Math.min(1, Math.max(0, data.winRate))
   484→        : 0.5;
   485→    const skill = Math.min(BOT_SKILL_MAX, Math.max(BOT_SKILL_MIN, BOT_SKILL_BASE + (winRate - 0.5) * 0.6));
   486→    const playerId = newPlayerId();
   487→    const botId = newPlayerId();
   488→    const room: Room = {
   489→      code: generateCode(),
   490→      hostId: playerId,
   491→      players: new Map(),
   492→      pack,
   493→      state: "lobby",
   494→      currentIndex: 0,
   495→      phase: "question",
   496→      answers: new Map(),
   497→      current: null,
   498→      revealAnswer: null,
   499→      questionTimer: null,
   500→      revealTimer: null,
   501→      graceTimer: null,
   502→      botTimer: null,
   503→      startedAt: null,
   504→      createdAt: now(),
   505→      lastActivity: now(),
   506→    };
   507→    room.players.set(playerId, {
   508→      id: playerId,
   509→      name: sanitizeName(data.name),
   510→      score: 0,
   511→      correct: 0,
   512→      lastDelta: 0,
   513→      lastCorrect: true,
   514→      connected: true,
   515→      socketId: socket.id,
   516→      disconnectedAt: null,
   517→    });
   518→    // بازیکن دوم = ربات داخلی — socket ندارد؛ امتیاز/پاسخ‌هایش با همان منطق registerAnswer ساخته می‌شود
   519→    room.players.set(botId, {
   520→      id: botId,
   521→      name: BOT_NAME,
   522→      score: 0,
   523→      correct: 0,
   524→      lastDelta: 0,
   525→      lastCorrect: true,
   526→      connected: true,
   527→      socketId: null,
   528→      disconnectedAt: null,
   529→      isBot: true,
   530→      botSkill: skill,
   531→    });
   532→    rooms.set(room.code, room);
   533→    socketToRoom.set(socket.id, { code: room.code, playerId });
   534→    socket.emit("created", { code: room.code, playerId, total: pack.length });
   535→    broadcastRoom(room); // ربات socket ندارد — فقط میزبان joined می‌گیرد
   536→    // لاگ فنی کوتاه (بدون PII) برای دیدن رفتار تطبیقی
   537→    console.log(
   538→      `[duel] bot room ${room.code} — skill=${skill.toFixed(2)} winRate=${winRate.toFixed(2)} (${pack.length}q)`,
   539→    );
   540→  });
   541→
   542→  // ── پیوستن با کد ──
   543→  socket.on("join", (raw: unknown) => {
   544→    const data = (raw ?? {}) as { code?: unknown; name?: unknown };
   545→    const code = typeof data.code === "string" ? data.code.replace(/\D/g, "").slice(0, 4) : "";
   546→    const room = rooms.get(code);
   547→    if (!room) {
   548→      emitError(socket, "کد اتاق پیدا نشد");
   549→      return;
   550→    }
   551→    if (room.players.size >= MAX_PLAYERS) {
   552→      emitError(socket, "اتاق پر است");
   553→      return;
   554→    }
   555→    if (room.state === "playing") {
   556→      emitError(socket, "بازی همین حالا در جریان است — منتظر دور بعد بمان");
   557→      return;
   558→    }
   559→    const playerId = newPlayerId();
   560→    room.players.set(playerId, {
   561→      id: playerId,
   562→      name: sanitizeName(data.name),
   563→      score: 0,
   564→      correct: 0,
   565→      lastDelta: 0,
   566→      lastCorrect: true,
   567→      connected: true,
   568→      socketId: socket.id,
   569→      disconnectedAt: null,
   570→    });
   571→    socketToRoom.set(socket.id, { code: room.code, playerId });
   572→    touch(room);
   573→    clearGraceIfAllConnected(room);
   574→    broadcastRoom(room);
   575→    console.log(`[duel] ${playerId} joined room ${room.code}`);
   576→  });
   577→
   578→  // ── اتصال دوباره بعد از قطعی (همان playerId) ──
   579→  socket.on("rejoin", (raw: unknown) => {
   580→    const data = (raw ?? {}) as { code?: unknown; playerId?: unknown };
   581→    const code = typeof data.code === "string" ? data.code.replace(/\D/g, "").slice(0, 4) : "";
   582→    const playerId = typeof data.playerId === "string" ? data.playerId : "";
   583→    const room = rooms.get(code);
   584→    if (!room) {
   585→      emitError(socket, "اتاق دیگر وجود ندارد — دور تمام شده است");
   586→      return;
   587→    }
   588→    const p = room.players.get(playerId);
   589→    if (!p) {
   590→      emitError(socket, "جایگاه شما در اتاق پیدا نشد");
   591→      return;
   592→    }
   593→    p.connected = true;
   594→    p.socketId = socket.id;
   595→    p.disconnectedAt = null;
   596→    socketToRoom.set(socket.id, { code: room.code, playerId });
   597→    touch(room);
   598→    clearGraceIfAllConnected(room);
   599→    // همگام‌سازی کامل وضعیت برای خودِ وصل‌شونده
   600→    socket.emit("joined", {
   601→      code: room.code,
   602→      players: playersPublic(room),
   603→      hostId: room.hostId,
   604→      state: room.state,
   605→      you: playerId,
   606→    });
   607→    if (room.state === "playing") {
   608→      if (room.phase === "question" && room.current) {
   609→        socket.emit("question", room.current);
   610→      } else if (room.phase === "reveal" && room.revealAnswer !== null) {
   611→        socket.emit("reveal", { index: room.currentIndex, answer: room.revealAnswer, scores: scoresOf(room) });
   612→      }
   613→    } else if (room.state === "ended") {
   614→      const duration = room.startedAt ? Math.max(1, Math.round((now() - room.startedAt) / 1000)) : 0;
   615→      const finalScores = [...room.players.values()]
   616→        .map((pl) => ({ id: pl.id, name: pl.name, score: pl.score, correct: pl.correct }))
   617→        .sort((a, b) => b.score - a.score || b.correct - a.correct);
   618→      socket.emit("finished", { finalScores, packSummary: room.pack, duration });
   619→    }
   620→    for (const other of room.players.values()) {
   621→      if (other.id !== playerId) {
   622→        emitToSocket(room, other.id, "opponent-back", {
   623→          name: p.name,
   624→          message: `${p.name} دوباره وصل شد — ادامه می‌دهیم!`,
   625→        });
   626→      }
   627→    }
   628→    console.log(`[duel] ${playerId} rejoined room ${room.code}`);
   629→  });
   630→
   631→  // ── شروع دوئل (فقط میزبان) ──
   632→  socket.on("start", () => {
   633→    const ctx = socketToRoom.get(socket.id);
   634→    if (!ctx) return;
   635→    const room = rooms.get(ctx.code);
   636→    if (!room) return;
   637→    if (room.hostId !== ctx.playerId) {
   638→      emitError(socket, "فقط میزبان می‌تواند دوئل را شروع کند");
   639→      return;
   640→    }
   641→    if (room.state !== "lobby") {
   642→      emitError(socket, "بازی قبلاً شروع شده است");
   643→      return;
   644→    }
   645→    if (room.players.size < 2 || [...room.players.values()].some((p) => !p.connected)) {
   646→      emitError(socket, "حریف هنوز آنلاین نیست — منتظر بمان");
   647→      return;
   648→    }
   649→    room.startedAt = now();
   650→    touch(room);
   651→    startQuestion(room);
   652→    console.log(`[duel] room ${room.code} started`);
   653→  });
   654→
   655→  // ── پاسخ به سؤال ──
   656→  socket.on("answer", (raw: unknown) => {
   657→    const data = (raw ?? {}) as { index?: unknown; choice?: unknown };
   658→    const ctx = socketToRoom.get(socket.id);
   659→    if (!ctx) return;
   660→    const room = rooms.get(ctx.code);
   661→    if (!room || room.state !== "playing" || room.phase !== "question" || !room.current) return;
   662→    if (typeof data.index !== "number" || data.index !== room.currentIndex) return;
   663→    const q = room.pack[room.currentIndex];
   664→    const choice = typeof data.choice === "number" && Number.isInteger(data.choice) ? data.choice : -1;
   665→    if (choice < 0 || choice >= q.options.length) return;
   666→    const p = room.players.get(ctx.playerId);
   667→    if (!p || !p.connected) return;
   668→    if (room.answers.has(p.id)) return; // هر بازیکن فقط یک بار در هر سؤال
   669→
   670→    // فاز G5-03 — بدنهٔ امتیازدهی به registerAnswer منتقل شد (مشترک با ربات داخلی)
   671→    registerAnswer(room, p, choice);
   672→  });
   673→
   674→  // ── شروع دوباره با همان بستهٔ سؤال (فقط میزبان) ──
   675→  socket.on("rematch", () => {
   676→    const ctx = socketToRoom.get(socket.id);
   677→    if (!ctx) return;
   678→    const room = rooms.get(ctx.code);
   679→    if (!room || room.hostId !== ctx.playerId) return;
   680→    if (room.state !== "ended") return;
   681→    for (const p of room.players.values()) {
   682→      p.score = 0;
   683→      p.correct = 0;
   684→      p.lastDelta = 0;
   685→      p.lastCorrect = true;
   686→    }
   687→    room.currentIndex = 0;
   688→    room.startedAt = now();
   689→    touch(room);
   690→    emitToAll(room, "rematch", { message: "دور دوباره با همان سؤال‌ها شروع شد!" });
   691→    broadcastRoom(room);
   692→    startQuestion(room);
   693→    console.log(`[duel] room ${room.code} rematch`);
   694→  });
   695→
   696→  // ── قطع اتصال ──
   697→  socket.on("disconnect", () => {
   698→    console.log(`[duel] disconnected: ${socket.id}`);
   699→    const ctx = socketToRoom.get(socket.id);
   700→    if (!ctx) return;
   701→    socketToRoom.delete(socket.id);
   702→    const room = rooms.get(ctx.code);
   703→    if (!room) return;
   704→    const p = room.players.get(ctx.playerId);
   705→    if (!p || p.socketId !== socket.id) return;
   706→
   707→    p.connected = false;
   708→    p.socketId = null;
   709→    p.disconnectedAt = now();
   710→    touch(room);
   711→
   712→    if (room.state === "lobby") {
   713→      // در لابی چیزی برای حفظ کردن نیست — جایگاه آزاد می‌شود (اتاق فقط‌ربات هم بسته می‌شود)
   714→      room.players.delete(p.id);
   715→      if (room.players.size === 0 || allBots(room)) {
   716→        deleteRoom(room.code);
   717→        return;
   718→      }
   719→      if (room.hostId === p.id) {
   720→        room.hostId = [...room.players.keys()][0];
   721→        emitToAll(room, "error", { message: "میزبان رفت — میزبانی به بازیکن دیگر رسید", fatal: false });
   722→      }
   723→      broadcastRoom(room);
   724→      return;
   725→    }
   726→
   727→    // حین بازی یا بعد از پایان: ۶۰ ثانیه مهلت rejoin — امتیازها حفظ می‌شود
   728→    for (const other of room.players.values()) {
   729→      if (other.connected) {
   730→        emitToSocket(room, other.id, "opponent-left", {
   731→          name: p.name,
   732→          message: `حریف ${p.name} قطع شد — امتیاز شما حفظ شد`,
   733→        });
   734→      }
   735→    }
   736→    armGrace(room);
   737→  });
   738→
   739→  socket.on("error", (err: unknown) => {
   740→    console.error(`[duel] socket error (${socket.id}):`, err);
   741→  });
   742→});

   744→// ─── پاک‌سازی دوره‌ای اتاق‌های بی‌صاحب/کهنه ───
   745→
   746→setInterval(() => {
   747→  const t = now();
   748→  for (const room of [...rooms.values()]) {
   749→    if (room.players.size === 0 || allBots(room)) {
   750→      if (t - room.createdAt > GRACE_MS) deleteRoom(room.code);
   751→      continue;
   752→    }
   753→    const allDisconnected = [...room.players.values()].every((p) => !p.connected);
   754→    if (allDisconnected && t - Math.max(...[...room.players.values()].map((p) => p.disconnectedAt ?? t)) > GRACE_MS + 5_000) {
   755→      deleteRoom(room.code);
   756→      continue;
   757→    }
   758→    const ttl = room.state === "ended" ? CODE_TTL_ENDED_MS : CODE_TTL_LOBBY_MS;
   759→    if (room.state !== "playing" && t - room.lastActivity > ttl) {
   760→      emitToAll(room, "room-closed", { message: "اتاق به‌دلیل بی‌فعالیتی بسته شد" });
   761→      deleteRoom(room.code);
   762→    }
   763→  }
