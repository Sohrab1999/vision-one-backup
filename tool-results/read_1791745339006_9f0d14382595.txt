   303→/** مسیر مستطیل گرد (سازگار با همهٔ مرورگرها) */
   304→function rrPath(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
   305→  c.beginPath();
   306→  c.moveTo(x + r, y);
   307→  c.arcTo(x + w, y, x + w, y + h, r);
   308→  c.arcTo(x + w, y + h, x, y + h, r);
   309→  c.arcTo(x, y + h, x, y, r);
   310→  c.arcTo(x, y, x + w, y, r);
   311→  c.closePath();
   312→}

   314→/** بافت بیلبورد گزینه — قرص سفید با متن انگلیسی تیره (رویه‌ای، بدون فونت خارجی) */
   315→function makeLabelTexture(text: string): THREE.CanvasTexture {
   316→  const W = 640;
   317→  const Hc = 160;
   318→  const cv = document.createElement("canvas");
   319→  cv.width = W;
   320→  cv.height = Hc;
   321→  const c = cv.getContext("2d");
   322→  if (c) {
   323→    let fs = 64;
   324→    c.font = `800 ${fs}px system-ui, sans-serif`;
   325→    while (c.measureText(text).width > W - 96 && fs > 24) {
   326→      fs -= 3;
   327→      c.font = `800 ${fs}px system-ui, sans-serif`;
   328→    }
   329→    const tw = Math.min(c.measureText(text).width, W - 96);
   330→    const pw = Math.min(tw + 56, W - 24);
   331→    const ph = 102;
   332→    const px = (W - pw) / 2;
   333→    const py = (Hc - ph) / 2;
   334→    c.fillStyle = "rgba(255,255,255,0.95)";
   335→    rrPath(c, px, py, pw, ph, 34);
   336→    c.fill();
   337→    c.lineWidth = 6;
   338→    c.strokeStyle = "rgba(6,95,70,0.35)";
   339→    rrPath(c, px + 3, py + 3, pw - 6, ph - 6, 30);
   340→    c.stroke();
   341→    c.fillStyle = "#065f46";
   342→    c.textAlign = "center";
   343→    c.textBaseline = "middle";
   344→    c.fillText(text, W / 2, Hc / 2 + 4, W - 90);
   345→  }
   346→  const tex = new THREE.CanvasTexture(cv);
   347→  tex.colorSpace = THREE.SRGBColorSpace;
   348→  tex.anisotropy = 2;
   349→  return tex;
   350→}

   549→/** یک هدف متحرک: اوکتاهدر درخشان + بیلبورد برچسب انگلیسی */
   550→function Target({
   551→  spec,
   552→  registry,
   553→  bridge,
   554→}: {
   555→  spec: TargetSpec;
   556→  registry: React.RefObject<THREE.Mesh[]>;
   557→  bridge: React.RefObject<SceneBridge>;
   558→}) {
   559→  const groupRef = useRef<THREE.Group>(null);
   560→  const meshRef = useRef<THREE.Mesh>(null);
   561→  const tex = useMemo(() => makeLabelTexture(spec.text), [spec.text]);
   562→  const t0 = useMemo(() => Math.random() * 6, []);
   563→  useEffect(() => () => tex.dispose(), [tex]);
   564→
   565→  // ثبت/حذف مش برای raycast شلیک
   566→  useEffect(() => {
   567→    const m = meshRef.current;
   568→    const list = registry.current;
   569→    if (!m || !list) return;
   570→    m.userData.id = spec.id;
   571→    list.push(m);
   572→    return () => {
   573→      const i = list.indexOf(m);
   574→      if (i >= 0) list.splice(i, 1);
   575→    };
   576→  }, [registry, spec.id]);
   577→
   578→  useFrame((state) => {
   579→    const g = groupRef.current;
   580→    const m = meshRef.current;
   581→    if (!g || !m) return;
   582→    const t = state.clock.elapsedTime;
   583→    const b = bridge.current;
   584→    g.position.set(
   585→      spec.lane * b.laneK + Math.sin(t * 0.5 + t0) * spec.driftAmp * b.laneK,
   586→      spec.baseY + Math.sin(t * spec.bobSpeed + t0) * 0.4,
   587→      spec.baseZ,
   588→    );
   589→    m.rotation.y += 0.008 * spec.spin * 60 * 0.016;
   590→    m.rotation.x = Math.sin(t * 0.4 + t0) * 0.25;
   591→    // فلاش سبز/قرمز پس از پاسخ + تنفس ملایم
   592→    const mat = m.material as THREE.MeshStandardMaterial;
   593→    const now = t * 1000;
   594→    const f = b.flashes.find((fl) => fl.id === spec.id && now < fl.until);
   595→    if (f) {
   596→      mat.emissive.setHex(f.kind === "ok" ? 0x10b981 : 0xef4444);
   597→      mat.emissiveIntensity = 1.15;
   598→    } else {
   599→      mat.emissive.setHex(spec.colorHex);
   600→      mat.emissiveIntensity = 0.24 + Math.sin(now / 420 + t0) * 0.09;
   601→    }
   602→  });
   603→
   604→  return (
   605→    <group ref={groupRef} position={[spec.lane, spec.baseY, spec.baseZ]}>
   606→      <mesh ref={meshRef}>
   607→        <octahedronGeometry args={[TARGET_R, 0]} />
   608→        <meshStandardMaterial
   609→          color={spec.colorHex}
   610→          flatShading
   611→          roughness={0.35}
   612→          metalness={0.15}
   613→          emissive={spec.colorHex}
   614→          emissiveIntensity={0.25}
   615→        />
   616→      </mesh>
   617→      <sprite position={[0, 1.55, 0]} scale={[2.9, 0.73, 1]}>
   618→        <spriteMaterial map={tex} transparent depthWrite={false} />
   619→      </sprite>
   620→    </group>
   621→  );
   622→}

   624→/** استخر انفجار رنگین‌کمانی — InstancedMesh با رنگ هر ذره */
   625→function Burst({ apiRef, count }: { apiRef: React.RefObject<BurstApi | null>; count: number }) {
   626→  const meshRef = useRef<THREE.InstancedMesh>(null);
   627→  const parts = useMemo(() => {
   628→    const dirs = new Float32Array(count * 3);
   629→    const speeds = new Float32Array(count);
   630→    for (let i = 0; i < count; i++) {
   631→      const u = Math.random() * Math.PI * 2;
   632→      const v = Math.acos(2 * Math.random() - 1);
   633→      dirs[i * 3] = Math.sin(v) * Math.cos(u);
   634→      dirs[i * 3 + 1] = Math.cos(v);
   635→      dirs[i * 3 + 2] = Math.sin(v) * Math.sin(u);
   636→      speeds[i] = 1.7 + Math.random() * 2.8;
   637→    }
   638→    return { dirs, speeds };
   639→  }, [count]);
   640→  const st = useRef({ active: false, t0: 0, origin: new THREE.Vector3() });
   641→  const dummy = useMemo(() => new THREE.Object3D(), []);
   642→
   643→  useEffect(() => {
   644→    const mesh = meshRef.current;
   645→    if (!mesh) return;
   646→    // رنگ هر ذره — پالت رنگین‌کمان
   647→    const c = new THREE.Color();
   648→    const rainbow = [0xf43f5e, 0xf97316, 0xfacc15, 0x22c55e, 0x14b8a6, 0x38bdf8, 0xa78bfa, 0xf472b6];
   649→    for (let i = 0; i < count; i++) {
   650→      c.setHex(rainbow[i % rainbow.length]);
   651→      mesh.setColorAt(i, c);
   652→    }
   653→    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
   654→    // پنهان‌سازی اولیه
   655→    const m0 = new THREE.Matrix4().makeScale(0, 0, 0);
   656→    for (let i = 0; i < count; i++) mesh.setMatrixAt(i, m0);
   657→    mesh.instanceMatrix.needsUpdate = true;
   658→  }, [count]);
   659→
   660→  useEffect(() => {
   661→    apiRef.current = {
   662→      fire: (origin) => {
   663→        st.current.active = true;
   664→        st.current.t0 = performance.now();
   665→        st.current.origin.copy(origin);
   666→      },
   667→    };
   668→    return () => {
   669→      apiRef.current = null;
   670→    };
   671→  }, [apiRef]);
   672→
   673→  useFrame(() => {
   674→    const mesh = meshRef.current;
   675→    if (!mesh || !st.current.active) return;
   676→    const t = (performance.now() - st.current.t0) / 950;
   677→    if (t >= 1) {
   678→      st.current.active = false;
   679→      const m0 = new THREE.Matrix4().makeScale(0, 0, 0);
   680→      for (let i = 0; i < count; i++) mesh.setMatrixAt(i, m0);
   681→      mesh.instanceMatrix.needsUpdate = true;
   682→      return;
   683→    }
   684→    const e = easeOutCubic(t);
   685→    for (let i = 0; i < count; i++) {
   686→      const d = e * parts.speeds[i] * 2.1;
   687→      dummy.position.set(
   688→        st.current.origin.x + parts.dirs[i * 3] * d,
   689→        st.current.origin.y + parts.dirs[i * 3 + 1] * d - t * t * 1.5,
   690→        st.current.origin.z + parts.dirs[i * 3 + 2] * d,
   691→      );
   692→      const s = (1 - t) * 0.3;
   693→      dummy.scale.setScalar(Math.max(0.001, s));
   694→      dummy.rotation.set(t * 5 + i, t * 4, i * 0.7);
   695→      dummy.updateMatrix();
   696→      mesh.setMatrixAt(i, dummy.matrix);
   697→    }
   698→    mesh.instanceMatrix.needsUpdate = true;
   699→  });
   700→
   701→  return (
   702→    <instancedMesh ref={meshRef} args={[undefined, undefined, count]} frustumCulled={false}>
   703→      <tetrahedronGeometry args={[1, 0]} />
   704→      <meshBasicMaterial toneMapped={false} />
   705→    </instancedMesh>
   706→  );
   707→}

   710→function World({

   738→  });
   739→  return null;
