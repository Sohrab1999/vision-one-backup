    76→    76→/** فقط واژه‌های تک‌کلمه‌ای لاتینِ ۳ تا ۱۲ حرفی وارد بازی می‌شوند */
    77→    77→const WORD_RE = /^[A-Za-z]{3,12}$/;
    78→    78→
    79→    79→/** واژهٔ پایه — حذف نشانه‌های دستوری داخل پرانتز: "dry (adj.)" → "dry" */
    80→    80→function baseWord(w: string): string {
    81→    81→  return w.replace(/\s*\([^)]*\)\s*/g, " ").trim();

    83→    83→
    84→    84→export function ScrambleGame() {
    85→    85→  const data = useBookData();
    86→    86→  const { bookId: activeBookId } = useBook();
    87→    87→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا تضمینش می‌کند) — در عمل هرگز تهی نیست
    88→    88→  const bookId: string = activeBookId ?? "";
    89→    89→  const { speed } = useVoiceSpeed();
    90→    90→
    91→    91→  const [phase, setPhase] = useState<Phase>("setup");
    92→    92→  const [lesson, setLesson] = useState<number>(ALL_LESSONS);
    93→    93→  const [rounds, setRounds] = useState<number>(8);
    94→    94→
    95→    95→  const [deck, setDeck] = useState<VocabEntry[]>([]);
    96→    96→  const [idx, setIdx] = useState(0);
    97→    97→  const [tiles, setTiles] = useState<Tile[]>([]);
    98→    98→  const [slots, setSlots] = useState<(number | null)[]>([]);
    99→    99→
   100→   100→  const [solved, setSolved] = useState(0);
   101→   101→  const [skipped, setSkipped] = useState(0);
   102→   102→  const [hinted, setHinted] = useState(0);
   103→   103→  const [detail, setDetail] = useState<DetailRow[]>([]);
   104→   104→  const [secs, setSecs] = useState(0);
   105→   105→  const [isRecord, setIsRecord] = useState(false);
   106→   106→
   107→   107→  /** انیمیشن match-pop روی ردیف پاسخ هنگام حل درست */
   108→   108→  const [pop, setPop] = useState(false);
   109→   109→  /** واژهٔ درستی که بعد از رد کردن برای ۱ ثانیه نشان داده می‌شود */
   110→   110→  const [reveal, setReveal] = useState<string | null>(null);
   111→   111→
   112→   112→  const locked = useRef(false);
   113→   113→  const timer = useRef<number | undefined>(undefined);
   114→   114→  const secsRef = useRef(0);
   115→   115→
   116→   116→  const busy = pop || reveal !== null;
   117→   117→
   118→   118→  // استخر قابل بازی — فقط واژه‌های تک‌کلمه‌ای ۳ تا ۱۲ حرفی
   119→   119→  const pool = useMemo(() => {
   120→   120→    const base = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   121→   121→    return base.filter((w) => WORD_RE.test(baseWord(w.word)));
   122→   122→  }, [data.vocab, lesson]);
   123→   123→
   124→   124→  // رکورد شخصی — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   125→   125→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   126→   126→  const best = useSyncExternalStore(
   127→   127→    subscribeGameData,
   128→   128→    () => {
   129→   129→      const b = loadBest<BestRec>(bookId, "scramble", scope);
   130→   130→      return b ? b.time : null;
   131→   131→    },
   132→   132→    () => null,
   133→   133→  );
   134→   134→
   135→   135→  // ── تایمر (functional) + شمارندهٔ دقیق برای رکورد ──
   136→   136→  useEffect(() => {
   137→   137→    if (phase !== "playing") return;
   138→   138→    const t = window.setInterval(() => {
   139→   139→      secsRef.current += 1;
   140→   140→      setSecs((s) => s + 1);
   141→   141→    }, 1000);
   142→   142→    return () => window.clearInterval(t);
   143→   143→  }, [phase]);
   144→   144→
   145→   145→  // ── پاک‌سازی نهایی: تلفظ + تایمرهای معلق ──
   146→   146→  useEffect(() => {
   147→   147→    return () => {
   148→   148→      stopSpeaking();
   149→   149→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   150→   150→    };
   151→   151→  }, []);
   152→   152→
   153→   153→  /** بارگذاری واژهٔ بعدی — حروف به‌هم‌ریخته + خانه‌های خالی */
   154→   154→  const loadWord = useCallback((entry: VocabEntry) => {
   155→   155→    const base = baseWord(entry.word);
   156→   156→    setTiles(shuffle(base.split("").map((letter, i) => ({ id: i, letter }))));
   157→   157→    setSlots(Array.from({ length: base.length }, () => null as number | null));
   158→   158→    setPop(false);
   159→   159→  }, []);
   160→   160→
   161→   161→  /** پایان دور — استریک/جلسه/Attempt/رکورد */
   162→   162→  const finishRun = useCallback(
   163→   163→    (finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   164→   164→      touchStreak();
   165→   165→      recordStudySession();
   166→   166→      bumpPlays(bookId, "scramble");
   167→   167→      saveAttempt(bookId, "scramble", {
   168→   168→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   169→   169→        total: deck.length,
   170→   170→        correct: finalSolved,
   171→   171→        duration: secsRef.current,
   172→   172→        detail: finalDetail,
   173→   173→      });
   174→   174→      // رکورد فقط برای اجرای کاملِ بدون رد کردن و بدون راهنما
   175→   175→      if (finalSkipped === 0 && hinted === 0) {
   176→   176→        const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   177→   177→        const prev = loadBest<BestRec>(bookId, "scramble", recScope);
   178→   178→        if (!prev || secsRef.current < prev.time) {
   179→   179→          saveBest<BestRec>(bookId, "scramble", recScope, { time: secsRef.current });
   180→   180→          setIsRecord(true);
   181→   181→        }
   182→   182→      }
   183→   183→      setPhase("done");
   184→   184→    },
   185→   185→    [bookId, lesson, rounds, deck.length, hinted],
   186→   186→  );
   187→   187→
   188→   188→  /** ادامه به واژهٔ بعد یا پایان دور */
   189→   189→  const advanceOrFinish = useCallback(
   190→   190→    (nextIdx: number, finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   191→   191→      if (nextIdx >= deck.length) {
   192→   192→        finishRun(finalSolved, finalSkipped, finalDetail);
   193→   193→        return;
   194→   194→      }
   195→   195→      setIdx(nextIdx);
   196→   196→      loadWord(deck[nextIdx]);
   197→   197→    },
   198→   198→    [deck, loadWord, finishRun],
   199→   199→  );
   200→   200→
   201→   201→  /** حل درست — پاپ + تلفظ + SRS(✓) و بعد واژهٔ بعد */
   202→   202→  const solveWord = useCallback(
   203→   203→    (entry: VocabEntry, finalSolved: number, finalDetail: DetailRow[]) => {
   204→   204→      locked.current = true;
   205→   205→      setPop(true);
   206→   206→      const base = baseWord(entry.word);
   207→   207→      recordSRS(base, true);
   208→   208→      speak(base, { speed });
   209→   209→      timer.current = window.setTimeout(() => {
   210→   210→        locked.current = false;
   211→   211→        setPop(false);
   212→   212→        advanceOrFinish(idx + 1, finalSolved, skipped, finalDetail);
   213→   213→      }, 650);
   214→   214→    },
   215→   215→    [speed, idx, skipped, advanceOrFinish],
   216→   216→  );
   217→   217→
   218→   218→  /** گذاشتن یک حرف از استخر در اولین خانهٔ خالی + بررسی حل شدن */
   219→   219→  const placeTile = useCallback(
   220→   220→    (tileId: number) => {
   221→   221→      if (locked.current) return;
   222→   222→      const current = deck[idx];
   223→   223→      if (!current) return;
   224→   224→      const empty = slots.indexOf(null);
   225→   225→      if (empty === -1) return;
   226→   226→      const tile = tiles.find((t) => t.id === tileId);
   227→   227→      if (!tile) return;
   228→   228→      const nextSlots = [...slots];
   229→   229→      nextSlots[empty] = tileId;
   230→   230→      setSlots(nextSlots);
   231→   231→      // همهٔ خانه‌ها پر شد؟ تطبیق با واژهٔ پایه را بسنجیم
   232→   232→      if (nextSlots.every((s) => s !== null)) {
   233→   233→        const base = baseWord(current.word);
   234→   234→        const attempt = nextSlots
   235→   235→          .map((s) => tiles.find((t) => t.id === s)?.letter ?? "")
   236→   236→          .join("");
   237→   237→        if (attempt === base) {
   238→   238→          const finalSolved = solved + 1;
   239→   239→          const nextDetail = [...detail, { word: base, ok: true }];
   240→   240→          setSolved(finalSolved);
   241→   241→          setDetail(nextDetail);
   242→   242→          solveWord(current, finalSolved, nextDetail);
   243→   243→        }
   244→   244→      }
   245→   245→    },
   246→   246→    [deck, idx, slots, tiles, solved, detail, solveWord],
   247→   247→  );
   248→   248→
   249→   249→  /** برگرداندن حرفِ یک خانه به استخر (کلیک روی خانهٔ پر) */
   250→   250→  const returnAt = useCallback((index: number) => {
   251→   251→    if (locked.current) return;
   252→   252→    setSlots((prev) => {
   253→   253→      if (prev[index] === null) return prev;
   254→   254→      const next = [...prev];
   255→   255→      next[index] = null;
   256→   256→      return next;
   257→   257→    });
   258→   258→  }, []);
   259→   259→
   260→   260→  /** پاک کردن آخرین حرف (دکمه یا Backspace) */
   261→   261→  const returnLast = useCallback(() => {
   262→   262→    if (locked.current) return;
   263→   263→    setSlots((prev) => {
   264→   264→      for (let i = prev.length - 1; i >= 0; i--) {
   265→   265→        if (prev[i] !== null) {
   266→   266→          const next = [...prev];
   267→   267→          next[i] = null;
   268→   268→          return next;
   269→   269→        }
   270→   270→      }
   271→   271→      return prev;
   272→   272→    });
   273→   273→  }, []);
   274→   274→
   275→   275→  /** راهنما — یک حرف درستِ بعدی را جای‌گذاری می‌کند (رکورد دور می‌سوزد) */
   276→   276→  const applyHint = useCallback(() => {
   277→   277→    if (locked.current) return;
   278→   278→    const current = deck[idx];
   279→   279→    if (!current) return;
   280→   280→    const empty = slots.indexOf(null);
   281→   281→    if (empty === -1) return;
   282→   282→    const base = baseWord(current.word);
   283→   283→    const used = new Set(slots.filter((s): s is number => s !== null));
   284→   284→    const tile = tiles.find((t) => !used.has(t.id) && t.letter === base[empty]);
   285→   285→    if (!tile) return;
   286→   286→    setHinted((h) => h + 1);
   287→   287→    placeTile(tile.id);
   288→   288→  }, [deck, idx, slots, tiles, placeTile]);
   289→   289→
   290→   290→  /** رد کردن — SRS(✗) + نمایش واژهٔ درست برای ۱ ثانیه (رکورد دور می‌سوزد) */
   291→   291→  const skipWord = useCallback(() => {
   292→   292→    if (locked.current) return;
   293→   293→    const current = deck[idx];
   294→   294→    if (!current) return;
   295→   295→    locked.current = true;
   296→   296→    const base = baseWord(current.word);
   297→   297→    recordSRS(base, false);
   298→   298→    const nextSkipped = skipped + 1;
   299→   299→    const nextDetail = [...detail, { word: base, ok: false }];
   300→   300→    setSkipped(nextSkipped);
   301→   301→    setDetail(nextDetail);
   302→   302→    setReveal(base);
   303→   303→    timer.current = window.setTimeout(() => {
   304→   304→      locked.current = false;
   305→   305→      setReveal(null);
   306→   306→      advanceOrFinish(idx + 1, solved, nextSkipped, nextDetail);
   307→   307→    }, 1000);
   308→   308→  }, [deck, idx, skipped, detail, solved, advanceOrFinish]);
   309→   309→
   310→   310→  /** انصراف — تایمرهای معلق پاک و به تنظیمات برگردیم */
   311→   311→  const cancel = useCallback(() => {
   312→   312→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   313→   313→    locked.current = false;
   314→   314→    stopSpeaking();
   315→   315→    setPhase("setup");
   316→   316→  }, []);
   317→   317→
   318→   318→  /** شروع دور — انتخاب واژه‌های یکتا از استخر قابل بازی */
   319→   319→  const start = useCallback(() => {
   320→   320→    const chosen = pickUnique(pool, rounds, (w) => baseWord(w.word));
   321→   321→    if (chosen.length < 3) {
   322→   322→      toast.error("واژه قابل بازی در این محدوده پیدا نشد");
   323→   323→      return;
   324→   324→    }
   325→   325→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   326→   326→    locked.current = false;
   327→   327→    secsRef.current = 0;
   328→   328→    setDeck(chosen);
   329→   329→    setIdx(0);
   330→   330→    setSolved(0);
   331→   331→    setSkipped(0);
   332→   332→    setHinted(0);
   333→   333→    setDetail([]);
   334→   334→    setSecs(0);
   335→   335→    setIsRecord(false);
   336→   336→    setReveal(null);
   337→   337→    loadWord(chosen[0]);
   338→   338→    setPhase("playing");
   339→   339→  }, [pool, rounds, loadWord]);
   340→   340→
   341→   341→  // ── کیبورد: فقط در فاز بازی؛ حروف → جای‌گذاری، Backspace → برگرداندن ──
   342→   342→  useEffect(() => {
   343→   343→    if (phase !== "playing") return;
   344→   344→    const onKey = (e: KeyboardEvent) => {
   345→   345→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   346→   346→      if (e.key === "Backspace") {
   347→   347→        e.preventDefault();
   348→   348→        returnLast();
   349→   349→        return;
   350→   350→      }
   351→   351→      if (e.key === "Enter") return; // حل خودکار است — اینتر کاری نمی‌کند
   352→   352→      if (/^[a-zA-Z]$/.test(e.key)) {
   353→   353→        const letter = e.key.toLowerCase();
   354→   354→        const used = new Set(slots.filter((s): s is number => s !== null));
   355→   355→        const tile = tiles.find((t) => !used.has(t.id) && t.letter.toLowerCase() === letter);
   356→   356→        if (tile) placeTile(tile.id);
   357→   357→      }
   358→   358→    };
   359→   359→    window.addEventListener("keydown", onKey);
   360→   360→    return () => window.removeEventListener("keydown", onKey);
   361→   361→  }, [phase, slots, tiles, placeTile, returnLast]);
   362→   362→
   363→   363→  const current = deck[idx];
   364→   364→  const usedIds = new Set(slots.filter((s): s is number => s !== null));
   365→   365→  const letterOf = (id: number): string => tiles.find((t) => t.id === id)?.letter ?? "";
   366→   366→  const pct = Math.round(((solved + skipped) / Math.max(deck.length, 1)) * 100);
   367→   367→
   368→   368→  // ── Setup ──
   369→   369→  if (phase === "setup") {
   370→   370→    return (
   371→   371→      <Card className="mx-auto max-w-xl rise-in">
   372→   372→        <CardContent className="space-y-6 p-6">
   373→   373→          <div className="flex items-center gap-3">
   374→   374→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-amber-500/15 text-amber-600 dark:text-amber-400">
   375→   375→              <Shuffle size={22} />
   376→   376→            </span>
   377→   377→            <div>
   378→   378→              <h2 className="text-lg font-black">حروف جابه‌جا</h2>
   379→   379→              <p className="text-xs text-muted-foreground">
   380→   380→                حروف واژه‌ها ریخته شده — با معنی فارسی سرهمشان کن!
   381→   381→              </p>
   382→   382→            </div>
   383→   383→          </div>
   384→   384→
   385→   385→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   386→   386→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   387→   387→            <p>
   388→   388→              روی حروف به ترتیب بزن تا واژه ساخته شود؛ کیبورد هم کار می‌کند. واژه‌های سخت را می‌توانی رد کنی ولی رکورد می‌سوزد!
   389→   389→            </p>
   390→   390→          </div>
   391→   391→
   392→   392→          <div>
   393→   393→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   394→   394→            <div className="flex flex-wrap gap-2">
   395→   395→              <button
   396→   396→                type="button"
   397→   397→                onClick={() => setLesson(ALL_LESSONS)}
   398→   398→                className={cn(
   399→   399→                  "rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   400→   400→                  lesson === ALL_LESSONS
