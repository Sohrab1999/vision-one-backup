     3→/**
     4→ * بازی «دوئل دونفره» (Same-device Duel) — فاز ۴ نقشه راه ویژن‌وان.
     5→ *
     6→ * یک صفحه، دو بازیکن، همان سؤال‌ها — رو در رو:
     7→ * - Setup: نام دو بازیکن + تعداد دور (۵/۷/۹) + دسته + محزودهٔ درس + Badge رکورد
     8→ *   «بیشترین برد دور بازیکن ۱» (اسکوپ درس|all:تعداد دور).
     9→ * - نیمهٔ بالا = بازیکن ۱ (emerald)، نیمهٔ پایین = بازیکن ۲ (amber)؛ وسط، نوار سؤال
    10→ *   مشترک با Badge شمارهٔ دور و تایمر ۱۰ ثانیه‌ای هر دور (نوار width).
    11→ * - هر نیمه ۴ دکمهٔ گزینه دارد؛ ترتیب گزینه‌ها بین دو نیمه shuffle متفاوت است تا
    12→ *   رقیب نتواند از جای انگشت دیگری تقلب کند.
    13→ * - هر بازیکن یک‌بار پاسخ می‌دهد؛ دور وقتی داوری می‌شود که هر دو پاسخ بدهند یا وقت
    14→ *   تمام شود (بی‌پاسخ = غلط). هر دو درست → سریع‌تر برنده؛ یکی درست → همان؛ هیچ‌کدام
    15→ *   → تساوی بدون امتیاز. نتیجهٔ دور ~۱.۲ ثانیه نمایش داده می‌شود و دور بعد می‌آید.
    16→ * - واژه‌های غلط‌زدهٔ واژگانی (فقط وقتی بازیکن واقعاً گزینه‌ای زده باشد) با
    17→ *   seedReviewFromQuiz — متن گزینهٔ درست = خود واژهٔ انگلیسی — به صف مرور برمی‌گردند.
    18→ * - پایان دورها: صفحهٔ برنده + جدول امتیاز (برد دور / پاسخ درست / دقت) + «دوئل جدید» و «تنظیمات».
    19→ * - پایان: touchStreak + recordStudySession + bumpPlays + saveAttempt (total = تعداد دور،
    20→ *   correct = دورهای بردهٔ بازیکن ۱، detail {p1,p2,p1Name,p2Name,winner}) + saveBest
    21→ *   {score} = بیشترین برد دور بازیکن ۱ با اسکوپ `${درس|all}:${تعداد دور}`.
    22→ *
    23→ * نکتهٔ تایمر: interval هر دور مالک شمارش است؛ مقدار واقعی در timeRef (با شروع هر دور
    24→ * ریست می‌شود) و setTimeLeft فقط نمایش را همگام نگه می‌دارد. داوریِ هر دور دقیقاً
    25→ * یک‌بار با judgedRef قفل می‌شود؛ زنجیرهٔ «نمایش نتیجه → دور بعد» در timersRef است و
    26→ * در انصراف/unmount کامل پاک می‌شود (الگوی react-hooks v6: صفر setState در بدنهٔ effect).
    27→ */

    78→type RoundWinner = 0 | 1 | -1;

   104→export function DuelGame() {
   105→  const data = useBookData();
   106→  const { bookId } = useBook();
   107→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا گارد است) — bookId عملاً هرگز null نیست
   108→  const activeBookId: string = bookId ?? "";
   109→
   110→  const [phase, setPhase] = useState<Phase>("setup");
   111→  const [p1Name, setP1Name] = useState("بازیکن ۱");
   112→  const [p2Name, setP2Name] = useState("بازیکن ۲");
   113→  const [rounds, setRounds] = useState<number>(7);
   114→  const [lesson, setLesson] = useState<number>(ALL_LESSON);
   115→  const [cat, setCat] = useState<CatKey>("all");
   116→
   117→  // ── stateهای حین بازی ──
   118→  const [q, setQ] = useState<QuizQuestionSeed | null>(null);
   119→  const [roundNum, setRoundNum] = useState(0);
   120→  const [timeLeft, setTimeLeft] = useState(ROUND_TIME);
   121→  const [p1Order, setP1Order] = useState<number[]>(BASE_ORDER);
   122→  const [p2Order, setP2Order] = useState<number[]>(BASE_ORDER);
   123→  /** گزینهٔ انتخابی هر بازیکن بر اساس اندیس اصلی q.options (null = هنوز پاسخ نداده) */
   124→  const [p1Pick, setP1Pick] = useState<number | null>(null);
   125→  const [p2Pick, setP2Pick] = useState<number | null>(null);
   126→  /** نتیجهٔ داوری دور جاری — null یعنی هنوز در حال بازی */
   127→  const [result, setResult] = useState<RoundOutcome | null>(null);
   128→  const [wins1, setWins1] = useState(0);
   129→  const [wins2, setWins2] = useState(0);
   130→  /** پاسخ‌های درست هر بازیکن (برای دقت — با تعداد بردها یکی نیست) */
   131→  const [correct1, setCorrect1] = useState(0);
   132→  const [correct2, setCorrect2] = useState(0);
   133→
   134→  // ── stateهای فاز done ──
   135→  const [winner, setWinner] = useState<RoundWinner>(-1);
   136→  const [isRecord, setIsRecord] = useState(false);
   137→
   138→  // ── refs — منبع حقیقت منطق بازی (همهٔ callbackهای تایمر از این‌ها می‌خوانند) ──
   139→  const qRef = useRef<QuizQuestionSeed | null>(null);
   140→  const deckRef = useRef<QuizQuestionSeed[]>([]);
   141→  const qiRef = useRef(0);
   142→  /** شمارهٔ دور نمایشی (۱-مبنا) */
   143→  const roundNoRef = useRef(0);
   144→  /** تایمر دور — مالک واقعی زمان (با شروع هر دور ریست می‌شود) */
   145→  const timeRef = useRef(ROUND_TIME);
   146→  /** شروع دور برای محاسبهٔ سرعت پاسخ هر بازیکن */
   147→  const roundStartRef = useRef(0);
   148→  const p1PickRef = useRef<number | null>(null);
   149→  const p2PickRef = useRef<number | null>(null);
   150→  /** زمان پاسخ (میلی‌ثانیه از شروع دور) — ملاک برنده در «هر دو درست» */
   151→  const p1MsRef = useRef(0);
   152→  const p2MsRef = useRef(0);
   153→  const p1OrderRef = useRef<number[]>(BASE_ORDER);
   154→  const p2OrderRef = useRef<number[]>(BASE_ORDER);
   155→  /** قفل داوری — هر دور دقیقاً یک‌بار داوری می‌شود */
   156→  const judgedRef = useRef(false);
   157→  /** مسابقه تمام/لغو شده؟ */
   158→  const overRef = useRef(false);
   159→  const wins1Ref = useRef(0);
   160→  const wins2Ref = useRef(0);
   161→  const corr1Ref = useRef(0);
   162→  const corr2Ref = useRef(0);
   163→  const configRef = useRef<DuelConfig>({ lesson: ALL_LESSON, rounds: 7, p1: "بازیکن ۱", p2: "بازیکن ۲" });
   164→  /** لحظهٔ شروع مسابقه برای محاسبهٔ duration */
   165→  const startAtRef = useRef(0);
   166→  /** زنجیرهٔ همهٔ setTimeoutها (نمایش نتیجه → دور بعد) — در انصراف/unmount پاک می‌شود */
   167→  const timersRef = useRef<number[]>([]);
   168→  /** آخرین نسخهٔ judge برای فراخوانی تازه از تایمر (الگوی latest-ref) */
   169→  const judgeRef = useRef<(() => void) | null>(null);
   170→
   171→  // ── داده‌های مشتق ──
   172→  const pool = useMemo(
   173→    () =>
   174→      data.quiz.seed.filter(
   175→        (item) =>
   176→          (lesson === ALL_LESSON || item.lesson === lesson) &&
   177→          (cat === "all" || item.category === cat),
   178→      ),
   179→    [data.quiz.seed, lesson, cat],
   180→  );
   181→
   182→  const lessons = useMemo(
   183→    () => [...new Set(data.quiz.seed.map((item) => item.lesson))].sort((a, b) => a - b),
   184→    [data.quiz.seed],
   185→  );
   186→
   187→  /** تعداد واقعی سؤال هر چیپ محدوده (با اعمال دستهٔ فعال) */
   188→  const countFor = (l: number) =>
   189→    data.quiz.seed.filter(
   190→      (item) => (l === ALL_LESSON || item.lesson === l) && (cat === "all" || item.category === cat),
   191→    ).length;
   192→
   193→  // رکورد «بیشترین برد دور بازیکن ۱» این اسکوپ — اسنپ‌شات عددی پایدار (الگوی match-game)
   194→  const bestScope = `${lesson === ALL_LESSON ? "all" : lesson}:${rounds}`;
   195→  const best = useSyncExternalStore(
   196→    subscribeGameData,
   197→    () => loadBest<BestRecord>(activeBookId, GAME_KEY, bestScope)?.score ?? null,
   198→    () => null,
   199→  );
   200→
   201→  // نام‌های نهایی (پیش‌فرض اگر خالی باشند) — محاسبهٔ خالص در رندر
   202→  const name1 = p1Name.trim() || "بازیکن ۱";
   203→  const name2 = p2Name.trim() || "بازیکن ۲";
   204→
   205→  // ── مدیریت تایمرها ──
   206→
   207→  /** زمان‌بندی داخل زنجیرهٔ ref‌شده — بعد از اجرا خودش را از فهرست حذف می‌کند */
   208→  function later(fn: () => void, ms: number): void {
   209→    const t = window.setTimeout(() => {
   210→      timersRef.current = timersRef.current.filter((x) => x !== t);
   211→      fn();
   212→    }, ms);
   213→    timersRef.current.push(t);
   214→  }
   215→
   216→  function clearTimers(): void {
   217→    for (const t of timersRef.current) window.clearTimeout(t);
   218→    timersRef.current = [];
   219→  }
   220→
   221→  // ── جریان بازی ──
   222→
   223→  /** شروع یک دور: سؤال بعدی + دو ترتیب گزینهٔ متفاوت + ریست تایمر و قفل‌ها */
   224→  function startRound(idx: number): void {
   225→    let deck = deckRef.current;
   226→    let i = idx;
   227→    if (i >= deck.length) {
   228→      // استخر تمام شد → از نو شافل و ادامه (چرخهٔ بی‌پایان مثل speed-quiz)
   229→      deck = shuffle(deck);
   230→      deckRef.current = deck;
   231→      i = 0;
   232→    }
   233→    qiRef.current = i;
   234→    const nq = deck[i];
   235→    qRef.current = nq;
   236→    // ترتیب گزینه‌ها بین دو نیمه باید متفاوت باشد
   237→    const o1 = shuffle(BASE_ORDER);
   238→    let o2 = shuffle(BASE_ORDER);
   239→    let tries = 0;
   240→    while (o2.join(",") === o1.join(",") && tries < 8) {
   241→      o2 = shuffle(BASE_ORDER);
   242→      tries += 1;
   243→    }
   244→    p1OrderRef.current = o1;
   245→    p2OrderRef.current = o2;
   246→    p1PickRef.current = null;
   247→    p2PickRef.current = null;
   248→    p1MsRef.current = 0;
   249→    p2MsRef.current = 0;
   250→    judgedRef.current = false;
   251→    timeRef.current = ROUND_TIME;
   252→    roundStartRef.current = Date.now();
   253→    roundNoRef.current += 1;
   254→    setQ(nq);
   255→    setP1Order(o1);
   256→    setP2Order(o2);
   257→    setP1Pick(null);
   258→    setP2Pick(null);
   259→    setResult(null);
   260→    setTimeLeft(ROUND_TIME);
   261→    setRoundNum(roundNoRef.current);
   262→  }
   263→
   264→  /** داوری دور — دقیقاً یک‌بار (قفل judgedRef): هر دو درست → سریع‌تر، یکی درست → همان، هیچ‌کدام → تساوی */
   265→  function judge(): void {
   266→    if (judgedRef.current || !qRef.current) return;
   267→    judgedRef.current = true;
   268→    const nq = qRef.current;
   269→    const a1 = p1PickRef.current;
   270→    const a2 = p2PickRef.current;
   271→    const c1 = a1 !== null && a1 === nq.answer;
   272→    const c2 = a2 !== null && a2 === nq.answer;
   273→    let w: RoundWinner;
   274→    if (c1 && c2) w = p1MsRef.current <= p2MsRef.current ? 0 : 1;
   275→    else if (c1) w = 0;
   276→    else if (c2) w = 1;
   277→    else w = -1;
   278→    // واژه‌های غلط‌زدهٔ واژگانی به صف مرور برمی‌گردند (گزینهٔ درست = خود واژهٔ انگلیسی)
   279→    if (nq.category === "vocabulary") {
   280→      if (a1 !== null && !c1) seedReviewFromQuiz(nq.options[nq.answer]);
   281→      if (a2 !== null && !c2) seedReviewFromQuiz(nq.options[nq.answer]);
   282→    }
   283→    if (w === 0) wins1Ref.current += 1;
   284→    else if (w === 1) wins2Ref.current += 1;
   285→    if (c1) corr1Ref.current += 1;
   286→    if (c2) corr2Ref.current += 1;
   287→    setWins1(wins1Ref.current);
   288→    setWins2(wins2Ref.current);
   289→    setCorrect1(corr1Ref.current);
   290→    setCorrect2(corr2Ref.current);
   291→    setResult({ c1, c2, winner: w });
   292→    // نمایش نتیجه ~۱.۲ ثانیه → دور بعد یا پایان مسابقه
   293→    later(() => {
   294→      if (overRef.current) return;
   295→      if (qiRef.current + 1 >= configRef.current.rounds) finish();
   296→      else startRound(qiRef.current + 1);
   297→    }, RESULT_MS);
   298→  }
   299→
   300→  /** ثبت پاسخ یک بازیکن (displayIdx = جای گزینه در نیمهٔ او؛ به اندیس اصلی ترجمه می‌شود) */
   301→  function pick(player: 0 | 1, displayIdx: number): void {
   302→    if (overRef.current || judgedRef.current || !qRef.current) return;
   303→    const order = player === 0 ? p1OrderRef.current : p2OrderRef.current;
   304→    const orig = order[displayIdx];
   305→    if (orig === undefined) return;
   306→    if (player === 0) {
   307→      if (p1PickRef.current !== null) return;
   308→      p1PickRef.current = orig;
   309→      p1MsRef.current = Math.max(0, Date.now() - roundStartRef.current);
   310→      setP1Pick(orig);
   311→    } else {
   312→      if (p2PickRef.current !== null) return;
   313→      p2PickRef.current = orig;
   314→      p2MsRef.current = Math.max(0, Date.now() - roundStartRef.current);
   315→      setP2Pick(orig);
   316→    }
   317→    // هر دو پاسخ دادند؟ دور همان‌جا داوری می‌شود
   318→    if (p1PickRef.current !== null && p2PickRef.current !== null) judge();
   319→  }
   320→
   321→  /** پایان مسابقه: آمار، Attempt، رکورد و صفحهٔ نتیجه */
   322→  function finish(): void {
   323→    if (overRef.current) return;
   324→    overRef.current = true;
   325→    judgedRef.current = true;
   326→    clearTimers();
   327→    const cf = configRef.current;
   328→    const w1 = wins1Ref.current;
   329→    const w2 = wins2Ref.current;
   330→    const finalWinner: RoundWinner = w1 > w2 ? 0 : w2 > w1 ? 1 : -1;
   331→    const spent = Math.max(1, Math.round((Date.now() - startAtRef.current) / 1000));
   332→    touchStreak();
   333→    recordStudySession();
   334→    bumpPlays(activeBookId, GAME_KEY);
   335→    saveAttempt(activeBookId, GAME_KEY, {
   336→      lesson: cf.lesson === ALL_LESSON ? 0 : cf.lesson,
   337→      total: cf.rounds,
   338→      correct: w1,
   339→      duration: spent,
   340→      detail: { p1: w1, p2: w2, p1Name: cf.p1, p2Name: cf.p2, winner: finalWinner },
   341→    });
   342→    // رکورد: بیشترین برد دور بازیکن ۱ — اسکوپ درس|all:تعداد دور
   343→    const scope = `${cf.lesson === ALL_LESSON ? "all" : cf.lesson}:${cf.rounds}`;
   344→    const prev = loadBest<BestRecord>(activeBookId, GAME_KEY, scope);
   345→    const rec = w1 > 0 && (prev === null || w1 > prev.score);
   346→    if (rec) saveBest(activeBookId, GAME_KEY, scope, { score: w1 });
   347→    setIsRecord(rec);
   348→    setWinner(finalWinner);
   349→    setPhase("done");
   350→  }
   351→
   352→  /** شروع مسابقه — گارد استخر سؤال یکتا < تعداد دور */
   353→  function start(): void {
   354→    const uniq = pickUnique(pool, pool.length, (item) => item.text);
   355→    if (uniq.length < rounds) {
   356→      toast.error("سؤال یکتای کافی در این محدوده نیست — محدودهٔ دیگری را انتخاب کن");
   357→      return;
   358→    }
   359→    clearTimers();
   360→    deckRef.current = uniq;
   361→    qiRef.current = 0;
   362→    roundNoRef.current = 0;
   363→    overRef.current = false;
   364→    judgedRef.current = false;
   365→    wins1Ref.current = 0;
   366→    wins2Ref.current = 0;
   367→    corr1Ref.current = 0;
   368→    corr2Ref.current = 0;
   369→    configRef.current = { lesson, rounds, p1: name1, p2: name2 };
   370→    startAtRef.current = Date.now();
   371→    timeRef.current = ROUND_TIME;
   372→    setWins1(0);
   373→    setWins2(0);
   374→    setCorrect1(0);
   375→    setCorrect2(0);
   376→    setResult(null);
   377→    setWinner(-1);
   378→    setIsRecord(false);
   379→    setPhase("playing");
   380→    startRound(0);
   381→  }
   382→
   383→  function quit(): void {
   384→    overRef.current = true;
   385→    clearTimers();
   386→    setPhase("setup");
   387→  }
