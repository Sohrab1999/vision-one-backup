   167→   167→ * آسان ۳ گزینه (حذف گزینهٔ انحرافی اضافه — پاسخ هرگز حذف نمی‌شود) · عادی ۴ گزینه
   168→   168→ * (بدون هیچ دخل‌و‌تصرفی — همان سؤال خام بانک تست) · سخت ۵ گزینه (مزاحم از استخر
   169→   169→ * گزینه‌های سایر سؤال‌ها، هرگز برابر گزینه‌های موجود). اگر استخر اجازهٔ افزودن
   170→   170→ * ندهد، سؤال بدون تغییر می‌ماند (شکنندگی صفر).
   171→   171→ */
   172→   172→function shapeOptions(
   173→   173→  q: QuizQuestionSeed,
   174→   174→  level: DifficultyKey,
   175→   175→  trapPool: string[],
   176→   176→): QuizQuestionSeed {
   177→   177→  const target = optionsFor(level, 3, 5);
   178→   178→  const opts = [...q.options];
   179→   179→  const ans = opts[q.answer];
   180→   180→  if (!ans || opts.length === target) return q;
   181→   181→  if (opts.length > target) {
   182→   182→    // آسان — حذف گزینه‌های غلط اضافه
   183→   183→    const wrongIdx = opts.map((_, i) => i).filter((i) => i !== q.answer);
   184→   184→    const removed = new Set(shuffle(wrongIdx).slice(0, opts.length - target));
   185→   185→    const next = opts.filter((_, i) => !removed.has(i));
   186→   186→    return { ...q, options: next, answer: Math.max(0, next.indexOf(ans)) };
   187→   187→  }
   188→   188→  // سخت — افزودن گزینهٔ مزاحم از استخر مشترک (شافل عادی — همان الگوی قبلی traps)
   189→   189→  const taken = new Set(opts.map((o) => o.trim().toLowerCase()));
   190→   190→  const extra: string[] = [];
   191→   191→  for (const cand of trapPool) {
   192→   192→    if (extra.length >= target - opts.length) break;
   193→   193→    const key = cand.trim().toLowerCase();
   194→   194→    if (!key || taken.has(key)) continue;
   195→   195→    taken.add(key);
   196→   196→    extra.push(cand);
   197→   197→  }
   198→   198→  if (extra.length === 0) return q; // استخر اجازه نداد — بدون تغییر
   199→   199→  const next = shuffle([...opts, ...extra]);
   200→   200→  return { ...q, options: next, answer: Math.max(0, next.indexOf(ans)) };
   201→   201→}

   205→   205→ * (نمونه‌گیری وزن‌دار کیت) و بقیه با شافل عادی — هیچ سؤالی حذف نمی‌شود و چرخهٔ
   206→   206→ * بی‌پایان دست‌نخورده است؛ فقط ترتیب ورود هوشمندتر است. شروع سرد (SRS خالی) =
   207→   207→ * دقیقاً همان شافل قبلی. فیلتر بر اساس هویت شیء است تا تکراری‌های کلید‌دار از دست نروند.
   208→   208→ */
   209→   209→function smartOrderDeck<T>(pool: T[], keyOf: (item: T) => string): T[] {
   210→   210→  if (pool.length <= 1) return shuffle(pool);
   211→   211→  const n = Math.max(1, Math.min(pool.length - 1, Math.ceil(pool.length / 2)));
   212→   212→  const head = smartPick(pool, n, keyOf);
   213→   213→  const inHead = new Set<T>(head);
   214→   214→  const tail = shuffle(pool.filter((x) => !inHead.has(x)));
   215→   215→  return [...head, ...tail];
   216→   216→}

   289→   289→/** سؤال تولیدی API — شکل اکید قرارداد */
   290→   290→interface AiQuestion {
   291→   291→  word: string;
   292→   292→  sentence: string;
   293→   293→  options: string[];
   294→   294→  answerIndex: number;
   295→   295→}

   297→   297→/** سؤال هوشمند در قالب seed بازی + واژهٔ انگلیسیِ خود جزوه برای مسیر SRS/تلفظ */
   298→   298→type AiQuizSeed = QuizQuestionSeed & { aiWord: string; source: "ai" };

   300→   300→/** گارد سؤال هوشمند داخل deck (سؤال‌های عادی پرچم ندارند) */
   301→   301→function isAiSeed(q: QuizQuestionSeed): q is AiQuizSeed {
   302→   302→  const probe = q as { source?: unknown; aiWord?: unknown };
   303→   303→  return probe.source === "ai" && typeof probe.aiWord === "string";
   304→   304→}
   305→   305→
   306→   306→/** گارد شکل سؤال تولیدی — دفاع دوم روی والیدیشن سرور (بدون any) */
   307→   307→function isAiQuestionShape(v: unknown): v is AiQuestion {
   308→   308→  if (typeof v !== "object" || v === null) return false;
   309→   309→  const o = v as Record<string, unknown>;
   310→   310→  return (
   311→   311→    typeof o.word === "string" &&
   312→   312→    o.word.trim().length > 0 &&
   313→   313→    typeof o.sentence === "string" &&
   314→   314→    o.sentence.trim().length > 0 &&
   315→   315→    Array.isArray(o.options) &&
   316→   316→    o.options.length >= 2 &&
   317→   317→    o.options.length <= 6 &&
   318→   318→    o.options.every((x) => typeof x === "string" && x.trim().length > 0) &&
   319→   319→    typeof o.answerIndex === "number" &&
   320→   320→    Number.isInteger(o.answerIndex) &&
   321→   321→    o.answerIndex >= 0 &&
   322→   322→    o.answerIndex < o.options.length
   323→   323→  );
   324→   324→}

   326→   326→/**
   327→   327→ * صدا زدن سؤال‌ساز با مهلت ۸ ثانیه (AbortController + Promise.race با تایمر).
   328→   328→ * هر خطا/تایم‌اوت/پاسخ بد → null (بدون پرتاب استثنا؛ دور عادی ادامه می‌یابد).
   329→   329→ */
   330→   330→async function fetchAiQuestions(
   331→   331→  game: "cloze" | "speed-quiz",
   332→   332→  pool: { word: string; meaning: string }[],
   333→   333→  ctrl: AbortController,
   334→   334→): Promise<AiQuestion[] | null> {
   335→   335→  let timerId: number | null = null;
   336→   336→  const timeout = new Promise<null>((resolve) => {
   337→   337→    timerId = window.setTimeout(() => {
   338→   338→      ctrl.abort();
   339→   339→      resolve(null);
   340→   340→    }, AI_TIMEOUT_MS);
   341→   341→  });
   342→   342→  try {
   343→   343→    const resPromise = fetch("/api/game-questions", {
   344→   344→      method: "POST",
   345→   345→      headers: { "Content-Type": "application/json" },
   346→   346→      body: JSON.stringify({ game, pool, count: AI_COUNT }),
   347→   347→      signal: ctrl.signal,
   348→   348→    })
   349→   349→      .then(async (res) => (res.ok ? ((await res.json()) as { questions?: unknown }) : null))
   350→   350→      .catch(() => null); // لغو/شبکه → بی‌صدا (مسابقه منتظر نمی‌ماند)
   351→   351→    const res = await Promise.race([resPromise, timeout]);
   352→   352→    const raw = res && typeof res === "object" ? (res as { questions?: unknown }).questions : null;
   353→   353→    if (!Array.isArray(raw)) return null;
   354→   354→    const valid = raw.filter(isAiQuestionShape);
   355→   355→    return valid.length > 0 ? valid : null;
   356→   356→  } catch {
   357→   357→    return null;
   358→   358→  } finally {
   359→   359→    if (timerId !== null) window.clearTimeout(timerId);
   360→   360→  }
   361→   361→}
   362→   362→
   363→   363→/** نرمال‌سازی مقایسه‌ای معنی‌ها (نیم‌فاصله/ی/ک عربی/فاصله) — فقط برای سنجش تعلق */
   364→   364→function aiNorm(s: string): string {
   365→   365→  return s
   366→   366→    .replace(/\u200c/g, "")
   367→   367→    .replace(/ي/g, "ی")
   368→   368→    .replace(/ك/g, "ک")
   369→   369→    .replace(/\s+/g, " ")
   370→   370→    .trim()
   371→   371→    .toLowerCase();
   372→   372→}
   373→   373→
   374→   374→/**
   375→   375→ * تبدیل سؤال‌های تولیدی به seed سؤال واژگانی همین بازی — دفاع دوم سند: واژه باید
   376→   376→ * دقیقاً عضو استخر واژگان همین دور باشد و گزینهٔ درست عین معنی همان واژه در جزوه؛
   377→   377→ * سؤال‌های مشکوک بی‌صدا رد می‌شوند. گزینه‌ها با shapeOptions به سطح دور شکل می‌گیرند
   378→   378→ * (مزاحم سطح سخت فقط از معنی‌های واقعی همان استخر — هیچ محتوای ساختگی).
   379→   379→ */
   380→   380→function aiQuestionsToSeeds(
   381→   381→  questions: AiQuestion[],
   382→   382→  vocab: { word: string; meaning: string }[],
   383→   383→  lesson: number,
   384→   384→  level: DifficultyKey,
   385→   385→): QuizQuestionSeed[] {
   386→   386→  const byWord = new Map(vocab.map((v) => [v.word.toLowerCase(), v]));
   387→   387→  const ownerOf = new Map<string, string>();
   388→   388→  for (const v of vocab) {
   389→   389→    const k = aiNorm(v.meaning);
   390→   390→    if (k && !ownerOf.has(k)) ownerOf.set(k, v.word);
   391→   391→  }
   392→   392→  const built: QuizQuestionSeed[] = [];
   393→   393→  for (const q of questions) {
   394→   394→    const entry = byWord.get(q.word.trim().toLowerCase());
   395→   395→    if (!entry) continue; // واژه بیرون از استخر این دور → رد
   396→   396→    const answer = q.options[q.answerIndex]?.trim() ?? "";
   397→   397→    if (!answer) continue;
   398→   398→    if (ownerOf.get(aiNorm(answer)) !== entry.word) continue; // پاسخِ درست‌نما → رد
   399→   399→    const seed: AiQuizSeed = {
   400→   400→      category: "vocabulary",
   401→   401→      lesson,
   402→   402→      scope: "هوشمند",
   403→   403→      text: q.sentence,
   404→   404→      options: q.options.map((o) => o.trim()),
   405→   405→      answer: q.answerIndex,
   406→   406→      level: "استاندارد",
   407→   407→      aiWord: entry.word,
   408→   408→      source: "ai",
   409→   409→    };
   410→   410→    built.push(shapeOptions(seed, level, []));
   411→   411→  }
   412→   412→  if (built.length <= 1) return built;
   413→   413→  // سطح سخت — مزاحم پنجم فقط از معنی‌های واقعی همان استخر
   414→   414→  const traps = shuffle([...new Set(built.flatMap((s) => s.options))]);
   415→   415→  return built.map((s) => shapeOptions(s, level, traps));
   416→   416→}
