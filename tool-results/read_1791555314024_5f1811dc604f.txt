    81→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
    82→function Confetti() {
    83→  const pieces = useMemo(
    84→    () =>
    85→      Array.from({ length: 32 }, (_, i) => ({
    86→        id: i,
    87→        left: Math.random() * 100,
    88→        delay: Math.random() * 1.2,
    89→        duration: 2.4 + Math.random() * 1.6,
    90→        size: 6 + Math.random() * 6,
    91→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
    92→        drift: Math.round((Math.random() - 0.5) * 120),
    93→        round: Math.random() > 0.5,
    94→      })),
    95→    [],
    96→  );
    97→  return (
    98→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
    99→      {pieces.map((p) => (
   100→        <span
   101→          key={p.id}
   102→          className="confetti-piece absolute top-[-14px] block"
   103→          style={{
   104→            left: `${p.left}%`,
   105→            width: p.size,
   106→            height: p.size * (p.round ? 1 : 0.45),
   107→            background: p.color,
   108→            borderRadius: p.round ? "50%" : 2,
   109→            animationDelay: `${p.delay}s`,
   110→            animationDuration: `${p.duration}s`,
   111→            "--confetti-drift": `${p.drift}px`,
   112→          } as React.CSSProperties}
   113→        />
   114→      ))}
   115→    </div>
   116→  );
   117→}

   152→  async function startPracticeRound(ids: string[], nextMode: Mode = "normal") {

   162→        return;
   163→      }
   164→      setQuestions(data.questions as Question[]);

   167→      setSelected(null);
   168→      setAiExplanation(null);
   169→      setElapsed(0);
   170→      deadlineRef.current = Date.now() + timed * 1000;
   171→      if (advanceRef.current) clearTimeout(advanceRef.current);
   172→      setPhase("playing");
   173→      clearPracticeIds();

   175→      toast.error("خطا در دریافت سؤالات");

   177→      setLoading(false);

   228→      setSelected(null);
   229→      setAiExplanation(null);
   230→      setElapsed(0);
   231→      deadlineRef.current = Date.now() + timed * 1000;

   241→  /** هنگام اتمام زمان یک سؤال: بدون پاسخ ثبت و اشتباه، سپس نمایش پاسخ و رفتن خودکار */
   242→  function handleTimeout() {
   243→    if (!current || selected !== null) return;
   244→    toast.info("وقت این سؤال تمام شد ⏰", { duration: 1900 });
   245→    recordMistake(current.id);
   246→    setSelected(-1);
   247→    setAnswers((prev) => {
   248→      const next = [...prev];
   249→      next[index] = -1;
   250→      return next;
   251→    });
   252→    setAiExplanation(null);
   253→    advanceRef.current = setTimeout(() => nextQuestion(), 1900);
   254→  }
   255→
   256→  // شمارش معکوس حالت زمان‌دار — دی‌دلاین هر سؤال با تغییر سؤال/فاز بازنشانی می‌شود
   257→  useEffect(() => {
   258→    if (phase !== "playing" || timed <= 0 || selected !== null) return;
   259→    if (deadlineRef.current === 0 || deadlineRef.current < Date.now() - timed * 1000) {
   260→      deadlineRef.current = Date.now() + timed * 1000;
   261→    }
   262→    const iv = setInterval(() => {
   263→      if (Date.now() >= deadlineRef.current) {
   264→        clearInterval(iv);
   265→        handleTimeout();
   266→      } else {
   267→        setQTick((t) => t + 1);
   268→      }
   269→    }, 1000);
   270→    return () => clearInterval(iv);
   271→  }, [phase, timed, selected, index]);

   273→  function chooseOption(optIdx: number) {
   274→    if (selected !== null) return;
   275→    if (timed > 0 && deadlineRef.current > 0) {
   276→      answeredLeftRef.current = Math.max(0, Math.ceil((deadlineRef.current - Date.now()) / 1000));
   277→    }
   278→    setSelected(optIdx);
   279→    setAnswers((prev) => {
   280→      const next = [...prev];
   281→      next[index] = optIdx;
   282→      return next;
   283→    });
   284→    if (current) {
   285→      if (optIdx === current.answer) {
   286→        if (mode === "review") resolveMistake(current.id);
   287→      } else {
   288→        recordMistake(current.id);
   289→      }
   290→    }
   291→  }

   293→  async function askAiExplanation() {
   294→    if (!current) return;

   298→        method: "POST",

   300→        body: JSON.stringify({
   301→          text: current.text,
   302→          options: current.options,
   303→          answer: current.answer,
   304→        }),
   305→      });

   307→      if (res.ok) setAiExplanation(data.explanation);
   308→      else toast.error(data.error ?? "خطا در توضیح هوشمند");
   309→    } catch {
   310→      toast.error("خطا در ارتباط با هوش مصنوعی");

   312→      setAiExplaining(false);
   313→    }

   316→  function nextQuestion() {
   317→    if (advanceRef.current) clearTimeout(advanceRef.current);
   318→    if (index + 1 >= questions.length) {
   319→      finish();
   320→      return;
   321→    }
   322→    setIndex((i) => i + 1);
   323→    setSelected(answers[index + 1]);
   324→    setAiExplanation(null);
   325→    deadlineRef.current = Date.now() + timed * 1000;
   326→  }

   328→  function finish() {
   329→    if (timerRef.current) clearInterval(timerRef.current);
   330→    if (advanceRef.current) clearTimeout(advanceRef.current);
   331→    setPhase("result");
   332→    recordStudySession();
   333→    const streak = touchStreak();
   334→    const detail = questions.map((q, i) => ({ id: q.id, chosen: answers[i], answer: q.answer }));
   335→    fetch("/api/attempts", {
   336→      method: "POST",
   337→      headers: { "Content-Type": "application/json" },
   338→      body: JSON.stringify({
   339→        category: mode === "review" ? "review" : category,
   340→        lesson: lesson === "all" ? 0 : Number(lesson),
   341→        total: questions.length,
   342→        correct: correctCount,
   343→        duration: elapsed,
   344→        name: userName || undefined,
   345→        detail,
   346→      }),
   347→    })
   348→      .then(() =>
   349→        toast.success(
   350→          mode === "review"
   351→            ? `دور مرور تمام شد! ${mistakeCount.toLocaleString("fa-IR")} سؤال دیگر در دفتر اشتباهات مانده`
   352→            : `نتیجه آزمون ذخیره شد ${userName ? `— موفق باشی ${userName}!` : "🎉"}${streak.count > 1 ? ` 🔥 زنجیره ${streak.count.toLocaleString("fa-IR")} روزه` : ""}`,
   353→        ),
   354→      )
   355→      .catch(() => undefined);
   356→  }

   358→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   359→  async function exportMistakesCsv() {
   360→    if (!mistakeIds.length || exporting) return;
   361→    setExporting(true);
   362→    try {
   363→      const res = await fetch(`/api/questions?count=${mistakeIds.length}&ids=${mistakeIds.join(",")}`);
   364→      const data = await res.json();
   365→      const qs = (data.questions ?? []) as Question[];
   366→      if (!qs.length) {
   367→        toast.error("سؤالی برای خروجی یافت نشد.");
   368→        return;
   369→      }
   370→      const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   371→      const rows = [
   372→        ["ردیف", "متن سؤال", "پاسخ صحیح", "دسته", "مبحث", "سطح"].join(","),
   373→        ...qs.map((q, i) =>
   374→          [
   375→            i + 1,
   376→            esc(q.text),
   377→            esc(q.options[q.answer] ?? ""),
   378→            q.category === "grammar" ? "گرامر" : "واژگان",
   379→            esc(q.scope),
   380→            esc(q.level),
   381→          ].join(","),
   382→        ),
   383→      ];
   384→      const blob = new Blob(["\uFEFF" + rows.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   385→      const url = URL.createObjectURL(blob);
   386→      const a = document.createElement("a");
   387→      a.href = url;
   388→      a.download = "vision-one-mistakes.csv";
   389→      document.body.appendChild(a);
   390→      a.click();
   391→      a.remove();
   392→      URL.revokeObjectURL(url);
   393→      toast.success(`فایل CSV با ${qs.length.toLocaleString("fa-IR")} سؤال دانلود شد.`);
   394→    } catch {
   395→      toast.error("خطا در ساخت فایل CSV");
   396→    } finally {
   397→      setExporting(false);
   398→    }
   399→  }
