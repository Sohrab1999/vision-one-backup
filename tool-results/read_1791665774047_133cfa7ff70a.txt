    80→    80→const WORD_RE = /^[A-Za-z]{3,12}$/;
    81→    81→
    82→    82→/** واژهٔ پایه — حذف نشانه‌های دستوری داخل پرانتز: "dry (adj.)" → "dry" */
    83→    83→function baseWord(w: string): string {
    84→    84→  return w.replace(/\s*\([^)]*\)\s*/g, " ").trim();

    87→    87→export function ScrambleGame() {
    88→    88→  const data = useBookData();
    89→    89→  const { bookId: activeBookId } = useBook();
    90→    90→  // بازی فقط داخل کتاب فعال رندر می‌شود (useBookData بالا تضمینش می‌کند) — در عمل هرگز تهی نیست
    91→    91→  const bookId: string = activeBookId ?? "";
    92→    92→  const { speed } = useVoiceSpeed();
    93→    93→
    94→    94→  const [phase, setPhase] = useState<Phase>("setup");
    95→    95→  const [lesson, setLesson] = useState<number>(ALL_LESSONS);
    96→    96→  const [rounds, setRounds] = useState<number>(8);
    97→    97→
    98→    98→  const [deck, setDeck] = useState<VocabEntry[]>([]);
    99→    99→  const [idx, setIdx] = useState(0);
   100→   100→  const [tiles, setTiles] = useState<Tile[]>([]);
   101→   101→  const [slots, setSlots] = useState<(number | null)[]>([]);
   102→   102→
   103→   103→  const [solved, setSolved] = useState(0);
   104→   104→  const [skipped, setSkipped] = useState(0);
   105→   105→  const [hinted, setHinted] = useState(0);
   106→   106→  const [detail, setDetail] = useState<DetailRow[]>([]);
   107→   107→  const [secs, setSecs] = useState(0);
   108→   108→  const [isRecord, setIsRecord] = useState(false);
   109→   109→
   110→   110→  /** انیمیشن match-pop روی ردیف پاسخ هنگام حل درست */
   111→   111→  const [pop, setPop] = useState(false);
   112→   112→  /** واژهٔ درستی که بعد از رد کردن برای ۱ ثانیه نشان داده می‌شود */
   113→   113→  const [reveal, setReveal] = useState<string | null>(null);
   114→   114→
   115→   115→  const locked = useRef(false);
   116→   116→  const timer = useRef<number | undefined>(undefined);
   117→   117→  const secsRef = useRef(0);
   118→   118→
   119→   119→  const busy = pop || reveal !== null;
   120→   120→
   121→   121→  // استخر قابل بازی — فقط واژه‌های تک‌کلمه‌ای ۳ تا ۱۲ حرفی
   122→   122→  const pool = useMemo(() => {
   123→   123→    const base = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   124→   124→    return base.filter((w) => WORD_RE.test(baseWord(w.word)));
   125→   125→  }, [data.vocab, lesson]);
   126→   126→
   127→   127→  // رکورد شخصی — snapshot عددی (پایدار برای useSyncExternalStore؛ سرور null)
   128→   128→  const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   129→   129→  const best = useSyncExternalStore(
   130→   130→    subscribeGameData,
   131→   131→    () => {
   132→   132→      const b = loadBest<BestRec>(bookId, GAME_KEY, scope);
   133→   133→      return b ? b.time : null;
   134→   134→    },
   135→   135→    () => null,
   136→   136→  );
   137→   137→
   138→   138→  // ── تایمر (functional) + شمارندهٔ دقیق برای رکورد ──
   139→   139→  useEffect(() => {
   140→   140→    if (phase !== "playing") return;
   141→   141→    const t = window.setInterval(() => {
   142→   142→      secsRef.current += 1;
   143→   143→      setSecs((s) => s + 1);
   144→   144→    }, 1000);
   145→   145→    return () => window.clearInterval(t);
   146→   146→  }, [phase]);
   147→   147→
   148→   148→  // ── پاک‌سازی نهایی: تلفظ + تایمرهای معلق ──
   149→   149→  useEffect(() => {
   150→   150→    return () => {
   151→   151→      stopSpeaking();
   152→   152→      if (timer.current !== undefined) window.clearTimeout(timer.current);
   153→   153→    };
   154→   154→  }, []);
   155→   155→
   156→   156→  /** بارگذاری واژهٔ بعدی — حروف به‌هم‌ریخته + خانه‌های خالی */
   157→   157→  const loadWord = useCallback((entry: VocabEntry) => {
   158→   158→    const base = baseWord(entry.word);
   159→   159→    setTiles(shuffle(base.split("").map((letter, i) => ({ id: i, letter }))));
   160→   160→    setSlots(Array.from({ length: base.length }, () => null as number | null));
   161→   161→    setPop(false);
   162→   162→  }, []);
   163→   163→
   164→   164→  /** پایان دور — استریک/جلسه/Attempt/رکورد */
   165→   165→  const finishRun = useCallback(
   166→   166→    (finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   167→   167→      touchStreak();
   168→   168→      recordStudySession();
   169→   169→      bumpPlays(bookId, GAME_KEY);
   170→   170→      saveAttempt(bookId, GAME_KEY, {
   171→   171→        lesson: lesson === ALL_LESSONS ? 0 : lesson,
   172→   172→        total: deck.length,
   173→   173→        correct: finalSolved,
   174→   174→        duration: secsRef.current,
   175→   175→        detail: finalDetail,
   176→   176→      });
   177→   177→      // رکورد فقط برای اجرای کاملِ بدون رد کردن و بدون راهنما
   178→   178→      if (finalSkipped === 0 && hinted === 0) {
   179→   179→        const recScope = `${lesson === ALL_LESSONS ? "all" : lesson}:${rounds}`;
   180→   180→        const prev = loadBest<BestRec>(bookId, GAME_KEY, recScope);
   181→   181→        if (!prev || secsRef.current < prev.time) {
   182→   182→          saveBest<BestRec>(bookId, GAME_KEY, recScope, { time: secsRef.current });
   183→   183→          setIsRecord(true);
   184→   184→        }
   185→   185→      }
   186→   186→      setPhase("done");
   187→   187→    },
   188→   188→    [bookId, lesson, rounds, deck.length, hinted],
   189→   189→  );
   190→   190→
   191→   191→  /** ادامه به واژهٔ بعد یا پایان دور */
   192→   192→  const advanceOrFinish = useCallback(
   193→   193→    (nextIdx: number, finalSolved: number, finalSkipped: number, finalDetail: DetailRow[]) => {
   194→   194→      if (nextIdx >= deck.length) {
   195→   195→        finishRun(finalSolved, finalSkipped, finalDetail);
   196→   196→        return;
   197→   197→      }
   198→   198→      setIdx(nextIdx);
   199→   199→      loadWord(deck[nextIdx]);
   200→   200→    },
   201→   201→    [deck, loadWord, finishRun],
   202→   202→  );
   203→   203→
   204→   204→  /** حل درست — پاپ + تلفظ + SRS(✓) و بعد واژهٔ بعد */
   205→   205→  const solveWord = useCallback(
   206→   206→    (entry: VocabEntry, finalSolved: number, finalDetail: DetailRow[]) => {
   207→   207→      locked.current = true;
   208→   208→      setPop(true);
   209→   209→      const base = baseWord(entry.word);
   210→   210→      recordSRS(base, true);
   211→   211→      speak(base, { speed });
   212→   212→      timer.current = window.setTimeout(() => {
   213→   213→        locked.current = false;
   214→   214→        setPop(false);
   215→   215→        advanceOrFinish(idx + 1, finalSolved, skipped, finalDetail);
   216→   216→      }, 650);
   217→   217→    },
   218→   218→    [speed, idx, skipped, advanceOrFinish],
   219→   219→  );
   220→   220→
   221→   221→  /** گذاشتن یک حرف از استخر در اولین خانهٔ خالی + بررسی حل شدن */
   222→   222→  const placeTile = useCallback(
   223→   223→    (tileId: number) => {
   224→   224→      if (locked.current) return;
   225→   225→      const current = deck[idx];
   226→   226→      if (!current) return;
   227→   227→      const empty = slots.indexOf(null);
   228→   228→      if (empty === -1) return;
   229→   229→      const tile = tiles.find((t) => t.id === tileId);
   230→   230→      if (!tile) return;
   231→   231→      const nextSlots = [...slots];
   232→   232→      nextSlots[empty] = tileId;
   233→   233→      setSlots(nextSlots);
   234→   234→      // همهٔ خانه‌ها پر شد؟ تطبیق با واژهٔ پایه را بسنجیم
   235→   235→      if (nextSlots.every((s) => s !== null)) {
   236→   236→        const base = baseWord(current.word);
   237→   237→        const attempt = nextSlots
   238→   238→          .map((s) => tiles.find((t) => t.id === s)?.letter ?? "")
   239→   239→          .join("");
   240→   240→        if (attempt === base) {
   241→   241→          const finalSolved = solved + 1;
   242→   242→          const nextDetail = [...detail, { word: base, ok: true }];
   243→   243→          setSolved(finalSolved);
   244→   244→          setDetail(nextDetail);
   245→   245→          solveWord(current, finalSolved, nextDetail);
   246→   246→        }
   247→   247→      }
   248→   248→    },
   249→   249→    [deck, idx, slots, tiles, solved, detail, solveWord],
   250→   250→  );
   251→   251→
   252→   252→  /** برگرداندن حرفِ یک خانه به استخر (کلیک روی خانهٔ پر) */
   253→   253→  const returnAt = useCallback((index: number) => {
   254→   254→    if (locked.current) return;
   255→   255→    setSlots((prev) => {
   256→   256→      if (prev[index] === null) return prev;
   257→   257→      const next = [...prev];
   258→   258→      next[index] = null;
   259→   259→      return next;
   260→   260→    });
   261→   261→  }, []);
   262→   262→
   263→   263→  /** پاک کردن آخرین حرف (دکمه یا Backspace) */
   264→   264→  const returnLast = useCallback(() => {
   265→   265→    if (locked.current) return;
   266→   266→    setSlots((prev) => {
   267→   267→      for (let i = prev.length - 1; i >= 0; i--) {
   268→   268→        if (prev[i] !== null) {
   269→   269→          const next = [...prev];
   270→   270→          next[i] = null;
   271→   271→          return next;
   272→   272→        }
   273→   273→      }
   274→   274→      return prev;
   275→   275→    });
   276→   276→  }, []);
   277→   277→
   278→   278→  /** راهنما — یک حرف درستِ بعدی را جای‌گذاری می‌کند (رکورد دور می‌سوزد) */
   279→   279→  const applyHint = useCallback(() => {
   280→   280→    if (locked.current) return;
   281→   281→    const current = deck[idx];
   282→   282→    if (!current) return;
   283→   283→    const empty = slots.indexOf(null);
   284→   284→    if (empty === -1) return;
   285→   285→    const base = baseWord(current.word);
   286→   286→    const used = new Set(slots.filter((s): s is number => s !== null));
   287→   287→    const tile = tiles.find((t) => !used.has(t.id) && t.letter === base[empty]);
   288→   288→    if (!tile) return;
   289→   289→    setHinted((h) => h + 1);
   290→   290→    placeTile(tile.id);
   291→   291→  }, [deck, idx, slots, tiles, placeTile]);
   292→   292→
   293→   293→  /** رد کردن — SRS(✗) + نمایش واژهٔ درست برای ۱ ثانیه (رکورد دور می‌سوزد) */
   294→   294→  const skipWord = useCallback(() => {
   295→   295→    if (locked.current) return;
   296→   296→    const current = deck[idx];
   297→   297→    if (!current) return;
   298→   298→    locked.current = true;
   299→   299→    const base = baseWord(current.word);
   300→   300→    recordSRS(base, false);
   301→   301→    const nextSkipped = skipped + 1;
   302→   302→    const nextDetail = [...detail, { word: base, ok: false }];
   303→   303→    setSkipped(nextSkipped);
   304→   304→    setDetail(nextDetail);
   305→   305→    setReveal(base);
   306→   306→    timer.current = window.setTimeout(() => {
   307→   307→      locked.current = false;
   308→   308→      setReveal(null);
   309→   309→      advanceOrFinish(idx + 1, solved, nextSkipped, nextDetail);
   310→   310→    }, 1000);
   311→   311→  }, [deck, idx, skipped, detail, solved, advanceOrFinish]);
   312→   312→
   313→   313→  /** انصراف — تایمرهای معلق پاک و به تنظیمات برگردیم */
   314→   314→  const cancel = useCallback(() => {
   315→   315→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   316→   316→    locked.current = false;
   317→   317→    stopSpeaking();
   318→   318→    setPhase("setup");
   319→   319→  }, []);
   320→   320→
   321→   321→  /** شروع دور — انتخاب واژه‌های یکتا از استخر قابل بازی */
   322→   322→  const start = useCallback(() => {
   323→   323→    const chosen = pickUnique(pool, rounds, (w) => baseWord(w.word));
   324→   324→    if (chosen.length < 3) {
   325→   325→      toast.error("واژه قابل بازی در این محدوده پیدا نشد");
   326→   326→      return;
   327→   327→    }
   328→   328→    if (timer.current !== undefined) window.clearTimeout(timer.current);
   329→   329→    locked.current = false;
   330→   330→    secsRef.current = 0;
   331→   331→    setDeck(chosen);
   332→   332→    setIdx(0);
   333→   333→    setSolved(0);
   334→   334→    setSkipped(0);
   335→   335→    setHinted(0);
   336→   336→    setDetail([]);
   337→   337→    setSecs(0);
   338→   338→    setIsRecord(false);
   339→   339→    setReveal(null);
   340→   340→    loadWord(chosen[0]);
   341→   341→    setPhase("playing");
   342→   342→  }, [pool, rounds, loadWord]);
   343→   343→
   344→   344→  // ── کیبورد: فقط در فاز بازی؛ حروف → جای‌گذاری، Backspace → برگرداندن ──
   345→   345→  useEffect(() => {
   346→   346→    if (phase !== "playing") return;
   347→   347→    const onKey = (e: KeyboardEvent) => {
   348→   348→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   349→   349→      if (e.key === "Backspace") {
   350→   350→        e.preventDefault();
   351→   351→        returnLast();
   352→   352→        return;
   353→   353→      }
   354→   354→      if (e.key === "Enter") return; // حل خودکار است — اینتر کاری نمی‌کند
   355→   355→      if (/^[a-zA-Z]$/.test(e.key)) {
   356→   356→        const letter = e.key.toLowerCase();
   357→   357→        const used = new Set(slots.filter((s): s is number => s !== null));
   358→   358→        const tile = tiles.find((t) => !used.has(t.id) && t.letter.toLowerCase() === letter);
   359→   359→        if (tile) placeTile(tile.id);
   360→   360→      }
   361→   361→    };
   362→   362→    window.addEventListener("keydown", onKey);
   363→   363→    return () => window.removeEventListener("keydown", onKey);
   364→   364→  }, [phase, slots, tiles, placeTile, returnLast]);
   365→   365→
   366→   366→  const current = deck[idx];
   367→   367→  const usedIds = new Set(slots.filter((s): s is number => s !== null));
   368→   368→  const letterOf = (id: number): string => tiles.find((t) => t.id === id)?.letter ?? "";
   369→   369→  const pct = Math.round(((solved + skipped) / Math.max(deck.length, 1)) * 100);
   370→   370→
   371→   371→  // ── Setup ──
   372→   372→  if (phase === "setup") {
   373→   373→    return (
   374→   374→      <Card className="mx-auto max-w-xl rise-in">
   375→   375→        <CardContent className="space-y-6 p-6">
   376→   376→          <div className="flex items-center gap-3">
   377→   377→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-amber-500/15 text-amber-600 dark:text-amber-400">
   378→   378→              <Shuffle size={22} />
   379→   379→            </span>
   380→   380→            <div>
   381→   381→              <h2 className="text-lg font-black">حروف جابه‌جا</h2>
   382→   382→              <p className="text-xs text-muted-foreground">
   383→   383→                حروف واژه‌ها ریخته شده — با معنی فارسی سرهمشان کن!
   384→   384→              </p>
   385→   385→            </div>
   386→   386→          </div>
   387→   387→
   388→   388→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   389→   389→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   390→   390→            <p>
   391→   391→              روی حروف به ترتیب بزن تا واژه ساخته شود؛ کیبورد هم کار می‌کند. واژه‌های سخت را می‌توانی رد کنی ولی رکورد می‌سوزد!
   392→   392→            </p>
   393→   393→          </div>
   394→   394→
   395→   395→          <div>
   396→   396→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   397→   397→            <div className="flex flex-wrap gap-2">
   398→   398→              <button
   399→   399→                type="button"
