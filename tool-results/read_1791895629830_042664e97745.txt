   151→   151→function hashStr(s: string): number {
   152→   152→  let h = 5381;
   153→   153→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   154→   154→  return Math.abs(h);
   155→   155→}

   163→   163→function judgeVerdicts(guess: string, answer: string, len: number): Verdict[] {
   164→   164→  const res: Verdict[] = Array<Verdict>(len).fill("absent");
   165→   165→  const remaining: Record<string, number> = {};
   166→   166→  for (let i = 0; i < len; i++) {
   167→   167→    if (guess[i] === answer[i]) {
   168→   168→      res[i] = "correct";
   169→   169→    } else {
   170→   170→      remaining[answer[i]] = (remaining[answer[i]] ?? 0) + 1;
   171→   171→    }
   172→   172→  }
   173→   173→  for (let i = 0; i < len; i++) {
   174→   174→    if (res[i] === "correct") continue;
   175→   175→    const ch = guess[i];
   176→   176→    if ((remaining[ch] ?? 0) > 0) {
   177→   177→      res[i] = "present";
   178→   178→      remaining[ch] -= 1;
   179→   179→    }
   180→   180→  }
   181→   181→  return res;
   182→   182→}
