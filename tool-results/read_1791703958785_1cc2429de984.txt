   165→/** واژه هدف بدون پرانتز توضیح — «order (n)» → «order» */
   166→function toBase(word: string): string {
   167→  return word.replace(/\s*\([^)]*\)\s*/g, " ").trim();
   168→}
   169→
   170→function escapeRe(s: string): string {
   171→  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   172→}

   174→function wordCount(s: string): number {
   175→  return s.trim().split(/\s+/).length;
   176→}

   178→/** همه جمله‌های کوتاه کتاب: متن‌ها (پاراگراف‌ها) + مکالمه‌ها (خط‌ها) */
   179→function collectSentences(data: BookData): SourceSentence[] {
   180→  const out: SourceSentence[] = [];
   181→  for (const r of data.readings) {
   182→    for (const p of r.paragraphs) {
   183→      for (const s of p.en.split(/(?<=[.!?])\s+/)) {
   184→        const t = s.trim();
   185→        if (t) out.push({ en: t, fa: p.fa });
   186→      }
   187→    }
   188→  }
   189→  for (const c of data.conversations) {
   190→    for (const line of c.lines) {
   191→      for (const s of line.en.split(/(?<=[.!?])\s+/)) {
   192→        const t = s.trim();
   193→        if (t) out.push({ en: t, fa: line.fa });
   194→      }
   195→    }
   196→  }
   197→  return out;
   198→}

   200→/**
   201→ * هسته ساخت پازل — تابع خالص: برای هر واژه (بدون تکرار) اولین جمله مناسب کتاب را
   202→ * پیدا می‌کند، واژه را با ______ جایگزین و ۲ حواس‌پرتی از همان استخر درس می‌چیند.
   203→ */
   204→function buildPool(data: BookData, lesson: number, distractorCount = 2): ClozeItem[] {
   205→  const vocabPool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   206→  const sentences = collectSentences(data);
   207→
   208→  // استخر واژه پاکسازی‌شده — فقط واژه‌های تک‌کلمه‌ای لاتین، یکتا بر اساس baseWord
   209→  const seenBase = new Set<string>();
   210→  const words: { base: string; entry: VocabEntry }[] = [];
   211→  for (const entry of vocabPool) {
   212→    const base = toBase(entry.word);
   213→    if (!WORD_RE.test(base) || seenBase.has(base)) continue;
   214→    seenBase.add(base);
   215→    words.push({ base, entry });
   216→  }
   217→
   218→  const items: ClozeItem[] = [];
   219→  for (const { base, entry } of words) {
   220→    if (items.length >= MAX_ITEMS) break;
   221→    const re = new RegExp("\\b" + escapeRe(base) + "\\b", "i");
   222→    const found = sentences.find((s) => {
   223→      if (s.en.length < 30 || s.en.length > 160) return false;
   224→      const wc = wordCount(s.en);
   225→      if (wc < 6 || wc > 35) return false;
   226→      return re.test(s.en);
   227→    });
   228→    if (!found) continue;
   229→
   230→    // حواس‌پرتی‌ها از همان استخر درس — baseWord متفاوت + معنی یکتا (بدون ابهام)
   231→    // (فاز G3-01 — تعداد حواس‌پرتی سطح‌دار؛ پیش‌فرض ۲ = همان رفتار قبلی)
   232→    const distractorPool = vocabPool.filter((w) => {
   233→      const b = toBase(w.word);
   234→      return b !== base && WORD_RE.test(b) && w.meaning !== entry.meaning;
   235→    });
   236→    const distractors = pickUnique(distractorPool, distractorCount, (w) => `${toBase(w.word)}|${w.meaning}`);
   237→    if (distractors.length < distractorCount) continue;
   238→
   239→    items.push({
   240→      id: `${lesson}:${base}`,
   241→      sentence: found.en.replace(re, BLANK),
   242→      fa: found.fa,
   243→      answer: base,
   244→      options: shuffle([base, ...distractors.map((d) => toBase(d.word))]),
   245→      meaning: entry.meaning,
   246→      lessonLabel: entry.lessonLabel,
   247→    });
   248→  }
   249→  return shuffle(items);
   250→}

   254→let bestCacheKey = "";
   255→let bestCacheValue: BestRecord | null = null;
   256→
   257→function readBest(bookId: string, scope: string): BestRecord | null {
   258→  let raw: string | null = null;
   259→  try {
   260→    raw = localStorage.getItem(bestKeyOf(bookId, GAME_KEY, scope));
   261→  } catch {
   262→    return null;
   263→  }
   264→  const cacheKey = `${bookId}|${scope}|${raw ?? ""}`;
   265→  if (cacheKey === bestCacheKey) return bestCacheValue;
   266→  let value: BestRecord | null = null;
   267→  if (raw) {
   268→    try {
   269→      const parsed = JSON.parse(raw) as Partial<BestRecord> | null;
   270→      if (parsed && typeof parsed.score === "number" && typeof parsed.total === "number") {
   271→        value = { score: parsed.score, total: parsed.total };
   272→      }
   273→    } catch {
   274→      value = null;
   275→    }
   276→  }
   277→  bestCacheKey = cacheKey;
   278→  bestCacheValue = value;
   279→  return value;
   280→}
