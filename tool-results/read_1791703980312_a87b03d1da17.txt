   135→/** تاریخ امروز به شکل YYYY-MM-DD (تقویم محلی — پایدار بین رفرش‌ها) */
   136→function todayKey(): string {

   141→/** جابه‌جایی تقویمی تاریخ YYYY-MM-DD (بدون وابستگی به منطقهٔ زمانی) */
   142→function shiftDate(date: string, days: number): string {
   143→  const parts = date.split("-").map(Number);
   144→  const t = new Date(Date.UTC(parts[0], (parts[1] ?? 1) - 1, parts[2] ?? 1) + days * 86400000);
   145→  return `${t.getUTCFullYear()}-${pad2(t.getUTCMonth() + 1)}-${pad2(t.getUTCDate())}`;
   146→}

   148→/** هش ساده و پایدار (djb2) برای انتخاب واژهٔ روز */
   149→function hashStr(s: string): number {
   150→  let h = 5381;
   151→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   152→  return Math.abs(h);
   153→}

   155→/**
   156→ * رنگ‌بندی دو-گذرهٔ وردل — گذر اول فقط سبزها را می‌شمارد، گذر دوم
   157→ * زرد/خاکستری را با شمارش حروف باقی‌ماندهٔ پاسخ تعیین می‌کند
   158→ * (پاسخ کلاسیک: answer "speed" × guess "erase" → Eسبز، Rخاکستری، Aخاکستری، Sسبز، Eزرد).
   159→ * فاز G2-d — طول پارامتری شد (۴/۵/۶ حرفی).
   160→ */
   161→function judgeVerdicts(guess: string, answer: string, len: number): Verdict[] {
   162→  const res: Verdict[] = Array<Verdict>(len).fill("absent");
   163→  const remaining: Record<string, number> = {};
   164→  for (let i = 0; i < len; i++) {
   165→    if (guess[i] === answer[i]) {
   166→      res[i] = "correct";
   167→    } else {
   168→      remaining[answer[i]] = (remaining[answer[i]] ?? 0) + 1;
   169→    }
   170→  }
   171→  for (let i = 0; i < len; i++) {
   172→    if (res[i] === "correct") continue;
   173→    const ch = guess[i];
   174→    if ((remaining[ch] ?? 0) > 0) {
   175→      res[i] = "present";
   176→      remaining[ch] -= 1;
   177→    }
   178→  }
   179→  return res;
   180→}

   221→function loadDailyState(bookId: string): DailyState | null {
   222→  try {
   223→    const raw = localStorage.getItem(dailyKeyOf(bookId));
   224→    if (!raw) return null;
   225→    const p = JSON.parse(raw) as Partial<DailyState> | null;
   226→    if (!p || typeof p.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(p.date)) return null;
   227→    if (p.status !== "playing" && p.status !== "won" && p.status !== "lost") return null;
   228→    if (!Array.isArray(p.guesses)) return null;
   229→    const guesses = p.guesses
   230→      .filter((g): g is string => typeof g === "string" && WORD_RE.test(g))
   231→      .slice(0, MAX_ROWS);
   232→    return { date: p.date, guesses, status: p.status };
   233→  } catch {
   234→    return null;
   235→  }
   236→}
