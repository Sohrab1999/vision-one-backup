   165→/**
   166→ * فاز G5-03 — مهارت ربات تطبیقی مار و پله برای دور تازه (فقط mode="bot").
   167→ * منبع نرخ برد w (بدون ساخت هیچ کلید/امضای جدید):
   168→ *  ۱) برد/باخت دورهای مقابل رباتِ همین نشست مرورگر (vsBotSession) — اگر ≥۳ دور؛
   169→ *  ۲) وگرنه تخمین محافظه‌کارانه از دادهٔ موجود: wins («visionone:game-wins» کلید
   170→ *     "snakes-ladders" — دور کامل بی‌نقص به‌عنوان پروکسی برد) ÷ plays (کل دورهای
   171→ *     این کتاب از شمارندهٔ مشترک — هر دو اسکوپ)؛
   172→ *  ۳) دادهٔ <۳ دور → BOT_SKILL (رفتار قبلی بایت‌به‌بایت).
   173→ */
   174→function computeAdaptiveBotSkill(bookId: string): number {
   175→  const rounds = vsBotSession.wins + vsBotSession.losses;
   176→  if (rounds >= ADAPTIVE_MIN_ROUNDS) {
   177→    return clampBotSkill(BOT_SKILL + (vsBotSession.wins / rounds - 0.5) * 0.6);
   178→  }
   179→  try {
   180→    const wins = readGameWins()[GAME_KEY] ?? 0;
   181→    const plays = getPlays(bookId)[GAME_KEY] ?? 0;
   182→    if (plays >= ADAPTIVE_MIN_ROUNDS) {
   183→      return clampBotSkill(BOT_SKILL + (Math.min(1, wins / plays) - 0.5) * 0.6);
   184→    }
   185→  } catch {
   186→    /* localStorage در دسترس نیست — پیش‌فرض */
   187→  }
   188→  return BOT_SKILL;
   189→}

   514→  function proceedAfterCorrect(p: number): void {

   524→      flips += 1;
   525→      if (flips <= DICE_FLIPS) {

   527→        setDicePop((k) => k + 1);

   533→        later(() => hop(p, d), 420);
   534→      }
   535→    }
   536→    later(cycle, 350);

   539→  /** حرکت مهره خانه‌به‌خانه با زنجیرهٔ پله‌ای (~۱۲۰ms) — هر گام یک transition است (فاز G2-e) */
   540→  function hop(p: number, d: number): void {
   541→    const from = posRef.current[p];
   542→    const to = Math.min(100, from + d);
   543→    let at = from;
   544→    function step(): void {
   545→      if (at < to) {
   546→        at += 1;
   547→        posRef.current[p] = at;
   548→        setPositions([...posRef.current]);
   549→        later(step, STEP_MS);
   550→      } else {
   551→        land(p, d);
   552→      }
   553→    }
   554→    step();
   555→  }

   557→  /** بررسی مار/پله در خانهٔ فرود و اعلام انتقال سه‌کاناله (toast + صدا/لرزش + liveMsg) */
   558→  function land(p: number, d: number): void {
   559→    const pos = posRef.current[p];
   560→    const snakeTo = SNAKES[pos];
   561→    const ladderTo = LADDERS[pos];
   562→    const name = playersRef.current[p]?.name ?? "";
   563→    if (snakeTo !== undefined) {
   564→      // فاز G2-e-1 — مار = سقوط: صدای نزولی + لرزش خطا
   565→      playSound("wrong");
   566→      buzz("error");
   567→      const jr = jumpsRef.current[p];
   568→      if (jr) jr.down += 1;
   569→      toast.error(`مار! برگشتی به خانهٔ ${faNum(snakeTo)}`);
   570→      setLiveMsg(`مار! ${name} به خانهٔ ${faNum(snakeTo)} برگشت`);
   571→      // سُرش واحد ۵۲۰ms به مقصد — الگوی دو-فازی مدت transition (فاز G2-e)
   572→      setSlideMs(JUMP_MS);
   573→      later(() => {
   574→        posRef.current[p] = snakeTo;
   575→        setPositions([...posRef.current]);
   576→        later(() => {
   577→          setSlideMs(STEP_MS);
   578→          afterJump(p, d);
   579→        }, JUMP_MS);
   580→      }, 60);
   581→    } else if (ladderTo !== undefined) {
   582→      // فاز G2-e-1 — پله = صعود: صدای صعودی + لرزش موفقیت
   583→      playSound("correct");
   584→      buzz("success");
   585→      const jr = jumpsRef.current[p];
   586→      if (jr) jr.up += 1;
   587→      toast.success(`پله! رفتی به خانهٔ ${faNum(ladderTo)}`);
   588→      setLiveMsg(`پله! ${name} به خانهٔ ${faNum(ladderTo)} رفت`);
   589→      setSlideMs(JUMP_MS);
   590→      later(() => {
   591→        posRef.current[p] = ladderTo;
   592→        setPositions([...posRef.current]);
   593→        later(() => {
   594→          setSlideMs(STEP_MS);
   595→          afterJump(p, d);
   596→        }, JUMP_MS);
   597→      }, 60);
   598→    } else {
   599→      afterJump(p, d);
   600→    }
   601→  }

   603→  /** بعد از مار/پله: پایان بازی یا نوبت بعدی (تاس ۶ = نوبت اضافه) */

   612→      beginTurn(p);
   613→    } else {
   614→      beginTurn((p + 1) % playersRef.current.length);
   615→    }
   616→  }
