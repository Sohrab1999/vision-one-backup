   118→
   119→const cellKey = (r: number, c: number): string => `${r},${c}`;
   120→
   121→const parseKey = (k: string): [number, number] => {
   122→  const i = k.indexOf(",");
   123→  return [Number(k.slice(0, i)), Number(k.slice(i + 1))];
   124→};
   125→
   126→/** خانهٔ jام یک جای‌گذاری روی بوم خام */
   127→const draftCellKey = (p: { row: number; col: number; dir: CrosswordDir }, j: number): string =>
   128→  p.dir === "across" ? cellKey(p.row, p.col + j) : cellKey(p.row + j, p.col);
   129→
   130→/**
   131→ * سازگاری یک جای‌گذاری را بررسی می‌کند:
   132→ * - حرف‌های از قبل اشغال‌شده باید دقیقاً برابر باشند (حداکثر یک تقاطع)
   133→ * - سلول قبل از آغاز و بعد از پایان واژه خالی باشد (واژه به واژهٔ دیگر نمی‌چسبد)
   134→ * - برای سلول‌های غیرتقاطع، هر دو همسایهٔ عمود بر جهت واژه خالی باشند
   135→ */
   136→function canPlace(
   137→  word: string,
   138→  r0: number,
   139→  c0: number,
   140→  dir: CrosswordDir,
   141→  cells: Map<string, string>,
   142→): boolean {
   143→  const dr = dir === "down" ? 1 : 0;
   144→  const dc = dir === "across" ? 1 : 0;
   145→  const len = word.length;
   146→  if (cells.has(cellKey(r0 - dr, c0 - dc))) return false;
   147→  if (cells.has(cellKey(r0 + dr * len, c0 + dc * len))) return false;
   148→  let crossings = 0;
   149→  for (let j = 0; j < len; j++) {
   150→    const r = r0 + dr * j;
   151→    const c = c0 + dc * j;
   152→    const cur = cells.get(cellKey(r, c));
   153→    if (cur !== undefined) {
   154→      if (cur !== word[j] || crossings > 0) return false;
   155→      crossings += 1;
   156→    } else if (cells.has(cellKey(r - dc, c - dr)) || cells.has(cellKey(r + dc, c + dr))) {
   157→      return false;
   158→    }
   159→  }
   160→  return crossings === 1;
   161→}

   163→/** یک تلاش خام ساخت — خروجی روی بوم نامحدود با مختصات ممکن منفی */
   164→interface BuildDraft {
   165→  placed: { entry: CrosswordEntry; row: number; col: number; dir: CrosswordDir }[];
   166→  minR: number;
   167→  minC: number;
   168→  maxR: number;
   169→  maxC: number;
   170→}
   171→
   172→/** یک تلاش ساخت: واژهٔ اول افقی در مبدأ، هر واژهٔ بعدی در بهترین تقاطع یا حذف */
   173→function tryBuild(entries: CrosswordEntry[], maxSize: number): BuildDraft | null {
   174→  if (entries.length === 0) return null;
   175→  const cells = new Map<string, string>();
   176→  const placed: BuildDraft["placed"] = [];
   177→
   178→  // واژهٔ اول افقی وسط بوم — بقیه در چهار جهت دورش رشد می‌کنند و برش آخر وسطش می‌گذارد
   179→  const first = entries[0];
   180→  for (let j = 0; j < first.word.length; j++) cells.set(cellKey(0, j), first.word[j]);
   181→  placed.push({ entry: first, row: 0, col: 0, dir: "across" });
   182→  let minR = 0;
   183→  let maxR = 0;
   184→  let minC = 0;
   185→  let maxC = first.word.length - 1;
   186→
   187→  for (let i = 1; i < entries.length; i++) {
   188→    const word = entries[i].word;
   189→    let best: BuildDraft["placed"][number] | null = null;
   190→    let bestScore = Infinity;
   191→    for (const [key, ch] of cells) {
   192→      if (!word.includes(ch)) continue;
   193→      const [r, c] = parseKey(key);
   194→      for (let j = 0; j < word.length; j++) {
   195→        if (word[j] !== ch) continue;
   196→        for (const dir of ["across", "down"] as const) {
   197→          const r0 = dir === "across" ? r : r - j;
   198→          const c0 = dir === "across" ? c - j : c;
   199→          if (!canPlace(word, r0, c0, dir, cells)) continue;
   200→          const nMinR = Math.min(minR, r0);
   201→          const nMinC = Math.min(minC, c0);
   202→          const nMaxR = Math.max(maxR, dir === "across" ? r0 : r0 + word.length - 1);
   203→          const nMaxC = Math.max(maxC, dir === "across" ? c0 + word.length - 1 : c0);
   204→          if (nMaxR - nMinR + 1 > maxSize || nMaxC - nMinC + 1 > maxSize) continue;
   205→          // کمترین ابعاد افزوده + کمی تصادف برای تنوع چیدمان
   206→          const added = nMaxR - minR + nMaxC - minC;
   207→          const score = added * 100 + Math.random() * 7;
   208→          if (score < bestScore) {
   209→            bestScore = score;
   210→            best = { entry: entries[i], row: r0, col: c0, dir };
   211→          }
   212→        }
   213→      }
   214→    }
   215→    if (best) {
   216→      const w = best.entry.word;
   217→      const dr = best.dir === "down" ? 1 : 0;
   218→      const dc = best.dir === "across" ? 1 : 0;
   219→      for (let j = 0; j < w.length; j++) {
   220→        cells.set(cellKey(best.row + dr * j, best.col + dc * j), w[j]);
   221→      }
   222→      minR = Math.min(minR, best.row);
   223→      minC = Math.min(minC, best.col);
   224→      maxR = Math.max(maxR, best.dir === "down" ? best.row + w.length - 1 : best.row);
   225→      maxC = Math.max(maxC, best.dir === "across" ? best.col + w.length - 1 : best.col);
   226→      placed.push(best);
   227→    }
   228→    // واژه‌ای که هیچ تقاطع سازگاری ندارد حذف می‌شود
   229→  }
   230→  return { placed, minR, minC, maxR, maxC };
   231→}

   233→/** برش به bounding box + شماره‌گذاری استاندارد سرنخ‌ها (پیمایش ردیفی) */
   234→function finalize(build: BuildDraft, maxSize: number): CrosswordPuzzle | null {
   235→  const { placed, minR, minC, maxR, maxC } = build;
   236→  const rows = maxR - minR + 1;
   237→  const cols = maxC - minC + 1;
   238→  if (rows > maxSize || cols > maxSize) return null;
   239→  const grid: CrosswordCell[][] = [];
   240→  for (let r = 0; r < rows; r++) {
   241→    const row: CrosswordCell[] = [];
   242→    for (let c = 0; c < cols; c++) row.push({ letter: "", num: null, acrossId: null, downId: null });
   243→    grid.push(row);
   244→  }
   245→  const placements: CrosswordPlacement[] = placed.map((p, id) => ({
   246→    id,
   247→    word: p.entry.word,
   248→    meaning: p.entry.meaning,
   249→    lesson: p.entry.lesson,
   250→    lessonLabel: p.entry.lessonLabel,
   251→    row: p.row - minR,
   252→    col: p.col - minC,
   253→    dir: p.dir,
   254→    num: 0,
   255→  }));
   256→  for (const p of placements) {
   257→    for (let j = 0; j < p.word.length; j++) {
   258→      const [r, c] = parseKey(draftCellKey(p, j));
   259→      const cell = grid[r][c];
   260→      cell.letter = p.word[j];
   261→      if (p.dir === "across") cell.acrossId = p.id;
   262→      else cell.downId = p.id;
   263→    }
   264→  }
   265→  const startNums = new Map<string, number>();
   266→  const starts = [...new Set(placements.map((p) => cellKey(p.row, p.col)))];
   267→  starts.sort((a, b) => {
   268→    const [r1, c1] = parseKey(a);
   269→    const [r2, c2] = parseKey(b);
   270→    return r1 - r2 || c1 - c2;
   271→  });
   272→  starts.forEach((k, i) => startNums.set(k, i + 1));
   273→  for (const p of placements) {
   274→    p.num = startNums.get(cellKey(p.row, p.col)) ?? 0;
   275→    grid[p.row][p.col].num = p.num;
   276→  }
   277→  return { rows, cols, grid, placements };
   278→}

   280→/**
   281→ * تولید جدول متقاطع از استخر واژه:
   282→ * ۱) واژهٔ اول افقی وسط بوم ۲) هر واژهٔ بعدی در بهترین تقاطع (حرف مشترک + سازگاری
   283→ * اطراف + کمترین ابعاد افزوده) ۳) واژه‌های جا نشده حذف ۴) اگر کمتر از minWords
   284→ * واژه جا شد، با seed جدید تا maxTries بار تلاش و بهترین نتیجه برگردانده می‌شود.
   285→ * جدول بزرگ‌تر از maxSize در هر بُعد رد می‌شود.
   286→ */
   287→export function generateCrossword(
   288→  entries: CrosswordEntry[],
   289→  opts: GenerateOptions = {},
   290→): CrosswordPuzzle | null {
   291→  const minWords = Math.max(opts.minWords ?? 5, 1);
   292→  const maxTries = Math.max(opts.maxTries ?? 60, 1);
   293→  const maxSize = Math.max(opts.maxSize ?? 13, 3);
   294→  if (entries.length < minWords) return null;
   295→
   296→  let bestPuzzle: CrosswordPuzzle | null = null;
   297→  for (let t = 0; t < maxTries; t++) {
   298→    const order = shuffle(entries);
   299→    const build = tryBuild(order, maxSize);
   300→    if (!build || build.placed.length < minWords) continue;
   301→    const puzzle = finalize(build, maxSize);
   302→    if (!puzzle) continue;
   303→    if (
   304→      !bestPuzzle ||
   305→      puzzle.placements.length > bestPuzzle.placements.length ||
   306→      (puzzle.placements.length === bestPuzzle.placements.length &&
   307→        puzzle.rows * puzzle.cols < bestPuzzle.rows * bestPuzzle.cols)
   308→    ) {
   309→      bestPuzzle = puzzle;
   310→    }
   311→    if (puzzle.placements.length === entries.length) break; // همه واژه‌ها جا گرفتند
   312→  }
   313→  return bestPuzzle;
   314→}

   435→      const scope = `${lesson === ALL_LESSONS ? "all" : lesson}:${wordCount}`;
   436→      const prev = loadBest<BestRecord>(activeBookId, GAME_KEY, scope);
   437→      if (!prev || finalSecs < prev.time) {
   438→        saveBest(activeBookId, GAME_KEY, scope, { time: finalSecs });
   439→        setIsRecord(true);
   440→        toast.success(`رکورد جدید! جدول در ${fmtTime(finalSecs)} حل شد 🏆`);
   441→      } else {
   442→        toast.success(`جدول کامل شد — ${fmtTime(finalSecs)} 🎉`);
   443→      }
   444→    } else {
   445→      toast.success(`جدول کامل شد — ${fmtTime(finalSecs)} با ${faNum(finalHints)} راهنما 🎉`);
   446→    }
   447→    setPhase("done");
   448→  };
   449→
   450→  /** ثبت واژه‌های تازه‌حل‌شده: قفل سلول‌ها + pop + تلفظ + پایان دور */
   451→  const applySolved = (ids: number[], hCount: number) => {
   452→    if (!puzzle) return;
   453→    const cells: string[] = [];
   454→    ids.forEach((id) => cells.push(...cellsOf(puzzle.placements[id])));
   455→    setSolved((prev) => {
   456→      const next = new Set(prev);
   457→      ids.forEach((id) => next.add(id));
   458→      return next;
   459→    });
   460→    setLocked((prev) => {
   461→      const next = new Set(prev);
   462→      cells.forEach((k) => next.add(k));
   463→      return next;
   464→    });
   465→    setPopKeys(cells);
   466→    later(() => setPopKeys([]), POP_MS);
   467→    speak(puzzle.placements[ids[0]].word, { speed });
   468→    if (solved.size + ids.length >= puzzle.placements.length) {
   469→      later(() => finish(secsRef.current, hCount), FINISH_MS);
   470→    }
   471→  };
   472→
   473→  const start = () => {
   474→    const pool = lesson === ALL_LESSONS ? data.vocab.all : data.vocab.ofLesson(lesson);
   475→    // گارد واژهٔ جدول‌پذیر: فقط حروف انگلیسی، طول ۳ تا ۹
   476→    const valid = pool.filter((w) => /^[A-Za-z]{3,9}$/.test(w.word));
   477→    const normalized = valid.map((w) => ({ ...w, word: w.word.toUpperCase() }));
   478→    const chosen = pickUnique(normalized, wordCount, (w) => w.word);
   479→    if (chosen.length < wordCount) {
   480→      toast.error(
   481→        `واژهٔ انگلیسی ۳ تا ۹ حرفیِ کافی پیدا نشد (${faNum(chosen.length)} از ${faNum(wordCount)}) — درس یا تعداد دیگری را امتحان کن`,
   482→      );
   483→      return;
   484→    }
   485→    const entries: CrosswordEntry[] = chosen.map((w) => ({
   486→      word: w.word,
   487→      meaning: w.meaning,
   488→      lesson: w.lesson,
   489→      lessonLabel: w.lessonLabel,
   490→    }));
   491→    const built = generateCrossword(entries, { minWords: MIN_PLACED, maxTries: 60, maxSize: 13 });
   492→    if (!built || built.placements.length < MIN_PLACED) {
   493→      toast.error("ساخت جدول متقاطع با این واژه‌ها ممکن نشد — دوباره شروع کن یا درس دیگری را انتخاب کن");
   494→      return;
   495→    }
   496→    setPuzzle(built);
   497→    lettersRef.current = {};
   498→    setLetters({});
   499→    setLocked(new Set());
   500→    setPopKeys([]);
   501→    setShakeKeys([]);
   502→    setSolved(new Set());
   503→    setHints(0);
   504→    setSecs(0);
   505→    secsRef.current = 0;
   506→    setIsRecord(false);
   507→    setDir("across");
   508→    const startP = built.placements.find((p) => p.dir === "across") ?? built.placements[0];
   509→    setActive({ r: startP.row, c: startP.col });
   510→    setPhase("playing");
   511→  };
   512→
   513→  /** تایپ یک حرف در سلول فعال + پیشروی + تشخیص خودکار واژهٔ کامل درست */
   514→  const typeLetter = (ch: string) => {
   515→    if (!puzzle || !active || !activeWord) return;
   516→    const up = ch.toUpperCase();
   517→    if (!/^[A-Z]$/.test(up)) return;
   518→    const key = cellKey(active.r, active.c);
   519→    if (locked.has(key)) return;
   520→    const next = { ...lettersRef.current, [key]: up };
   521→    lettersRef.current = next;
   522→    setLetters(next);
   523→
   524→    const completed = completionsIn(next, solved);
   525→    // قفل‌های پس از این تایپ — سلول‌های واژه‌های تازه‌حل‌شده هم رد می‌شوند
   526→    const lockPlus = new Set(locked);
   527→    completed.forEach((id) => cellsOf(puzzle.placements[id]).forEach((k) => lockPlus.add(k)));
   528→
   529→    // پیشروی به بعدیِ آزاد در همان واژه
   530→    const cells = cellsOf(activeWord);
   531→    const idx = cells.indexOf(key);
   532→    let target: string | null = null;
   533→    for (let j = idx + 1; j < cells.length; j++) {
   534→      if (!lockPlus.has(cells[j])) {
   535→        target = cells[j];
   536→        break;
   537→      }
   538→    }
   539→    let jumpDir: CrosswordDir | null = null;
   540→    if (completed.length > 0) {
   541→      applySolved(completed, hints);
   542→      // اگر مقصد داخل واژهٔ تازه‌حل‌شده قفل شد → اولین سلول آزادِ واژهٔ حل‌نشدهٔ بعدی
   543→      if (!target || lockPlus.has(target)) {
   544→        const unsolved = puzzle.placements.find(
   545→          (p) => !solved.has(p.id) && !completed.includes(p.id),
   546→        );
   547→        const free = unsolved ? cellsOf(unsolved).find((k) => !lockPlus.has(k)) : undefined;
   548→        if (free && unsolved) {
   549→          target = free;
   550→          jumpDir = unsolved.dir;
   551→        }
   552→      }
   553→    }
   554→    if (!target) return;
   555→    const [tr, tc] = parseKey(target);
   556→    if (jumpDir) {
   557→      setDir(jumpDir);
   558→    } else {
   559→      const tcell = puzzle.grid[tr][tc];
   560→      const okDir = dir === "across" ? tcell.acrossId !== null : tcell.downId !== null;
   561→      if (!okDir) setDir(dir === "across" ? "down" : "across");
   562→    }
   563→    setActive({ r: tr, c: tc });
   564→  };
   565→
   566→  /** Backspace: پاک‌کردن سلول فعال، و اگر خالی بود پرش به سلول آزاد قبلی و پاک‌کردنش */
   567→  const backspace = () => {
   568→    if (!puzzle || !active || !activeWord) return;
   569→    const key = cellKey(active.r, active.c);
   570→    if (locked.has(key)) return;
   571→    if (lettersRef.current[key]) {
   572→      const next = { ...lettersRef.current };
   573→      delete next[key];
   574→      lettersRef.current = next;
   575→      setLetters(next);
   576→      return;
   577→    }
   578→    const cells = cellsOf(activeWord);
   579→    const idx = cells.indexOf(key);
   580→    for (let j = idx - 1; j >= 0; j--) {
   581→      if (!locked.has(cells[j])) {
   582→        const pk = cells[j];
   583→        const next = { ...lettersRef.current };
   584→        delete next[pk];
   585→        lettersRef.current = next;
   586→        setLetters(next);
   587→        const [pr, pc] = parseKey(pk);
   588→        setActive({ r: pr, c: pc });
   589→        return;
   590→      }
   591→    }
   592→  };
   593→
   594→  /** راهنما: حرف سلول فعال (یا نزدیک‌ترین سلول نیازمند) آشکار می‌شود +۱ شمارش */
   595→  const hint = () => {
   596→    if (!puzzle) return;
   597→    const isFree = (k: string) => !locked.has(k);
   598→    const needsLetter = (k: string) => {
   599→      const [r, c] = parseKey(k);
   600→      return isFree(k) && lettersRef.current[k] !== puzzle.grid[r][c].letter;
   601→    };
   602→    let target: { key: string; dir: CrosswordDir } | null = null;
   603→    if (active && activeWord) {
   604→      const cur = cellKey(active.r, active.c);
   605→      if (needsLetter(cur)) target = { key: cur, dir: activeWord.dir };
   606→      else {
   607→        const free = cellsOf(activeWord).find(needsLetter) ?? cellsOf(activeWord).find(isFree);
   608→        if (free) target = { key: free, dir: activeWord.dir };
   609→      }
   610→    }
   611→    if (!target) {
   612→      const unsolved = puzzle.placements.find((p) => !solved.has(p.id));
   613→      const free = unsolved ? cellsOf(unsolved).find(isFree) : undefined;
   614→      if (free && unsolved) target = { key: free, dir: unsolved.dir };
   615→    }
   616→    if (!target) return;
   617→    const [tr, tc] = parseKey(target.key);
   618→    const next = { ...lettersRef.current, [target.key]: puzzle.grid[tr][tc].letter };
   619→    lettersRef.current = next;
   620→    setLetters(next);
   621→    setHints((h) => h + 1);
   622→    setDir(target.dir);
   623→    setActive({ r: tr, c: tc });
   624→    const completed = completionsIn(next, solved);
   625→    if (completed.length > 0) applySolved(completed, hints + 1);
   626→  };
   627→
   628→  /** بررسی: حروف درست قفل، غلط‌ها لرزش و خالی */
   629→  const check = () => {
   630→    if (!puzzle) return;
   631→    const wrong: string[] = [];
   632→    const right: string[] = [];
   633→    for (const [key, ch] of Object.entries(lettersRef.current)) {
   634→      if (locked.has(key)) continue;
   635→      const [r, c] = parseKey(key);
   636→      const cell = puzzle.grid[r]?.[c];
   637→      if (cell && cell.letter === ch) right.push(key);
   638→      else wrong.push(key);
   639→    }
   640→    if (right.length > 0) {
   641→      setLocked((prev) => {
   642→        const next = new Set(prev);
   643→        right.forEach((k) => next.add(k));
   644→        return next;
   645→      });
   646→    }
   647→    if (wrong.length > 0) {
   648→      setShakeKeys(wrong);
   649→      later(() => {
   650→        setShakeKeys([]);
   651→        const next = { ...lettersRef.current };
   652→        wrong.forEach((k) => delete next[k]);
   653→        lettersRef.current = next;
   654→        setLetters(next);
   655→      }, SHAKE_MS);
   656→      toast.error(`${faNum(wrong.length)} حرف غلط پاک شد${right.length > 0 ? " — درست‌ها قفل شدند" : ""}`);
   657→    } else if (right.length > 0) {
   658→      toast.success(`${faNum(right.length)} حرف درست قفل شد`);
   659→    } else {
   660→      toast("چیزی برای بررسی وارد نشده");
   661→    }
   662→  };
   663→
   664→  /** tap سلول: انتخاب؛ دوبار tap روی همان سلول = تغییر جهت */
   665→  const onCellTap = (r: number, c: number) => {
   666→    if (!puzzle) return;
   667→    const cell = puzzle.grid[r][c];
   668→    if (cell.acrossId === null && cell.downId === null) return;
   669→    if (active && active.r === r && active.c === c) {
   670→      if (cell.acrossId !== null && cell.downId !== null) {
   671→        setDir((d) => (d === "across" ? "down" : "across"));
   672→      }
   673→      return;
   674→    }
   675→    setActive({ r, c });
   676→    const curOk = dir === "across" ? cell.acrossId !== null : cell.downId !== null;
   677→    if (!curOk) setDir(dir === "across" ? "down" : "across");
   678→  };
   679→
   680→  /** tap سرنخ: فعال‌سازی و هایلایت مسیر واژه */
   681→  const focusWord = (p: CrosswordPlacement) => {
   682→    const cells = cellsOf(p);
   683→    const target = cells.find((k) => !locked.has(k)) ?? cells[0];
   684→    const [r, c] = parseKey(target);
   685→    setActive({ r, c });
   686→    setDir(p.dir);
   687→  };
   688→
   689→  // واژهٔ فعال بر اساس سلول و جهت (اگر جهت در این سلول معتبر نباشد، جهت دیگر)
   690→  // بدون useMemo دستی — React Compiler خودش memo می‌کند
   691→  const activeWord: CrosswordPlacement | null = (() => {
   692→    if (!puzzle || !active) return null;
   693→    const cell = puzzle.grid[active.r]?.[active.c];
   694→    if (!cell) return null;
   695→    const primary = dir === "across" ? cell.acrossId : cell.downId;
   696→    if (primary !== null) return puzzle.placements[primary];
   697→    const other = dir === "across" ? cell.downId : cell.acrossId;
   698→    return other !== null ? puzzle.placements[other] : null;
   699→  })();
   700→
   701→  // تایمر — بدون setState مستقیم در effect (react-hooks v6)؛ secsRef برای زمان دقیق پایان
   702→  useEffect(() => {
   703→    if (phase !== "playing") return;
   704→    const t = setInterval(() => {
   705→      secsRef.current += 1;
   706→      setSecs((s) => s + 1);
   707→    }, 1000);
   708→    return () => clearInterval(t);
   709→  }, [phase]);
   710→
   711→  // cleanup کامل unmount: صدا + تایمرهای معلق
   712→  useEffect(() => {
   713→    return () => {
   714→      stopSpeaking();
   715→      timersRef.current.forEach((id) => window.clearTimeout(id));
   716→    };
   717→  }, []);
   718→
   719→  // کیبورد فیزیکی — الگوی latest-ref: شنونده فقط با [phase] ثبت می‌شود و بدنهٔ آن
   720→  // همیشه آخرین نسخهٔ توابع را صدا می‌زند (صفر setState در بدنهٔ effect)
   721→  const keyRef = useRef<(e: KeyboardEvent) => void>(() => {});
   722→  useEffect(() => {
   723→    keyRef.current = (e: KeyboardEvent) => {
   724→      if (exitAskRef.current) return; // نگهبان خروج — تایپ حروف حین دیالوگ تأیید خروج
   725→      if (e.ctrlKey || e.metaKey || e.altKey) return;
   726→      if (e.key === "Backspace") {
   727→        e.preventDefault();
   728→        backspace();
   729→        return;
   730→      }
   731→      if (/^[a-zA-Z]$/.test(e.key)) typeLetter(e.key);
   732→    };
   733→  });
   734→  useEffect(() => {
   735→    if (phase !== "playing") return;
   736→    const onKey = (e: KeyboardEvent) => keyRef.current(e);
   737→    window.addEventListener("keydown", onKey);
   738→    return () => window.removeEventListener("keydown", onKey);
   739→  }, [phase]);
   740→
   741→  // ── Setup ──
   742→  if (phase === "setup") {
   743→    return (
   744→      <Card className="mx-auto max-w-xl rise-in">
   745→        <CardContent className="space-y-6 p-6">
   746→          <div className="flex items-center gap-3">
   747→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-teal-500/15 text-teal-600 dark:text-teal-400">
   748→              <Puzzle size={22} />
   749→            </span>
   750→            <div>
   751→              <h2 className="text-lg font-black">جدول کلمات متقاطع</h2>
   752→              <p className="text-xs text-muted-foreground">
   753→                جدول از واژه‌های خود درس ساخته می‌شود — با سرنخ فارسی حروف را در خانه‌ها بنشین!
   754→              </p>
   755→            </div>
   756→          </div>
   757→
   758→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   759→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   760→            <p>
   761→              معنی فارسی هر واژه را بخوان و حروفش را در جدول بگذار؛ روی خانه بزن تا جهت افقی/عمودی عوض شود.
   762→              سریع‌ترین حلِ بدون راهنما = رکورد!
   763→            </p>
   764→          </div>
   765→
   766→          <div>
   767→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   768→            <div className="flex flex-wrap gap-2" role="group" aria-label="انتخاب درس">
   769→              <button
   770→                type="button"
   771→                onClick={() => setLesson(ALL_LESSONS)}
   772→                aria-pressed={lesson === ALL_LESSONS}
   773→                className={cn(
   774→                  "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   775→                  lesson === ALL_LESSONS
   776→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   777→                    : "text-muted-foreground hover:bg-muted",
   778→                )}
   779→              >
   780→                کل کتاب ({faNum(data.vocab.all.length)} واژه)
   781→              </button>
   782→              {data.vocab.lessons
   783→                .filter((l) => l.lesson > 0)
   784→                .map((l) => (
   785→                  <button
   786→                    key={l.lesson}
   787→                    type="button"
   788→                    onClick={() => setLesson(l.lesson)}
   789→                    aria-pressed={lesson === l.lesson}
   790→                    className={cn(
   791→                      "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   792→                      lesson === l.lesson
   793→                        ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   794→                        : "text-muted-foreground hover:bg-muted",
   795→                    )}
   796→                  >
   797→                    {l.label}
   798→                  </button>
   799→                ))}
   800→            </div>
   801→          </div>
   802→
   803→          <div>
   804→            <p className="mb-2 text-[13px] font-bold">چند واژه؟</p>
   805→            <div className="flex flex-wrap gap-2" role="group" aria-label="تعداد واژه‌ها">
   806→              {COUNT_OPTIONS.map((n) => (
   807→                <button
   808→                  key={n}
   809→                  type="button"
   810→                  onClick={() => setWordCount(n)}
   811→                  aria-pressed={wordCount === n}
   812→                  className={cn(
   813→                    "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   814→                    wordCount === n
   815→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   816→                      : "text-muted-foreground hover:bg-muted",
   817→                  )}
   818→                >
   819→                  {faNum(n)} واژه
   820→                </button>
   821→              ))}
   822→            </div>
   823→          </div>
   824→
   825→          {best && (
   826→            <Badge
   827→              variant="outline"
   828→              className="gap-1.5 border-amber-500/40 text-amber-600 dark:text-amber-400"
   829→            >
   830→              <Trophy size={12} />
   831→              سریع‌ترین حل: {fmtTime(best.time)} (بدون راهنما)
   832→            </Badge>
   833→          )}
   834→
   835→          <Button size="lg" className="h-12 w-full gap-2 text-base" onClick={start}>
   836→            <Play size={18} />
   837→            ساخت جدول و شروع
   838→          </Button>
   839→        </CardContent>
   840→      </Card>
   841→    );
   842→  }
   843→
   844→  if (!puzzle) return null;
   845→
   846→  // ── Done ──
   847→  if (phase === "done") {
   848→    return (
   849→      <Card className="mx-auto max-w-xl rise-in border-amber-500/30">
   850→        <CardContent className="flex flex-col items-center gap-4 p-8 text-center">
   851→          <span className="flex h-16 w-16 items-center justify-center rounded-full bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   852→            <PartyPopper size={30} />
   853→          </span>
   854→          <h2 className="text-xl font-black">جدول کامل شد!</h2>
   855→          <p className="text-xs text-muted-foreground">
   856→            هر {faNum(puzzle.placements.length)} واژه سر جایش نشست — آفرین!

   858→          {isRecord && (
   859→            <Badge className="gap-1.5 bg-amber-500 text-amber-950 shadow-md shadow-amber-500/30">

   881→              <p className="mt-1 text-[10px] text-muted-foreground">راهنما</p>
