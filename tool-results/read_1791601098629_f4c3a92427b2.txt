   257→   257→function fmtDuration(sec: number): string {
   258→   258→  if (!sec) return "—";
   259→   259→  const fa = (v: number | string) => String(v).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   260→   260→  const m = Math.floor(sec / 60);
   261→   261→  const s = sec % 60;
   262→   262→  if (m === 0) return `${fa(s)} ثانیه`;
   263→   263→  return `${fa(m)}:${fa(String(s).padStart(2, "0"))}`;
   264→   264→}

   269→   269→  } catch {
   270→   270→    return "";
   271→   271→  }
