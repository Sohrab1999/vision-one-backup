   100→/** Lightweight CSS confetti — shown when the quiz score is ۸۰٪ or higher. */
   101→function Confetti() {
   102→  const pieces = useMemo(
   103→    () =>
   104→      Array.from({ length: 32 }, (_, i) => ({
   105→        id: i,
   106→        left: Math.random() * 100,
   107→        delay: Math.random() * 1.2,
   108→        duration: 2.4 + Math.random() * 1.6,
   109→        size: 6 + Math.random() * 6,
   110→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   111→        drift: Math.round((Math.random() - 0.5) * 120),
   112→        round: Math.random() > 0.5,
   113→      })),
   114→    [],
   115→  );
   116→  return (
   117→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   118→      {pieces.map((p) => (
   119→        <span
   120→          key={p.id}
   121→          className="confetti-piece absolute top-[-14px] block"
   122→          style={{
   123→            left: `${p.left}%`,
   124→            width: p.size,
   125→            height: p.size * (p.round ? 1 : 0.45),
   126→            background: p.color,
   127→            borderRadius: p.round ? "50%" : 2,
   128→            animationDelay: `${p.delay}s`,
   129→            animationDuration: `${p.duration}s`,
   130→            "--confetti-drift": `${p.drift}px`,
   131→          } as React.CSSProperties}
   132→        />
   133→      ))}
   134→    </div>
   135→  );
   136→}

   350→  function chooseOption(optIdx: number) {
   351→    if (selected !== null) return;
   352→    if (timed > 0 && deadlineRef.current > 0) {
   353→      answeredLeftRef.current = Math.max(0, Math.ceil((deadlineRef.current - Date.now()) / 1000));

   357→      const next = [...prev];

   359→      return next;
   360→    });

   362→      if (optIdx === current.answer) {
   363→        if (mode === "review") resolveMistake(current.id);
   364→      } else {
   365→        recordMistake(current.id);
   366→      }
   367→    }

   393→  function nextQuestion() {
   394→    if (advanceRef.current) clearTimeout(advanceRef.current);
   395→    if (index + 1 >= questions.length) {
   396→      finish();
   397→      return;
   398→    }
   399→    setIndex((i) => i + 1);
   400→    setSelected(answers[index + 1]);
   401→    setAiExplanation(null);
   402→    deadlineRef.current = mode === "daily" ? 0 : Date.now() + timed * 1000;
   403→  }

   405→  // میان‌برهای کیبورد حین آزمون: ۱-۴ (فارسی/انگلیسی) انتخاب گزینه، Enter سؤال بعدی
   406→  useEffect(() => {
   407→    if (phase !== "playing") return;
   408→    function onKey(e: KeyboardEvent) {
   409→      const target = e.target as HTMLElement | null;
   410→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA")) return;
   411→      if (e.key === "Enter" && selected !== null) {
   412→        e.preventDefault();
   413→        nextQuestion();
   414→        return;
   415→      }
   416→      if (selected !== null) return;
   417→      const fa = "۱۲۳۴".indexOf(e.key);
   418→      const en = "1234".indexOf(e.key);
   419→      const digit = fa >= 0 ? fa : en;
   420→      if (digit >= 0 && digit < (current?.options.length ?? 0)) {
   421→        e.preventDefault();
   422→        chooseOption(digit);
   423→      }
   424→    }
   425→    window.addEventListener("keydown", onKey);
   426→    return () => window.removeEventListener("keydown", onKey);
   427→  }, [phase, selected, index, questions]);

   429→  function finish() {
   430→    if (timerRef.current) clearInterval(timerRef.current);
   431→    if (advanceRef.current) clearTimeout(advanceRef.current);

   433→    recordStudySession();
   434→    const streak = touchStreak();
   435→    if (mode === "daily") recordDailyResult(todayKey(), scorePct, questions.length);
   436→    const detail = questions.map((q, i) => ({ id: q.id, chosen: answers[i], answer: q.answer }));
   437→    fetch("/api/attempts", {
   438→      method: "POST",
   439→      headers: { "Content-Type": "application/json" },
   440→      body: JSON.stringify({
   441→        category: mode === "review" ? "review" : mode === "daily" ? "daily" : category,
   442→        lesson: lesson === "all" ? 0 : Number(lesson),
   443→        total: questions.length,
   444→        correct: correctCount,
   445→        duration: elapsed,
   446→        name: userName || undefined,
   447→        detail,
   448→      }),
   449→    })
   450→      .then(() =>
   451→        toast.success(
   452→          mode === "daily"
   453→            ? `چالش امروز ثبت شد — ${scorePct.toLocaleString("fa-IR")}٪ ${scorePct >= 80 ? "🏆" : "💪"}`
   454→            : mode === "review"
   455→              ? `دور مرور تمام شد! ${mistakeCount.toLocaleString("fa-IR")} سؤال دیگر در دفتر اشتباهات مانده`
   456→              : `نتیجه آزمون ذخیره شد ${userName ? `— موفق باشی ${userName}!` : "🎉"}${streak.count > 1 ? ` 🔥 زنجیره ${streak.count.toLocaleString("fa-IR")} روزه` : ""}`,
   457→        ),
   458→      )
   459→      .catch(() => undefined);
   460→  }

   462→  /** خروجی CSV از دفتر اشتباهات — فایل UTF-8 با BOM برای اکسل */
   463→  async function exportMistakesCsv() {
   464→    if (!mistakeIds.length || exporting) return;
   465→    setExporting(true);
   466→    try {
   467→      const res = await fetch(`/api/questions?count=${mistakeIds.length}&ids=${mistakeIds.join(",")}`);
   468→      const data = await res.json();
   469→      const qs = (data.questions ?? []) as Question[];
   470→      if (!qs.length) {
   471→        toast.error("سؤالی برای خروجی یافت نشد.");
   472→        return;
   473→      }
   474→      const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   475→      const rows = [
   476→        ["ردیف", "متن سؤال", "پاسخ صحیح", "دسته", "مبحث", "سطح"].join(","),
   477→        ...qs.map((q, i) =>
   478→          [
   479→            i + 1,
   480→            esc(q.text),
   481→            esc(q.options[q.answer] ?? ""),
   482→            q.category === "grammar" ? "گرامر" : "واژگان",
   483→            esc(q.scope),
   484→            esc(q.level),
   485→          ].join(","),
   486→        ),
   487→      ];
   488→      const blob = new Blob(["\uFEFF" + rows.join("\r\n")], { type: "text/csv;charset=utf-8;" });
   489→      const url = URL.createObjectURL(blob);
   490→      const a = document.createElement("a");
   491→      a.href = url;
   492→      a.download = "vision-one-mistakes.csv";
   493→      document.body.appendChild(a);
   494→      a.click();
   495→      a.remove();
   496→      URL.revokeObjectURL(url);
   497→      toast.success(`فایل CSV با ${qs.length.toLocaleString("fa-IR")} سؤال دانلود شد.`);
   498→    } catch {
