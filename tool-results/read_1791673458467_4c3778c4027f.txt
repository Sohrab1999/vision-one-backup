   201→  // فاز G0-05 — گارد visibilitychange: شمارش زمان سپری‌شده (ملاک رکورد) در تب مخفی می‌ایستد
   202→  // (interval پاک می‌شود) و در بازگشت با همان secsRef ادامه می‌یابد — شمارنده افزایشی است و
   203→  // از Date.now() محاسبه نمی‌شود، پس نیازی به offset نیست. بازگشت از bfcache با pageshow
   204→  // پوشش داده می‌شود؛ unmount همیشه پاک می‌کند.

   329→  const handleCardClick = useCallback(

   368→          });

   370→          busy.current = false;
   371→          if (willFinish) finish(finalMoves, secsRef.current);
   372→        }, MATCH_LOCK_MS);
   373→      } else {
   374→        // جفت غلط — لرزش و بسته‌شدن (کاوش است؛ هیچ چیزی ثبت نمی‌شود)
   375→        setWrongIds([first.id, card.id]);
   376→        setLiveMsg("جفت غلط — کارت‌ها بسته می‌شوند، دوباره تلاش کن");
   377→        playSound("wrong");
   378→        buzz("error");

   380→        setMissedWords((prev) =>
   381→          prev.some((m) => m.front === first.word)
   382→            ? prev
   383→            : [...prev, { front: first.word, back: first.meaning }],
   384→        );
   385→        setTimeout(() => {
   386→          setWrongIds([]);
   387→          setOpenIds([]);
   388→          busy.current = false;
   389→        }, WRONG_LOCK_MS);
   390→      }
   391→    },
   392→    [cards, finish, matchedIds, moves, openIds, paused, speed],
   393→  );
   394→
   395→  // ── فاز G2-a — کیبورد چهارجهته روی گرید کارت‌ها (رفع شکاف کیبورد سند §۵-G2-a) ──
   396→  // الگو: نشانگر سیار (roving cursor) — فقط wrapper گرید تاب‌پذیر است؛ فلش‌ها focusIdx را
   397→  // با گارد مرز سطر/ستون جابه‌جا می‌کنند و Enter/Space همان handleCardClick را صدا می‌زند
   398→  // (گاردهای busy/paused/matched/open داخل همان تابع موجود رعایت می‌شوند — هیچ منطق جدیدی نیست).
   399→  const handleGridKeyDown = useCallback(
   400→    (e: ReactKeyboardEvent<HTMLDivElement>) => {
   401→      if (paused || cards.length === 0) return;
   402→      const key = e.key;
   403→
   404→      // Enter/Space — برگرداندن کارتِ زیر نشانگر کیبورد
   405→      if (key === "Enter" || key === " ") {
   406→        e.preventDefault(); // مهار اسکرول Space و مهار فعال‌سازی دوبارهٔ native دکمه‌ها
   407→        if (focusIdx === null) return;
   408→        const card = cards[focusIdx];
   409→        if (!card) return;
   410→        // اعلام محتوای کارتِ برگشته — فقط وقتی واقعاً باز می‌شود (همان گاردهای handleCardClick)؛
   411→        // برای کارت دوم، پیام جفت درست/غلطِ خودِ بازی جایگزین همین پیام می‌شود
   412→        if (!busy.current && !matchedIds.has(card.id) && !openIds.includes(card.id)) {
   413→          setLiveMsg(
   414→            `کارت شمارهٔ ${faNum(focusIdx + 1)} — ${
   415→              card.side === "en" ? `واژه ${card.word}` : `معنی ${card.meaning}`
   416→            }`,
   417→          );
   418→        }
   419→        handleCardClick(card);
   420→        return;
   421→      }
   422→
   423→      if (!key.startsWith("Arrow")) return;
   424→      e.preventDefault(); // فلش‌ها نباید صفحه را اسکرول کنند
   425→      // ورود اول با کیبورد — از ابتدای گرید شروع کن
   426→      if (focusIdx === null) {
   427→        setFocusIdx(0);
   428→        setLiveMsg(`کارت شمارهٔ ${faNum(1)}`);
   429→        return;
   430→      }
   431→      const last = cards.length - 1;
   432→      let next = focusIdx;
   433→      switch (key) {
   434→        case "ArrowRight":
   435→          // فلش راست = +۱ — گارد مرز: از آخرین ستون عبور نکند
   436→          if (focusIdx % GRID_COLS < GRID_COLS - 1 && focusIdx < last) next = focusIdx + 1;
   437→          break;
   438→        case "ArrowLeft":
   439→          // فلش چپ = کاهش ایندکس — گارد مرز: از اولین ستون عبور نکند
   440→          if (focusIdx % GRID_COLS > 0) next = focusIdx - 1;
   441→          break;
   442→        case "ArrowDown":
   443→          // فلش پایین = +COLS با clamp به انتهای گرید
   444→          next = Math.min(last, focusIdx + GRID_COLS);
   445→          break;
   446→        case "ArrowUp":
   447→          // فلش بالا = منهای COLS با clamp به ابتدای گرید
   448→          next = Math.max(0, focusIdx - GRID_COLS);
   449→          break;
   450→        default:
   451→          return;
   452→      }
   453→      if (next === focusIdx) return;
   454→      setFocusIdx(next);
   455→      // حلقهٔ aria-live — اعلام مقصد؛ کارت بسته فقط شماره می‌گیرد (محتوایش لو نمی‌رود)
   456→      const dest = cards[next];
   457→      const face = dest.side === "en" ? `واژه ${dest.word}` : `معنی ${dest.meaning}`;
   458→      const destMatched = matchedIds.has(dest.id);
   459→      const revealed = destMatched || openIds.includes(dest.id);
   460→      setLiveMsg(
   461→        `کارت شمارهٔ ${faNum(next + 1)}${revealed ? ` — ${face}${destMatched ? " — جفت شد" : ""}` : ""}`,
   462→      );
   463→    },
   464→    [cards, focusIdx, handleCardClick, matchedIds, openIds, paused],
   465→  );
   466→
   467→  // فاز G2-a — گارد اسکرول: کارتِ نشانگرشده همیشه در دید بماند (block: nearest = بدون پرش صفحه)
   468→  useEffect(() => {
   469→    if (focusIdx === null) return;
   470→    gridRef.current
   471→      ?.querySelector<HTMLButtonElement>('button[data-focused="true"]')
   472→      ?.scrollIntoView({ block: "nearest" });
   473→  }, [focusIdx]);
   474→
   475→  // انصراف حین بازی — با پیشرفت، نگهبان خروج؛ بدون پیشرفت، خروج مستقیم
   476→  const handleExitClick = useCallback(() => {
   477→    stopSpeaking();
   478→    // نگهبان خروج (فاز G1-04) — هر پیشرفتی یعنی دور نیمه‌کاره
   479→    if (moves > 0 || openIds.length > 0 || matchedIds.size > 0) {
   480→      setExitAsk(true);
   481→    } else {
   482→      // فاز G1-10 — دور بدون پیشرفت هم یک جلسه است؛ همین‌جا رهاشده ثبت می‌شود
   483→      sessionRef.current?.abandon("exit");
   484→      setPhase("setup");
   485→    }
   486→  }, [matchedIds, moves, openIds]);
   487→
   488→  // ── Setup ──
   489→  if (phase === "setup") {
   490→    return (
   491→      <>
   492→      <Card className="mx-auto max-w-xl rise-in">
   493→        <CardContent className="space-y-6 p-6">
   494→          <div className="flex items-center gap-3">
   495→            <span className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   496→              <LayoutGrid size={22} />
   497→            </span>
   498→            <div>
   499→              <h2 className="text-lg font-black">تطبیق حافظه</h2>
   500→              <p className="text-xs text-muted-foreground">
   501→                کارت‌ها رو به پایین‌اند — هر واژه و معنی‌اش را از حفظ پیدا کن!
   502→              </p>
   503→            </div>
   504→          </div>
   505→
   506→          <div className="flex items-start gap-2.5 rounded-xl border border-amber-500/25 bg-amber-500/8 p-3 text-[11px] leading-5 text-muted-foreground">
   507→            <Sparkles size={15} className="mt-0.5 shrink-0 text-amber-500" />
   508→            <p>
   509→              دو کارت را برگردان؛ اگر واژه انگلیسی و معنی فارسی هم باشند می‌مانند. کمترین حرکت و زمان = رکورد!
   510→            </p>
   511→          </div>
   512→
   513→          <div>
   514→            <p className="mb-2 text-[13px] font-bold">کدام درس؟</p>
   515→            <div className="flex flex-wrap gap-2" role="group" aria-label="انتخاب درس">
   516→              <button
   517→                type="button"
   518→                onClick={() => setLesson(ALL_LESSONS)}
   519→                aria-pressed={lesson === ALL_LESSONS}
   520→                className={cn(
   521→                  "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   522→                  lesson === ALL_LESSONS
   523→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   524→                    : "text-muted-foreground hover:bg-muted",
   525→                )}
   526→              >
   527→                کل کتاب ({faNum(data.vocab.all.length)} واژه)
   528→              </button>
   529→              {data.vocab.lessons.filter((l) => l.lesson > 0).map((l) => (
   530→                <button
   531→                  key={l.lesson}
   532→                  type="button"
   533→                  onClick={() => setLesson(l.lesson)}
   534→                  aria-pressed={lesson === l.lesson}
   535→                  className={cn(
   536→                    "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   537→                    lesson === l.lesson
   538→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   539→                      : "text-muted-foreground hover:bg-muted",
   540→                  )}
   541→                >
   542→                  {l.label}
   543→                </button>
   544→              ))}
   545→            </div>
   546→          </div>
   547→
   548→          <div>
   549→            <p className="mb-2 text-[13px] font-bold">چند جفت؟</p>
   550→            <div className="flex flex-wrap gap-2" role="group" aria-label="تعداد جفت کارت‌ها">
   551→              {PAIR_OPTIONS.map((n) => (
   552→                <button
   553→                  key={n}
   554→                  type="button"
   555→                  onClick={() => setPairCount(n)}
   556→                  aria-pressed={pairCount === n}
   557→                  className={cn(
   558→                    "min-h-11 rounded-full border px-3.5 py-1.5 text-[12px] font-bold transition-all",
   559→                    pairCount === n
   560→                      ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/25"
   561→                      : "text-muted-foreground hover:bg-muted",
   562→                  )}
   563→                >
   564→                  {faNum(n)} جفت
   565→                </button>
   566→              ))}
   567→            </div>
   568→          </div>
   569→
   570→          {best && (
   571→            <Badge variant="outline" className="gap-1.5 border-amber-500/40 text-amber-600 dark:text-amber-400">
   572→              <Trophy size={12} />
   573→              بهترین رکورد: {faNum(best.moves)} حرکت در {fmtTime(best.time)}
   574→            </Badge>
   575→          )}
   576→
   577→          <Button size="lg" className="h-12 w-full gap-2 text-base" onClick={start}>
   578→            <Play size={18} />
   579→            شروع بازی
   580→          </Button>
   581→        </CardContent>
   582→      </Card>
   583→      {/* فاز G1-08 — آموزش یک‌بارنمایش در اولین ورود */}
   584→      <TutorialOverlay
   585→        game={GAME_KEY}
   586→        steps={TUTORIAL_STEPS}
   587→        open={tutorialOpen}
   588→        onOpenChange={handleTutorialOpenChange}
   589→      />
   590→      </>
   591→    );
   592→  }
   593→
   594→  // ── Done ──
   595→  if (phase === "done") {
   596→    // ستاره‌بندی per-game (فاز G1-06): بازی بی‌نقص = ۱۰۰٪ دقت
   597→    const acc = Math.round((pairCount / Math.max(moves, 1)) * 100);
   598→    const stars = acc >= 85 ? 3 : acc >= 60 ? 2 : 1;
   599→    return (
   600→      <ResultScreen
   601→        title="همه جفت‌ها پیدا شد!"
   602→        stars={stars}
   603→        isRecord={isRecord}
   604→        errorCount={missedWords.length}
   605→        reviewItems={missedWords}
   606→        stats={[
   607→          { label: "زمان", value: fmtTime(secs), icon: <Timer size={16} className="text-primary" /> },
   608→          { label: "حرکت", value: faNum(moves), icon: <MousePointerClick size={16} className="text-teal-500" /> },
   609→          { label: "دقت", value: `${faNum(acc)}٪`, icon: <Target size={16} className="text-amber-500" /> },
   610→        ]}
   611→        onReplay={start}
   612→        onExit={() => setPhase("setup")}
   613→        exitLabel="تنظیمات"
   614→      />
   615→    );
   616→  }
   617→
   618→  // ── Playing ──
   619→  const pct = Math.round((matchedIds.size / Math.max(cards.length, 1)) * 100);
   620→  return (
   621→    <>
   622→      {/* فاز G1-05 — هدر یکنواخت از کیت: نام + بج‌ها + راهنما/توقف/صدا/انصراف */}
   623→      <GameShell
   624→        title="تطبیق حافظه"
   625→        icon={<LayoutGrid size={18} />}
   626→        stats={
   627→          <>
   628→            <Badge variant="secondary" className="gap-1 text-[11px]">
   629→              <Timer size={11} />
   630→              {fmtTime(secs)}
   631→            </Badge>
   632→            <Badge variant="secondary" className="gap-1 text-[11px]">
   633→              <Target size={11} className="text-amber-500" />
   634→              {faNum(moves)} حرکت
   635→            </Badge>
   636→            <Badge variant="secondary" className="gap-1 text-[11px]">
   637→              <CheckCircle2 size={11} className="text-primary" />
   638→              {faNum(matchedIds.size / 2)} / {faNum(pairCount)}
   639→            </Badge>
   640→          </>
   641→        }
   642→        liveMsg={liveMsg}
   643→        progress={
   644→          <div className="h-1.5 overflow-hidden rounded-full bg-muted">
   645→            <div
   646→              className="h-full rounded-full bg-gradient-to-l from-primary to-emerald-400 transition-all duration-500"
   647→              style={{ width: `${pct}%` }}
   648→            />
   649→          </div>
   650→        }
   651→        paused={paused}
   652→        onPauseToggle={togglePause}
   653→        muted={muted}
   654→        onMuteToggle={() => setMuted(toggleSoundMuted())}
   655→        onHelp={handleHelp}
   656→        onExit={handleExitClick}
   657→      >
   658→      <div className="relative">
   659→      {/* فاز G2-a — wrapper گرید با tabIndex=0 نقطهٔ ورود Tab است؛ این‌جا onKeyDown می‌گذاریم نه روی هر کارت */}
   660→      <div
   661→        ref={gridRef}
   662→        role="group"
   663→        aria-label="گرید کارت‌ها — با فلش‌ها بین کارت‌ها جابه‌جا شو و با Enter کارت را برگردان"
   664→        tabIndex={paused ? -1 : 0}
   665→        onKeyDown={handleGridKeyDown}
   666→        className={cn("grid grid-cols-3 gap-2 sm:grid-cols-4 sm:gap-3", paused && "pointer-events-none blur-sm")}
   667→        aria-hidden={paused}
   668→      >
   669→        {cards.map((c, i) => {
   670→          const isMatched = matchedIds.has(c.id);
   671→          const isOpen = isMatched || openIds.includes(c.id);
   672→          const isWrong = wrongIds.includes(c.id);
   673→          // فاز G2-a — aria-label وضعیت‌مند (الگوی describeOption سند): کارت بسته فقط شماره؛
   674→          // کارت باز/جفت‌شده متن واقعی (واژهٔ انگلیسی یا معنی فارسی) با موقعیت روی گرید
   675→          const face = c.side === "en" ? `واژه ${c.word}` : `معنی ${c.meaning}`;
   676→          const label = isMatched
   677→            ? `کارت شمارهٔ ${faNum(i + 1)} — ${face} — جفت شد`
   678→            : isOpen
   679→              ? `کارت شمارهٔ ${faNum(i + 1)} — ${face}`
   680→              : `کارت شمارهٔ ${faNum(i + 1)}`;
   681→          // فاز G2-a — الگوی نشانگر سیار: دکمه‌ها از ترتیب Tab خارج‌اند تا دو نشانگر موازی نسازد
   682→          return (
   683→            <button
   684→              key={c.id}
   685→              type="button"
   686→              tabIndex={-1}
   687→              onClick={() => {
   688→                // فاز G2-a — همگام‌سازی نشانگر کیبورد با لمس/کلیک
   689→                setFocusIdx(i);
   690→                handleCardClick(c);
   691→              }}
   692→              disabled={isMatched}
   693→              aria-pressed={isOpen}
   694→              aria-label={label}
   695→              data-focused={i === focusIdx ? "true" : undefined}
   696→              className={cn(
   697→                "flip-scene h-24 sm:h-28 cursor-pointer select-none rounded-lg",
   698→                isMatched && "match-pop",
   699→                isWrong && "match-shake",
   700→                // فاز G2-a — حلقهٔ کهربایی نشانگر کیبورد چهارجهته
   701→                i === focusIdx && "ring-2 ring-amber-400 ring-offset-1",
   702→              )}
   703→            >
   704→              <div className={cn("flip-inner", isOpen && "flipped")}>
   705→                {/* روی کارت — رو به پایین */}
   706→                <div className="flip-face flex h-full w-full items-center justify-center rounded-xl border-2 border-primary/25 bg-gradient-to-br from-primary/15 to-amber-500/10 shadow-md">
   707→                  <span className="text-2xl font-black text-primary/40">؟</span>
   708→                </div>
   709→                {/* پشت کارت — محتوای واقعی */}
   710→                <div
   711→                  className={cn(
   712→                    "flip-face flip-back flex h-full w-full items-center justify-center rounded-xl border bg-card p-2 shadow-md",
   713→                    isMatched && "border-primary/50 opacity-60",
   714→                    isWrong && "border-destructive/60 bg-destructive/10",
   715→                  )}
   716→                >
   717→                  {c.side === "en" ? (
   718→                    <span dir="ltr" className="font-en text-center text-[13px] font-black sm:text-[15px]">
   719→                      {c.word}
   720→                    </span>
   721→                  ) : (
   722→                    <span className="text-center text-[12px] font-bold leading-6 sm:text-[13px]">
   723→                      {c.meaning}
   724→                    </span>
   725→                  )}
   726→                </div>
   727→              </div>
   728→            </button>
   729→          );
   730→        })}
   731→      </div>
   732→
   733→      {/* فاز G1-03 — اورلی توقف: کارت‌ها تار و غیرقابل‌کلیک؛ اعلام متنی در liveMsg سراسری */}
   734→      {paused && (
   735→        <div className="absolute inset-0 z-10 flex flex-col items-center justify-center gap-3 rounded-2xl bg-background/80 backdrop-blur-sm">
   736→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-primary/15 text-primary">
   737→            <Pause size={26} />
   738→          </span>
   739→          <p className="text-sm font-black">بازی متوقف شد</p>
   740→          <Button onClick={resume} className="h-11 gap-2">
   741→            <Play size={16} />
   742→            ادامهٔ بازی
   743→          </Button>
   744→        </div>
   745→      )}
   746→      </div>
   747→
   748→      {/* فاز G1-04 — نگهبان خروج استاندارد کیت */}
   749→      <GameExitGuard
   750→        open={exitAsk}
   751→        onOpenChange={setExitAsk}
   752→        onConfirm={() => {
   753→          // فاز G1-10 — رهاکردن دور نیمه‌کاره
   754→          sessionRef.current?.abandon("exit");
   755→          stopSpeaking();
   756→          setExitAsk(false);
   757→          resume();
   758→          setPhase("setup");
   759→        }}
   760→      />
   761→      </GameShell>
   762→      {/* فاز G1-08 — راهنمای دکمهٔ شِل (حین بازی با توقف باز می‌شود) */}
   763→      <TutorialOverlay
   764→        game={GAME_KEY}
   765→        steps={TUTORIAL_STEPS}
   766→        open={tutorialOpen}
   767→        onOpenChange={handleTutorialOpenChange}
   768→      />
   769→    </>
   770→  );
   771→}
