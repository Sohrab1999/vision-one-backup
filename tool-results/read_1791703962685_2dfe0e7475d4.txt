   478→   478→  /** حرکت مهره خانه‌به‌خانه با زنجیرهٔ پله‌ای (~۱۲۰ms) — هر گام یک transition است (فاز G2-e) */
   479→   479→  function hop(p: number, d: number): void {
   480→   480→    const from = posRef.current[p];
   481→   481→    const to = Math.min(100, from + d);
   482→   482→    let at = from;
   483→   483→    function step(): void {
   484→   484→      if (at < to) {
   485→   485→        at += 1;
   486→   486→        posRef.current[p] = at;
   487→   487→        setPositions([...posRef.current]);
   488→   488→        later(step, STEP_MS);
   489→   489→      } else {
   490→   490→        land(p, d);
   491→   491→      }
   492→   492→    }
   493→   493→    step();
   494→   494→  }

   497→   497→  function land(p: number, d: number): void {

   504→   504→      playSound("wrong");

   506→   506→      const jr = jumpsRef.current[p];
   507→   507→      if (jr) jr.down += 1;

   509→   509→      setLiveMsg(`مار! ${name} به خانهٔ ${faNum(snakeTo)} برگشت`);

   512→   512→      later(() => {

   517→   517→          afterJump(p, d);
   518→   518→        }, JUMP_MS);

   520→   520→    } else if (ladderTo !== undefined) {

   522→   522→      playSound("correct");

   527→   527→      setLiveMsg(`پله! ${name} به خانهٔ ${faNum(ladderTo)} رفت`);
   528→   528→      setSlideMs(JUMP_MS);
   529→   529→      later(() => {
   530→   530→        posRef.current[p] = ladderTo;
   531→   531→        setPositions([...posRef.current]);
   532→   532→        later(() => {
   533→   533→          setSlideMs(STEP_MS);
   534→   534→          afterJump(p, d);
   535→   535→        }, JUMP_MS);
   536→   536→      }, 60);
   537→   537→    } else {
   538→   538→      afterJump(p, d);
   539→   539→    }
   540→   540→  }
