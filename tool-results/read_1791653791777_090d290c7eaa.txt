    98→function pad2(n: number): string {
    99→  return n < 10 ? `0${n}` : String(n);
   100→}

   102→/** تاریخ امروز به شکل YYYY-MM-DD (تقویم محلی — پایدار بین رفرش‌ها) */
   103→function todayKey(): string {
   104→  const d = new Date();
   105→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}`;
   106→}

   108→/** جابه‌جایی تقویمی تاریخ YYYY-MM-DD (بدون وابستگی به منطقهٔ زمانی) */
   109→function shiftDate(date: string, days: number): string {
   110→  const parts = date.split("-").map(Number);
   111→  const t = new Date(Date.UTC(parts[0], (parts[1] ?? 1) - 1, parts[2] ?? 1) + days * 86400000);
   112→  return `${t.getUTCFullYear()}-${pad2(t.getUTCMonth() + 1)}-${pad2(t.getUTCDate())}`;
   113→}
   114→
   115→/** هش ساده و پایدار (djb2) برای انتخاب واژهٔ روز */
   116→function hashStr(s: string): number {
   117→  let h = 5381;
   118→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   119→  return Math.abs(h);
   120→}

   122→/**
   123→ * رنگ‌بندی دو-گذرهٔ وردل — گذر اول فقط سبزها را می‌شمارد، گذر دوم
   124→ * زرد/خاکستری را با شمارش حروف باقی‌ماندهٔ پاسخ تعیین می‌کند
   125→ * (پاسخ کلاسیک: answer "speed" × guess "erase" → Eسبز، Rخاکستری، Aخاکستری، Sسبز، Eزرد).
   126→ */
   127→function judgeVerdicts(guess: string, answer: string): Verdict[] {
   128→  const res: Verdict[] = Array<Verdict>(WORD_LEN).fill("absent");
   129→  const remaining: Record<string, number> = {};
   130→  for (let i = 0; i < WORD_LEN; i++) {
   131→    if (guess[i] === answer[i]) {
   132→      res[i] = "correct";
   133→    } else {
   134→      remaining[answer[i]] = (remaining[answer[i]] ?? 0) + 1;
   135→    }
   136→  }
   137→  for (let i = 0; i < WORD_LEN; i++) {
   138→    if (res[i] === "correct") continue;
   139→    const ch = guess[i];
   140→    if ((remaining[ch] ?? 0) > 0) {
   141→      res[i] = "present";
   142→      remaining[ch] -= 1;
   143→    }
   144→  }
   145→  return res;
   146→}

   181→function loadDailyState(bookId: string): DailyState | null {
   182→  try {
   183→    const raw = localStorage.getItem(dailyKeyOf(bookId));
   184→    if (!raw) return null;
   185→    const p = JSON.parse(raw) as Partial<DailyState> | null;
   186→    if (!p || typeof p.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(p.date)) return null;
   187→    if (p.status !== "playing" && p.status !== "won" && p.status !== "lost") return null;
   188→    if (!Array.isArray(p.guesses)) return null;
   189→    const guesses = p.guesses
   190→      .filter((g): g is string => typeof g === "string" && WORD_RE.test(g))
   191→      .slice(0, MAX_ROWS);
   192→    return { date: p.date, guesses, status: p.status };
   193→  } catch {
   194→    return null;
   195→  }
   196→}
   197→
   198→function writeDailyState(bookId: string, s: DailyState): void {
   199→  try {
   200→    localStorage.setItem(dailyKeyOf(bookId), JSON.stringify(s));
   201→  } catch {
   202→    /* noop */
   203→  }
   204→  notifyGameData();
   205→}

   207→function loadDailyStreak(bookId: string): DailyStreakState | null {
   208→  try {
   209→    const raw = localStorage.getItem(streakKeyOf(bookId));
   210→    if (!raw) return null;
   211→    const p = JSON.parse(raw) as Partial<DailyStreakState> | null;
   212→    if (!p || typeof p.date !== "string" || typeof p.streak !== "number") return null;
   213→    return { date: p.date, streak: p.streak };
   214→  } catch {
   215→    return null;
   216→  }
   217→}

   224→  }
   225→  notifyGameData();
   226→}

   228→export function WordleGame() {

   245→  /** چند ستونِ ردیف در حال فلیپ برگشته‌اند */
   246→  const [flipCols, setFlipCols] = useState(0);
   247→  /** کلید لرزش ردیف جاری — با تغییر key انیمیشن دوباره اجرا می‌شود */
   248→  const [shakeKey, setShakeKey] = useState(0);
   249→  const [isRecord, setIsRecord] = useState(false);
   250→
   251→  /** بازی روزانه بسته شده (کاربر به تنظیمات برگشته)؟ — بازیابی خودکار mount مستقل از این است */
   252→  const [dailyClosed, setDailyClosed] = useState(false);
   253→
   254→  const { speed } = useVoiceSpeed();
   255→  const locked = useRef(false);
   256→  const timers = useRef<number[]>([]);
   257→  const startedAt = useRef(0);
   258→
   259→  // ── استخر واژگان ۵حرفی — یکتا (lowercase) و مرتب (seed پایدار روزانه) ──
   260→  const pool = useMemo<VocabEntry[]>(() => {
   261→    const seen = new Set<string>();
   262→    const out: VocabEntry[] = [];
   263→    for (const w of data.vocab.all) {
   264→      if (!WORD_RE.test(w.word)) continue;
   265→      const key = w.word.toLowerCase();
   266→      if (seen.has(key)) continue;
   267→      seen.add(key);
   268→      out.push(w);
   269→    }
   270→    out.sort((a, b) => {
   271→      const x = a.word.toLowerCase();
   272→      const y = b.word.toLowerCase();
   273→      return x < y ? -1 : x > y ? 1 : 0;
   274→    });
   275→    return out;
   276→  }, [data.vocab]);
   277→
   278→  const poolSet = useMemo(() => new Set(pool.map((w) => w.word.toLowerCase())), [pool]);
   279→
   280→  /** واژهٔ روز — hash پایدار از `${bookId}:${YYYY-MM-DD}` روی استخر مرتب‌شده */
   281→  const dailyAnswer = useMemo(() => {
   282→    if (pool.length === 0) return "";
   283→    return pool[hashStr(`${activeBookId}:${todayKey()}`) % pool.length].word.toLowerCase();
   284→  }, [pool, activeBookId]);
   285→
   286→  // ── وضعیت روزانه از localStorage — استور بیرونی (بازیابی mount بدون setState در effect) ──
   287→  const dailyJson = useSyncExternalStore(
   288→    subscribeGameData,
   289→    () => {
   290→      const s = loadDailyState(activeBookId);
   291→      return s ? JSON.stringify(s) : "null";
   292→    },
   293→    () => "null",
   294→  );
   295→  const dailyRaw = useMemo<DailyState | null>(
   296→    () => (dailyJson === "null" ? null : (JSON.parse(dailyJson) as DailyState)),
   297→    [dailyJson],
   298→  );
   299→
   300→  /** وضعیت روزانهٔ «امروز» — با نرمال‌سازی (رفرش وسط انیمیشن حدس آخر = وضعیت نهایی) */
   301→  const dailyToday = useMemo<DailyState | null>(() => {
   302→    if (!dailyRaw || dailyRaw.date !== todayKey()) return null;
   303→    if (pool.length === 0 || !dailyAnswer) return null;
   304→    if (dailyRaw.status === "playing") {
   305→      if (dailyRaw.guesses.some((g) => g === dailyAnswer)) return { ...dailyRaw, status: "won" };
   306→      if (dailyRaw.guesses.length >= MAX_ROWS) return { ...dailyRaw, status: "lost" };
   307→    }
   308→    return dailyRaw;
   309→  }, [dailyRaw, pool.length, dailyAnswer]);
   310→
   311→  // ── رکوردها و زنجیره — اسنپ‌شات رشته‌ای پایدار (الگوی match-game) ──
   312→  const metaJson = useSyncExternalStore(
   313→    subscribeGameData,
   314→    () =>
   315→      JSON.stringify({
   316→        infBest: loadBest<BestMoves>(activeBookId, GAME_KEY, "infinite"),
   317→        dailyBest: loadBest<BestScore>(activeBookId, GAME_KEY, "daily-streak"),
   318→        streak: loadDailyStreak(activeBookId),
   319→      }),
   320→    () => "null",
   321→  );
   322→  const meta = useMemo<{ infBest: BestMoves | null; dailyBest: BestScore | null; streak: DailyStreakState | null }>(
   323→    () =>
   324→      metaJson === "null"
   325→        ? { infBest: null, dailyBest: null, streak: null }
   326→        : (JSON.parse(metaJson) as { infBest: BestMoves | null; dailyBest: BestScore | null; streak: DailyStreakState | null }),
   327→    [metaJson],
   328→  );
   329→
   330→  // ── تفکیک نماها ──
   331→  /** روزانهٔ امروز فعال است (بازیابی mount یا شروع دستی) */
   332→  const dailyLive = mode === "daily" && !dailyClosed && dailyToday !== null;
   333→  /** نمای بازی روزانه — حتی حین انیمیشن حدس آخر (وضعیت استور هنوز «playing» یا فلیپ تمام‌نشده) */
   334→  const dailyPlayingView = dailyLive && (dailyToday.status === "playing" || revealRow !== -1);
   335→  const dailyDoneView = dailyLive && !dailyPlayingView;
   336→  const infinitePlaying = phase === "playing" && !dailyLive;
   337→
   338→  const activeAnswer = dailyLive ? dailyAnswer : answer;
   339→  const activeCommitted = dailyPlayingView ? dailyToday.guesses : guesses;
   340→
   341→  const answerEntry = useMemo<VocabEntry | null>(
   342→    () => pool.find((w) => w.word.toLowerCase() === activeAnswer) ?? null,
   343→    [pool, activeAnswer],
   344→  );
   345→
   346→  /** بهترین وضعیت هر حرف روی کیبورد — از همهٔ حدس‌های ثبت‌شده */
   347→  const keyStatus = useMemo(() => {
   348→    const map = new Map<string, Verdict>();
   349→    for (const g of activeCommitted) {
   350→      const vs = judgeVerdicts(g, activeAnswer);
   351→      for (let i = 0; i < WORD_LEN; i++) {
   352→        const prev = map.get(g[i]);
   353→        if (!prev || VERDICT_RANK[vs[i]] > VERDICT_RANK[prev]) map.set(g[i], vs[i]);
   354→      }
   355→    }
   356→    return map;
   357→  }, [activeCommitted, activeAnswer]);
   358→
   359→  // ── تایمرهای پله‌ای فلیپ — همیشه پاک‌شونده ──
   360→  const clearTimers = useCallback(() => {
   361→    for (const t of timers.current) window.clearTimeout(t);
   362→    timers.current = [];
   363→  }, []);
   364→
   365→  useEffect(() => {
   366→    return () => {
   367→      stopSpeaking();
   368→      for (const t of timers.current) window.clearTimeout(t);
   369→    };
   370→  }, []);
   371→
   372→  // ── پایان دور — استریک/جلسه/Attempt/SRS/رکورد ──
   373→  const finishGame = useCallback(
   374→    (win: boolean, rowsUsed: number, gameMode: Mode, word: string) => {
   375→      touchStreak();
   376→      recordStudySession();
   377→      bumpPlays(activeBookId, GAME_KEY);
   378→      const started = startedAt.current || Date.now();
   379→      startedAt.current = 0;
   380→      const duration = Math.max(1, Math.round((Date.now() - started) / 1000));
   381→      saveAttempt(activeBookId, GAME_KEY, {
   382→        lesson: 0, // وردل کتاب‌محور است (بدون انتخاب درس)
   383→        total: 1,
   384→        correct: win ? 1 : 0,
   385→        duration,
   386→        detail: { word, rows: rowsUsed, mode: gameMode },
   387→      });
   388→      recordSRS(word, win);
   389→      if (win) {
   390→        speak(word, { speed });
   391→        if (gameMode === "infinite") {
   392→          const prev = loadBest<BestMoves>(activeBookId, GAME_KEY, "infinite");
   393→          if (!prev || rowsUsed < prev.moves) {
   394→            saveBest<BestMoves>(activeBookId, GAME_KEY, "infinite", { moves: rowsUsed });
   395→            setIsRecord(true);
   396→            toast.success(`رکورد جدید! برد با ${faNum(rowsUsed)} حدس 🏆`);
   397→          } else {
   398→            toast.success(`واژه را با ${faNum(rowsUsed)} حدس پیدا کردی! 🎉`);
   399→          }
   400→        } else {
   401→          // زنجیرهٔ برد پیاپی روزانه — از localStorage خود بازی
   402→          const today = todayKey();
   403→          const prevStreak = loadDailyStreak(activeBookId);
   404→          const streak = prevStreak && prevStreak.date === shiftDate(today, -1) ? prevStreak.streak + 1 : 1;
   405→          saveDailyStreak(activeBookId, { date: today, streak });
   406→          const prevBest = loadBest<BestScore>(activeBookId, GAME_KEY, "daily-streak");
   407→          if (!prevBest || streak > prevBest.score) {
   408→            saveBest<BestScore>(activeBookId, GAME_KEY, "daily-streak", { score: streak });
   409→            setIsRecord(true);
   410→            toast.success(`رکورد زنجیره! ${faNum(streak)} برد پیاپی 🏆`);
   411→          } else {
   412→            toast.success(`واژهٔ امروز پیدا شد — زنجیرهٔ ${faNum(streak)} روزه! 🎉`);
   413→          }
   414→        }
   415→      } else {
   416→        toast.error("این‌بار پیدا نشد — واژهٔ درست را ببین");
   417→      }
   418→      if (gameMode === "infinite") setPhase("done"); // روزانه: نما از استور مشتق می‌شود
   419→    },
   420→    [activeBookId, speed],
   421→  );
   422→
   423→  // ── ارسال حدس ──
   424→  const submitGuess = useCallback(() => {
   425→    if (locked.current || !activeAnswer) return;
   426→    const rowIdx = activeCommitted.length;
   427→    if (rowIdx >= MAX_ROWS) return;
   428→    const guess = input.toLowerCase();
   429→    if (guess.length < WORD_LEN) {
   430→      setShakeKey((k) => k + 1);
   431→      toast.error("حدس باید ۵ حرف باشد");
