   156→/**
   157→ * فاز G3-01 — شکل‌دهی گزینه‌های یک سؤال به تعداد سطح:
   158→ * آسان ۳ گزینه (حذف گزینهٔ انحرافی اضافه — پاسخ هرگز حذف نمی‌شود) · عادی ۴ گزینه
   159→ * (بدون هیچ دخل‌و‌تصرفی — همان سؤال خام بانک تست) · سخت ۵ گزینه (مزاحم از استخر
   160→ * گزینه‌های سایر سؤال‌ها، هرگز برابر گزینه‌های موجود). اگر استخر اجازهٔ افزودن
   161→ * ندهد، سؤال بدون تغییر می‌ماند (شکنندگی صفر).
   162→ */
   163→function shapeOptions(
   164→  q: QuizQuestionSeed,
   165→  level: DifficultyKey,
   166→  trapPool: string[],
   167→): QuizQuestionSeed {
   168→  const target = optionsFor(level, 3, 5);
   169→  const opts = [...q.options];
   170→  const ans = opts[q.answer];
   171→  if (!ans || opts.length === target) return q;
   172→  if (opts.length > target) {
   173→    // آسان — حذف گزینه‌های غلط اضافه
   174→    const wrongIdx = opts.map((_, i) => i).filter((i) => i !== q.answer);
   175→    const removed = new Set(shuffle(wrongIdx).slice(0, opts.length - target));
   176→    const next = opts.filter((_, i) => !removed.has(i));
   177→    return { ...q, options: next, answer: Math.max(0, next.indexOf(ans)) };
   178→  }
   179→  // سخت — افزودن گزینهٔ مزاحم از استخر مشترک (شافل عادی — همان الگوی قبلی traps)
   180→  const taken = new Set(opts.map((o) => o.trim().toLowerCase()));
   181→  const extra: string[] = [];
   182→  for (const cand of trapPool) {
   183→    if (extra.length >= target - opts.length) break;
   184→    const key = cand.trim().toLowerCase();
   185→    if (!key || taken.has(key)) continue;
   186→    taken.add(key);
   187→    extra.push(cand);
   188→  }
   189→  if (extra.length === 0) return q; // استخر اجازه نداد — بدون تغییر
   190→  const next = shuffle([...opts, ...extra]);
   191→  return { ...q, options: next, answer: Math.max(0, next.indexOf(ans)) };
   192→}

   194→/**
   195→ * فاز G3-02 — چیدن کارت دور با اولویت SRS ضعیف/سررسید/نو: نیمهٔ اول از smartPick
   196→ * (نمونه‌گیری وزن‌دار کیت) و بقیه با شافل عادی — هیچ سؤالی حذف نمی‌شود و چرخهٔ
   197→ * بی‌پایان دست‌نخورده است؛ فقط ترتیب ورود هوشمندتر است. شروع سرد (SRS خالی) =
   198→ * دقیقاً همان شافل قبلی. فیلتر بر اساس هویت شیء است تا تکراری‌های کلید‌دار از دست نروند.
   199→ */
   200→function smartOrderDeck<T>(pool: T[], keyOf: (item: T) => string): T[] {
   201→  if (pool.length <= 1) return shuffle(pool);
   202→  const n = Math.max(1, Math.min(pool.length - 1, Math.ceil(pool.length / 2)));
   203→  const head = smartPick(pool, n, keyOf);
   204→  const inHead = new Set<T>(head);
   205→  const tail = shuffle(pool.filter((x) => !inHead.has(x)));
   206→  return [...head, ...tail];
   207→}
